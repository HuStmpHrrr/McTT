A client of the Prelude library: Prelude::Arith::Equality, Prelude::Arith::Plus and Prelude::Function::Iter are separate units, loaded once each.

module NatTheory where

import Prelude::Arith::Equality use (Eq; refl; sym; zeroNeSucc)
import Prelude::Arith::Plus use (plus)
open Prelude::Arith::Plus.Algebra use (plusComm)
import Prelude::Function::Iter as Iter

Prelude::Function::Iter instantiated with succ: iterating it is addition.

def iterSucc : forall (n : Nat) (m : Nat) ->
    Eq (Iter.iter 0l Nat (fun (x : Nat) -> succ x) n m) (plus n m) :=
  fun (n : Nat) (m : Nat) ->
    rec n return x . Eq (Iter.iter 0l Nat (fun (y : Nat) -> succ y) x m) (plus x m)
    | zero => refl m
    | succ p, ih => ih
    end
end

A proof of a closed equation normalizes to true, of type True.

eval plusComm 2 3
output
Evaluate plusComm 2 3 --> true : True
eval sym 4 4 (refl 4)
output
Evaluate sym 4 4 (refl 4) --> true : True
eval zeroNeSucc 5
output
Evaluate zeroNeSucc 5
  --> fun (x1 : False) -> Prelude::Arith::Equality.zeroNeSucc 5 x1
  : forall (x1 : False) -> False
eval iterSucc 3 4
output
Evaluate iterSucc 3 4 --> true : True

The same unit, instantiated with doubling.

eval Iter.iter 0l Nat (fun (x : Nat) -> plus x x) 3 1
output
Evaluate Iter.iter 0l Nat (fun (x1 : Nat) -> plus x1 x1) 3 1 --> 8 : Nat

A let-bound number is known in the body, so true proves the equation; Doubling names an instance of Prelude::Function::Iter.

eval let module Doubling := Iter 0l Nat (fun (x : Nat) -> plus x x);
         n := 2;
         p : Eq (Doubling.iter n 1) 4 := true
     in p
     end
output
Evaluate let module M1 := Iter 0l Nat (fun (x1 : Nat) -> plus x1 x1);
             x2 := 2;
             x3 : Eq (M1.iter x2 1) 4 := true
         in x3
         end --> true : True
end
