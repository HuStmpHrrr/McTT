Halving, by the two induction principles; evals below compare them.

module Halving where

Halving by strong induction: half (succ (succ q)) calls itself at q, which is below succ (succ q) by leSucc q. The two case analyses give the result as a function of the hypothesis for all smaller numbers.

def half (n : Nat) : Nat :=
  strongInd (fun (k : Nat) -> Nat)
    (fun (k : Nat) (r : forall (m : Nat) (lt : Lt m k) -> Nat) ->
       caseNat (fun (x : Nat) -> forall (r : forall (m : Nat) (lt : Lt m x) -> Nat) -> Nat)
         (fun (r : forall (m : Nat) (lt : Lt m 0) -> Nat) -> 0)
         (fun (p : Nat) ->
            caseNat (fun (y : Nat) -> forall (r : forall (m : Nat) (lt : Lt m (succ y)) -> Nat) -> Nat)
              (fun (r : forall (m : Nat) (lt : Lt m 1) -> Nat) -> 0)
              (fun (q : Nat) (r : forall (m : Nat) (lt : Lt m (succ (succ q))) -> Nat) ->
                 succ (r q (leSucc q)))
              p)
         k r)
    n
end

Halving by two-step induction, which needs no order hypothesis.

def halfTwo (n : Nat) : Nat :=
  twoStepInd (fun (k : Nat) -> Nat) 0 0 (fun (k : Nat) (h : Nat) -> succ h) n
end

end

open Halving use (half; halfTwo)

eval half 7

eval half 20

eval halfTwo 7

eval halfTwo 20

Case analysis: the predecessor.

Decisions, eliminated into Nat: 1 for the positive branch, 0 for the negative one.

eval orElim (Eq 3 3) (Not (Eq 3 3)) Nat (decEq 3 3)
       (fun (h : Eq 3 3) -> 1) (fun (h : Not (Eq 3 3)) -> 0)

eval orElim (Eq 3 4) (Not (Eq 3 4)) Nat (decEq 3 4)
       (fun (h : Eq 3 4) -> 1) (fun (h : Not (Eq 3 4)) -> 0)

eval orElim (Le 2 5) (Not (Le 2 5)) Nat (decLe 2 5)
       (fun (h : Le 2 5) -> 1) (fun (h : Not (Le 2 5)) -> 0)

eval orElim (Lt 5 5) (Not (Lt 5 5)) Nat (decLt 5 5)
       (fun (h : Lt 5 5) -> 1) (fun (h : Not (Lt 5 5)) -> 0)
