A client of Prelude::Algebra::Monoid and its instances. The laws of powers are proved once, for every monoid; read in the additive, the multiplicative and the composition monoid, they become laws of multiplication, of exponentiation and of iteration.

module Algebra where

import Prelude::Arith::Equality use (Eq; refl; sym; trans)
import Prelude::Arith::Plus use (plus)
open Prelude::Arith::Plus.Basic use (plusZero)
open Prelude::Arith::Plus.Algebra use (plusAssoc; plusCongBoth)
import Prelude::Arith::Mult use (mult)
import Prelude::Arith::Pow use (pow)
import Prelude::Arith::MinMax use (max)
open Prelude::Arith::MinMax.Laws use (maxIdem; maxZero)
import Prelude::Algebra::Monoid as Monoid
import Prelude::Algebra::Instances use (Additive; AdditiveComm; Multiplicative; MultiplicativeComm; Maximum; Composition; Endo; Pointwise; compose)

The powers of the additive and multiplicative monoids are multiplication and exponentiation, by conversion.

module Unfold where

theorem additivePow : forall (x : Nat) (n : Nat) -> Eq (Additive.pow x n) (mult n x) :=
  fun (x : Nat) (n : Nat) -> refl (mult n x)
end

theorem multiplicativePow : forall (x : Nat) (n : Nat) -> Eq (Multiplicative.pow x n) (pow x n) :=
  fun (x : Nat) (n : Nat) -> refl (pow x n)
end

end

Laws of multiplication, as laws of powers in the additive monoid.

module Scaling where

Right distributivity is powPlus.

theorem multPlusRight : forall (m : Nat) (n : Nat) (x : Nat) ->
    Eq (mult (plus m n) x) (plus (mult m x) (mult n x)) :=
  fun (m : Nat) (n : Nat) (x : Nat) -> Additive.Power.powPlus x m n
end

Associativity is powMult.

theorem multAssocPow : forall (m : Nat) (n : Nat) (x : Nat) ->
    Eq (mult (mult m n) x) (mult m (mult n x)) :=
  fun (m : Nat) (n : Nat) (x : Nat) -> Additive.Power.powMult x m n
end

Left distributivity needs commutativity of addition: it is powOp in the commutative instance.

theorem multPlusLeft : forall (n : Nat) (x : Nat) (y : Nat) ->
    Eq (mult n (plus x y)) (plus (mult n x) (mult n y)) :=
  fun (n : Nat) (x : Nat) (y : Nat) -> AdditiveComm.powOp x y n
end

end

Laws of exponents, as laws of powers in the multiplicative monoid. Each proof names the submodule of laws it uses with a local module.

module Exponents where

theorem powPlus : forall (x : Nat) (m : Nat) (n : Nat) ->
    Eq (pow x (plus m n)) (mult (pow x m) (pow x n)) :=
  fun (x : Nat) (m : Nat) (n : Nat) ->
    let module P := Multiplicative.Power in P.powPlus x m n end
end

theorem powMult : forall (x : Nat) (m : Nat) (n : Nat) ->
    Eq (pow x (mult m n)) (pow (pow x n) m) :=
  fun (x : Nat) (m : Nat) (n : Nat) ->
    let module P := Multiplicative.Power in P.powMult x m n end
end

theorem powOfProduct : forall (x : Nat) (y : Nat) (n : Nat) ->
    Eq (pow (mult x y) n) (mult (pow x n) (pow y n)) :=
  fun (x : Nat) (y : Nat) (n : Nat) -> MultiplicativeComm.powOp x y n
end

x to the fourth is the square of its square: mult 2 2 computes to 4.

theorem fourthPower : forall (x : Nat) -> Eq (pow x 4) (pow (pow x 2) 2) :=
  fun (x : Nat) -> let module P := Multiplicative.Power in P.powMult x 2 2 end
end

end

Iteration, as powers in the composition monoid, compared pointwise.

module Iteration where

Iterating m + n times is iterating n times, then m times.

theorem iterPlus : forall (f : Endo) (m : Nat) (n : Nat) ->
    Pointwise (Composition.pow f (plus m n)) (compose (Composition.pow f m) (Composition.pow f n)) :=
  fun (f : Endo) (m : Nat) (n : Nat) -> Composition.Power.powPlus f m n
end

Iterating m * n times is iterating n times, m times over.

theorem iterMult : forall (f : Endo) (m : Nat) (n : Nat) ->
    Pointwise (Composition.pow f (mult m n)) (Composition.pow (Composition.pow f n) m) :=
  fun (f : Endo) (m : Nat) (n : Nat) -> Composition.Power.powMult f m n
end

end

A monoid built here, from a partial application of the unit: the setoid of numbers is fixed, and the operation and its laws come later. It is addition again, so its powers agree with Additive's.

module NatMonoid := Monoid Nat Eq refl sym trans

module Sums := NatMonoid plus plusCongBoth 0 plusAssoc refl plusZero

eval Sums.pow 6 3
output
Evaluate Sums.pow 6 3 --> 18 : Nat

In the maximum monoid every positive power of x is x: the first is max x 0, and each further factor is absorbed by maxIdem. The step names its rewriting under max x in a local module, which opens the congruence laws of the instance.

theorem maxPowSucc : forall (x : Nat) (n : Nat) -> Eq (Maximum.pow x (succ n)) x :=
  fun (x : Nat) (n : Nat) ->
    rec n return k . Eq (Maximum.pow x (succ k)) x
    | zero => maxZero x
    | succ p, ih =>
        let power := Maximum.pow x (succ p);
            module L where
              open Maximum.Cong use (congRight)
              (* One more factor [x] in front of a power that is already [x]. *)
              def front : Eq (max x power) (max x x) := congRight x power x ih end
            end
        in trans (max x power) (max x x) x L.front (maxIdem x)
        end
    end
end
