Equality of natural numbers, computed by recursion on both sides, with its basic laws.

module Prelude::Arith::Equality where

import Prelude::Logic use (Not; absurd)

def Eq : forall (n : Nat) (m : Nat) -> Type@0 :=
  fun (n : Nat) ->
    rec n return y . (forall (m : Nat) -> Type@0)
    | zero => fun (m : Nat) ->
        rec m return z . Type@0 | zero => True | succ q, s => False end
    | succ p, r => fun (m : Nat) ->
        rec m return z . Type@0 | zero => False | succ q, s => r q end
    end
end

theorem refl : forall (n : Nat) -> Eq n n :=
  fun (n : Nat) ->
    rec n return x . Eq x x
    | zero => true
    | succ p, ih => ih
    end
end

theorem zeroNeSucc : forall (n : Nat) -> Not (Eq 0 (succ n)) :=
  fun (n : Nat) (h : Eq 0 (succ n)) -> h
end

theorem sym : forall (a : Nat) (b : Nat) (h : Eq a b) -> Eq b a :=
  fun (a : Nat) ->
    rec a return x . (forall (b : Nat) (h : Eq x b) -> Eq b x)
    | zero => fun (b : Nat) ->
        rec b return z . (forall (h : Eq 0 z) -> Eq z 0)
        | zero => fun (h : True) -> h
        | succ q, s => fun (h : Eq 0 (succ q)) -> absurd (Eq (succ q) 0) h
        end
    | succ p, ih => fun (b : Nat) ->
        rec b return z . (forall (h : Eq (succ p) z) -> Eq z (succ p))
        | zero => fun (h : Eq (succ p) 0) -> h
        | succ q, s => fun (h : Eq (succ p) (succ q)) -> ih q h
        end
    end
end

theorem trans : forall (a : Nat) (b : Nat) (c : Nat) (hab : Eq a b) (hbc : Eq b c) -> Eq a c :=
  fun (a : Nat) ->
    rec a return x . (forall (b : Nat) (c : Nat) (hab : Eq x b) (hbc : Eq b c) -> Eq x c)
    | zero => fun (b : Nat) ->
        rec b return z . (forall (c : Nat) (hab : Eq 0 z) (hbc : Eq z c) -> Eq 0 c)
        | zero => fun (c : Nat) (hab : True) (hbc : Eq 0 c) -> hbc
        | succ q, s => fun (c : Nat) (hab : Eq 0 (succ q)) (hbc : Eq (succ q) c) ->
            absurd (Eq 0 c) hab
        end
    | succ p, ih => fun (b : Nat) ->
        rec b return z . (forall (c : Nat) (hab : Eq (succ p) z) (hbc : Eq z c) -> Eq (succ p) c)
        | zero => fun (c : Nat) (hab : Eq (succ p) 0) (hbc : Eq 0 c) ->
            absurd (Eq (succ p) c) hab
        | succ q, s => fun (c : Nat) ->
            rec c return w . (forall (hab : Eq (succ p) (succ q)) (hbc : Eq (succ q) w) -> Eq (succ p) w)
            | zero => fun (hab : Eq p q) (hbc : Eq (succ q) 0) -> absurd (Eq (succ p) 0) hbc
            | succ r, t => fun (hab : Eq p q) (hbc : Eq q r) -> ih q r hab hbc
            end
        end
    end
end

Every function on numbers respects equality. The proof is by induction on both numbers, with the function generalized so that the successor case can use the hypothesis at fun y -> f (succ y).

theorem cong : forall (f : forall (x : Nat) -> Nat) (a : Nat) (b : Nat) (h : Eq a b) -> Eq (f a) (f b) :=
  fun (f : forall (x : Nat) -> Nat) (a : Nat) ->
    (rec a return x . (forall (g : forall (y : Nat) -> Nat) (b : Nat) (h : Eq x b) -> Eq (g x) (g b))
     | zero => fun (g : forall (y : Nat) -> Nat) (b : Nat) ->
         rec b return z . (forall (h : Eq 0 z) -> Eq (g 0) (g z))
         | zero => fun (h : True) -> refl (g 0)
         | succ q, s => fun (h : Eq 0 (succ q)) -> absurd (Eq (g 0) (g (succ q))) h
         end
     | succ p, ih => fun (g : forall (y : Nat) -> Nat) (b : Nat) ->
         rec b return z . (forall (h : Eq (succ p) z) -> Eq (g (succ p)) (g z))
         | zero => fun (h : Eq (succ p) 0) -> absurd (Eq (g (succ p)) (g 0)) h
         | succ q, s => fun (h : Eq p q) -> ih (fun (y : Nat) -> g (succ y)) q h
         end
     end) f
end

Equal numbers satisfy the same predicates.

def transport : forall (P : forall (n : Nat) -> Type@0) (x : Nat) (y : Nat)
    (h : Eq x y) (px : P x) -> P y :=
  fun (P : forall (n : Nat) -> Type@0) (x : Nat) ->
    (rec x return u . (forall (Q : forall (n : Nat) -> Type@0) (y : Nat)
                         (h : Eq u y) (pu : Q u) -> Q y)
     | zero => fun (Q : forall (n : Nat) -> Type@0) (y : Nat) ->
         rec y return v . (forall (h : Eq 0 v) (pu : Q 0) -> Q v)
         | zero => fun (h : True) (pu : Q 0) -> pu
         | succ q, s => fun (h : Eq 0 (succ q)) (pu : Q 0) -> absurd (Q (succ q)) h
         end
     | succ p, ih => fun (Q : forall (n : Nat) -> Type@0) (y : Nat) ->
         rec y return v . (forall (h : Eq (succ p) v) (pu : Q (succ p)) -> Q v)
         | zero => fun (h : Eq (succ p) 0) (pu : Q (succ p)) -> absurd (Q 0) h
         | succ q, s => fun (h : Eq p q) (pu : Q (succ p)) ->
             ih (fun (n : Nat) -> Q (succ n)) q h pu
         end
     end) P
end
