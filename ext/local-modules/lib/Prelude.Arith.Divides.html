Divides k n says that k divides n: the remainder of n by k is 0. Since mod n 0 is n, Divides 0 n computes to Eq n 0. It computes, so a closed instance is True or False.

def Divides (k : Nat) (n : Nat) : Type@0 := Eq (mod n k) 0 end

Divisibility by and of zero and one, and reflexivity.

module Basic where

Zero divides only zero.

theorem zeroDivides : forall (n : Nat) (h : Divides 0 n) -> Eq n 0 :=
  fun (n : Nat) (h : Eq n 0) -> h
end

theorem dividesRefl : forall (n : Nat) -> Divides n n :=
  fun (n : Nat) ->
    rec n return x . Divides x x
    | zero => true
    | succ d, ih => modSelf d
    end
end

theorem dividesZero : forall (k : Nat) -> Divides k 0 :=
  fun (k : Nat) -> modZero k
end

theorem oneDivides : forall (n : Nat) -> Divides 1 n :=
  fun (n : Nat) -> modOne n
end

end

open Basic use (dividesRefl; dividesZero)

Divisibility is closed under sums, multiples and differences.

module Closure where

A divisor of two numbers divides their sum. For a positive divisor, the remainder of a + b is a steps of wrap from the remainder of b, which is 0 by hb, so it is the remainder of a, which is 0 by ha.

theorem dividesPlus : forall (k : Nat) (a : Nat) (b : Nat) (ha : Divides k a) (hb : Divides k b) ->
    Divides k (plus a b) :=
  fun (k : Nat) (a : Nat) (b : Nat) ->
    rec k return x . (forall (ha : Divides x a) (hb : Divides x b) -> Divides x (plus a b))
    | zero => fun (ha : Eq a 0) (hb : Eq b 0) ->
        transport (fun (y : Nat) -> Eq (plus y b) 0) 0 a (sym a 0 ha) hb
    | succ d, ih => fun (ha : Eq (mod a (succ d)) 0) (hb : Eq (mod b (succ d)) 0) ->
        let e := succ d;
            w := fun (r : Nat) -> wrap r d;
            split : Eq (mod (plus a b) e) (Generic.iterate Nat w a (mod b e)) := modPlus a b d;
            atZero : Eq (Generic.iterate Nat w a (mod b e)) (mod a e) :=
              cong (fun (y : Nat) -> Generic.iterate Nat w a y) (mod b e) 0 hb
        in trans (mod (plus a b) e) (mod a e) 0
             (trans (mod (plus a b) e) (Generic.iterate Nat w a (mod b e)) (mod a e) split atZero)
             ha
        end
    end
end

Every multiple of k is divisible by k.

theorem dividesMult : forall (k : Nat) (m : Nat) -> Divides k (mult m k) :=
  fun (k : Nat) (m : Nat) ->
    rec m return x . Divides k (mult x k)
    | zero => dividesZero k
    | succ p, ih => dividesPlus k k (mult p k) (dividesRefl k) ih
    end
end

A number divisible by succ d is its quotient times succ d: the division equation with a zero remainder.

theorem dividesQuot : forall (d : Nat) (b : Nat) (h : Divides (succ d) b) ->
    Eq (mult (div b (succ d)) (succ d)) b :=
  fun (d : Nat) (b : Nat) (h : Eq (mod b (succ d)) 0) ->
    let k := succ d;
        m := mult (div b k) k;
        dropZero : Eq m (plus m 0) := sym (plus m 0) m (plusZero m);
        toRem : Eq (plus m 0) (plus m (mod b k)) :=
          cong (fun (y : Nat) -> plus m y) 0 (mod b k) (sym (mod b k) 0 h)
    in trans m (plus m 0) b dropZero (trans (plus m 0) (plus m (mod b k)) b toRem (divModSpec b d))
    end
end

A divisor of b divides every multiple mult m b.

theorem dividesMultLeft : forall (k : Nat) (b : Nat) (m : Nat) (h : Divides k b) ->
    Divides k (mult m b) :=
  fun (k : Nat) (b : Nat) (m : Nat) ->
    rec k return x . (forall (h : Divides x b) -> Divides x (mult m b))
    | zero => fun (h : Eq b 0) ->
        transport (fun (y : Nat) -> Eq (mult m y) 0) 0 b (sym b 0 h) (multZero m)
    | succ d, ih => fun (h : Divides (succ d) b) ->
        let e := succ d;
            q := div b e;
            toB : Eq (mult (mult m q) e) (mult m b) :=
              trans (mult (mult m q) e) (mult m (mult q e)) (mult m b)
                (multAssoc m q e)
                (cong (fun (y : Nat) -> mult m y) (mult q e) b (dividesQuot d b h))
        in transport (fun (y : Nat) -> Divides e y) (mult (mult m q) e) (mult m b) toB
             (dividesMult e (mult m q))
        end
    end
end

A divisor of x and of plus x y divides y.

theorem dividesPlusCancel : forall (k : Nat) (x : Nat) (y : Nat)
    (hx : Divides k x) (hxy : Divides k (plus x y)) -> Divides k y :=
  fun (k : Nat) (x : Nat) (y : Nat) ->
    rec k return w . (forall (hx : Divides w x) (hxy : Divides w (plus x y)) -> Divides w y)
    | zero => fun (hx : Eq x 0) (hxy : Eq (plus x y) 0) -> plusEqZeroRight x y hxy
    | succ d, ih => fun (hx : Divides (succ d) x) (hxy : Eq (mod (plus x y) (succ d)) 0) ->
        let e := succ d;
            m := mult (div x e) e;
            atM : Eq (mod (plus m y) e) 0 :=
              transport (fun (z : Nat) -> Eq (mod (plus z y) e) 0) x m
                (sym m x (dividesQuot d x hx)) hxy
        in trans (mod y e) (mod (plus m y) e) 0
             (sym (mod (plus m y) e) (mod y e) (modPlusMult d (div x e) y)) atM
        end
    end
end

Divisibility by two is evenness.

module Two where

The remainder by two is the parity flag: 0 for even numbers and 1 for odd ones. Both halves are proved together, since the step exchanges them.

let modTwoParity : forall (n : Nat) ->
    And (forall (h : Even n) -> Eq (mod n 2) 0) (forall (h : Odd n) -> Eq (mod n 2) 1) :=
  fun (n : Nat) ->
    rec n return x . And (forall (h : Even x) -> Eq (mod x 2) 0)
                         (forall (h : Odd x) -> Eq (mod x 2) 1)
    | zero => andIntro (forall (h : True) -> True) (forall (h : False) -> False)
                (fun (h : True) -> true) (fun (h : False) -> h)
    | succ p, ih =>
        let r := mod p 2;
            fromOdd : forall (h : Odd p) -> Eq (wrap r 1) 0 :=
              fun (h : Odd p) ->
                transport (fun (y : Nat) -> Eq (wrap y 1) 0) 1 r
                  (sym r 1 (andSnd (forall (h : Even p) -> Eq r 0) (forall (h : Odd p) -> Eq r 1) ih h))
                  true;
            fromEven : forall (h : Even p) -> Eq (wrap r 1) 1 :=
              fun (h : Even p) ->
                transport (fun (y : Nat) -> Eq (wrap y 1) 1) 0 r
                  (sym r 0 (andFst (forall (h : Even p) -> Eq r 0) (forall (h : Odd p) -> Eq r 1) ih h))
                  true
        in andIntro (forall (h : Odd p) -> Eq (wrap r 1) 0) (forall (h : Even p) -> Eq (wrap r 1) 1)
             fromOdd fromEven
        end
    end
end

An even number is divisible by two.

theorem evenDividesTwo : forall (n : Nat) (h : Even n) -> Divides 2 n :=
  fun (n : Nat) (h : Even n) ->
    andFst (forall (h : Even n) -> Eq (mod n 2) 0) (forall (h : Odd n) -> Eq (mod n 2) 1)
      (modTwoParity n) h
end

A number divisible by two is even: were it odd, its remainder would be 1 rather than 0.

theorem dividesTwoEven : forall (n : Nat) (h : Divides 2 n) -> Even n :=
  fun (n : Nat) (h : Eq (mod n 2) 0) ->
    orElim (Even n) (Odd n) (Even n) (evenOrOdd n)
      (fun (e : Even n) -> e)
      (fun (o : Odd n) ->
         absurd (Even n)
           (trans 1 (mod n 2) 0
              (sym (mod n 2) 1
                 (andSnd (forall (h : Even n) -> Eq (mod n 2) 0) (forall (h : Odd n) -> Eq (mod n 2) 1)
                    (modTwoParity n) o))
              h))
end
