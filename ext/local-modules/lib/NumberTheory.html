A client of Prelude::Arith::Gcd, its laws, and Prelude::Arith::Prime: greatest common divisors, the primes below 30, and closed instances of the theorems, which normalize to true. The numbers in the proofs stay small, since they compute with remainders and equalities of numbers.

Enumerating the primes. The search for the next prime is a private helper; only nthPrime is exported.

module Primes where

The least prime above n, searched among the numbers from n + 1 to 2 n + 2, which contain one by Bertrand's postulate.

let nextPrime (n : Nat) : Nat :=
  (rec n return y . (forall (c : Nat) -> Nat)
   | zero => fun (c : Nat) -> pick (isPrime c) (fun (u : Nat) -> succ c) (fun (u : Nat) -> c)
   | succ p, r => fun (c : Nat) -> pick (isPrime c) (fun (u : Nat) -> r (succ c)) (fun (u : Nat) -> c)
   end) (succ n)
end

The k-th prime, counting from 0.

def nthPrime (k : Nat) : Nat := Generic.iterate Nat nextPrime (succ k) 0 end

end

open Primes use (nthPrime)
