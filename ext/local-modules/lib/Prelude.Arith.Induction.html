Derived induction principles on natural numbers: case analysis, two-step induction and strong (course-of-values) induction, all from rec.

module Prelude::Arith::Induction where

import Prelude::Logic use (absurd)
open Prelude::Logic.Disj use (orElim)
open Prelude::Logic.Conj use (And; andIntro; andFst; andSnd)
import Prelude::Arith::Equality use (Eq; sym; transport)
import Prelude::Arith::Order use (Le; Lt)
open Prelude::Arith::Order.LtLaws use (leCases)

Case analysis: a property of 0 and of every successor holds of every number. It is rec without the induction hypothesis.

def caseNat (P : forall (n : Nat) -> Type@0) (pz : P 0)
    (ps : forall (n : Nat) -> P (succ n)) : forall (n : Nat) -> P n :=
  fun (n : Nat) ->
    rec n return x . P x
    | zero => pz
    | succ p, ih => ps p
    end
end

Two-step induction: from P 0, P 1, and P n implying P (succ (succ n)), P holds of every number. The induction carries the pair P n and P (succ n); the motive is a conjunction, in Type@1.

def twoStepInd (P : forall (n : Nat) -> Type@0) (pz : P 0) (po : P 1)
    (step : forall (n : Nat) (h : P n) -> P (succ (succ n))) : forall (n : Nat) -> P n :=
  fun (n : Nat) ->
    andFst (P n) (P (succ n))
      (rec n return x . And (P x) (P (succ x))
       | zero => andIntro (P 0) (P 1) pz po
       | succ p, ih =>
           andIntro (P (succ p)) (P (succ (succ p)))
             (andSnd (P p) (P (succ p)) ih)
             (step p (andFst (P p) (P (succ p)) ih))
       end)
end

Strong induction: if P n follows from P m for all m below n, then P holds of every number. The proof strengthens the goal to "P m for all m below n", proved by rec on n: nothing is below 0, and a number below succ p is either below p, covered by the hypothesis, or p itself, given by step.

def strongInd (P : forall (n : Nat) -> Type@0)
    (step : forall (n : Nat) (h : forall (m : Nat) (lt : Lt m n) -> P m) -> P n)
    : forall (n : Nat) -> P n :=
  fun (n : Nat) ->
    step n
      (rec n return x . (forall (m : Nat) (lt : Lt m x) -> P m)
       | zero => fun (m : Nat) (lt : False) -> absurd (P m) lt
       | succ p, ih => fun (m : Nat) (lt : Le m p) ->
           orElim (Lt m p) (Eq m p) (P m) (leCases m p lt)
             (fun (h : Lt m p) -> ih m h)
             (fun (e : Eq m p) -> transport P p m (sym m p e) (step p ih))
       end)
end
