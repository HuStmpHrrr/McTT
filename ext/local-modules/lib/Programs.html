Contents

    A client of the programs in the Prelude library: subtraction, minimum and maximum, generic combinators, and the Ackermann function, run on closed instances.

    module Programs where

import Prelude::Arith::Sub use (pred; sub)
open Prelude::Arith::Sub.Laws use (subSelf; subSuccRight)
open Prelude::Arith::Sub.PlusSub use (plusSub)
import Prelude::Arith::MinMax use (min; max)
open Prelude::Arith::MinMax.Laws use (minComm; maxComm; plusMinMax)
import Prelude::Function::Combinators use (Generic)
open Prelude::Function::Combinators.IterateLaws use (iteratePlus; iterateComm)
import Prelude::Arith::Ackermann use (ack)
open Prelude::Arith::Ackermann.Equations use (ackZero)
open Prelude::Arith::Ackermann.Rows use (ackTwo)
import Prelude::Arith::Plus use (plus)

eval pred 5

eval sub 10 3

    Subtraction is truncated at 0.

    eval sub 3 10

eval min 4 9

eval max 4 9

eval ack 2 3

eval ack 3 3

    A proof of a closed equation normalizes to true, of type True.

    eval plusSub 6 4

eval subSelf 7

eval subSuccRight 9 4

eval plusMinMax 3 8

eval minComm 2 5

eval maxComm 6 1

eval ackZero 4

eval ackTwo 5

eval iteratePlus (fun (x : Nat) -> plus x 3) 2 3 1

eval iterateComm (fun (x : Nat) -> succ (succ x)) 3 4 0

    The generic combinators, instantiated at Nat.

    eval Generic.compose Nat Nat Nat (sub 20) (fun (x : Nat) -> max x 5) 2

eval Generic.const Nat Nat 7 100

eval Generic.iterate Nat (ack 1) 3 0
