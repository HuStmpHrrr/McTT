Propositions as types in Type@0: True holds, False has no proof; negation, Church-encoded conjunction, disjunction, equivalence and existence over numbers, and decidability.

module Prelude::Logic where

def Not (A : Type@0) : Type@0 := forall (a : A) -> False end

From a proof of False, anything follows.

def absurd (A : Type@0) (f : False) : A := exfalso f return x . A end

Conjunction: its introduction, projections and symmetry.

module Conj where

Conjunction, Church-encoded: a proof of And A B eliminates into any proposition C. It quantifies over Type@0, so it lives in Type@1.

def And (A : Type@0) (B : Type@0) : Type@1 :=
  forall (C : Type@0) (f : forall (a : A) (b : B) -> C) -> C
end

def andIntro (A : Type@0) (B : Type@0) (a : A) (b : B) : And A B :=
  fun (C : Type@0) (f : forall (x : A) (y : B) -> C) -> f a b
end

def andFst (A : Type@0) (B : Type@0) (h : And A B) : A :=
  h A (fun (a : A) (b : B) -> a)
end

def andSnd (A : Type@0) (B : Type@0) (h : And A B) : B :=
  h B (fun (a : A) (b : B) -> b)
end

def andSwap (A : Type@0) (B : Type@0) (h : And A B) : And B A :=
  fun (C : Type@0) (f : forall (b : B) (a : A) -> C) ->
    h C (fun (a : A) (b : B) -> f b a)
end

end

open Conj use (And; andIntro; andFst; andSnd; andSwap)

Disjunction: its injections, case analysis and symmetry.

module Disj where

Disjunction, Church-encoded like And, and likewise in Type@1.

def Or (A : Type@0) (B : Type@0) : Type@1 :=
  forall (C : Type@0) (f : forall (a : A) -> C) (g : forall (b : B) -> C) -> C
end

def orInl (A : Type@0) (B : Type@0) (a : A) : Or A B :=
  fun (C : Type@0) (f : forall (x : A) -> C) (g : forall (y : B) -> C) -> f a
end

def orInr (A : Type@0) (B : Type@0) (b : B) : Or A B :=
  fun (C : Type@0) (f : forall (x : A) -> C) (g : forall (y : B) -> C) -> g b
end

Case analysis on a disjunction, into a proposition C.

def orElim (A : Type@0) (B : Type@0) (C : Type@0) (h : Or A B)
    (f : forall (a : A) -> C) (g : forall (b : B) -> C) : C :=
  h C f g
end

Swapping needs no elimination into Type@1: the result is itself a function of the target proposition.

def orSwap (A : Type@0) (B : Type@0) (h : Or A B) : Or B A :=
  fun (C : Type@0) (f : forall (b : B) -> C) (g : forall (a : A) -> C) -> h C g f
end

end

open Disj use (Or; orElim)

Logical equivalence and its rules.

module Equiv where

Logical equivalence: implications both ways, as a conjunction. Like And, it lives in Type@1.

def Iff (A : Type@0) (B : Type@0) : Type@1 :=
  And (forall (a : A) -> B) (forall (b : B) -> A)
end

def iffIntro (A : Type@0) (B : Type@0) (f : forall (a : A) -> B) (g : forall (b : B) -> A)
    : Iff A B :=
  andIntro (forall (a : A) -> B) (forall (b : B) -> A) f g
end

The forward implication of an equivalence.

def iffFwd (A : Type@0) (B : Type@0) (h : Iff A B) (a : A) : B :=
  andFst (forall (x : A) -> B) (forall (y : B) -> A) h a
end

The backward implication of an equivalence.

def iffBwd (A : Type@0) (B : Type@0) (h : Iff A B) (b : B) : A :=
  andSnd (forall (x : A) -> B) (forall (y : B) -> A) h b
end

def iffRefl (A : Type@0) : Iff A A :=
  iffIntro A A (fun (a : A) -> a) (fun (a : A) -> a)
end

def iffSym (A : Type@0) (B : Type@0) (h : Iff A B) : Iff B A :=
  andSwap (forall (a : A) -> B) (forall (b : B) -> A) h
end

end

Existence of a number, and its rules.

module Exist where

Existence of a number satisfying P, Church-encoded: a proof eliminates into any proposition C that follows from some witness and its proof. It quantifies over Type@0, so it lives in Type@1; hence P must be a predicate into Type@0, and an Exists cannot be nested in another.

def Exists (P : forall (n : Nat) -> Type@0) : Type@1 :=
  forall (C : Type@0) (f : forall (n : Nat) (p : P n) -> C) -> C
end

def existsIntro (P : forall (n : Nat) -> Type@0) (n : Nat) (p : P n) : Exists P :=
  fun (C : Type@0) (f : forall (m : Nat) (q : P m) -> C) -> f n p
end

Elimination of an existence proof into a proposition C.

def existsElim (P : forall (n : Nat) -> Type@0) (C : Type@0) (h : Exists P)
    (f : forall (n : Nat) (p : P n) -> C) : C :=
  h C f
end

end

Decidability, and the propositions it makes classical.

module Decidable where

A proposition is decidable when it holds or is refuted. Dec A is a disjunction, so it lives in Type@1, and is eliminated with orElim.

def Dec (A : Type@0) : Type@1 := Or A (Not A) end

A decidable proposition is stable under double negation.

def decStable (A : Type@0) (d : Dec A) (h : Not (Not A)) : A :=
  orElim A (Not A) A d (fun (a : A) -> a) (fun (n : Not A) -> absurd A (h n))
end

Decidability is preserved by negation. As for orSwap, the result is built as a function of the target proposition, since Dec is in Type@1 and orElim only targets Type@0.

def decNot (A : Type@0) (d : Dec A) : Dec (Not A) :=
  fun (C : Type@0) (f : forall (n : Not A) -> C) (g : forall (nn : Not (Not A)) -> C) ->
    d C (fun (a : A) -> g (fun (n : Not A) -> n a)) (fun (n : Not A) -> f n)
end

end
