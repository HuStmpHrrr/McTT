The order on natural numbers, computed by recursion on both sides like Prelude::Arith::Equality.Eq, with its basic laws, case analysis, and a decision function.

module Prelude::Arith::Order where

import Prelude::Logic use (Not; absurd)
open Prelude::Logic.Disj use (Or; orInl; orInr)
import Prelude::Arith::Equality use (Eq; sym; transport)
import Prelude::Arith::Plus use (plus)

Le n m computes to True or False: Le 0 m is True, Le (succ n) 0 is False, and Le (succ n) (succ m) is Le n m.

def Le : forall (n : Nat) (m : Nat) -> Type@0 :=
  fun (n : Nat) ->
    rec n return y . (forall (m : Nat) -> Type@0)
    | zero => fun (m : Nat) -> True
    | succ p, r => fun (m : Nat) ->
        rec m return z . Type@0 | zero => False | succ q, s => r q end
    end
end

Strict order.

def Lt (n : Nat) (m : Nat) : Type@0 := Le (succ n) m end

Le is a total order, and its relation to successors and sums.

module LeLaws where

def leRefl : forall (n : Nat) -> Le n n :=
  fun (n : Nat) ->
    rec n return x . Le x x
    | zero => true
    | succ p, ih => ih
    end
end

def leTrans : forall (a : Nat) (b : Nat) (c : Nat) (hab : Le a b) (hbc : Le b c) -> Le a c :=
  fun (a : Nat) ->
    rec a return x . (forall (b : Nat) (c : Nat) (hab : Le x b) (hbc : Le b c) -> Le x c)
    | zero => fun (b : Nat) (c : Nat) (hab : True) (hbc : Le b c) -> true
    | succ p, ih => fun (b : Nat) ->
        rec b return z . (forall (c : Nat) (hab : Le (succ p) z) (hbc : Le z c) -> Le (succ p) c)
        | zero => fun (c : Nat) (hab : False) (hbc : True) -> absurd (Le (succ p) c) hab
        | succ q, s => fun (c : Nat) ->
            rec c return w . (forall (hab : Le p q) (hbc : Le (succ q) w) -> Le (succ p) w)
            | zero => fun (hab : Le p q) (hbc : False) -> hbc
            | succ r, t => fun (hab : Le p q) (hbc : Le q r) -> ih q r hab hbc
            end
        end
    end
end

theorem leAntisym : forall (a : Nat) (b : Nat) (hab : Le a b) (hba : Le b a) -> Eq a b :=
  fun (a : Nat) ->
    rec a return x . (forall (b : Nat) (hab : Le x b) (hba : Le b x) -> Eq x b)
    | zero => fun (b : Nat) ->
        rec b return z . (forall (hab : True) (hba : Le z 0) -> Eq 0 z)
        | zero => fun (hab : True) (hba : True) -> true
        | succ q, s => fun (hab : True) (hba : False) -> hba
        end
    | succ p, ih => fun (b : Nat) ->
        rec b return z . (forall (hab : Le (succ p) z) (hba : Le z (succ p)) -> Eq (succ p) z)
        | zero => fun (hab : False) (hba : True) -> hab
        | succ q, s => fun (hab : Le p q) (hba : Le q p) -> ih q hab hba
        end
    end
end

Zero is the least number.

theorem leZero : forall (n : Nat) -> Le 0 n :=
  fun (n : Nat) -> true
end

Only zero is below zero.

theorem leZeroInv : forall (n : Nat) (h : Le n 0) -> Eq n 0 :=
  fun (n : Nat) ->
    rec n return x . (forall (h : Le x 0) -> Eq x 0)
    | zero => fun (h : True) -> h
    | succ p, ih => fun (h : False) -> h
    end
end

def leSucc : forall (n : Nat) -> Le n (succ n) :=
  fun (n : Nat) ->
    rec n return x . Le x (succ x)
    | zero => true
    | succ p, ih => ih
    end
end

Weakening of the upper bound by one.

def leStep : forall (a : Nat) (b : Nat) (h : Le a b) -> Le a (succ b) :=
  fun (a : Nat) (b : Nat) (h : Le a b) -> leTrans a b (succ b) h (leSucc b)
end

def lePlusRight : forall (a : Nat) (b : Nat) -> Le a (plus a b) :=
  fun (a : Nat) (b : Nat) ->
    rec a return x . Le x (plus x b)
    | zero => true
    | succ p, ih => ih
    end
end

theorem lePlusLeft : forall (a : Nat) (b : Nat) -> Le b (plus a b) :=
  fun (a : Nat) (b : Nat) ->
    rec a return x . Le b (plus x b)
    | zero => leRefl b
    | succ p, ih => leStep b (plus p b) ih
    end
end

Any two numbers are comparable. In the step, Le (succ p) (succ q) is Le p q, so the hypothesis already has the goal's type.

def leTotal : forall (a : Nat) (b : Nat) -> Or (Le a b) (Le b a) :=
  fun (a : Nat) ->
    rec a return x . (forall (b : Nat) -> Or (Le x b) (Le b x))
    | zero => fun (b : Nat) -> orInl (Le 0 b) (Le b 0) true
    | succ p, ih => fun (b : Nat) ->
        rec b return z . Or (Le (succ p) z) (Le z (succ p))
        | zero => orInr (Le (succ p) 0) (Le 0 (succ p)) true
        | succ q, s => ih q
        end
    end
end

end

Lt is irreflexive and has no element below zero.

module LtLaws where

theorem ltIrrefl : forall (n : Nat) -> Not (Lt n n) :=
  fun (n : Nat) ->
    rec n return x . Not (Lt x x)
    | zero => fun (h : False) -> h
    | succ p, ih => ih
    end
end

theorem notLtZero : forall (n : Nat) -> Not (Lt n 0) :=
  fun (n : Nat) (h : False) -> h
end

A number below another is different from it.

theorem ltNotEq : forall (a : Nat) (b : Nat) (h : Lt a b) (e : Eq a b) -> False :=
  fun (a : Nat) (b : Nat) (h : Lt a b) (e : Eq a b) ->
    ltIrrefl a (transport (fun (x : Nat) -> Lt a x) b a (sym a b e) h)
end

A number at most p is either below p or equal to it. In the step, Lt (succ m) (succ q) and Eq (succ m) (succ q) compute to Lt m q and Eq m q, so the hypothesis already has the goal's type. The impossible case uses exfalso directly, since the goal is in Type@1.

def leCases : forall (m : Nat) (p : Nat) (h : Le m p) -> Or (Lt m p) (Eq m p) :=
  fun (m : Nat) ->
    rec m return x . (forall (p : Nat) (h : Le x p) -> Or (Lt x p) (Eq x p))
    | zero => fun (p : Nat) ->
        rec p return z . (forall (h : True) -> Or (Lt 0 z) (Eq 0 z))
        | zero => fun (h : True) -> orInr (Lt 0 0) (Eq 0 0) true
        | succ q, s => fun (h : True) -> orInl (Lt 0 (succ q)) (Eq 0 (succ q)) true
        end
    | succ r, ih => fun (p : Nat) ->
        rec p return z . (forall (h : Le (succ r) z) -> Or (Lt (succ r) z) (Eq (succ r) z))
        | zero => fun (h : False) -> exfalso h return w . Or (Lt (succ r) 0) (Eq (succ r) 0)
        | succ q, s => fun (h : Le r q) -> ih q h
        end
    end
end

end

Decision of Le: leb a b is 1 when a is at most b, and 0 otherwise.

def leb : forall (n : Nat) (m : Nat) -> Nat :=
  fun (n : Nat) ->
    rec n return y . (forall (m : Nat) -> Nat)
    | zero => fun (m : Nat) -> 1
    | succ p, r => fun (m : Nat) ->
        rec m return z . Nat | zero => 0 | succ q, s => r q end
    end
end

The boolean test leb agrees with Le.

module LebLaws where

leb answers 1 exactly on Le: soundness.

theorem lebSound : forall (a : Nat) (b : Nat) (h : Eq (leb a b) 1) -> Le a b :=
  fun (a : Nat) ->
    rec a return x . (forall (b : Nat) (h : Eq (leb x b) 1) -> Le x b)
    | zero => fun (b : Nat) (h : True) -> true
    | succ p, ih => fun (b : Nat) ->
        rec b return z . (forall (h : Eq (leb (succ p) z) 1) -> Le (succ p) z)
        | zero => fun (h : False) -> h
        | succ q, s => fun (h : Eq (leb p q) 1) -> ih q h
        end
    end
end

leb answers 1 exactly on Le: completeness.

theorem lebComplete : forall (a : Nat) (b : Nat) (h : Le a b) -> Eq (leb a b) 1 :=
  fun (a : Nat) ->
    rec a return x . (forall (b : Nat) (h : Le x b) -> Eq (leb x b) 1)
    | zero => fun (b : Nat) (h : True) -> true
    | succ p, ih => fun (b : Nat) ->
        rec b return z . (forall (h : Le (succ p) z) -> Eq (leb (succ p) z) 1)
        | zero => fun (h : False) -> h
        | succ q, s => fun (h : Le p q) -> ih q h
        end
    end
end

When leb a b answers 0, the order holds strictly the other way.

theorem lebZero : forall (a : Nat) (b : Nat) (h : Eq (leb a b) 0) -> Lt b a :=
  fun (a : Nat) ->
    rec a return x . (forall (b : Nat) (h : Eq (leb x b) 0) -> Lt b x)
    | zero => fun (b : Nat) (h : False) -> absurd (Lt b 0) h
    | succ p, ih => fun (b : Nat) ->
        rec b return z . (forall (h : Eq (leb (succ p) z) 0) -> Lt z (succ p))
        | zero => fun (h : True) -> true
        | succ q, s => fun (h : Eq (leb p q) 0) -> ih q h
        end
    end
end

end
