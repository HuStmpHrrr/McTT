The order on natural numbers, computed by recursion on both sides like Prelude›Arith›Equality.Eq, with its basic laws, case analysis, and a decision function.

module Prelude›Arith›Order where

import Prelude›Logic use (Not; absurd)
open Prelude›Logic.Negation as Neg
open Prelude›Logic.Disj 0l use (Or; orInl; orInr)
import Prelude›Arith›Equality use (Eq; sym; transport)
import Prelude›Arith›Plus use (plus)

Le n m computes to ⊤ or ⊥: Le 0 m is ⊤, Le (succ n) 0 is ⊥, and Le (succ n) (succ m) is Le n m.

def Le : ∀ (n : ℕ) (m : ℕ) → Type@0 ≔
  λ (n : ℕ) →
    rec n return y . (∀ (m : ℕ) → Type@0)
    | zero ⇒ λ (m : ℕ) → ⊤
    | succ p, r ⇒ λ (m : ℕ) →
        rec m return z . Type@0 | zero ⇒ ⊥ | succ q, s ⇒ r q end
    end
end

Strict order.

def Lt (n : ℕ) (m : ℕ) : Type@0 ≔ Le (succ n) m end

Le is a total order, and its relation to successors and sums.

module LeLaws where

def leRefl : ∀ (n : ℕ) → Le n n ≔
  λ (n : ℕ) →
    rec n return x . Le x x
    | zero ⇒ ⋆
    | succ p, ih ⇒ ih
    end
end

def leTrans : ∀ (a : ℕ) (b : ℕ) (c : ℕ) (hab : Le a b) (hbc : Le b c) → Le a c ≔
  λ (a : ℕ) →
    rec a return x . (∀ (b : ℕ) (c : ℕ) (hab : Le x b) (hbc : Le b c) → Le x c)
    | zero ⇒ λ (b : ℕ) (c : ℕ) (hab : ⊤) (hbc : Le b c) → ⋆
    | succ p, ih ⇒ λ (b : ℕ) →
        rec b return z . (∀ (c : ℕ) (hab : Le (succ p) z) (hbc : Le z c) → Le (succ p) c)
        | zero ⇒ λ (c : ℕ) (hab : ⊥) (hbc : ⊤) → absurd (Le (succ p) c) hab
        | succ q, s ⇒ λ (c : ℕ) →
            rec c return w . (∀ (hab : Le p q) (hbc : Le (succ q) w) → Le (succ p) w)
            | zero ⇒ λ (hab : Le p q) (hbc : ⊥) → hbc
            | succ r, t ⇒ λ (hab : Le p q) (hbc : Le q r) → ih q r hab hbc
            end
        end
    end
end

theorem leAntisym : ∀ (a : ℕ) (b : ℕ) (hab : Le a b) (hba : Le b a) → Eq a b ≔
  λ (a : ℕ) →
    rec a return x . (∀ (b : ℕ) (hab : Le x b) (hba : Le b x) → Eq x b)
    | zero ⇒ λ (b : ℕ) →
        rec b return z . (∀ (hab : ⊤) (hba : Le z 0) → Eq 0 z)
        | zero ⇒ λ (hab : ⊤) (hba : ⊤) → ⋆
        | succ q, s ⇒ λ (hab : ⊤) (hba : ⊥) → hba
        end
    | succ p, ih ⇒ λ (b : ℕ) →
        rec b return z . (∀ (hab : Le (succ p) z) (hba : Le z (succ p)) → Eq (succ p) z)
        | zero ⇒ λ (hab : ⊥) (hba : ⊤) → hab
        | succ q, s ⇒ λ (hab : Le p q) (hba : Le q p) → ih q hab hba
        end
    end
end

Zero is the least number.

theorem leZero : ∀ (n : ℕ) → Le 0 n ≔
  λ (n : ℕ) → ⋆
end

Only zero is below zero.

theorem leZeroInv : ∀ (n : ℕ) (h : Le n 0) → Eq n 0 ≔
  λ (n : ℕ) →
    rec n return x . (∀ (h : Le x 0) → Eq x 0)
    | zero ⇒ λ (h : ⊤) → h
    | succ p, ih ⇒ λ (h : ⊥) → h
    end
end

def leSucc : ∀ (n : ℕ) → Le n (succ n) ≔
  λ (n : ℕ) →
    rec n return x . Le x (succ x)
    | zero ⇒ ⋆
    | succ p, ih ⇒ ih
    end
end

Weakening of the upper bound by one.

def leStep : ∀ (a : ℕ) (b : ℕ) (h : Le a b) → Le a (succ b) ≔
  λ (a : ℕ) (b : ℕ) (h : Le a b) → leTrans a b (succ b) h (leSucc b)
end

def lePlusRight : ∀ (a : ℕ) (b : ℕ) → Le a (plus a b) ≔
  λ (a : ℕ) (b : ℕ) →
    rec a return x . Le x (plus x b)
    | zero ⇒ ⋆
    | succ p, ih ⇒ ih
    end
end

theorem lePlusLeft : ∀ (a : ℕ) (b : ℕ) → Le b (plus a b) ≔
  λ (a : ℕ) (b : ℕ) →
    rec a return x . Le b (plus x b)
    | zero ⇒ leRefl b
    | succ p, ih ⇒ leStep b (plus p b) ih
    end
end

Any two numbers are comparable. In the step, Le (succ p) (succ q) is Le p q, so the hypothesis already has the goal's type.

def leTotal : ∀ (a : ℕ) (b : ℕ) → Or (Le a b) (Le b a) ≔
  λ (a : ℕ) →
    rec a return x . (∀ (b : ℕ) → Or (Le x b) (Le b x))
    | zero ⇒ λ (b : ℕ) → orInl (Le 0 b) (Le b 0) ⋆
    | succ p, ih ⇒ λ (b : ℕ) →
        rec b return z . Or (Le (succ p) z) (Le z (succ p))
        | zero ⇒ orInr (Le (succ p) 0) (Le 0 (succ p)) ⋆
        | succ q, s ⇒ ih q
        end
    end
end

Lt is irreflexive and has no element below zero.

module LtLaws where

theorem ltIrrefl : ∀ (n : ℕ) → Not (Lt n n) ≔
  λ (n : ℕ) →
    rec n return x . Not (Lt x x)
    | zero ⇒ λ (h : ⊥) → h
    | succ p, ih ⇒ ih
    end
end

theorem notLtZero : ∀ (n : ℕ) → Not (Lt n 0) ≔
  λ (n : ℕ) (h : ⊥) → h
end

A number below another is different from it.

theorem ltNotEq : ∀ (a : ℕ) (b : ℕ) (h : Lt a b) (e : Eq a b) → ⊥ ≔
  λ (a : ℕ) (b : ℕ) (h : Lt a b) (e : Eq a b) →
    ltIrrefl a (transport (λ (x : ℕ) → Lt a x) b a (sym a b e) h)
end

A number at most p is either below p or equal to it, as a disjunction at any level u, so that it eliminates into motives of Type@{u}. In the step, Lt (succ m) (succ q) and Eq (succ m) (succ q) compute to Lt m q and Eq m q, so the hypothesis already has the goal's type. The impossible case uses absurd one level up, where the disjunction lives.

module Cases (u : Level) where

open Prelude›Logic.Disj u as D

def leCases : ∀ (m : ℕ) (p : ℕ) (h : Le m p) → D.Or (Lt m p) (Eq m p) ≔
  λ (m : ℕ) →
    rec m return x . (∀ (p : ℕ) (h : Le x p) → D.Or (Lt x p) (Eq x p))
    | zero ⇒ λ (p : ℕ) →
        rec p return z . (∀ (h : ⊤) → D.Or (Lt 0 z) (Eq 0 z))
        | zero ⇒ λ (h : ⊤) → D.orInr (Lt 0 0) (Eq 0 0) ⋆
        | succ q, s ⇒ λ (h : ⊤) → D.orInl (Lt 0 (succ q)) (Eq 0 (succ q)) ⋆
        end
    | succ r, ih ⇒ λ (p : ℕ) →
        rec p return z . (∀ (h : Le (succ r) z) → D.Or (Lt (succ r) z) (Eq (succ r) z))
        | zero ⇒ λ (h : ⊥) →
            Neg.absurd (succl u) (D.Or (Lt (succ r) 0) (Eq (succ r) 0)) h
        | succ q, s ⇒ λ (h : Le r q) → ih q h
        end
    end
end

end

Decision of Le: leb a b is 1 when a is at most b, and 0 otherwise.

def leb : ∀ (n : ℕ) (m : ℕ) → ℕ ≔
  λ (n : ℕ) →
    rec n return y . (∀ (m : ℕ) → ℕ)
    | zero ⇒ λ (m : ℕ) → 1
    | succ p, r ⇒ λ (m : ℕ) →
        rec m return z . ℕ | zero ⇒ 0 | succ q, s ⇒ r q end
    end
end

The boolean test leb agrees with Le.

module LebLaws where

leb answers 1 exactly on Le: soundness.

theorem lebSound : ∀ (a : ℕ) (b : ℕ) (h : Eq (leb a b) 1) → Le a b ≔
  λ (a : ℕ) →
    rec a return x . (∀ (b : ℕ) (h : Eq (leb x b) 1) → Le x b)
    | zero ⇒ λ (b : ℕ) (h : ⊤) → ⋆
    | succ p, ih ⇒ λ (b : ℕ) →
        rec b return z . (∀ (h : Eq (leb (succ p) z) 1) → Le (succ p) z)
        | zero ⇒ λ (h : ⊥) → h
        | succ q, s ⇒ λ (h : Eq (leb p q) 1) → ih q h
        end
    end
end

leb answers 1 exactly on Le: completeness.

theorem lebComplete : ∀ (a : ℕ) (b : ℕ) (h : Le a b) → Eq (leb a b) 1 ≔
  λ (a : ℕ) →
    rec a return x . (∀ (b : ℕ) (h : Le x b) → Eq (leb x b) 1)
    | zero ⇒ λ (b : ℕ) (h : ⊤) → ⋆
    | succ p, ih ⇒ λ (b : ℕ) →
        rec b return z . (∀ (h : Le (succ p) z) → Eq (leb (succ p) z) 1)
        | zero ⇒ λ (h : ⊥) → h
        | succ q, s ⇒ λ (h : Le p q) → ih q h
        end
    end
end

When leb a b answers 0, the order holds strictly the other way.

theorem lebZero : ∀ (a : ℕ) (b : ℕ) (h : Eq (leb a b) 0) → Lt b a ≔
  λ (a : ℕ) →
    rec a return x . (∀ (b : ℕ) (h : Eq (leb x b) 0) → Lt b x)
    | zero ⇒ λ (b : ℕ) (h : ⊥) → absurd (Lt b 0) h
    | succ p, ih ⇒ λ (b : ℕ) →
        rec b return z . (∀ (h : Eq (leb (succ p) z) 0) → Lt z (succ p))
        | zero ⇒ λ (h : ⊤) → ⋆
        | succ q, s ⇒ λ (h : Eq (leb p q) 0) → ih q h
        end
    end
end
