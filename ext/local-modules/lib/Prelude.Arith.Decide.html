Decision procedures for equality and order.

module Deciders where

Equality of numbers is decidable. In the step, Eq (succ p) (succ q) computes to Eq p q, so the hypothesis already has the goal's type.

def decEq : forall (n : Nat) (m : Nat) -> Dec (Eq n m) :=
  fun (n : Nat) ->
    rec n return x . (forall (m : Nat) -> Dec (Eq x m))
    | zero => fun (m : Nat) ->
        rec m return z . Dec (Eq 0 z)
        | zero => orInl True (Not True) true
        | succ q, s => orInr False (Not False) (fun (h : False) -> h)
        end
    | succ p, ih => fun (m : Nat) ->
        rec m return z . Dec (Eq (succ p) z)
        | zero => orInr False (Not False) (fun (h : False) -> h)
        | succ q, s => ih q
        end
    end
end

The order is decidable, by the same recursion as decEq.

def decLe : forall (n : Nat) (m : Nat) -> Dec (Le n m) :=
  fun (n : Nat) ->
    rec n return x . (forall (m : Nat) -> Dec (Le x m))
    | zero => fun (m : Nat) -> orInl True (Not True) true
    | succ p, ih => fun (m : Nat) ->
        rec m return z . Dec (Le (succ p) z)
        | zero => orInr False (Not False) (fun (h : False) -> h)
        | succ q, s => ih q
        end
    end
end

The strict order is decidable, as an instance of decLe.

def decLt (n : Nat) (m : Nat) : Dec (Lt n m) := decLe (succ n) m end

end

The conditional on a test: pick b z s is z 0 when b is 0, and s 0 otherwise. The branches are functions of a dummy argument, so only the chosen one is computed.

def pick (b : Nat) (z : forall (u : Nat) -> Nat) (s : forall (u : Nat) -> Nat) : Nat :=
  (rec b return y . (forall (u : Nat) -> Nat)
   | zero => z
   | succ k, t => s
   end) 0
end

The equality test: eqb n m is 1 when n and m are equal, and 0 otherwise.

def eqb : forall (n : Nat) (m : Nat) -> Nat :=
  fun (n : Nat) ->
    rec n return y . (forall (m : Nat) -> Nat)
    | zero => fun (m : Nat) ->
        rec m return z . Nat | zero => 1 | succ q, s => 0 end
    | succ p, r => fun (m : Nat) ->
        rec m return z . Nat | zero => 0 | succ q, s => r q end
    end
end

The boolean equality test eqb agrees with Eq.

module EqbLaws where

eqb answers 1 only on equal numbers.

theorem eqbSound : forall (n : Nat) (m : Nat) (h : Eq (eqb n m) 1) -> Eq n m :=
  fun (n : Nat) ->
    rec n return x . (forall (m : Nat) (h : Eq (eqb x m) 1) -> Eq x m)
    | zero => fun (m : Nat) ->
        rec m return z . (forall (h : Eq (eqb 0 z) 1) -> Eq 0 z)
        | zero => fun (h : True) -> h
        | succ q, s => fun (h : False) -> h
        end
    | succ p, ih => fun (m : Nat) ->
        rec m return z . (forall (h : Eq (eqb (succ p) z) 1) -> Eq (succ p) z)
        | zero => fun (h : False) -> h
        | succ q, s => fun (h : Eq (eqb p q) 1) -> ih q h
        end
    end
end

eqb answers 1 on all equal numbers.

theorem eqbComplete : forall (n : Nat) (m : Nat) (h : Eq n m) -> Eq (eqb n m) 1 :=
  fun (n : Nat) ->
    rec n return x . (forall (m : Nat) (h : Eq x m) -> Eq (eqb x m) 1)
    | zero => fun (m : Nat) ->
        rec m return z . (forall (h : Eq 0 z) -> Eq (eqb 0 z) 1)
        | zero => fun (h : True) -> h
        | succ q, s => fun (h : False) -> h
        end
    | succ p, ih => fun (m : Nat) ->
        rec m return z . (forall (h : Eq (succ p) z) -> Eq (eqb (succ p) z) 1)
        | zero => fun (h : False) -> h
        | succ q, s => fun (h : Eq p q) -> ih q h
        end
    end
end

The specification of eqb: it answers 1 exactly on equal numbers.

def eqbSpec (n : Nat) (m : Nat) : Iff (Eq (eqb n m) 1) (Eq n m) :=
  iffIntro (Eq (eqb n m) 1) (Eq n m) (eqbSound n m) (eqbComplete n m)
end

When eqb answers 0, the numbers differ.

theorem eqbZero : forall (n : Nat) (m : Nat) (h : Eq (eqb n m) 0) -> Not (Eq n m) :=
  fun (n : Nat) ->
    rec n return x . (forall (m : Nat) (h : Eq (eqb x m) 0) -> Not (Eq x m))
    | zero => fun (m : Nat) ->
        rec m return z . (forall (h : Eq (eqb 0 z) 0) -> Not (Eq 0 z))
        | zero => fun (h : False) (e : True) -> h
        | succ q, s => fun (h : True) (e : False) -> e
        end
    | succ p, ih => fun (m : Nat) ->
        rec m return z . (forall (h : Eq (eqb (succ p) z) 0) -> Not (Eq (succ p) z))
        | zero => fun (h : True) (e : False) -> e
        | succ q, s => fun (h : Eq (eqb p q) 0) -> ih q h
        end
    end
end

theorem eqbRefl : forall (n : Nat) -> Eq (eqb n n) 1 :=
  fun (n : Nat) ->
    rec n return x . Eq (eqb x x) 1
    | zero => true
    | succ p, ih => ih
    end
end

Case analysis on eqb a b for a predicate P: it suffices to prove P 0 when eqb a b is 0, and P 1 when a and b are equal.

def eqbCases : forall (a : Nat) (b : Nat) (P : forall (m : Nat) -> Type@0)
    (hz : forall (h : Eq (eqb a b) 0) -> P 0) (hs : forall (h : Eq a b) -> P 1) ->
    P (eqb a b) :=
  fun (a : Nat) ->
    rec a return x . (forall (b : Nat) (P : forall (m : Nat) -> Type@0)
        (hz : forall (h : Eq (eqb x b) 0) -> P 0) (hs : forall (h : Eq x b) -> P 1) ->
        P (eqb x b))
    | zero => fun (b : Nat) ->
        rec b return z . (forall (P : forall (m : Nat) -> Type@0)
            (hz : forall (h : Eq (eqb 0 z) 0) -> P 0) (hs : forall (h : Eq 0 z) -> P 1) ->
            P (eqb 0 z))
        | zero => fun (P : forall (m : Nat) -> Type@0)
            (hz : forall (h : False) -> P 0) (hs : forall (h : True) -> P 1) -> hs true
        | succ q, s => fun (P : forall (m : Nat) -> Type@0)
            (hz : forall (h : True) -> P 0) (hs : forall (h : False) -> P 1) -> hz true
        end
    | succ p, ih => fun (b : Nat) ->
        rec b return z . (forall (P : forall (m : Nat) -> Type@0)
            (hz : forall (h : Eq (eqb (succ p) z) 0) -> P 0)
            (hs : forall (h : Eq (succ p) z) -> P 1) ->
            P (eqb (succ p) z))
        | zero => fun (P : forall (m : Nat) -> Type@0)
            (hz : forall (h : True) -> P 0) (hs : forall (h : False) -> P 1) -> hz true
        | succ q, s => fun (P : forall (m : Nat) -> Type@0)
            (hz : forall (h : Eq (eqb p q) 0) -> P 0) (hs : forall (h : Eq p q) -> P 1) ->
            ih q P hz hs
        end
    end
end

A number at most b for which eqb answers 0 is strictly below b.

theorem leEqbZero : forall (a : Nat) (b : Nat) (h : Le a b) (e : Eq (eqb a b) 0) -> Lt a b :=
  fun (a : Nat) ->
    rec a return x . (forall (b : Nat) (h : Le x b) (e : Eq (eqb x b) 0) -> Lt x b)
    | zero => fun (b : Nat) ->
        rec b return z . (forall (h : Le 0 z) (e : Eq (eqb 0 z) 0) -> Lt 0 z)
        | zero => fun (h : True) (e : False) -> e
        | succ q, s => fun (h : True) (e : True) -> true
        end
    | succ p, ih => fun (b : Nat) ->
        rec b return z . (forall (h : Le (succ p) z) (e : Eq (eqb (succ p) z) 0) -> Lt (succ p) z)
        | zero => fun (h : False) (e : True) -> h
        | succ q, s => fun (h : Le p q) (e : Eq (eqb p q) 0) -> ih q h e
        end
    end
end
