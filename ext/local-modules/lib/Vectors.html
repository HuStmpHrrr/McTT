A client of Prelude::Data::Vec and Prelude::Data::Vec::Properties: small vectors of numbers are built, mapped, zipped, appended, split, reversed, folded, read and summed, and closed instances of the laws normalize to true. Index bounds on closed numbers compute to True, so true proves them.

module Vectors where

import Prelude::Arith::Plus use (plus)
import Prelude::Arith::Mult use (mult)
import Prelude::Arith::Order use (Lt)
import Prelude::Data::Vec as V
import Prelude::Data::Vec::Properties use (sumVec)
open Prelude::Data::Vec::Properties.Nth use (nthMap; nthReplicate; nthAppendLeft; nthAppendRight; nthTabulate; nthZipWith; lastSnoc; nthReverse)
open Prelude::Data::Vec::Properties.Sums use (sumReplicate; sumAppend; sumMapPlus; sumMapScale; sumZipWithPlus; foldlPlus; sumSnoc; sumInitLast; sumReverse; sumTabulate)
open Prelude::Data::Vec::Properties.Slices use (takeAppend; dropAppend; appendTakeDrop)
open Prelude::Data::Vec::Properties.Tests use (allbSound; allbComplete; anybSound)

Vectors of numbers.

module NV := V Nat

The vector 1, 2, 3.

def oneTwoThree : NV.Vec 3 :=
  NV.cons 2 1 (NV.cons 1 2 (NV.cons 0 3 (NV.nil)))
end

The vector 4, 5.

def fourFive : NV.Vec 2 :=
  NV.cons 1 4 (NV.cons 0 5 (NV.nil))
end

def square (x : Nat) : Nat := mult x x end
