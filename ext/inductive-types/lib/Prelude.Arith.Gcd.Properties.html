The invariants of Euclid's algorithm with enough fuel, from which Divisor below reads the laws of gcd.

private module Fuel where

Euclid's algorithm with fuel above b gives a common divisor. Both halves are proved together: the step divides a because it divides succ d and the remainder mod a (succ d).

def divides : ∀ (f : ℕ) (a : ℕ) (b : ℕ) (h : Lt b f) →
    And (Divides (gcdFuel f a b) a) (Divides (gcdFuel f a b) b) ≔
  λ (f : ℕ) →
    elim f return x . (∀ (a : ℕ) (b : ℕ) (h : Lt b x) →
                         And (Divides (gcdFuel x a b) a)
                             (Divides (gcdFuel x a b) b))
    | zero ⇒ λ (a : ℕ) (b : ℕ) (h : ⊥) →
        Neg.absurd 1l
          (And (Divides (gcdFuel 0 a b) a) (Divides (gcdFuel 0 a b) b)) h
    | succ p, ih ⇒ λ (a : ℕ) (b : ℕ) →
        elim b return z . (∀ (h : Lt z (succ p)) →
                             And (Divides (gcdFuel (succ p) a z) a)
                                 (Divides (gcdFuel (succ p) a z) z))
        | zero ⇒ λ (h : ⊤) →
            andIntro (Divides a a) (Divides a 0) (dividesRefl a) (dividesZero a)
        | succ d, s ⇒ λ (h : Le (succ d) p) →
            let e ≔ succ d;
                r ≔ mod a e;
                q ≔ div a e;
                g ≔ gcdFuel p e r;
                both : And (Divides g e) (Divides g r) ≔
                  ih e r (leTrans (succ r) e p (modLt a d) h);
                ge : Divides g e ≔ andFst (Divides g e) (Divides g r) both;
                gr : Divides g r ≔ andSnd (Divides g e) (Divides g r) both;
                ga : Divides g a ≔
                  transport (λ (y : ℕ) → Divides g y) (plus (mult q e) r) a
                    (divModSpec a d)
                    (dividesPlus g (mult q e) r (dividesMultLeft g e q ge) gr)
            in andIntro (Divides g a) (Divides g e) ga ge
            end
        end
    end
end

With fuel above b, every common divisor of a and b divides the result. The step passes the divisor to succ d and to the remainder, which it divides by dividesPlusCancel.

abstract def greatest : ∀ (k : ℕ) (f : ℕ) (a : ℕ) (b : ℕ) (h : Lt b f)
    (ha : Divides k a) (hb : Divides k b) → Divides k (gcdFuel f a b) ≔
  λ (k : ℕ) (f : ℕ) →
    elim f return x . (∀ (a : ℕ) (b : ℕ) (h : Lt b x)
                         (ha : Divides k a) (hb : Divides k b) →
                         Divides k (gcdFuel x a b))
    | zero ⇒ λ (a : ℕ) (b : ℕ) (h : ⊥) (ha : Divides k a) (hb : Divides k b) →
        absurd (Divides k (gcdFuel 0 a b)) h
    | succ p, ih ⇒ λ (a : ℕ) (b : ℕ) →
        elim b return z . (∀ (h : Lt z (succ p)) (ha : Divides k a)
                             (hb : Divides k z) →
                             Divides k (gcdFuel (succ p) a z))
        | zero ⇒ λ (h : ⊤) (ha : Divides k a) (hb : Divides k 0) → ha
        | succ d, s ⇒ λ (h : Le (succ d) p) (ha : Divides k a)
                        (hb : Divides k (succ d)) →
            let e ≔ succ d;
                r ≔ mod a e;
                q ≔ div a e;
                hsum : Divides k (plus (mult q e) r) ≔
                  transport (λ (y : ℕ) → Divides k y) a (plus (mult q e) r)
                    (sym (plus (mult q e) r) a (divModSpec a d)) ha;
                hr : Divides k r ≔
                  dividesPlusCancel k (mult q e) r
                    (dividesMultLeft k e q hb) hsum
            in ih e r (leTrans (succ r) e p (modLt a d) h) hb hr
            end
        end
    end
end

end

The greatest common divisor divides both numbers, and every common divisor divides it.

module Divisor where

The gcd divides its first argument.

abstract def gcdDividesLeft : ∀ (a : ℕ) (b : ℕ) → Divides (gcd a b) a ≔
  λ (a : ℕ) (b : ℕ) →
    andFst (Divides (gcd a b) a) (Divides (gcd a b) b)
      (Fuel.divides (succ b) a b (leRefl b))
end

The gcd divides its second argument.

abstract def gcdDividesRight : ∀ (a : ℕ) (b : ℕ) → Divides (gcd a b) b ≔
  λ (a : ℕ) (b : ℕ) →
    andSnd (Divides (gcd a b) a) (Divides (gcd a b) b)
      (Fuel.divides (succ b) a b (leRefl b))
end

The gcd is the greatest common divisor: every common divisor divides it.

abstract def gcdGreatest : ∀ (k : ℕ) (a : ℕ) (b : ℕ)
    (ha : Divides k a) (hb : Divides k b) →
    Divides k (gcd a b) ≔
  λ (k : ℕ) (a : ℕ) (b : ℕ) (ha : Divides k a) (hb : Divides k b) →
    Fuel.greatest k (succ b) a b (leRefl b) ha hb
end

end

Any two fuels above b give the same result.

abstract def gcdFuelEnough : ∀ (f : ℕ) (g : ℕ) (a : ℕ) (b : ℕ)
    (hf : Lt b f) (hg : Lt b g) → Eq (gcdFuel f a b) (gcdFuel g a b) ≔
  λ (f : ℕ) →
    elim f return x . (∀ (g : ℕ) (a : ℕ) (b : ℕ) (hf : Lt b x)
                         (hg : Lt b g) →
                         Eq (gcdFuel x a b) (gcdFuel g a b))
    | zero ⇒ λ (g : ℕ) (a : ℕ) (b : ℕ) (hf : ⊥) (hg : Lt b g) →
        absurd (Eq (gcdFuel 0 a b) (gcdFuel g a b)) hf
    | succ p, ih ⇒ λ (g : ℕ) →
        elim g return y . (∀ (a : ℕ) (b : ℕ) (hf : Lt b (succ p))
                             (hg : Lt b y) →
                             Eq (gcdFuel (succ p) a b) (gcdFuel y a b))
        | zero ⇒ λ (a : ℕ) (b : ℕ) (hf : Lt b (succ p)) (hg : ⊥) →
            absurd (Eq (gcdFuel (succ p) a b) (gcdFuel 0 a b)) hg
        | succ q, t ⇒ λ (a : ℕ) (b : ℕ) →
            elim b return z . (∀ (hf : Lt z (succ p)) (hg : Lt z (succ q)) →
                                 Eq (gcdFuel (succ p) a z)
                                    (gcdFuel (succ q) a z))
            | zero ⇒ λ (hf : ⊤) (hg : ⊤) → refl a
            | succ d, s ⇒ λ (hf : Le (succ d) p) (hg : Le (succ d) q) →
                let e ≔ succ d;
                    r ≔ mod a e;
                    hr : Le (succ r) e ≔ modLt a d
                in ih q e r (leTrans (succ r) e p hr hf)
                     (leTrans (succ r) e q hr hg)
                end
            end
        end
    end
end

Below b, the first step of Euclid's algorithm swaps the arguments: mod a b is a.

private abstract def gcdSwapSmall : ∀ (a : ℕ) (b : ℕ) (h : Lt a b) →
    Eq (gcd a b) (gcd b a) ≔
  λ (a : ℕ) (b : ℕ) →
    elim b return x . (∀ (h : Lt a x) → Eq (gcd a x) (gcd x a))
    | zero ⇒ λ (h : Lt a 0) → absurd (Eq (gcd a 0) (gcd 0 a)) h
    | succ d, ih ⇒ λ (h : Lt a (succ d)) →
        let e ≔ succ d;
            swap : Eq (gcdFuel e e (mod a e)) (gcdFuel e e a) ≔
              cong (λ (y : ℕ) → gcdFuel e e y) (mod a e) a (modSmall a d h)
        in trans (gcdFuel e e (mod a e)) (gcdFuel e e a) (gcdFuel (succ a) e a)
             swap
             (gcdFuelEnough e (succ a) e a h (leRefl a))
        end
    end
end

Values at zero, one and equal arguments, and commutativity.

module Values where

The gcd with 0 on the left is the other number.

abstract def gcdZeroLeft : ∀ (b : ℕ) → Eq (gcd 0 b) b ≔
  λ (b : ℕ) →
    elim b return x . Eq (gcd 0 x) x
    | zero ⇒ ⋆
    | succ d, ih ⇒ refl (succ d)
    end
end

abstract def gcdZeroRight : ∀ (a : ℕ) → Eq (gcd a 0) a ≔
  λ (a : ℕ) → refl a
end

The gcd of a number with itself is the number: the first step leaves mod n n, which is 0.

abstract def gcdSelf : ∀ (n : ℕ) → Eq (gcd n n) n ≔
  λ (n : ℕ) →
    elim n return x . Eq (gcd x x) x
    | zero ⇒ ⋆
    | succ d, ih ⇒
        cong (λ (y : ℕ) → gcdFuel (succ d) (succ d) y)
          (mod (succ d) (succ d)) 0 (modSelf d)
    end
end

The gcd with 1 is 1: the first step leaves mod a 1, which is 0.

abstract def gcdOneRight : ∀ (a : ℕ) → Eq (gcd a 1) 1 ≔
  λ (a : ℕ) → cong (λ (y : ℕ) → gcdFuel 1 1 y) (mod a 1) 0 (modOne a)
end

The gcd is commutative: compare the arguments, and swap the smaller one to the right with gcdSwapSmall.

abstract def gcdComm : ∀ (a : ℕ) (b : ℕ) → Eq (gcd a b) (gcd b a) ≔
  λ (a : ℕ) (b : ℕ) →
    orElim (Le a b) (Le b a) (Eq (gcd a b) (gcd b a)) (leTotal a b)
      (λ (hab : Le a b) →
         orElim (Lt a b) (Eq a b) (Eq (gcd a b) (gcd b a)) (leCases a b hab)
           (λ (lt : Lt a b) → gcdSwapSmall a b lt)
           (λ (e : Eq a b) →
              transport (λ (y : ℕ) → Eq (gcd a y) (gcd y a)) a b e
                (refl (gcd a a))))
      (λ (hba : Le b a) →
         orElim (Lt b a) (Eq b a) (Eq (gcd a b) (gcd b a)) (leCases b a hba)
           (λ (lt : Lt b a) → sym (gcd b a) (gcd a b) (gcdSwapSmall b a lt))
           (λ (e : Eq b a) →
              transport (λ (y : ℕ) → Eq (gcd y b) (gcd b y)) b a e
                (refl (gcd b b))))
end

end
