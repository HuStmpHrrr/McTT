Prelude›Data›Church

Church encodings at the universe level u: numerals, booleans and lists.

A Church encoding at level u quantifies over the types of Type@{u}, so it lives one level up, in Type@{succl u}. Every operation here stays at u, so the level is the unit's parameter: Prelude›Data›Church 0l has the numerals over the types of Type@0. The operations that relate two levels, such as exponentiation, are in Prelude›Data›Church›Levels. Pairs and sums at level u are Prelude›Logic.Conj u and Prelude›Logic.Disj u. The laws are in Prelude›Data›Church›Properties.

module Prelude›Data›Church (u : Level) where

import Prelude›Builtin›Nat use (succ)
import Prelude›Arith›Plus use (plus)

Numerals: n applies a function n times.

def Numeral : Type@{succl u} ≔
  ∀ (A : Type@{u}) (f : ∀ (a : A) → A) (x : A) → A
end

def czero : Numeral ≔ λ (A : Type@{u}) (f : ∀ (a : A) → A) (x : A) → x end

def csucc (n : Numeral) : Numeral ≔
  λ (A : Type@{u}) (f : ∀ (a : A) → A) (x : A) → f (n A f x)
end

cplus m n applies f n times, then m times.

def cplus (m : Numeral) (n : Numeral) : Numeral ≔
  λ (A : Type@{u}) (f : ∀ (a : A) → A) (x : A) → m A f (n A f x)
end

cmult m n applies n-fold f m times.

def cmult (m : Numeral) (n : Numeral) : Numeral ≔
  λ (A : Type@{u}) (f : ∀ (a : A) → A) → m A (n A f)
end

The numeral of a number, by recursion.

def fromNat (n : ℕ) : Numeral ≔
  elim n return y . Numeral | zero ⇒ czero | succ p, r ⇒ csucc r end
end

The number of a numeral: ℕ is in Type@0, hence in every Type@{u}.

def toNat (n : Numeral) : ℕ ≔ n ℕ (λ (k : ℕ) → succ k) 0 end

Booleans: a boolean chooses one of two elements.

def Boolean : Type@{succl u} ≔ ∀ (A : Type@{u}) (t : A) (f : A) → A end

def ctrue : Boolean ≔ λ (A : Type@{u}) (t : A) (f : A) → t end

def cfalse : Boolean ≔ λ (A : Type@{u}) (t : A) (f : A) → f end

def cif (A : Type@{u}) (b : Boolean) (t : A) (f : A) : A ≔ b A t f end

def cnot (b : Boolean) : Boolean ≔
  λ (A : Type@{u}) (t : A) (f : A) → b A f t
end

def cand (b : Boolean) (c : Boolean) : Boolean ≔
  λ (A : Type@{u}) (t : A) (f : A) → b A (c A t f) f
end

The boolean of a boolean, as 1 or 0.

def boolToNat (b : Boolean) : ℕ ≔ b ℕ 1 0 end

Lists over A: a list is its right fold.

def List (A : Type@{u}) : Type@{succl u} ≔
  ∀ (C : Type@{u}) (n : C) (c : ∀ (a : A) (r : C) → C) → C
end

def nil (A : Type@{u}) : List A ≔
  λ (C : Type@{u}) (n : C) (c : ∀ (a : A) (r : C) → C) → n
end

def cons (A : Type@{u}) (a : A) (l : List A) : List A ≔
  λ (C : Type@{u}) (n : C) (c : ∀ (x : A) (r : C) → C) → c a (l C n c)
end

Right fold, with the step before the start as in Prelude›Data›List.Ops.foldr: foldr A C c n l is l C n c.

def foldr (A : Type@{u}) (C : Type@{u}) (c : ∀ (a : A) (r : C) → C) (n : C)
    (l : List A) : C ≔
  l C n c
end

def append (A : Type@{u}) (k : List A) (l : List A) : List A ≔
  λ (C : Type@{u}) (n : C) (c : ∀ (a : A) (r : C) → C) → k C (l C n c) c
end

def map (A : Type@{u}) (B : Type@{u}) (f : ∀ (a : A) → B) (l : List A) :
    List B ≔
  λ (C : Type@{u}) (n : C) (c : ∀ (b : B) (r : C) → C) →
    l C n (λ (a : A) (r : C) → c (f a) r)
end

def length (A : Type@{u}) (l : List A) : ℕ ≔
  l ℕ 0 (λ (a : A) (r : ℕ) → succ r)
end

The list of n copies of a.

def replicate (A : Type@{u}) (n : ℕ) (a : A) : List A ≔
  elim n return y . List A | zero ⇒ nil A | succ p, r ⇒ cons A a r end
end

The sum of a list of numbers.

def sum (l : List ℕ) : ℕ ≔ l ℕ 0 plus end
