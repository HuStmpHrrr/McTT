Prelude›Algebra›Integers

The integers, as differences of numbers. An integer is a pair (a, b) of numbers that stands for a - b, and two pairs are the same integer, Same x y, when they stand for the same difference: a + d = c + b for (a, b) and (c, d). Addition is componentwise and negation swaps the pair. With ofNat 0 as zero they form a commutative group, Additive, an instance of Prelude›Algebra›Group; its powers are the multiples of an integer.

Same computes, so a closed instance is ⊤ or ⊥. magnitude and negative read a pair back as a sign and a number.

A pair of numbers, as a function from the index 0 or 1.

def Int : Type@0 ≔ ∀ (i : ℕ) → ℕ end

def pair (a : ℕ) (b : ℕ) : Int ≔
  λ (i : ℕ) → elim i return y . ℕ | zero ⇒ a | succ j, r ⇒ b end
end

x and y stand for the same difference.

def Same (x : Int) (y : Int) : Type@0 ≔
  Eq (plus (x 0) (y 1)) (plus (y 0) (x 1))
end

The number n as the integer n - 0.

def ofNat (n : ℕ) : Int ≔ pair n 0 end

def add (x : Int) (y : Int) : Int ≔
  pair (plus (x 0) (y 0)) (plus (x 1) (y 1))
end

def negate (x : Int) : Int ≔ pair (x 1) (x 0) end

x - y.

def minus (x : Int) (y : Int) : Int ≔ add x (negate y) end

The absolute value of the difference, read back by truncated subtraction.

def magnitude (x : Int) : ℕ ≔
  plus (monus (x 0) (x 1)) (monus (x 1) (x 0))
end

tt when the difference is negative, that is when x 1 is not at most x 0, and ff otherwise.

def negative (x : Int) : Bool ≔ not (leb (x 1) (x 0)) end

Same is an equivalence.

module Setoid where

theorem sameRefl : ∀ (x : Int) → Same x x ≔
  λ (x : Int) → refl (plus (x 0) (x 1))
end

theorem sameSym : ∀ (x : Int) (y : Int) (h : Same x y) → Same y x ≔
  λ (x : Int) (y : Int) (h : Same x y) →
    sym (plus (x 0) (y 1)) (plus (y 0) (x 1)) h
end

Adding y 1 to both sides of the goal, the two hypotheses rewrite it into an equation that holds by rearranging; y 1 then cancels.

theorem sameTrans : ∀ (x : Int) (y : Int) (z : Int)
    (hxy : Same x y) (hyz : Same y z) →
    Same x z ≔
  λ (x : Int) (y : Int) (z : Int) (hxy : Same x y) (hyz : Same y z) →
    let a ≔ x 0;
        b ≔ x 1;
        c ≔ y 0;
        d ≔ y 1;
        f ≔ z 0;
        g ≔ z 1;
        start : Eq (plus (plus a g) d) (plus (plus a d) g) ≔
          plusRightComm a g d;
        first : Eq (plus (plus a d) g) (plus (plus c b) g) ≔
          cong (λ (w : ℕ) → plus w g) (plus a d) (plus c b) hxy;
        middle : Eq (plus (plus c b) g) (plus (plus c g) b) ≔
          plusRightComm c b g;
        second : Eq (plus (plus c g) b) (plus (plus f d) b) ≔
          cong (λ (w : ℕ) → plus w b) (plus c g) (plus f d) hyz;
        finish : Eq (plus (plus f d) b) (plus (plus f b) d) ≔
          plusRightComm f d b
    in plusCancelRight (plus a g) (plus f b) d
         (trans (plus (plus a g) d) (plus (plus a d) g) (plus (plus f b) d)
            start
            (trans (plus (plus a d) g) (plus (plus c b) g) (plus (plus f b) d)
               first
               (trans (plus (plus c b) g) (plus (plus c g) b)
                  (plus (plus f b) d) middle
                  (trans (plus (plus c g) b) (plus (plus f d) b)
                     (plus (plus f b) d) second finish))))
    end
end

end

open Setoid use (sameRefl; sameSym; sameTrans)

Addition and negation respect Same, and satisfy the group laws.

module Addition where

theorem addCong : ∀ (a : Int) (b : Int) (c : Int) (d : Int)
    (hac : Same a c) (hbd : Same b d) →
    Same (add a b) (add c d) ≔
  λ (a : Int) (b : Int) (c : Int) (d : Int)
    (hac : Same a c) (hbd : Same b d) →
    let lhs ≔ plus (plus (a 0) (b 0)) (plus (c 1) (d 1));
        rhs ≔ plus (plus (c 0) (d 0)) (plus (a 1) (b 1))
    in trans lhs (plus (plus (a 0) (c 1)) (plus (b 0) (d 1))) rhs
         (plusSwapMiddle (a 0) (b 0) (c 1) (d 1))
         (trans (plus (plus (a 0) (c 1)) (plus (b 0) (d 1)))
            (plus (plus (c 0) (a 1)) (plus (d 0) (b 1))) rhs
            (plusCongBoth (plus (a 0) (c 1)) (plus (b 0) (d 1))
               (plus (c 0) (a 1)) (plus (d 0) (b 1)) hac hbd)
            (plusSwapMiddle (c 0) (a 1) (d 0) (b 1)))
    end
end

theorem addAssoc : ∀ (x : Int) (y : Int) (z : Int) →
    Same (add (add x y) z) (add x (add y z)) ≔
  λ (x : Int) (y : Int) (z : Int) →
    plusCongBoth (plus (plus (x 0) (y 0)) (z 0)) (plus (x 1) (plus (y 1) (z 1)))
      (plus (x 0) (plus (y 0) (z 0))) (plus (plus (x 1) (y 1)) (z 1))
      (plusAssoc (x 0) (y 0) (z 0))
      (sym (plus (plus (x 1) (y 1)) (z 1)) (plus (x 1) (plus (y 1) (z 1)))
         (plusAssoc (x 1) (y 1) (z 1)))
end

Zero on the left holds by conversion: plus 0 n is n.

theorem addIdentityLeft : ∀ (x : Int) → Same (add (ofNat 0) x) x ≔
  λ (x : Int) → refl (plus (x 0) (x 1))
end

theorem addIdentityRight : ∀ (x : Int) → Same (add x (ofNat 0)) x ≔
  λ (x : Int) →
    plusCongBoth (plus (x 0) 0) (x 1) (x 0) (plus (x 1) 0) (plusIdentityRight (x 0))
      (sym (plus (x 1) 0) (x 1) (plusIdentityRight (x 1)))
end

theorem addComm : ∀ (x : Int) (y : Int) → Same (add x y) (add y x) ≔
  λ (x : Int) (y : Int) →
    plusCongBoth (plus (x 0) (y 0)) (plus (y 1) (x 1))
      (plus (y 0) (x 0)) (plus (x 1) (y 1))
      (plusComm (x 0) (y 0)) (plusComm (y 1) (x 1))
end

theorem negateCong : ∀ (a : Int) (b : Int) (h : Same a b) →
    Same (negate a) (negate b) ≔
  λ (a : Int) (b : Int) (h : Same a b) →
    trans (plus (a 1) (b 0)) (plus (b 0) (a 1)) (plus (b 1) (a 0))
      (plusComm (a 1) (b 0))
      (trans (plus (b 0) (a 1)) (plus (a 0) (b 1)) (plus (b 1) (a 0))
         (sym (plus (a 0) (b 1)) (plus (b 0) (a 1)) h) (plusComm (a 0) (b 1)))
end

(-x) + x stands for (x 1 + x 0) - (x 0 + x 1), which is zero.

theorem negateLeft : ∀ (x : Int) → Same (add (negate x) x) (ofNat 0) ≔
  λ (x : Int) →
    trans (plus (plus (x 1) (x 0)) 0) (plus (x 1) (x 0)) (plus (x 0) (x 1))
      (plusIdentityRight (plus (x 1) (x 0))) (plusComm (x 1) (x 0))
end

theorem negateRight : ∀ (x : Int) → Same (add x (negate x)) (ofNat 0) ≔
  λ (x : Int) →
    trans (plus (plus (x 0) (x 1)) 0) (plus (x 0) (x 1)) (plus (x 1) (x 0))
      (plusIdentityRight (plus (x 0) (x 1))) (plusComm (x 0) (x 1))
end

end

The integers under addition: Additive.Base.pow x n is n times x.

module Additive ≔
  Group 0l Int Same sameRefl sameSym sameTrans add addCong (ofNat 0)
    addAssoc addIdentityLeft addIdentityRight negate negateCong negateLeft negateRight

module AdditiveComm ≔ Additive.Base.Commutative addComm

The numbers embed into the integers.

module Embedding where

theorem ofNatAdd : ∀ (m : ℕ) (n : ℕ) →
    Same (ofNat (plus m n)) (add (ofNat m) (ofNat n)) ≔
  λ (m : ℕ) (n : ℕ) → refl (plus (plus m n) 0)
end

Different numbers are different integers.

theorem ofNatInj : ∀ (m : ℕ) (n : ℕ) (h : Same (ofNat m) (ofNat n)) →
    Eq m n ≔
  λ (m : ℕ) (n : ℕ) (h : Same (ofNat m) (ofNat n)) →
    trans m (plus m 0) n (sym (plus m 0) m (plusIdentityRight m))
      (trans (plus m 0) (plus n 0) n h (plusIdentityRight n))
end

end
