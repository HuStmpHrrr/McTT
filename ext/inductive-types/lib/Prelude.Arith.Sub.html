Prelude›Arith›Sub

Predecessor and truncated subtraction, with their laws stated with Prelude›Arith›Equality.

module Prelude›Arith›Sub where

import Prelude›Builtin›Nat use (succ)
import Prelude›Arith›Equality use (Eq; refl; sym; transport)
import Prelude›Arith›Plus use (plus)
import Prelude›Function›Iter as Iter
open Prelude›Arith›Plus.Algebra use (plusComm)

The predecessor; pred 0 is 0.

def pred (n : ℕ) : ℕ ≔
  elim n return y . ℕ
  | zero ⇒ 0
  | succ p, r ⇒ p
  end
end

Truncated subtraction sub a b, which is 0 when b exceeds a. It recurses on b and then on a, so sub (succ a) (succ b) is sub a b by computation.

def sub (a : ℕ) (b : ℕ) : ℕ ≔
  (elim b return y . (∀ (x : ℕ) → ℕ)
   | zero ⇒ λ (x : ℕ) → x
   | succ q, r ⇒ λ (x : ℕ) →
       elim x return z . ℕ
       | zero ⇒ 0
       | succ p, s ⇒ r p
       end
   end) a
end

Truncated subtraction again, as b predecessors of a: monus a b is sub a b, but recurses on b alone, so monus a (succ b) is pred (monus a b). Programs that compare by subtracting use it, since the evaluation of sub inside them takes exponential time.

def monus (a : ℕ) (b : ℕ) : ℕ ≔ Iter.iter 0l ℕ pred b a end

Predecessor and subtraction at zero, successors and equal arguments.

module Laws where

theorem predSucc : ∀ (n : ℕ) → Eq (pred (succ n)) n ≔
  λ (n : ℕ) → refl n
end

theorem subIdentityRight : ∀ (n : ℕ) → Eq (sub n 0) n ≔
  λ (n : ℕ) → refl n
end

theorem subZeroLeft : ∀ (n : ℕ) → Eq (sub 0 n) 0 ≔
  λ (n : ℕ) →
    elim n return x . Eq (sub 0 x) 0
    | zero ⇒ ⋆
    | succ p, ih ⇒ ⋆
    end
end

theorem subSelf : ∀ (n : ℕ) → Eq (sub n n) 0 ≔
  λ (n : ℕ) →
    elim n return x . Eq (sub x x) 0
    | zero ⇒ ⋆
    | succ p, ih ⇒ ih
    end
end

Subtracting successors is subtracting the numbers.

theorem subSucc : ∀ (a : ℕ) (b : ℕ) → Eq (sub (succ a) (succ b)) (sub a b) ≔
  λ (a : ℕ) (b : ℕ) → refl (sub a b)
end

Subtracting one more takes one more predecessor.

theorem subSuccRight : ∀ (a : ℕ) (b : ℕ) →
    Eq (sub a (succ b)) (pred (sub a b)) ≔
  λ (a : ℕ) →
    elim a return x . (∀ (b : ℕ) → Eq (sub x (succ b)) (pred (sub x b)))
    | zero ⇒ λ (b : ℕ) →
        transport (λ (m : ℕ) → Eq 0 (pred m)) 0 (sub 0 b)
          (sym (sub 0 b) 0 (subZeroLeft b)) ⋆
    | succ p, ih ⇒ λ (b : ℕ) →
        elim b return y . Eq (sub (succ p) (succ y)) (pred (sub (succ p) y))
        | zero ⇒ refl p
        | succ q, s ⇒ ih q
        end
    end
end

end

Subtraction undoes addition.

module PlusSub where

Adding b on the left and subtracting it again gives back a.

theorem plusSubLeft : ∀ (b : ℕ) (a : ℕ) → Eq (sub (plus b a) b) a ≔
  λ (b : ℕ) (a : ℕ) →
    elim b return x . Eq (sub (plus x a) x) a
    | zero ⇒ refl a
    | succ p, ih ⇒ ih
    end
end

theorem plusSub : ∀ (a : ℕ) (b : ℕ) → Eq (sub (plus a b) b) a ≔
  λ (a : ℕ) (b : ℕ) →
    transport (λ (m : ℕ) → Eq (sub m b) a) (plus b a) (plus a b)
      (plusComm b a) (plusSubLeft b a)
end

end
