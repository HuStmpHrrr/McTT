Prelude›Data›Vec›Properties

Laws of Prelude›Data›Vec, stated with Prelude›Arith›Equality on number-valued observations: elements read with nth, sums of vectors of numbers (sumVec of Prelude›Data›Vec›Nat), and with IsTrue on the answers of the boolean tests allb and anyb. Bound proofs are irrelevant to the elements read (nthIrrel), which, with sumExt, lets laws relate vectors that agree only up to those proofs.

private module NV ≔ V 0l ℕ

The elements of the operations.

module Nth where

Reading an element of a mapped vector applies the function to it.

theorem nthMap : ∀ (u : Level) (A : Type@{u}) (f : ∀ (a : A) → ℕ) (n : ℕ)
    (v : V.Vec u A n) (i : ℕ) (h : Lt i n) →
    Eq (NV.nth n (V.map u A ℕ f n v) i h) (f (V.nth u A n v i h)) ≔
  λ (u : Level) (A : Type@{u}) (f : ∀ (a : A) → ℕ) (n : ℕ) (v : V.Vec u A n)
    (i : ℕ) (h : Lt i n) → refl (f (v i h))
end

theorem nthReplicate : ∀ (n : ℕ) (c : ℕ) (i : ℕ) (h : Lt i n) →
    Eq (NV.nth n (NV.replicate n c) i h) c ≔
  λ (n : ℕ) (c : ℕ) (i : ℕ) (h : Lt i n) → refl c
end

An index below m reads the first part of an append, whichever proofs of the two bounds are given. The induction on m and i computes both bounds to ⊤ at index 0, where eta for ⊤ identifies them.

theorem nthAppendLeft : ∀ (m : ℕ) (n : ℕ) (u : NV.Vec m) (v : NV.Vec n)
    (i : ℕ) (h : Lt i (plus m n)) (hm : Lt i m) →
    Eq (NV.nth (plus m n) (NV.append m n u v) i h) (NV.nth m u i hm) ≔
  λ (m : ℕ) (n : ℕ) (u : NV.Vec m) (v : NV.Vec n) →
    (elim m return k . (∀ (w : NV.Vec k) (i : ℕ) (h : Lt i (plus k n))
                          (hm : Lt i k) →
                          Eq (NV.nth (plus k n) (NV.append k n w v) i h)
                             (NV.nth k w i hm))
     | zero ⇒ λ (w : NV.Vec 0) (i : ℕ) (h : Lt i n) (hm : Lt i 0) →
         absurd (Eq (v i h) (w i hm)) hm
     | succ p, ih ⇒ λ (w : NV.Vec (succ p)) (i : ℕ) →
         elim i return x . (∀ (h : Lt x (succ (plus p n)))
                              (hm : Lt x (succ p)) →
                              Eq (NV.nth (succ (plus p n))
                                    (NV.append (succ p) n w v) x h)
                                 (NV.nth (succ p) w x hm))
         | zero ⇒ λ (h : ⊤) (hm : ⊤) → refl (w 0 ⋆)
         | succ q, s ⇒ λ (h : Lt q (plus p n)) (hm : Lt q p) →
             ih (NV.tail p w) q h hm
         end
     end) u
end

The index plus m j reads element j of the second part of an append.

theorem nthAppendRight : ∀ (m : ℕ) (n : ℕ) (u : NV.Vec m) (v : NV.Vec n)
    (j : ℕ) (h : Lt j n) →
    Eq (NV.nth (plus m n) (NV.append m n u v) (plus m j)
          (ltPlusMonoRight m j n h))
       (NV.nth n v j h) ≔
  λ (m : ℕ) (n : ℕ) (u : NV.Vec m) (v : NV.Vec n) (j : ℕ) (h : Lt j n) →
    (elim m return k . (∀ (w : NV.Vec k) →
                          Eq (NV.nth (plus k n) (NV.append k n w v) (plus k j)
                                (ltPlusMonoRight k j n h))
                             (NV.nth n v j h))
     | zero ⇒ λ (w : NV.Vec 0) → refl (v j h)
     | succ p, ih ⇒ λ (w : NV.Vec (succ p)) → ih (NV.tail p w)
     end) u
end

Proof irrelevance for bounds: reading index i gives the same element whichever proof of Lt i n is given. At index 0 both proofs are of ⊤, which eta identifies; at succ i the claim is the hypothesis for the tail.

theorem nthIrrel : ∀ (n : ℕ) (v : NV.Vec n) (i : ℕ) (h : Lt i n)
    (g : Lt i n) →
    Eq (NV.nth n v i h) (NV.nth n v i g) ≔
  λ (n : ℕ) (v : NV.Vec n) →
    (elim n return k . (∀ (w : NV.Vec k) (i : ℕ) (h : Lt i k) (g : Lt i k) →
                          Eq (w i h) (w i g))
     | zero ⇒ λ (w : NV.Vec 0) (i : ℕ) (h : Lt i 0) (g : Lt i 0) →
         absurd (Eq (w i h) (w i g)) h
     | succ p, ih ⇒ λ (w : NV.Vec (succ p)) (i : ℕ) →
         elim i return x . (∀ (h : Lt x (succ p)) (g : Lt x (succ p)) →
                              Eq (w x h) (w x g))
         | zero ⇒ λ (h : ⊤) (g : ⊤) → refl (w 0 ⋆)
         | succ q, s ⇒ λ (h : Lt q p) (g : Lt q p) → ih (NV.tail p w) q h g
         end
     end) v
end

theorem nthTabulate : ∀ (f : ∀ (i : ℕ) → ℕ) (n : ℕ) (i : ℕ) (h : Lt i n) →
    Eq (NV.nth n (NV.tabulate f n) i h) (f i) ≔
  λ (f : ∀ (i : ℕ) → ℕ) (n : ℕ) (i : ℕ) (h : Lt i n) → refl (f i)
end

Reading an element of zipWith f s v applies f to the two elements.

theorem nthZipWith : ∀ (u : Level) (A : Type@{u}) (B : Type@{u})
    (f : ∀ (a : A) (b : B) → ℕ) (n : ℕ)
    (s : V.Vec u A n) (v : V.Vec u B n) (i : ℕ) (h : Lt i n) →
    Eq (NV.nth n (V.zipWith u A B ℕ f n s v) i h)
       (f (V.nth u A n s i h) (V.nth u B n v i h)) ≔
  λ (u : Level) (A : Type@{u}) (B : Type@{u}) (f : ∀ (a : A) (b : B) → ℕ)
    (n : ℕ) (s : V.Vec u A n) (v : V.Vec u B n) (i : ℕ) (h : Lt i n) →
    refl (f (s i h) (v i h))
end

The last element of snoc n v a is a.

theorem lastSnoc : ∀ (n : ℕ) (v : NV.Vec n) (a : ℕ) →
    Eq (NV.last n (NV.snoc n v a)) a ≔
  λ (n : ℕ) (v : NV.Vec n) (a : ℕ) →
    (elim n return k . (∀ (w : NV.Vec k) → Eq (NV.last k (NV.snoc k w a)) a)
     | zero ⇒ λ (w : NV.Vec 0) → refl a
     | succ p, ih ⇒ λ (w : NV.Vec (succ p)) → ih (NV.tail p w)
     end) v
end

Element i of reverse n v is element sub n (succ i) of v, for any proof that this index is in bounds.

theorem nthReverse : ∀ (n : ℕ) (v : NV.Vec n) (i : ℕ) (h : Lt i n)
    (g : Lt (sub n (succ i)) n) →
    Eq (NV.nth n (NV.reverse n v) i h) (NV.nth n v (sub n (succ i)) g) ≔
  λ (n : ℕ) →
    elim n return k . (∀ (v : NV.Vec k) (i : ℕ) (h : Lt i k)
                         (g : Lt (sub k (succ i)) k) →
                         Eq (NV.nth k (NV.reverse k v) i h)
                            (NV.nth k v (sub k (succ i)) g))
    | zero ⇒ λ (v : NV.Vec 0) (i : ℕ) (h : Lt i 0)
               (g : Lt (sub 0 (succ i)) 0) →
        absurd (Eq (NV.nth 0 (NV.reverse 0 v) i h) (v (sub 0 (succ i)) g)) h
    | succ p, ih ⇒ λ (v : NV.Vec (succ p)) (i : ℕ) (h : Lt i (succ p))
                     (g : Lt (sub p i) (succ p)) →
        nthIrrel (succ p) v (sub p i) (subLe p i) g
    end
end

end

Sums of the operations.

module Sums where

A vector of n copies of c sums to mult n c.

theorem sumReplicate : ∀ (n : ℕ) (c : ℕ) →
    Eq (sumVec n (NV.replicate n c)) (mult n c) ≔
  λ (n : ℕ) (c : ℕ) →
    elim n return x . Eq (sumVec x (NV.replicate x c)) (mult x c)
    | zero ⇒ ⋆
    | succ p, ih ⇒
        cong (λ (y : ℕ) → plus c y)
          (sumVec p (NV.replicate p c)) (mult p c) ih
    end
end

The sum of an append is the sum of the parts. In the step, the tail of append (succ p) n u v is append p n (tail u) v by eta, and the head is moved out by plusAssoc.

theorem sumAppend : ∀ (m : ℕ) (n : ℕ) (u : NV.Vec m) (v : NV.Vec n) →
    Eq (sumVec (plus m n) (NV.append m n u v))
       (plus (sumVec m u) (sumVec n v)) ≔
  λ (m : ℕ) (n : ℕ) (u : NV.Vec m) (v : NV.Vec n) →
    (elim m return k . (∀ (w : NV.Vec k) →
                          Eq (sumVec (plus k n) (NV.append k n w v))
                             (plus (sumVec k w) (sumVec n v)))
     | zero ⇒ λ (w : NV.Vec 0) → refl (sumVec n v)
     | succ p, ih ⇒ λ (w : NV.Vec (succ p)) →
         let a ≔ NV.head p w;
             t ≔ NV.tail p w;
             rest ≔ sumVec (plus p n) (NV.append p n t v);
             step : Eq (plus a rest) (plus a (plus (sumVec p t) (sumVec n v))) ≔
               cong (λ (y : ℕ) → plus a y)
                 rest (plus (sumVec p t) (sumVec n v)) (ih t);
             assoc : Eq (plus a (plus (sumVec p t) (sumVec n v)))
                        (plus (plus a (sumVec p t)) (sumVec n v)) ≔
               sym (plus (plus a (sumVec p t)) (sumVec n v))
                 (plus a (plus (sumVec p t) (sumVec n v)))
                 (plusAssoc a (sumVec p t) (sumVec n v))
         in trans (plus a rest) (plus a (plus (sumVec p t) (sumVec n v)))
              (plus (plus a (sumVec p t)) (sumVec n v)) step assoc
         end
     end) u
end

Vectors that agree at every index have the same sum.

theorem sumExt : ∀ (n : ℕ) (u : NV.Vec n) (v : NV.Vec n)
    (e : ∀ (i : ℕ) (h : Lt i n) → Eq (u i h) (v i h)) →
    Eq (sumVec n u) (sumVec n v) ≔
  λ (n : ℕ) →
    elim n return k . (∀ (u : NV.Vec k) (v : NV.Vec k)
                         (e : ∀ (i : ℕ) (h : Lt i k) → Eq (u i h) (v i h)) →
                         Eq (sumVec k u) (sumVec k v))
    | zero ⇒ λ (u : NV.Vec 0) (v : NV.Vec 0)
               (e : ∀ (i : ℕ) (h : Lt i 0) → Eq (u i h) (v i h)) → ⋆
    | succ p, ih ⇒ λ (u : NV.Vec (succ p)) (v : NV.Vec (succ p))
                     (e : ∀ (i : ℕ) (h : Lt i (succ p)) → Eq (u i h) (v i h)) →
        let a ≔ u 0 ⋆;
            b ≔ v 0 ⋆;
            su ≔ sumVec p (NV.tail p u);
            sv ≔ sumVec p (NV.tail p v)
        in trans (plus a su) (plus b su) (plus b sv)
             (cong (λ (y : ℕ) → plus y su) a b (e 0 ⋆))
             (cong (λ (y : ℕ) → plus b y) su sv
                (ih (NV.tail p u) (NV.tail p v)
                   (λ (i : ℕ) (h : Lt i p) → e (succ i) h)))
        end
    end
end

Adding c to every element adds mult n c to the sum.

theorem sumMapPlus : ∀ (c : ℕ) (n : ℕ) (v : NV.Vec n) →
    Eq (sumVec n (NV.map ℕ (plus c) n v)) (plus (sumVec n v) (mult n c)) ≔
  λ (c : ℕ) (n : ℕ) (v : NV.Vec n) →
    (elim n return k . (∀ (w : NV.Vec k) →
                          Eq (sumVec k (NV.map ℕ (plus c) k w))
                             (plus (sumVec k w) (mult k c)))
     | zero ⇒ λ (w : NV.Vec 0) → ⋆
     | succ p, ih ⇒ λ (w : NV.Vec (succ p)) →
         let a ≔ w 0 ⋆;
             t ≔ NV.tail p w;
             sm ≔ sumVec p (NV.map ℕ (plus c) p t);
             s ≔ sumVec p t;
             m ≔ mult p c;
             step : Eq (plus (plus c a) sm) (plus (plus c a) (plus s m)) ≔
               cong (λ (y : ℕ) → plus (plus c a) y) sm (plus s m) (ih t);
             swap : Eq (plus (plus c a) (plus s m))
                       (plus (plus a c) (plus s m)) ≔
               cong (λ (y : ℕ) → plus y (plus s m))
                 (plus c a) (plus a c) (plusComm c a)
         in trans (plus (plus c a) sm) (plus (plus a c) (plus s m))
              (plus (plus a s) (plus c m))
              (trans (plus (plus c a) sm) (plus (plus c a) (plus s m))
                 (plus (plus a c) (plus s m)) step swap)
              (plusSwapMiddle a c s m)
         end
     end) v
end

Multiplying every element by c multiplies the sum by c.

theorem sumMapScale : ∀ (c : ℕ) (n : ℕ) (v : NV.Vec n) →
    Eq (sumVec n (NV.map ℕ (mult c) n v)) (mult c (sumVec n v)) ≔
  λ (c : ℕ) (n : ℕ) (v : NV.Vec n) →
    (elim n return k . (∀ (w : NV.Vec k) →
                          Eq (sumVec k (NV.map ℕ (mult c) k w))
                             (mult c (sumVec k w)))
     | zero ⇒ λ (w : NV.Vec 0) → sym (mult c 0) 0 (multZeroRight c)
     | succ p, ih ⇒ λ (w : NV.Vec (succ p)) →
         let a ≔ w 0 ⋆;
             t ≔ NV.tail p w;
             sm ≔ sumVec p (NV.map ℕ (mult c) p t);
             s ≔ sumVec p t
         in trans (plus (mult c a) sm) (plus (mult c a) (mult c s))
              (mult c (plus a s))
              (cong (λ (y : ℕ) → plus (mult c a) y) sm (mult c s) (ih t))
              (sym (mult c (plus a s)) (plus (mult c a) (mult c s))
                 (multDistribLeft c a s))
         end
     end) v
end

The sum of zipWith plus u v is the sum of the sums.

theorem sumZipWithPlus : ∀ (n : ℕ) (u : NV.Vec n) (v : NV.Vec n) →
    Eq (sumVec n (NV.zipWith ℕ ℕ plus n u v))
       (plus (sumVec n u) (sumVec n v)) ≔
  λ (n : ℕ) →
    elim n return k . (∀ (u : NV.Vec k) (v : NV.Vec k) →
                         Eq (sumVec k (NV.zipWith ℕ ℕ plus k u v))
                            (plus (sumVec k u) (sumVec k v)))
    | zero ⇒ λ (u : NV.Vec 0) (v : NV.Vec 0) → ⋆
    | succ p, ih ⇒ λ (u : NV.Vec (succ p)) (v : NV.Vec (succ p)) →
        let a ≔ u 0 ⋆;
            b ≔ v 0 ⋆;
            tu ≔ NV.tail p u;
            tv ≔ NV.tail p v;
            sz ≔ sumVec p (NV.zipWith ℕ ℕ plus p tu tv);
            su ≔ sumVec p tu;
            sv ≔ sumVec p tv
        in trans (plus (plus a b) sz) (plus (plus a b) (plus su sv))
             (plus (plus a su) (plus b sv))
             (cong (λ (y : ℕ) → plus (plus a b) y) sz (plus su sv) (ih tu tv))
             (plusSwapMiddle a b su sv)
        end
    end
end

A left fold of plus from a adds the sum to a.

theorem foldlPlus : ∀ (a : ℕ) (n : ℕ) (v : NV.Vec n) →
    Eq (NV.foldl ℕ plus a n v) (plus a (sumVec n v)) ≔
  λ (a : ℕ) (n : ℕ) (v : NV.Vec n) →
    (elim n return k . (∀ (w : NV.Vec k) (acc : ℕ) →
                          Eq (NV.foldl ℕ plus acc k w) (plus acc (sumVec k w)))
     | zero ⇒ λ (w : NV.Vec 0) (acc : ℕ) → sym (plus acc 0) acc (plusIdentityRight acc)
     | succ p, ih ⇒ λ (w : NV.Vec (succ p)) (acc : ℕ) →
         let b ≔ w 0 ⋆;
             s ≔ sumVec p (NV.tail p w)
         in trans (NV.foldl ℕ plus (plus acc b) p (NV.tail p w))
              (plus (plus acc b) s) (plus acc (plus b s))
              (ih (NV.tail p w) (plus acc b)) (plusAssoc acc b s)
         end
     end) v a
end

The sum of snoc n v a is the sum of v plus a.

theorem sumSnoc : ∀ (n : ℕ) (v : NV.Vec n) (a : ℕ) →
    Eq (sumVec (succ n) (NV.snoc n v a)) (plus (sumVec n v) a) ≔
  λ (n : ℕ) (v : NV.Vec n) (a : ℕ) →
    (elim n return k . (∀ (w : NV.Vec k) →
                          Eq (sumVec (succ k) (NV.snoc k w a))
                             (plus (sumVec k w) a))
     | zero ⇒ λ (w : NV.Vec 0) → plusIdentityRight a
     | succ p, ih ⇒ λ (w : NV.Vec (succ p)) →
         let b ≔ w 0 ⋆;
             t ≔ NV.tail p w;
             r ≔ sumVec (succ p) (NV.snoc p t a);
             s ≔ sumVec p t
         in trans (plus b r) (plus b (plus s a)) (plus (plus b s) a)
              (cong (λ (y : ℕ) → plus b y) r (plus s a) (ih t))
              (sym (plus (plus b s) a) (plus b (plus s a)) (plusAssoc b s a))
         end
     end) v
end

A vector sums to the sum of its init plus its last. In the step, the init of the tail and the tail of the init agree up to bound proofs.

theorem sumInitLast : ∀ (n : ℕ) (v : NV.Vec (succ n)) →
    Eq (sumVec (succ n) v) (plus (sumVec n (NV.init n v)) (NV.last n v)) ≔
  λ (n : ℕ) →
    elim n return k . (∀ (v : NV.Vec (succ k)) →
                         Eq (sumVec (succ k) v)
                            (plus (sumVec k (NV.init k v)) (NV.last k v)))
    | zero ⇒ λ (v : NV.Vec 1) → plusIdentityRight (v 0 ⋆)
    | succ p, ih ⇒ λ (w : NV.Vec (succ (succ p))) →
        let a ≔ w 0 ⋆;
            t ≔ NV.tail (succ p) w;
            l ≔ NV.last (succ p) w;
            sa ≔ sumVec p (NV.init p t);
            sb ≔ sumVec p (NV.tail p (NV.init (succ p) w));
            step : Eq (plus a (sumVec (succ p) t)) (plus a (plus sa l)) ≔
              cong (λ (y : ℕ) → plus a y)
                (sumVec (succ p) t) (plus sa l) (ih t);
            assoc : Eq (plus a (plus sa l)) (plus (plus a sa) l) ≔
              sym (plus (plus a sa) l) (plus a (plus sa l)) (plusAssoc a sa l);
            ext : Eq sa sb ≔
              sumExt p (NV.init p t) (NV.tail p (NV.init (succ p) w))
                (λ (i : ℕ) (h : Lt i p) →
                   nthIrrel (succ (succ p)) w (succ i) (leStep (succ i) p h)
                     (leStep (succ (succ i)) (succ p) h));
            swap : Eq (plus (plus a sa) l) (plus (plus a sb) l) ≔
              cong (λ (y : ℕ) → plus (plus a y) l) sa sb ext
        in trans (plus a (sumVec (succ p) t)) (plus a (plus sa l))
             (plus (plus a sb) l) step
             (trans (plus a (plus sa l)) (plus (plus a sa) l)
                (plus (plus a sb) l) assoc swap)
        end
    end
end

The tail of reverse (succ p) w is the reverse of init p w, pointwise.

fact tailReverse : ∀ (p : ℕ) (w : NV.Vec (succ p)) (i : ℕ) (h : Lt i p) →
    Eq (NV.tail p (NV.reverse (succ p) w) i h)
       (NV.reverse p (NV.init p w) i h) ≔
  λ (p : ℕ) →
    elim p return k . (∀ (w : NV.Vec (succ k)) (i : ℕ) (h : Lt i k) →
                         Eq (NV.tail k (NV.reverse (succ k) w) i h)
                            (NV.reverse k (NV.init k w) i h))
    | zero ⇒ λ (w : NV.Vec 1) (i : ℕ) (h : Lt i 0) →
        absurd (Eq (NV.tail 0 (NV.reverse 1 w) i h)
                   (NV.reverse 0 (NV.init 0 w) i h)) h
    | succ q, ih ⇒ λ (w : NV.Vec (succ (succ q))) (i : ℕ)
                     (h : Lt i (succ q)) →
        nthIrrel (succ (succ q)) w (sub q i) (subLe (succ q) (succ i))
          (leStep (succ (sub q i)) (succ q) (subLe q i))
    end
end

Reversal preserves the sum: the head of the reverse is the last, and its tail is the reverse of the init.

theorem sumReverse : ∀ (n : ℕ) (v : NV.Vec n) →
    Eq (sumVec n (NV.reverse n v)) (sumVec n v) ≔
  λ (n : ℕ) →
    elim n return k . (∀ (v : NV.Vec k) →
                         Eq (sumVec k (NV.reverse k v)) (sumVec k v))
    | zero ⇒ λ (v : NV.Vec 0) → ⋆
    | succ p, ih ⇒ λ (w : NV.Vec (succ p)) →
        let l ≔ NV.last p w;
            iw ≔ NV.init p w;
            t ≔ NV.tail p (NV.reverse (succ p) w);
            st ≔ sumVec p t;
            si ≔ sumVec p iw;
            sr ≔ sumVec p (NV.reverse p iw);
            toInit : Eq st si ≔
              trans st sr si (sumExt p t (NV.reverse p iw) (tailReverse p w))
                (ih iw);
            step : Eq (plus l st) (plus l si) ≔
              cong (λ (y : ℕ) → plus l y) st si toInit;
            back : Eq (plus si l) (sumVec (succ p) w) ≔
              sym (sumVec (succ p) w) (plus si l) (sumInitLast p w)
        in trans (plus l st) (plus l si) (sumVec (succ p) w) step
             (trans (plus l si) (plus si l) (sumVec (succ p) w)
                (plusComm l si) back)
        end
    end
end

A tabulated vector sums like sumTo, from the other end.

theorem sumTabulate : ∀ (f : ∀ (i : ℕ) → ℕ) (n : ℕ) →
    Eq (sumVec n (NV.tabulate f n)) (sumTo f n) ≔
  λ (f : ∀ (i : ℕ) → ℕ) (n : ℕ) →
    (elim n return k . (∀ (g : ∀ (i : ℕ) → ℕ) →
                          Eq (sumVec k (NV.tabulate g k)) (sumTo g k))
     | zero ⇒ λ (g : ∀ (i : ℕ) → ℕ) → ⋆
     | succ p, ih ⇒ λ (g : ∀ (i : ℕ) → ℕ) →
         let s ≔ sumVec p (NV.tabulate (λ (i : ℕ) → g (succ i)) p);
             t ≔ sumTo (λ (i : ℕ) → g (succ i)) p
         in trans (plus (g 0) s) (plus (g 0) t) (sumTo g (succ p))
              (cong (λ (y : ℕ) → plus (g 0) y) s t
                 (ih (λ (i : ℕ) → g (succ i))))
              (sym (sumTo g (succ p)) (plus (g 0) t) (sumToShift g p))
         end
     end) f
end

Taking and dropping from an append.

module Slices where

The first m elements of append m n u v are those of u.

theorem takeAppend : ∀ (m : ℕ) (n : ℕ) (u : NV.Vec m) (v : NV.Vec n)
    (i : ℕ) (h : Lt i m) →
    Eq (NV.nth m (NV.take m n (NV.append m n u v)) i h) (NV.nth m u i h) ≔
  λ (m : ℕ) (n : ℕ) (u : NV.Vec m) (v : NV.Vec n) (i : ℕ) (h : Lt i m) →
    nthAppendLeft m n u v i (ltLeTrans i m (plus m n) h (lePlusRight m n)) h
end

The last n elements of append m n u v are those of v.

theorem dropAppend : ∀ (m : ℕ) (n : ℕ) (u : NV.Vec m) (v : NV.Vec n)
    (j : ℕ) (h : Lt j n) →
    Eq (NV.nth n (NV.drop m n (NV.append m n u v)) j h) (NV.nth n v j h) ≔
  λ (m : ℕ) (n : ℕ) (u : NV.Vec m) (v : NV.Vec n) (j : ℕ) (h : Lt j n) →
    nthAppendRight m n u v j h
end

Appending the take and the drop of a vector gives it back. By induction on m: the tail of take (succ p) n w is take p n of the tail, and likewise for drop, by conversion.

theorem appendTakeDrop : ∀ (m : ℕ) (n : ℕ) (v : NV.Vec (plus m n))
    (i : ℕ) (h : Lt i (plus m n)) →
    Eq (NV.nth (plus m n)
          (NV.append m n (NV.take m n v) (NV.drop m n v)) i h)
       (NV.nth (plus m n) v i h) ≔
  λ (m : ℕ) (n : ℕ) →
    elim m return k . (∀ (w : NV.Vec (plus k n)) (i : ℕ) (h : Lt i (plus k n)) →
                         Eq (NV.nth (plus k n)
                               (NV.append k n (NV.take k n w) (NV.drop k n w))
                               i h)
                            (w i h))
    | zero ⇒ λ (w : NV.Vec n) (i : ℕ) (h : Lt i n) → refl (w i h)
    | succ p, ih ⇒ λ (w : NV.Vec (succ (plus p n))) (i : ℕ) →
        elim i return x . (∀ (h : Lt x (succ (plus p n))) →
                             Eq (NV.nth (succ (plus p n))
                                   (NV.append (succ p) n (NV.take (succ p) n w)
                                      (NV.drop (succ p) n w)) x h)
                                (w x h))
        | zero ⇒ λ (h : ⊤) → refl (w 0 ⋆)
        | succ q, s ⇒ λ (h : Lt q (plus p n)) → ih (NV.tail (plus p n) w) q h
        end
    end
end

The tests allb and anyb are sound and complete.

module Tests where

One step of each test, taken apart and put together. The proofs below are by induction on the length, and use one of these at each element.

private module Step where

One step of allb: and x b holds exactly when x and b do. These three lemmas take the step apart and put it together.

lemma allHead : ∀ (x : Bool) (b : Bool) (e : IsTrue (and x b)) → IsTrue x ≔
  λ (x : Bool) (b : Bool) →
    elim x return z . (∀ (e : IsTrue (and z b)) → IsTrue z)
    | tt ⇒ λ (e : IsTrue b) → ⋆
    | ff ⇒ λ (e : ⊥) → e
    end
end

lemma allTail : ∀ (x : Bool) (b : Bool) (e : IsTrue (and x b)) → IsTrue b ≔
  λ (x : Bool) (b : Bool) →
    elim x return z . (∀ (e : IsTrue (and z b)) → IsTrue b)
    | tt ⇒ λ (e : IsTrue b) → e
    | ff ⇒ λ (e : ⊥) → absurd (IsTrue b) e
    end
end

lemma allIntro : ∀ (x : Bool) (b : Bool) (hx : IsTrue x) (hb : IsTrue b) →
    IsTrue (and x b) ≔
  λ (x : Bool) (b : Bool) →
    elim x return z . (∀ (hx : IsTrue z) (hb : IsTrue b) → IsTrue (and z b))
    | tt ⇒ λ (hx : ⊤) (hb : IsTrue b) → hb
    | ff ⇒ λ (hx : ⊥) (hb : IsTrue b) → hx
    end
end

One step of anyb: when or x b holds, x or b does. The conclusion is eliminated into any C, of any level u.

def anyCases : ∀ (u : Level) (x : Bool) (b : Bool) (e : IsTrue (or x b))
    (C : Type@{u}) (l : ∀ (hx : IsTrue x) → C) (r : ∀ (hb : IsTrue b) → C) →
    C ≔
  λ (u : Level) (x : Bool) (b : Bool) →
    elim x return z . (∀ (e : IsTrue (or z b)) (C : Type@{u})
                         (l : ∀ (hx : IsTrue z) → C)
                         (r : ∀ (hb : IsTrue b) → C) → C)
    | tt ⇒ λ (e : ⊤) (C : Type@{u}) (l : ∀ (hx : ⊤) → C)
             (r : ∀ (hb : IsTrue b) → C) → l ⋆
    | ff ⇒ λ (e : IsTrue b) (C : Type@{u}) (l : ∀ (hx : ⊥) → C)
             (r : ∀ (hb : IsTrue b) → C) → r e
    end
end

end

allb answers tt only when the test holds of every element.

def allbSound : ∀ (u : Level) (A : Type@{u}) (p : ∀ (a : A) → Bool) (n : ℕ)
    (v : V.Vec u A n) (e : IsTrue (V.allb u A p n v)) →
    V.All u A (λ (a : A) → IsTrue (p a)) n v ≔
  λ (u : Level) (A : Type@{u}) (p : ∀ (a : A) → Bool) (n : ℕ) →
    elim n return k . (∀ (v : V.Vec u A k) (e : IsTrue (V.allb u A p k v)) →
                         V.All u A (λ (a : A) → IsTrue (p a)) k v)
    | zero ⇒ λ (v : V.Vec u A 0) (e : ⊤) (i : ℕ) (h : Lt i 0) →
        absurd (IsTrue (p (v i h))) h
    | succ m, ih ⇒ λ (w : V.Vec u A (succ m)) →
        let b ≔ V.allb u A p m (V.tail u A m w)
        in λ (e : IsTrue (and (p (w 0 ⋆)) b)) (i : ℕ) →
             elim i return x . (∀ (h : Lt x (succ m)) → IsTrue (p (w x h)))
             | zero ⇒ λ (h : ⊤) → Step.allHead (p (w 0 ⋆)) b e
             | succ j, s ⇒ λ (h : Lt j m) →
                 ih (V.tail u A m w) (Step.allTail (p (w 0 ⋆)) b e) j h
             end
        end
    end
end

allb answers tt when the test holds of every element.

theorem allbComplete : ∀ (u : Level) (A : Type@{u}) (p : ∀ (a : A) → Bool)
    (n : ℕ) (v : V.Vec u A n) (h : V.All u A (λ (a : A) → IsTrue (p a)) n v) →
    IsTrue (V.allb u A p n v) ≔
  λ (u : Level) (A : Type@{u}) (p : ∀ (a : A) → Bool) (n : ℕ) →
    elim n return k . (∀ (v : V.Vec u A k)
                         (h : V.All u A (λ (a : A) → IsTrue (p a)) k v) →
                         IsTrue (V.allb u A p k v))
    | zero ⇒ λ (v : V.Vec u A 0)
               (h : V.All u A (λ (a : A) → IsTrue (p a)) 0 v) → ⋆
    | succ m, ih ⇒ λ (w : V.Vec u A (succ m))
                     (h : V.All u A (λ (a : A) → IsTrue (p a)) (succ m) w) →
        Step.allIntro (p (w 0 ⋆)) (V.allb u A p m (V.tail u A m w)) (h 0 ⋆)
          (ih (V.tail u A m w) (λ (i : ℕ) (hi : Lt i m) → h (succ i) hi))
    end
end

When anyb answers tt, the test holds of some element.

def anybSound : ∀ (u : Level) (A : Type@{u}) (p : ∀ (a : A) → Bool) (n : ℕ)
    (v : V.Vec u A n) (e : IsTrue (V.anyb u A p n v)) →
    V.Any u A (λ (a : A) → IsTrue (p a)) n v ≔
  λ (u : Level) (A : Type@{u}) (p : ∀ (a : A) → Bool) (n : ℕ) →
    elim n return k . (∀ (v : V.Vec u A k) (e : IsTrue (V.anyb u A p k v)) →
                         V.Any u A (λ (a : A) → IsTrue (p a)) k v)
    | zero ⇒ λ (v : V.Vec u A 0) (e : ⊥) →
        Neg.absurd (succl u) (V.Any u A (λ (a : A) → IsTrue (p a)) 0 v) e
    | succ m, ih ⇒ λ (w : V.Vec u A (succ m)) →
        let b ≔ V.anyb u A p m (V.tail u A m w)
        in λ (e : IsTrue (or (p (w 0 ⋆)) b)) (C : Type@{u})
             (k : ∀ (i : ℕ) (h : Lt i (succ m)) (q : IsTrue (p (w i h))) → C) →
             Step.anyCases u (p (w 0 ⋆)) b e C
               (λ (hx : IsTrue (p (w 0 ⋆))) → k 0 ⋆ hx)
               (λ (hb : IsTrue b) →
                  ih (V.tail u A m w) hb C
                    (λ (i : ℕ) (h : Lt i m) (q : IsTrue (p (w (succ i) h))) →
                       k (succ i) h q))
        end
    end
end
