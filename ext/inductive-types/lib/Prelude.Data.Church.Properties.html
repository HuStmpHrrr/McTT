Prelude›Data›Church›Properties

Laws of Prelude›Data›Church u: the arithmetic of numerals agrees with that of ℕ, and the length and sum of lists.

A Church numeral is any function of the right type, and without parametricity nothing can be said of an arbitrary one. So the laws are about the numerals of numbers, fromNat n, and are proved by induction on n. Laws that only unfold definitions, like lengthMap, hold for every list. The laws across two levels are in Prelude›Data›Church›Levels›Properties.

The successor on numbers, the function toNat iterates.

private def S (k : ℕ) : ℕ ≔ succ k end

Numerals of numbers.

module Numerals where

theorem toNatFromNat : ∀ (n : ℕ) → Eq (C.toNat (C.fromNat n)) n ≔
  λ (n : ℕ) →
    elim n return x . Eq (C.toNat (C.fromNat x)) x
    | zero ⇒ ⋆
    | succ p, ih ⇒ ih
    end
end

The numeral of m, iterating the successor from y, adds m.

theorem fromNatAdds : ∀ (m : ℕ) (y : ℕ) → Eq (C.fromNat m ℕ S y) (plus m y) ≔
  λ (m : ℕ) (y : ℕ) →
    elim m return x . Eq (C.fromNat x ℕ S y) (plus x y)
    | zero ⇒ refl y
    | succ p, ih ⇒ ih
    end
end

If g adds c, then the numeral of m, iterating g from y, adds m times c.

theorem fromNatAddsMult : ∀ (c : ℕ) (g : ∀ (k : ℕ) → ℕ)
    (hg : ∀ (k : ℕ) → Eq (g k) (plus c k)) (m : ℕ) (y : ℕ) →
    Eq (C.fromNat m ℕ g y) (plus (mult m c) y) ≔
  λ (c : ℕ) (g : ∀ (k : ℕ) → ℕ) (hg : ∀ (k : ℕ) → Eq (g k) (plus c k))
    (m : ℕ) (y : ℕ) →
    elim m return x . Eq (C.fromNat x ℕ g y) (plus (mult x c) y)
    | zero ⇒ refl y
    | succ p, ih ⇒
        let r ≔ C.fromNat p ℕ g y
        in trans (g r) (plus c r) (plus (mult (succ p) c) y)
             (hg r)
             (trans (plus c r) (plus c (plus (mult p c) y))
                (plus (plus c (mult p c)) y)
                (cong (λ (k : ℕ) → plus c k) r (plus (mult p c) y) ih)
                (sym (plus (plus c (mult p c)) y) (plus c (plus (mult p c) y))
                   (plusAssoc c (mult p c) y)))
        end
    end
end

end

open Numerals use (toNatFromNat; fromNatAdds; fromNatAddsMult)

The operations on numerals compute the operations on numbers.

module Arithmetic where

theorem toNatPlus : ∀ (m : ℕ) (n : ℕ) →
    Eq (C.toNat (C.cplus (C.fromNat m) (C.fromNat n))) (plus m n) ≔
  λ (m : ℕ) (n : ℕ) →
    let k ≔ C.toNat (C.fromNat n)
    in trans (C.fromNat m ℕ S k) (plus m k) (plus m n)
         (fromNatAdds m k)
         (cong (λ (j : ℕ) → plus m j) k n (toNatFromNat n))
    end
end

theorem toNatMult : ∀ (m : ℕ) (n : ℕ) →
    Eq (C.toNat (C.cmult (C.fromNat m) (C.fromNat n))) (mult m n) ≔
  λ (m : ℕ) (n : ℕ) →
    trans (C.fromNat m ℕ (C.fromNat n ℕ S) 0) (plus (mult m n) 0)
      (mult m n)
      (fromNatAddsMult n (C.fromNat n ℕ S) (fromNatAdds n) m 0)
      (plusIdentityRight (mult m n))
end

end

Length and sum of lists.

module Lists where

Mapping keeps the length of every list, by conversion.

theorem lengthMap : ∀ (A : Type@{u}) (B : Type@{u}) (f : ∀ (a : A) → B)
    (l : C.List A) → Eq (C.length B (C.map A B f l)) (C.length A l) ≔
  λ (A : Type@{u}) (B : Type@{u}) (f : ∀ (a : A) → B) (l : C.List A) →
    refl (C.length A l)
end

theorem lengthReplicate : ∀ (A : Type@{u}) (n : ℕ) (a : A) →
    Eq (C.length A (C.replicate A n a)) n ≔
  λ (A : Type@{u}) (n : ℕ) (a : A) →
    elim n return x . Eq (C.length A (C.replicate A x a)) x
    | zero ⇒ ⋆
    | succ p, ih ⇒ ih
    end
end

Appending n copies of a in front adds n to the length.

theorem lengthAppendReplicate : ∀ (A : Type@{u}) (n : ℕ) (a : A)
    (l : C.List A) →
    Eq (C.length A (C.append A (C.replicate A n a) l))
       (plus n (C.length A l)) ≔
  λ (A : Type@{u}) (n : ℕ) (a : A) (l : C.List A) →
    elim n return x . Eq (C.length A (C.append A (C.replicate A x a) l))
                         (plus x (C.length A l))
    | zero ⇒ refl (C.length A l)
    | succ p, ih ⇒ ih
    end
end

theorem sumReplicate : ∀ (n : ℕ) (c : ℕ) →
    Eq (C.sum (C.replicate ℕ n c)) (mult n c) ≔
  λ (n : ℕ) (c : ℕ) →
    elim n return x . Eq (C.sum (C.replicate ℕ x c)) (mult x c)
    | zero ⇒ ⋆
    | succ p, ih ⇒
        cong (λ (k : ℕ) → plus c k) (C.sum (C.replicate ℕ p c)) (mult p c)
          ih
    end
end

end
