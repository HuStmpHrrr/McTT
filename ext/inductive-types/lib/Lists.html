The recursor counts the even positions: the motive is a function of the parity of the position.

def evens (l : NatList) : ℕ ≔
  List.rec 0l ℕ 0l (λ (l : NatList) → ∀ (b : Bool) → ℕ)
    (λ (b : Bool) → 0)
    (λ (x : ℕ) (xs : NatList) (r : ∀ (b : Bool) → ℕ) (b : Bool) →
       if ℕ b (succ (r (not b))) (r (not b)))
    l tt
end

eval evens oneTwoThree
output
Evaluate evens oneTwoThree --> 2 : ℕ

Options: the first element of a list, if any.

def head (l : NatList) : Option 0l ℕ ≔
  elim l return z . Option 0l ℕ
  | nil ⇒ Option.none 0l ℕ
  | cons x xs, r ⇒ Option.some 0l ℕ x
  end
end
