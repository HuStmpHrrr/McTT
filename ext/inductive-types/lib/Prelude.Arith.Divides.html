Divides k n says that k divides n: the remainder of n by k is 0. Since mod n 0 is n, Divides 0 n computes to Eq n 0. It computes, so a closed instance is ⊤ or ⊥.

def Divides (k : ℕ) (n : ℕ) : Type@0 ≔ Eq (mod n k) 0 end

Divisibility by and of zero and one, and reflexivity.

module Basic where

Zero divides only zero.

theorem zeroDivides : ∀ (n : ℕ) (h : Divides 0 n) → Eq n 0 ≔
  λ (n : ℕ) (h : Eq n 0) → h
end

theorem dividesRefl : ∀ (n : ℕ) → Divides n n ≔
  λ (n : ℕ) →
    elim n return x . Divides x x
    | zero ⇒ ⋆
    | succ d, ih ⇒ modSelf d
    end
end

theorem dividesZero : ∀ (k : ℕ) → Divides k 0 ≔
  λ (k : ℕ) → modZero k
end

theorem oneDivides : ∀ (n : ℕ) → Divides 1 n ≔
  λ (n : ℕ) → modOne n
end

end

open Basic use (dividesRefl; dividesZero)

Divisibility is closed under sums, multiples and differences.

module Closure where

A divisor of two numbers divides their sum. For a positive divisor, the remainder of a + b is a steps of wrap from the remainder of b, which is 0 by hb, so it is the remainder of a, which is 0 by ha.

theorem dividesPlus : ∀ (k : ℕ) (a : ℕ) (b : ℕ)
    (ha : Divides k a) (hb : Divides k b) →
    Divides k (plus a b) ≔
  λ (k : ℕ) (a : ℕ) (b : ℕ) →
    elim k return x . (∀ (ha : Divides x a) (hb : Divides x b) →
                         Divides x (plus a b))
    | zero ⇒ λ (ha : Eq a 0) (hb : Eq b 0) →
        transport (λ (y : ℕ) → Eq (plus y b) 0) 0 a (sym a 0 ha) hb
    | succ d, ih ⇒ λ (ha : Eq (mod a (succ d)) 0)
                     (hb : Eq (mod b (succ d)) 0) →
        let e ≔ succ d;
            w ≔ λ (r : ℕ) → wrap r d;
            split : Eq (mod (plus a b) e)
                       (Generic.iterate 0l ℕ w a (mod b e)) ≔
              modPlus a b d;
            atZero : Eq (Generic.iterate 0l ℕ w a (mod b e)) (mod a e) ≔
              cong (λ (y : ℕ) → Generic.iterate 0l ℕ w a y) (mod b e) 0 hb
        in trans (mod (plus a b) e) (mod a e) 0
             (trans (mod (plus a b) e) (Generic.iterate 0l ℕ w a (mod b e))
                (mod a e) split atZero)
             ha
        end
    end
end

Every multiple of k is divisible by k.

theorem dividesMult : ∀ (k : ℕ) (m : ℕ) → Divides k (mult m k) ≔
  λ (k : ℕ) (m : ℕ) →
    elim m return x . Divides k (mult x k)
    | zero ⇒ dividesZero k
    | succ p, ih ⇒ dividesPlus k k (mult p k) (dividesRefl k) ih
    end
end

A number divisible by succ d is its quotient times succ d: the division equation with a zero remainder.

theorem dividesQuot : ∀ (d : ℕ) (b : ℕ) (h : Divides (succ d) b) →
    Eq (mult (div b (succ d)) (succ d)) b ≔
  λ (d : ℕ) (b : ℕ) (h : Eq (mod b (succ d)) 0) →
    let k ≔ succ d;
        m ≔ mult (div b k) k;
        dropZero : Eq m (plus m 0) ≔ sym (plus m 0) m (plusZero m);
        toRem : Eq (plus m 0) (plus m (mod b k)) ≔
          cong (λ (y : ℕ) → plus m y) 0 (mod b k) (sym (mod b k) 0 h)
    in trans m (plus m 0) b dropZero
         (trans (plus m 0) (plus m (mod b k)) b toRem (divModSpec b d))
    end
end

A divisor of b divides every multiple mult m b.

theorem dividesMultLeft : ∀ (k : ℕ) (b : ℕ) (m : ℕ) (h : Divides k b) →
    Divides k (mult m b) ≔
  λ (k : ℕ) (b : ℕ) (m : ℕ) →
    elim k return x . (∀ (h : Divides x b) → Divides x (mult m b))
    | zero ⇒ λ (h : Eq b 0) →
        transport (λ (y : ℕ) → Eq (mult m y) 0) 0 b (sym b 0 h) (multZero m)
    | succ d, ih ⇒ λ (h : Divides (succ d) b) →
        let e ≔ succ d;
            q ≔ div b e;
            toB : Eq (mult (mult m q) e) (mult m b) ≔
              trans (mult (mult m q) e) (mult m (mult q e)) (mult m b)
                (multAssoc m q e)
                (cong (λ (y : ℕ) → mult m y) (mult q e) b (dividesQuot d b h))
        in transport (λ (y : ℕ) → Divides e y) (mult (mult m q) e) (mult m b)
             toB (dividesMult e (mult m q))
        end
    end
end

A divisor of x and of plus x y divides y.

theorem dividesPlusCancel : ∀ (k : ℕ) (x : ℕ) (y : ℕ)
    (hx : Divides k x) (hxy : Divides k (plus x y)) → Divides k y ≔
  λ (k : ℕ) (x : ℕ) (y : ℕ) →
    elim k return w . (∀ (hx : Divides w x) (hxy : Divides w (plus x y)) →
                         Divides w y)
    | zero ⇒ λ (hx : Eq x 0) (hxy : Eq (plus x y) 0) → plusEqZeroRight x y hxy
    | succ d, ih ⇒ λ (hx : Divides (succ d) x)
                     (hxy : Eq (mod (plus x y) (succ d)) 0) →
        let e ≔ succ d;
            m ≔ mult (div x e) e;
            atM : Eq (mod (plus m y) e) 0 ≔
              transport (λ (z : ℕ) → Eq (mod (plus z y) e) 0) x m
                (sym m x (dividesQuot d x hx)) hxy
        in trans (mod y e) (mod (plus m y) e) 0
             (sym (mod (plus m y) e) (mod y e) (modPlusMult d (div x e) y)) atM
        end
    end
end

end

Divisibility by two is evenness.

module Two where

The remainder by two is the parity flag: 0 for even numbers and 1 for odd ones. Both halves are proved together, since the step exchanges them.

let modTwoParity : ∀ (n : ℕ) →
    And (∀ (h : Even n) → Eq (mod n 2) 0) (∀ (h : Odd n) → Eq (mod n 2) 1) ≔
  λ (n : ℕ) →
    elim n return x . And (∀ (h : Even x) → Eq (mod x 2) 0)
                          (∀ (h : Odd x) → Eq (mod x 2) 1)
    | zero ⇒ andIntro (∀ (h : ⊤) → ⊤) (∀ (h : ⊥) → ⊥)
                (λ (h : ⊤) → ⋆) (λ (h : ⊥) → h)
    | succ p, ih ⇒
        let r ≔ mod p 2;
            fromOdd : ∀ (h : Odd p) → Eq (wrap r 1) 0 ≔
              λ (h : Odd p) →
                transport (λ (y : ℕ) → Eq (wrap y 1) 0) 1 r
                  (sym r 1
                     (andSnd (∀ (h : Even p) → Eq r 0) (∀ (h : Odd p) → Eq r 1)
                        ih h))
                  ⋆;
            fromEven : ∀ (h : Even p) → Eq (wrap r 1) 1 ≔
              λ (h : Even p) →
                transport (λ (y : ℕ) → Eq (wrap y 1) 1) 0 r
                  (sym r 0
                     (andFst (∀ (h : Even p) → Eq r 0) (∀ (h : Odd p) → Eq r 1)
                        ih h))
                  ⋆
        in andIntro (∀ (h : Odd p) → Eq (wrap r 1) 0)
             (∀ (h : Even p) → Eq (wrap r 1) 1) fromOdd fromEven
        end
    end
end

An even number is divisible by two.

theorem evenDividesTwo : ∀ (n : ℕ) (h : Even n) → Divides 2 n ≔
  λ (n : ℕ) (h : Even n) →
    andFst (∀ (h : Even n) → Eq (mod n 2) 0) (∀ (h : Odd n) → Eq (mod n 2) 1)
      (modTwoParity n) h
end

A number divisible by two is even: were it odd, its remainder would be 1 rather than 0.

theorem dividesTwoEven : ∀ (n : ℕ) (h : Divides 2 n) → Even n ≔
  λ (n : ℕ) (h : Eq (mod n 2) 0) →
    orElim (Even n) (Odd n) (Even n) (evenOrOdd n)
      (λ (e : Even n) → e)
      (λ (o : Odd n) →
         absurd (Even n)
           (trans 1 (mod n 2) 0
              (sym (mod n 2) 1
                 (andSnd (∀ (h : Even n) → Eq (mod n 2) 0)
                    (∀ (h : Odd n) → Eq (mod n 2) 1) (modTwoParity n) o))
              h))
end

end
