Laws of Prelude›Data›Ordering, stated with the Leibniz equality of Prelude›Logic›Leibniz at Ordering: flip is an involution and reverses compare; compare is correct, Spec m n (compare m n), and complete for each outcome; thenCmp is a monoid with unit equal.

Spec m n o is the evidence the constructors of Agda's indexed Ordering m n carry: Lt m n for less, Eq m n for equal and Lt n m for greater. Since Lt and Eq compute on successors, Spec (succ p) (succ q) o is Spec p q o, and the step case of compareSpec is its hypothesis.

module Prelude›Data›Ordering›Properties where

import Prelude›Builtin›Nat use (succ)
import Prelude›Logic›Leibniz 0l use (Id; idRefl; idTrans; idCong)
import Prelude›Arith›Equality use (Eq)
import Prelude›Arith›Order use (Lt)
import Prelude›Data›Ordering use (Ordering; less; equal; greater; flip;
  compare; thenCmp)

def Spec (m : ℕ) (n : ℕ) (o : Ordering) : Type@0 ≔
  elim o return z . Type@0
  | less ⇒ Lt m n
  | equal ⇒ Eq m n
  | greater ⇒ Lt n m
  end
end

flip.

module Flip where

theorem flipFlip : ∀ (o : Ordering) → Id Ordering (flip (flip o)) o ≔
  λ (o : Ordering) →
    elim o return z . Id Ordering (flip (flip z)) z
    | less ⇒ idRefl Ordering less
    | equal ⇒ idRefl Ordering equal
    | greater ⇒ idRefl Ordering greater
    end
end

A comparison the other way round is flipped.

theorem compareFlip : ∀ (m : ℕ) (n : ℕ) →
    Id Ordering (compare n m) (flip (compare m n)) ≔
  λ (m : ℕ) →
    elim m return x .
      (∀ (n : ℕ) → Id Ordering (compare n x) (flip (compare x n)))
    | zero ⇒ λ (n : ℕ) →
        elim n return y . Id Ordering (compare y 0) (flip (compare 0 y))
        | zero ⇒ idRefl Ordering equal
        | succ q, s ⇒ idRefl Ordering greater
        end
    | succ p, ih ⇒ λ (n : ℕ) →
        elim n return y . Id Ordering (compare y (succ p))
                            (flip (compare (succ p) y))
        | zero ⇒ idRefl Ordering less
        | succ q, s ⇒ ih q
        end
    end
end

end

compare against the order and the equality of numbers.

module Compare where

theorem compareRefl : ∀ (n : ℕ) → Id Ordering (compare n n) equal ≔
  λ (n : ℕ) →
    elim n return x . Id Ordering (compare x x) equal
    | zero ⇒ idRefl Ordering equal
    | succ p, ih ⇒ ih
    end
end

Correctness: the outcome of compare comes with its evidence.

theorem compareSpec : ∀ (m : ℕ) (n : ℕ) → Spec m n (compare m n) ≔
  λ (m : ℕ) →
    elim m return x . (∀ (n : ℕ) → Spec x n (compare x n))
    | zero ⇒ λ (n : ℕ) →
        elim n return y . Spec 0 y (compare 0 y)
        | zero ⇒ ⋆
        | succ q, s ⇒ ⋆
        end
    | succ p, ih ⇒ λ (n : ℕ) →
        elim n return y . Spec (succ p) y (compare (succ p) y)
        | zero ⇒ ⋆
        | succ q, s ⇒ ih q
        end
    end
end

Completeness at less: the impossible cases have a hypothesis of type ⊥.

theorem compareLt : ∀ (m : ℕ) (n : ℕ) (h : Lt m n) →
    Id Ordering (compare m n) less ≔
  λ (m : ℕ) →
    elim m return x . (∀ (n : ℕ) (h : Lt x n) → Id Ordering (compare x n) less)
    | zero ⇒ λ (n : ℕ) →
        elim n return y . (∀ (h : Lt 0 y) → Id Ordering (compare 0 y) less)
        | zero ⇒ λ (h : ⊥) → exfalso h return w . Id Ordering equal less
        | succ q, s ⇒ λ (h : ⊤) → idRefl Ordering less
        end
    | succ p, ih ⇒ λ (n : ℕ) →
        elim n return y . (∀ (h : Lt (succ p) y) →
                             Id Ordering (compare (succ p) y) less)
        | zero ⇒ λ (h : ⊥) → exfalso h return w . Id Ordering greater less
        | succ q, s ⇒ λ (h : Lt p q) → ih q h
        end
    end
end

Completeness at equal.

theorem compareEq : ∀ (m : ℕ) (n : ℕ) (h : Eq m n) →
    Id Ordering (compare m n) equal ≔
  λ (m : ℕ) →
    elim m return x . (∀ (n : ℕ) (h : Eq x n) → Id Ordering (compare x n) equal)
    | zero ⇒ λ (n : ℕ) →
        elim n return y . (∀ (h : Eq 0 y) → Id Ordering (compare 0 y) equal)
        | zero ⇒ λ (h : ⊤) → idRefl Ordering equal
        | succ q, s ⇒ λ (h : ⊥) → exfalso h return w . Id Ordering less equal
        end
    | succ p, ih ⇒ λ (n : ℕ) →
        elim n return y . (∀ (h : Eq (succ p) y) →
                             Id Ordering (compare (succ p) y) equal)
        | zero ⇒ λ (h : ⊥) → exfalso h return w . Id Ordering greater equal
        | succ q, s ⇒ λ (h : Eq p q) → ih q h
        end
    end
end

Completeness at greater, from compareLt by compareFlip.

theorem compareGt : ∀ (m : ℕ) (n : ℕ) (h : Lt n m) →
    Id Ordering (compare m n) greater ≔
  λ (m : ℕ) (n : ℕ) (h : Lt n m) →
    idTrans Ordering (compare m n) (flip (compare n m)) greater
      (Flip.compareFlip n m)
      (idCong Ordering Ordering flip (compare n m) less (compareLt n m h))
end

end

thenCmp is a monoid with unit equal.

module Then where

theorem thenEqualRight : ∀ (o : Ordering) → Id Ordering (thenCmp o equal) o ≔
  λ (o : Ordering) →
    elim o return z . Id Ordering (thenCmp z equal) z
    | less ⇒ idRefl Ordering less
    | equal ⇒ idRefl Ordering equal
    | greater ⇒ idRefl Ordering greater
    end
end

theorem thenAssoc : ∀ (o : Ordering) (p : Ordering) (q : Ordering) →
    Id Ordering (thenCmp (thenCmp o p) q) (thenCmp o (thenCmp p q)) ≔
  λ (o : Ordering) (p : Ordering) (q : Ordering) →
    elim o return z . Id Ordering (thenCmp (thenCmp z p) q)
                        (thenCmp z (thenCmp p q))
    | less ⇒ idRefl Ordering less
    | equal ⇒ idRefl Ordering (thenCmp p q)
    | greater ⇒ idRefl Ordering greater
    end
end

flip distributes over thenCmp.

theorem flipThen : ∀ (o : Ordering) (p : Ordering) →
    Id Ordering (flip (thenCmp o p)) (thenCmp (flip o) (flip p)) ≔
  λ (o : Ordering) (p : Ordering) →
    elim o return z . Id Ordering (flip (thenCmp z p))
                        (thenCmp (flip z) (flip p))
    | less ⇒ idRefl Ordering greater
    | equal ⇒ idRefl Ordering (flip p)
    | greater ⇒ idRefl Ordering less
    end
end

end
