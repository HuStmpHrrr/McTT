Mctt.Core.Semantic.Inductives.IndCanon

The Clause of an Inductive Type, Canonically

Two related inductive types give their telescope relations as arguments of the clause of per_univ_elem_core. These are the canonical ones of Core.Semantic.Inductives.IndPER, at the uniform parameters of either type.

The relation of a related pair of types is the head PER at either.
  Lemma per_head_left : forall {S S' ρ ρ' i R a a'},
      ⟦ S ⟧ ρ ↘ a -> ⟦ S' ⟧ ρ' ↘ a' -> DF a ≈ a' ∈ per_univ_elem i ↘ R -> (R <~> per_head S S ρ ρ).
  Proof.
    intros * Ha _ HR; eapply per_head_iff; [ exact Ha | exact Ha |].
    eapply per_univ_trans; [ exact HR | apply per_univ_sym; exact HR ].
  Qed.

  Lemma per_head_right : forall {S S' ρ ρ' i R a a'},
      ⟦ S ⟧ ρ ↘ a -> ⟦ S' ⟧ ρ' ↘ a' -> DF a ≈ a' ∈ per_univ_elem i ↘ R -> (R <~> per_head S' S' ρ' ρ').
  Proof.
    intros * _ Ha' HR; eapply per_head_iff; [ exact Ha' | exact Ha' |].
    eapply per_univ_trans; [ apply per_univ_sym; exact HR | exact HR ].
  Qed.

    ; icp_arg_per : forall ms ms' c gs k A vs vs', tele_rel (canon_nparams_rel D ups) (List.rev (ind_nparams D)) ms ms' ->
        ind_con D c = Some gs -> List.nth_error (cargs_nonrec gs) k = Some A ->
        tele_prefix_rel (canon_args_rel D ups ms c) k vs vs' -> PER (canon_args_rel D ups ms c k vs)
    ; icp_arg_iff : forall ms ms' c gs k A vs vs', tele_rel (canon_nparams_rel D ups) (List.rev (ind_nparams D)) ms ms' ->
        ind_con D c = Some gs -> List.nth_error (cargs_nonrec gs) k = Some A ->
        tele_prefix_rel (canon_args_rel D ups ms c) k vs vs' -> (canon_args_rel D ups ms c k vs <~> canon_args_rel D ups ms' c k vs')

        gs = gs1 ++ ca_rec t :: gs2 -> tele_prefix_rel (canon_args_rel D ups ms c) (List.length (cargs_nonrec gs1)) ws ws' ->

Two families of relations agree where both are defined, as the section FamChange of Core.Semantic.PER.Tele needs.
  Record ind_fam_agree (D : ind_decl) (nr1 nr2 : nat -> list domain -> relation domain)
      (ar1 ar2 : list domain -> String.string -> nat -> list domain -> relation domain)
      (dr1 dr2 : list domain -> exp -> relation domain) : Prop := mk_ind_fam_agree
    { ifa_np : forall k vs vs1 vs2, k < List.length (List.rev (ind_nparams D)) ->
        tele_prefix_rel nr1 k vs vs1 -> tele_prefix_rel nr2 k vs vs2 -> (nr1 k vs <~> nr2 k vs)
    ; ifa_arg : forall ms ms1 ms2 c gs k vs vs1 vs2,
        tele_rel nr1 (List.rev (ind_nparams D)) ms ms1 -> tele_rel nr2 (List.rev (ind_nparams D)) ms ms2 ->
        ind_con D c = Some gs -> k < List.length (cargs_nonrec gs) ->
        tele_prefix_rel (ar1 ms c) k vs vs1 -> tele_prefix_rel (ar2 ms c) k vs vs2 ->
        (ar1 ms c k vs <~> ar2 ms c k vs)
    ; ifa_dom : forall ms ms1 ms2 c gs gs1 t gs2 ws ws1 ws2 k ys ys1 ys2,
        tele_rel nr1 (List.rev (ind_nparams D)) ms ms1 -> tele_rel nr2 (List.rev (ind_nparams D)) ms ms2 ->
        ind_con D c = Some gs -> gs = gs1 ++ ca_rec t :: gs2 ->
        tele_prefix_rel (ar1 ms c) (List.length (cargs_nonrec gs1)) ws ws1 ->
        tele_prefix_rel (ar2 ms c) (List.length (cargs_nonrec gs1)) ws ws2 ->
        k < List.length (ptyp_doms t) ->
        tele_prefix_rel (shape_doms_rel dr1 ms ws (ptyp_doms t)) k ys ys1 ->
        tele_prefix_rel (shape_doms_rel dr2 ms ws (ptyp_doms t)) k ys ys2 ->
        (shape_doms_rel dr1 ms ws (ptyp_doms t) k ys <~> shape_doms_rel dr2 ms ws (ptyp_doms t) k ys) }.

  Lemma ind_fam_agree_sym : forall D nr1 nr2 ar1 ar2 dr1 dr2,
      ind_fam_agree D nr1 nr2 ar1 ar2 dr1 dr2 -> ind_fam_agree D nr2 nr1 ar2 ar1 dr2 dr1.
  Proof.
    intros * [Hn Ha Hd]; constructor; intros; symmetry; [ eapply Hn | eapply Ha | eapply Hd ]; eassumption.
  Qed.

           DF a ≈ a' ∈ per_univ_elem i ↘ nr k vs) ->

         exists a a', ⟦ A ⟧ env_args nil (ups ++ ms ++ vs) ↘ a /\ ⟦ A ⟧ env_args nil (ups' ++ ms' ++ vs') ↘ a' /\
           DF a ≈ a' ∈ per_univ_elem i ↘ ar ms c k vs) ->
      (forall ms ms' c gs gs1 t gs2 ws ws' k B ys ys', tele_rel nr (List.rev (ind_nparams D)) ms ms' ->
         ind_con D c = Some gs -> gs = gs1 ++ ca_rec t :: gs2 ->
         tele_prefix_rel (ar ms c) (List.length (cargs_nonrec gs1)) ws ws' ->
         List.nth_error (ptyp_doms t) k = Some B -> tele_prefix_rel (shape_doms_rel dr ms ws (ptyp_doms t)) k ys ys' ->
         exists b b', ⟦ B ⟧ env_args nil (ups ++ ms ++ ws ++ ys) ↘ b /\ ⟦ B ⟧ env_args nil (ups' ++ ms' ++ ws' ++ ys') ↘ b' /\
           DF b ≈ b' ∈ per_univ_elem i ↘ dr (ms ++ ws ++ ys) B) ->
      ind_fam_agree D nr (canon_nparams_rel D ups) ar (canon_args_rel D ups) dr (canon_doms_rel ups).

      unfold canon_nparams_rel; rewrite (nth_ctyp_eq _ _ _ HA).
      exact (per_head_left Ea Ea' HR).
    - intros * Hm _ Hc Hk Hp _.
      destruct (nth_error_lt_some' _ _ Hk) as [A HA].
      destruct (Ha _ _ _ _ _ _ _ _ Hm Hc HA Hp) as (a & a' & Ea & Ea' & HR).
      unfold canon_args_rel; rewrite (nth_atyp_eq _ _ _ _ _ Hc HA).
      exact (per_head_left Ea Ea' HR).
    - intros * Hm _ Hc Heq Hw _ Hk Hy _.
      destruct (nth_error_lt_some' _ _ Hk) as [B HB].
      destruct (Hd _ _ _ _ _ _ _ _ _ _ _ _ _ Hm Hc Heq Hw HB Hy) as (b & b' & Eb & Eb' & HR).
      unfold shape_doms_rel; rewrite (List.nth_error_nth _ _ _ HB); unfold canon_doms_rel.
      exact (per_head_left Eb Eb' HR).
  Qed.
