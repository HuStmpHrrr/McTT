Mctt.Core.Semantic.Kinds.KindsEval

A Module Expression Has the Kinds of its Value

In a well-formed global context, the kinds read off a module expression in an environment (modexp_member_kind) are those of the module value it evaluates to (dmod_member_kind), and selecting or applying a module value keeps the kinds of what is left (kind_eval). A body evaluates to an environment with the kinds of the one the kinds read off it (kinds_env).

Renaming by the Identity


  Lemma modexp_member_kind_id : forall ρ ρ', kinds_rename Θ Ξ wk_id ρ ρ' -> forall H ch k, modexp_member_kind Θ Ξ ρ H ch k <-> modexp_member_kind Θ Ξ ρ' H ch k.
  Proof.
    intros * K H ch k.
    rewrite <- (proj1 (kind_rename Θ Ξ) H wk_id ρ ρ' K), modexp_wk_id_ext by (intros x; reflexivity).
    reflexivity.
  Qed.

Of an expanded unit, as kinds read units (gunit_expand).
  Lemma gunit_member_kind_id : forall ρ ρ', kinds_rename Θ Ξ wk_id ρ ρ' -> forall U ch k,
      gunit_member_kind Θ Ξ ρ (gunit_expand U) ch k <-> gunit_member_kind Θ Ξ ρ' (gunit_expand U) ch k.
  Proof.
    intros * K U ch k.
    rewrite <- (proj1 (proj2 (kind_rename Θ Ξ)) U wk_id ρ ρ' K), gunit_wk_id_ext by (intros x; reflexivity).
    reflexivity.
  Qed.

Environments with the Same Kinds and Shape

The sort of an environment entry: a term (0), a module (1), or a term with a module part (2).
  Definition dentry_tag (e : dentry) : nat :=
    match e with
    | de_term _ _ => 0
    | de_mod _ _ => 1
    | de_dm _ _ _ _ => 2
    end.

Two environments with the same kinds, and entries of the same sorts at the same places.
  Definition kinds_shape (ρ ρ' : env) : Prop :=
    kinds_rename Θ Ξ wk_id ρ ρ' /\ forall x, dentry_tag (env_entry ρ x) = dentry_tag (env_entry ρ' x).

  Lemma kinds_shape_cons_term : forall ρ ρ' a a' t t', kinds_shape ρ ρ' -> kinds_shape (de_term a t :: ρ) (de_term a' t' :: ρ').
  Proof. intros * [K S]; split; intros [| x]; first [ apply kinds_equiv_refl | apply K | reflexivity | apply S ]. Qed.

  Lemma kinds_shape_cons_mod : forall ρ ρ' h h' U U', kinds_shape ρ ρ' -> kinds_equiv Θ Ξ h h' ->
      kinds_shape (de_mod h U :: ρ) (de_mod h' U' :: ρ').
  Proof. intros * [K S] Hh; split; intros [| x]; first [ exact Hh | apply K | reflexivity | apply S ]. Qed.

A term with a module part, by the kinds of its module part.
  Lemma kinds_shape_cons_dm : forall ρ ρ' a a' t t' h h' U U', kinds_shape ρ ρ' -> kinds_equiv Θ Ξ h h' ->
      kinds_shape (de_dm a t h U :: ρ) (de_dm a' t' h' U' :: ρ').
  Proof. intros * [K S] Hh; split; intros [| x]; first [ exact Hh | apply K | reflexivity | apply S ]. Qed.

An environment extended by typed values is their entries, innermost first, over it.
  Lemma env_extend_typed_rev : forall vts ρ,
      env_extend_typed ρ vts = List.rev (List.map (fun vt => de_term (fst vt) (snd vt)) vts) ++ ρ.
  Proof.
    induction vts as [| vt vts IH]; intros; [ reflexivity |].
    unfold env_extend_typed in *; cbn [List.fold_left List.map List.rev]; rewrite IH, <- app_assoc; reflexivity.
  Qed.

  Lemma kinds_shape_terms : forall X Y ρ ρ', List.length X = List.length Y ->
      Forall (fun e => dentry_tag e = 0) X -> Forall (fun e => dentry_tag e = 0) Y ->
      kinds_shape ρ ρ' -> kinds_shape (X ++ ρ) (Y ++ ρ').
  Proof.
    induction X as [| e X IH]; intros [| e' Y] * Hl HX HY Hs; cbn in Hl; try discriminate; [ exact Hs |].
    inversion HX; inversion HY; subst.
    destruct e as [a t | h | a t h], e' as [a' t' | h' | a' t' h']; cbn in *; try discriminate.
    apply kinds_shape_cons_term, IH; auto.
  Qed.

  Lemma kinds_shape_refl : forall ρ, kinds_shape ρ ρ.
  Proof. intros ρ; split; [ intros x; apply kinds_equiv_refl | reflexivity ]. Qed.

An argument pushed under pads is one more pad.
  Lemma kinds_shape_pad_push : forall m ρ n t, kinds_shape (kinds_pads m (de_term n t :: ρ)) (kinds_pads (S m) ρ).
  Proof.
    intros; unfold kinds_pads.
    replace (List.repeat absent_entry m ++ de_term n t :: ρ)
      with ((List.repeat absent_entry m ++ de_term n t :: nil) ++ ρ)
      by (rewrite <- app_assoc; reflexivity).
    replace (List.repeat absent_entry (S m) ++ ρ)
      with ((List.repeat absent_entry m ++ absent_entry :: nil) ++ ρ)
      by (cbn [List.repeat]; rewrite List.repeat_cons, <- app_assoc; reflexivity).
    apply kinds_shape_terms.
    - rewrite !length_app, !repeat_length; reflexivity.
    - apply Forall_app; split; [ apply Forall_forall; intros e He; apply repeat_spec in He as ->; reflexivity | repeat constructor ].
    - apply Forall_app; split; [ apply Forall_forall; intros e He; apply repeat_spec in He as ->; reflexivity | repeat constructor ].
    - apply kinds_shape_refl.
  Qed.

  Lemma kinds_shape_args_pad : forall ρ vts n, List.length vts = n -> kinds_shape (env_extend_typed ρ vts) (kinds_pads n ρ).
  Proof.
    intros * <-; rewrite env_extend_typed_rev; unfold kinds_pads; apply kinds_shape_terms.
    - rewrite length_rev, length_map, repeat_length; reflexivity.
    - apply Forall_rev, Forall_map, Forall_forall; reflexivity.
    - apply Forall_forall; intros e He; apply repeat_spec in He as ->; reflexivity.
    - split; [ intros x; apply kinds_equiv_refl | reflexivity ].
  Qed.

Global Modules


  Lemma dmod_member_kind_global_args : forall ch p args args' k,
      dmod_member_kind Θ Ξ (dm_global p args) ch k -> dmod_member_kind Θ Ξ (dm_global p args') ch k.
  Proof.
    induction ch as [| y ch IH]; intros * Hk; inversion Hk; subst;
      first [ eapply dmod_member_kind_global_here_def; eassumption | eapply dmod_member_kind_global_here_mod; eassumption
            | eapply dmod_member_kind_global_here_dm; eassumption | eapply dmod_member_kind_global_alias_dm; eassumption
            | eapply dmod_member_kind_global_alias_case; eassumption | eapply dmod_member_kind_global_step; [ eassumption | eassumption | eapply IH; eassumption ] ].
  Qed.

A stored body is a module.
  Lemma dmod_member_kind_global_loc : forall p args TΦ, gc_loc Θ Ξ p = Some TΦ -> dmod_member_kind Θ Ξ (dm_global p args) nil member_kind_mod.
  Proof.
    intros * Hl; apply dmod_member_kind_global_here_mod; unfold gc_loc in Hl; destruct (gc_frame Ξ p) eqn:Hf; [ right; eauto |].
    destruct TΦ; left; destruct (gc_body_module _ _ _ _ _ Hl); eauto.
  Qed.

A stored submodule has the kinds of its parent's submodule closure: past the empty chain those of the longer chain, and at it a module.
  Lemma dmod_member_kind_global_snoc : forall p args y ch k TΦ,
      gc_loc Θ Ξ (qname_app p (y :: nil)) = Some TΦ ->
      dmod_member_kind Θ Ξ (dm_global (qname_app p (y :: nil)) args) ch k <-> dmod_member_kind Θ Ξ (dm_member (dm_global p args) (y :: nil)) ch k.
  Proof.
    intros * Hl; destruct ch as [| c ch].
    - rewrite dmod_member_kind_member_iff; cbn [app].
      pose proof (dmod_member_kind_global_loc _ args _ Hl) as Hm.
      split.
      + intros Hk; pose proof (proj1 (@functional_kind Θ Ξ) _ _ _ Hk _ Hm) as ->.
        destruct (dm_path_cases Θ Ξ (qname_app p (y :: nil))) as [Hd | Hd].
        * right; repeat split; eapply dmod_member_kind_global_here_dm; [ exact Hd | exact Hl ].
        * left; split; [ apply dmod_member_kind_global_step; [ left; eexists; exact Hl | right; exact Hd | exact Hm ] | right; discriminate ].
      + intros [[Hk Hc] | (_ & -> & _)]; [| exact Hm ].
        inversion Hk; subst;
          first [ destruct Hc as [Hc | Hc]; exfalso; exact (Hc eq_refl)
                | exfalso; eapply gc_loc_not_alias; eassumption | assumption ].
    - rewrite dmod_member_kind_member_cons_iff; cbn [app]; split; intros Hk.
      + apply dmod_member_kind_global_step; [ left; eexists; exact Hl | left; discriminate | assumption ].
      + inversion Hk; subst; first [ exfalso; eapply gc_loc_not_alias; eassumption | assumption ].
  Qed.

  Lemma dm_of_cases : forall k, (k = member_kind_mod /\ dm_of k = member_kind_dm) \/ (k <> member_kind_mod /\ dm_of k = k).
  Proof. intros []; cbn; [ right | left | right ]; split; first [ reflexivity | discriminate ]. Qed.

An alias submodule has the kinds of its parent's submodule closure: past the empty chain those of its unit, and at it a module.
  Lemma dmod_member_kind_global_alias : forall p args y U ch k,
      gc_module Θ Ξ (qname_app p (y :: nil)) = Some (mr_alias U) ->
      gunit_member_kind Θ Ξ nil (gunit_expand U) ch k <-> dmod_member_kind Θ Ξ (dm_member (dm_global p args) (y :: nil)) ch k.
  Proof.
    intros * Ha.
    assert (Hst : (exists TΦ, gc_loc Θ Ξ (qname_app p (y :: nil)) = Some TΦ) \/
                  (exists b pv ra A B, gc_resolve Θ Ξ (qname_app p (y :: nil)) = Some (ge_def b pv ra A B)) -> False)
      by (intros [(? & Hl) | (? & ? & ? & ? & ? & Hr)];
          [ exact (gc_loc_not_alias _ _ _ _ _ Hl Ha) | rewrite (gc_module_resolve _ _ _ _ Ha) in Hr; discriminate Hr ]).
    pose proof (proj1 (proj2 (@functional_kind Θ Ξ))) as Fu.
    pose proof (proj1 (proj2 (kind_nil_not_dm Θ Ξ))) as Nd.
    destruct ch as [| c ch].
    - rewrite dmod_member_kind_member_iff; cbn [app]; split.
      + intros Hk; pose proof (Nd _ _ _ _ Hk eq_refl) as Hk'.
        destruct (dm_path_cases Θ Ξ (qname_app p (y :: nil))) as [Hd | Hd].
        * destruct (dm_of_cases k) as [(-> & _) | (Hm & Ek)].
          -- right; repeat split; change member_kind_dm with (dm_of member_kind_mod); eapply dmod_member_kind_global_alias_dm; eassumption.
          -- left; split; [ rewrite <- Ek; eapply dmod_member_kind_global_alias_dm; eassumption | right; exact Hk' ].
        * left; split; [ eapply dmod_member_kind_global_alias_case; [ exact Ha | right; exact Hd | exact Hk ] | right; exact Hk' ].
      + intros [[Hk Hc] | (_ & -> & Hk)].
        * inversion Hk; subst;
            try (match goal with Ha' : gc_module _ _ _ = Some (mr_alias _) |- _ => rewrite Ha in Ha'; injection Ha' as <- end).
          -- exfalso; eapply gc_loc_not_alias; eassumption.
          -- assumption.
          -- match goal with Hk0 : gunit_member_kind _ _ nil (gunit_expand U) nil ?k0 |- _ =>
               destruct (dm_of_cases k0) as [(-> & ->) | (_ & ->)]; [ destruct Hc as [Hc | Hc]; exfalso; exact (Hc eq_refl) | exact Hk0 ] end.
          -- exfalso; eapply Hst; eassumption.
        * inversion Hk; subst;
            try (match goal with Ha' : gc_module _ _ _ = Some (mr_alias _) |- _ => rewrite Ha in Ha'; injection Ha' as <- end).
          -- exfalso; eapply gc_loc_not_alias; eassumption.
          -- exfalso; eapply Nd; [ eassumption | reflexivity | reflexivity ].
          -- match goal with Hk0 : gunit_member_kind _ _ nil (gunit_expand U) nil ?k0, Ek : dm_of ?k0 = member_kind_dm |- _ =>
               destruct (dm_of_cases k0) as [(-> & _) | (Hm & Ek')]; [ exact Hk0 |];
               rewrite Ek' in Ek; subst; exfalso; exact (Nd _ _ _ _ Hk0 eq_refl eq_refl) end.
          -- exfalso; eapply Hst; eassumption.
    - rewrite dmod_member_kind_member_cons_iff; cbn [app]; split; intros Hk.
      + eapply dmod_member_kind_global_alias_case; [ exact Ha | left; discriminate | exact Hk ].
      + inversion Hk; subst.
        * match goal with Ha' : gc_module _ _ _ = Some (mr_alias _) |- _ => rewrite Ha in Ha'; injection Ha' as <- end; assumption.
        * exfalso; eapply Hst; eassumption.
  Qed.

The Kinds a Body Gives Its Names


  Lemma gunit_member_kind_gens_other : forall gs Φ k ρ sm y ch kk, existsb (String.eqb y) (map generated_item_name gs) = false ->
      gunit_member_kind Θ Ξ ρ (gu_mk nil (md_body sm (gens_append Φ k gs))) (y :: ch) kk <->
      gunit_member_kind Θ Ξ ρ (gu_mk nil (md_body sm Φ)) (y :: ch) kk.
  Proof.
    induction gs as [| g gs IH]; intros * Hn; cbn [gens_append]; [ reflexivity |].
    cbn [map existsb] in Hn; apply Bool.orb_false_iff in Hn as [Hy Hn].
    unfold generated_item_name in Hy; destruct (generated_item_entry g) as [d E] eqn:Eg; cbn [fst] in Hy.
    rewrite IH by exact Hn; apply gunit_member_kind_body_ext_other.
    intros ->; rewrite String.eqb_refl in Hy; discriminate.
  Qed.

  Lemma gm_prefix_name : forall Φ y Φ0 z E, gm_prefix_upto Φ y = Some (gm_ext Φ0 z E) -> z = y.
  Proof.
    induction Φ as [| Φ IH x E0 | Φ IH H its]; intros * Hp; cbn in Hp; [ discriminate | |].
    - destruct (String.eqb_spec y x) as [-> |]; [ injection Hp as _ <- _; reflexivity | eauto ].
    - destruct (existsb _ _); [ discriminate | eauto ].
  Qed.

The kinds a body gives a name its own entry declares.
  Lemma gunit_member_kind_body_entry : forall ρ sm Φ y Φ' E ch k,
      gm_prefix_upto Φ y = Some (gm_ext Φ' y E) ->
      gunit_member_kind Θ Ξ ρ (gu_mk nil (md_body sm Φ)) (y :: ch) k <->
      (exists b pv ra A B, E = ge_def b pv ra A B /\ ch = nil /\ k = member_kind_term ra) \/
      (exists pm U ρ1, E = ge_mod pm U /\ kinds_env Θ Ξ ρ Φ' ρ1 /\ gunit_member_kind Θ Ξ ρ1 (gunit_expand U) ch k) \/
      (exists pv A M U ρ1 k0, E = ge_def_mod pv A M U /\ ch = nil /\ kinds_env Θ Ξ ρ Φ' ρ1 /\
         gunit_member_kind Θ Ξ ρ1 (gunit_expand U) nil k0 /\ k = dm_of k0) \/
      (exists pv A M U ρ1, E = ge_def_mod pv A M U /\ ch <> nil /\ kinds_env Θ Ξ ρ Φ' ρ1 /\
         gunit_member_kind Θ Ξ ρ1 (gunit_expand U) ch k).
  Proof.
    intros * Hp; split; intros Hu.
    - inversion Hu; subst;
        try (rewrite (gm_prefix_item_exclusive _ _ _ Hp) in *; discriminate);
        match goal with H2 : gm_prefix_upto _ _ = Some _ |- _ => rewrite Hp in H2; injection H2; intros; subst end.
      + left; do 5 eexists; repeat split; reflexivity.
      + right; left; do 3 eexists; repeat split; eassumption.
      + right; right; left; do 6 eexists; repeat split; eassumption.
      + right; right; right; do 5 eexists; repeat split; eassumption.
    - destruct Hu as [(b & pv & ra & A & B & -> & -> & ->) | [(pm & U & ρ1 & -> & Hk & Hu)
                     | [(pv & A & M & U & ρ1 & k0 & -> & -> & Hk & Hu & ->) | (pv & A & M & U & ρ1 & -> & Hc & Hk & Hu)]]].
      + eapply gunit_member_kind_def; exact Hp.
      + eapply gunit_member_kind_mod; [ exact Hp | exact Hk | exact Hu ].
      + eapply gunit_member_kind_dm_here; [ exact Hp | exact Hk | exact Hu ].
      + eapply gunit_member_kind_dm_step; [ exact Hp | exact Hc | exact Hk | exact Hu ].
  Qed.

The module part of a body's entry (of a submodule, or of a definition with a module part) has the kinds of the body's submodule closure at its name.
  Lemma gunit_member_kind_body_mod : forall ρ Φ, gm_noopen Φ ->
      forall y Φ' E U ρ1, gm_prefix_upto Φ y = Some (gm_ext Φ' y E) -> entry_unit E = Some U -> kinds_env Θ Ξ ρ Φ' ρ1 ->
      forall sm ch k, gunit_member_kind Θ Ξ ρ1 (gunit_expand U) ch k <->
        dmod_member_kind Θ Ξ (dm_member (dm_local nil (md_body sm Φ) ρ) (y :: nil)) ch k.
  Proof.
    intros * Hno * Hp HE Hk1 sm ch k.
    assert (Hloc : forall ch0 k0, dmod_member_kind Θ Ξ (dm_local nil (md_body sm Φ) ρ) ch0 k0 <->
                               gunit_member_kind Θ Ξ ρ (gu_mk nil (md_body sm Φ)) ch0 k0)
      by (intros; rewrite dmod_member_kind_local_iff; cbn [gunit_expand]; rewrite (gm_expand_noopen _ Hno); reflexivity).
    pose proof (proj2 (proj2 (proj2 (@functional_kind Θ Ξ)))) as Fk.
    pose proof (proj1 (proj2 (kind_nil_not_dm Θ Ξ))) as Nd.
    destruct E as [| pm U0 | | pv A M U0 |]; cbn in HE; try discriminate; injection HE as ->.
    all: destruct ch as [| c ch].
    all: rewrite ?dmod_member_kind_member_cons_iff, ?dmod_member_kind_member_iff; cbn [app];
           rewrite ?Hloc, ?(gunit_member_kind_body_entry _ _ _ _ _ _ _ _ Hp).
    - (* a submodule, itself: its unit, no member of anything *)
      split.
      + intros Hu; pose proof (Nd _ _ _ _ Hu eq_refl) as Hn.
        left; split; [ right; left; do 3 eexists; split; [ reflexivity | split; [ exact Hk1 | exact Hu ] ] | right; exact Hn ].
      + intros [[H0 _] | (_ & -> & H0)].
        * destruct H0 as [(? & ? & ? & ? & ? & [=] & _) | [(pm' & U' & ρ0 & [= <- <-] & Hk0 & Hu)
                         | [(? & ? & ? & ? & ? & ? & [=] & _) | (? & ? & ? & ? & ? & [=] & _)]]].
          pose proof (Fk _ _ _ Hk0 _ Hk1) as ->; exact Hu.
        * apply (gunit_member_kind_body_entry _ _ _ _ _ _ _ _ Hp) in H0.
          destruct H0 as [(? & ? & ? & ? & ? & [=] & _) | [(pm' & U' & ρ0 & [= <- <-] & Hk0 & Hu)
                         | [(? & ? & ? & ? & ? & ? & [=] & _) | (? & ? & ? & ? & ? & [=] & _)]]].
          pose proof (Fk _ _ _ Hk0 _ Hk1) as ->; exfalso; exact (Nd _ _ _ _ Hu eq_refl eq_refl).
    - (* a submodule, past its name *)
      split.
      + intros Hu; right; left; do 3 eexists; split; [ reflexivity | split; [ exact Hk1 | exact Hu ] ].
      + intros [(? & ? & ? & ? & ? & [=] & _) | [(pm' & U' & ρ0 & [= <- <-] & Hk0 & Hu)
               | [(? & ? & ? & ? & ? & ? & [=] & _) | (? & ? & ? & ? & ? & [=] & _)]]].
        pose proof (Fk _ _ _ Hk0 _ Hk1) as ->; exact Hu.
    - (* a definition with a module part, itself: selected as a module *)
      split.
      + intros Hu; pose proof (Nd _ _ _ _ Hu eq_refl) as Hn.
        destruct (dm_of_cases k) as [(-> & _) | (Hm & Ek)].
        * right; split; [ reflexivity | split; [ reflexivity |] ].
          apply (gunit_member_kind_body_entry _ _ _ _ _ _ _ _ Hp).
          right; right; left; do 6 eexists; split; [ reflexivity | split; [ reflexivity | split; [ exact Hk1 | split; [ exact Hu | reflexivity ] ] ] ].
        * left; split; [| right; exact Hn ].
          right; right; left; do 6 eexists; split; [ reflexivity | split; [ reflexivity | split; [ exact Hk1 | split; [ exact Hu | symmetry; exact Ek ] ] ] ].
      + intros [[H0 Hc] | (_ & -> & H0)].
        * destruct H0 as [(? & ? & ? & ? & ? & [=] & _) | [(? & ? & ? & [=] & _)
                         | [(pv' & A' & M' & U' & ρ0 & k0 & [= <- <- <- <-] & _ & Hk0 & Hu & ->) | (? & ? & ? & ? & ? & _ & Hn & _)]]].
          -- pose proof (Fk _ _ _ Hk0 _ Hk1) as ->.
             destruct (dm_of_cases k0) as [(-> & ->) | (_ & ->)]; [ destruct Hc as [Hc | Hc]; exfalso; exact (Hc eq_refl) | exact Hu ].
          -- exfalso; exact (Hn eq_refl).
        * apply (gunit_member_kind_body_entry _ _ _ _ _ _ _ _ Hp) in H0.
          destruct H0 as [(? & ? & ? & ? & ? & [=] & _) | [(? & ? & ? & [=] & _)
                         | [(pv' & A' & M' & U' & ρ0 & k0 & [= <- <- <- <-] & _ & Hk0 & Hu & Ek) | (? & ? & ? & ? & ? & _ & Hn & _)]]].

             destruct (dm_of_cases k0) as [(-> & _) | (_ & Ek')]; [ exact Hu |].
             rewrite Ek' in Ek; subst; exfalso; exact (Nd _ _ _ _ Hu eq_refl eq_refl).
          -- exfalso; exact (Hn eq_refl).
    - (* a definition with a module part, past its name *)
      split.
      + intros Hu; right; right; right; do 5 eexists; split; [ reflexivity | split; [ discriminate | split; [ exact Hk1 | exact Hu ] ] ].
      + intros [(? & ? & ? & ? & ? & [=] & _) | [(? & ? & ? & [=] & _) | [(? & ? & ? & ? & ? & ? & _ & [=] & _)
               | (pv' & A' & M' & U' & ρ0 & [= <- <- <- <-] & _ & Hk0 & Hu)]]].

  Qed.

  Hypothesis Hg : ⊢ᵍ Θ ⍮ Ξ.

  Lemma gm_item_prefix_exclusive : forall Φ x r, gm_item_upto Φ x = Some r -> gm_prefix_upto Φ x = None.
  Proof.
    intros * Hi; destruct (gm_prefix_upto Φ x) eqn:E; [| reflexivity ].
    rewrite (gm_prefix_item_exclusive _ _ _ E) in Hi; discriminate.
  Qed.

The Kinds of Values


  Theorem kind_eval :
    (forall M ρ m, ⟦ M ⟧ Θ ⍮ Ξ ⍮ ρ ↘ m -> True) /\
    (forall m n r, $| m & n | Θ ⍮ Ξ ↘ r -> True) /\
    (forall Ms ρ ms, ⟦ Ms ⟧* Θ ⍮ Ξ ⍮ ρ ↘ ms -> True) /\
    (forall m args r, $*| m & args | Θ ⍮ Ξ ↘ r -> True) /\
    (forall H ρ h, ⟦ H ⟧ᵐ Θ ⍮ Ξ ⍮ ρ ↘ h -> forall ch k, dmod_member_kind Θ Ξ h ch k <-> modexp_member_kind Θ Ξ ρ H ch k) /\
    (forall Δ D ρ h, ⟦ Δ ⊢ D ⟧ᶜ Θ ⍮ Ξ ⍮ ρ ↘ h -> forall ch k, dmod_member_kind Θ Ξ h ch k <-> gunit_member_kind Θ Ξ ρ (gunit_expand (gu_mk Δ D)) ch k) /\
    (forall h n r, $ᵐ| h & n | Θ ⍮ Ξ ↘ r -> forall ch k, dmod_member_kind Θ Ξ r ch k <-> dmod_member_kind Θ Ξ h ch k) /\
    (forall h ns r, $ᵐ*| h & ns | Θ ⍮ Ξ ↘ r -> forall ch k, dmod_member_kind Θ Ξ r ch k <-> dmod_member_kind Θ Ξ h ch k) /\
    (forall h x r, h ·ₜ x Θ ⍮ Ξ ↘ r -> True) /\
    (forall h y r, h ·ₘ y Θ ⍮ Ξ ↘ r -> forall ch k, dmod_member_kind Θ Ξ r ch k <-> dmod_member_kind Θ Ξ (dm_member h (y :: nil)) ch k) /\
    (forall h ch r, h ·ₜ* ch Θ ⍮ Ξ ↘ r -> True) /\
    (forall h ch0 r, h ·ₘ* ch0 Θ ⍮ Ξ ↘ r -> forall ch k, dmod_member_kind Θ Ξ r ch k <-> dmod_member_kind Θ Ξ (dm_member h ch0) ch k) /\
    (forall ρ Φ ρ', ⟦ Φ ⟧ᵇ Θ ⍮ Ξ ⍮ ρ ↘ ρ' -> forall σ, kinds_shape ρ σ -> exists σ', kinds_env Θ Ξ σ Φ σ' /\ kinds_shape ρ' σ') /\
    (forall ρ p ups Bks j m r, ⟦elim m at j of p ups Bks ⟧ Θ ⍮ Ξ ⍮ ρ ↘ r -> True) /\
    (forall ρ p ups Bks gs vs r, ⟦ihs vs of gs under p ups Bks ⟧ Θ ⍮ Ξ ⍮ ρ ↘ r -> True) /\
    (forall ρ p ups Bks t v r, ⟦ih v of t under p ups Bks ⟧ Θ ⍮ Ξ ⍮ ρ ↘ r -> True) /\
    (forall p ups j r, ⟦mref j of p ups ⟧ Θ ⍮ Ξ ↘ r -> True) /\
    (forall p ups ρ t a, ⟦shty t of p ups ⟧ Θ ⍮ Ξ ⍮ ρ ↘ a -> True) /\
    (forall p ups ρm As ρ t v a, ⟦ihty v of t at p ups under ρm As ⟧ Θ ⍮ Ξ ⍮ ρ ↘ a -> True) /\
    (forall f m r, eval_proj Θ Ξ f m r -> True) /\
    (forall m ρ fl rs, eval_neutral_fields Θ Ξ m ρ fl rs -> True) /\
    (forall fl m ws, ⟦projs fl of m ⟧ Θ ⍮ Ξ ↘ ws -> True) /\
    (forall p ups ρm As ρ gs vs rs, ⟦ihtys vs of gs at p ups under ρm As ⟧ Θ ⍮ Ξ ⍮ ρ ↘ rs -> True) /\
    (forall rs Ms ρ vs, eval_field_vals Θ Ξ rs Ms ρ vs -> True) /\
    (forall f Ms ρ r, eval_app_args Θ Ξ f Ms ρ r -> True).
  Proof.
    apply eval_mut_ind; intros; try exact I.
    (* module expressions *)
    - rewrite modexp_member_kind_var_iff; reflexivity.
    - split; intros Hk; [ constructor; exact Hk | inversion Hk; subst; assumption ].
    - rewrite H, modexp_member_kind_lit_iff; reflexivity.
    - rewrite H1, dmod_member_kind_member_iff, modexp_member_kind_mem_iff; cbn [app]; rewrite (H0 (y :: ch)), (H0 (y :: nil)); reflexivity.
    - rewrite modexp_member_kind_app_iff; etransitivity; [ apply H2 | apply H0 ].
    (* closures: a body closed expanded *)
    - rewrite dmod_member_kind_local_iff; cbn [gunit_expand]; rewrite gm_expand_idem; reflexivity.
    - apply dmod_member_kind_local_iff.
    - cbn [gunit_expand]; rewrite H, gunit_member_kind_alias_iff; reflexivity.
    (* application: the arguments of a module are no part of its kinds *)
    - split; apply dmod_member_kind_global_args.
    - rewrite H, dmod_member_kind_local_iff.
      destruct D as [sm Φ | E]; cbn [gunit_expand];
        rewrite (gunit_member_kind_pad _ _ (ρ ↦ n ⦂ entry_type_closure A ρ)), (gunit_member_kind_pad _ _ ρ (Δ ++ ce_ass κ A :: nil)),
          length_app, Nat.add_1_r;
        [ apply (gunit_member_kind_id _ _ (proj1 (kinds_shape_pad_push _ _ _ _)) (gu_mk nil (md_body sm Φ)))
        | apply (gunit_member_kind_id _ _ (proj1 (kinds_shape_pad_push _ _ _ _)) (gu_mk nil (md_alias E))) ].
    - (* a submodule closure: its parent applied, then the chain *)
      rewrite H0, (dmod_member_kind_member_iff Θ Ξ h'), (dmod_member_kind_member_iff Θ Ξ h), (H (ch ++ ch0)), (H ch); reflexivity.
    - reflexivity.
    - rewrite H0, H; reflexivity.
    (* selection of a submodule *)
    - eapply dmod_member_kind_global_snoc; exact e.
    - rewrite H0, H; apply dmod_member_kind_global_alias; assumption.
    - reflexivity.
    - (* a body's submodule *)
      destruct (H _ (kinds_shape_refl ρ)) as (σ & Hk & Hs).
      rewrite H0, (gunit_member_kind_id _ _ (proj1 Hs)), dmod_member_kind_member_local_expand.
      eapply gunit_member_kind_body_mod; [ apply gm_noopen_expand | eassumption.. ].
    - (* a submodule closure: a longer chain *)
      rewrite (dmod_member_kind_member_iff Θ Ξ h), (dmod_member_kind_member_iff Θ Ξ (dm_member h ch)); cbn [app].
      rewrite !dmod_member_kind_member_cons_iff, <- !app_assoc; reflexivity.
    (* chains *)
    - symmetry; apply dmod_member_kind_member_nil.
    - rewrite H0; apply dmod_member_kind_member_snoc; exact H.
    (* bodies *)
    - exists σ; split; [ constructor | assumption ].
    - (* a definition tagged relevant *)
      match goal with Hρ : kinds_shape ρ σ, IHΦ : forall σ0, kinds_shape ρ σ0 -> _ |- _ => destruct (IHΦ _ Hρ) as (σ1 & Hk & Hs) end.
      eexists; split; [ constructor; exact Hk |]; cbn [kinds_body_entry absent_entry].
      apply kinds_shape_cons_term; exact Hs.
    - (* a definition tagged a proposition: its kept proof, a pad *)
      match goal with Hρ : kinds_shape ρ σ, IHΦ : forall σ0, kinds_shape ρ σ0 -> _ |- _ => destruct (IHΦ _ Hρ) as (σ1 & Hk & Hs) end.
      eexists; split; [ constructor; exact Hk |]; cbn [kinds_body_entry absent_entry].
      apply kinds_shape_cons_term; exact Hs.
    - destruct (H _ H1) as (σ1 & Hk & Hs).
      eexists; split; [ constructor; exact Hk |]; cbn [kinds_body_entry].
      apply kinds_shape_cons_mod; [ exact Hs |].
      intros ch k; rewrite H0, dmod_member_kind_clo_iff; apply gunit_member_kind_id, Hs.
    - (* a definition with a module part: no term value, and its unit's
         closure *)
      match goal with Hρ : kinds_shape ρ σ, IHΦ : forall σ0, kinds_shape ρ σ0 -> _,
                      IHc : forall ch k, dmod_member_kind _ _ h ch k <-> _ |- _ =>
        destruct (IHΦ _ Hρ) as (σ1 & Hk & Hs);
        eexists; split; [ constructor; exact Hk |]; cbn [kinds_body_entry];
        apply kinds_shape_cons_dm; [ exact Hs |];
        intros ch k; rewrite IHc, dmod_member_kind_clo_iff; apply gunit_member_kind_id, Hs end.
  Qed.

  Corollary kind_eval_modexp : forall H ρ h, ⟦ H ⟧ᵐ Θ ⍮ Ξ ⍮ ρ ↘ h ->
