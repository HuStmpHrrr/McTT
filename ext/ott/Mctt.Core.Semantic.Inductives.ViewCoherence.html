The Elements of a View, as the Container's

A view of a group D at p is the declared member k' of the container's group Dq at qc at the uniform arguments Is: its type is the container's at ⟦Is⟧ups, so its elements are related by the container's relation there. The group relates them by its own relation at the view, whose telescopes are the group's own, filed free of the group (model P). Coherence: an element of the container's member i' is one of the group's member vm i' the view takes it to, when the two telescopes agree position by position (view_container_to_group). The agreement is stated over the two relations' families, at prefixes the container relates: what the view equations give, read in the PER model once the group is a type of it.

  Lemma cargs_correspond_app : forall vm gs1' gs1 gs2' gs2, cargs_correspond vm gs1' gs1 -> cargs_correspond vm gs2' gs2 ->
      cargs_correspond vm (gs1' ++ gs2') (gs1 ++ gs2).
  Proof.
    induction gs1' as [| [r' A' | t'] gs1' IH]; intros [| [r A | t] gs1] * H1 H2; cbn in *; try contradiction; auto;
      intuition auto.
  Qed.

  Lemma cargs_correspond_len : forall vm gs' gs, cargs_correspond vm gs' gs -> List.length gs' = List.length gs.
  Proof.
    induction gs' as [| [r' A' | t'] gs' IH]; intros [| [r A | t] gs] H; cbn in H |- *; try contradiction; try reflexivity.
    - f_equal; exact (IH _ (proj2 H)).
    - f_equal; exact (IH _ (proj2 H)).
    - f_equal; exact (IH _ (proj2 (proj2 H))).
  Qed.

The member v of the group the container's member i' is taken to: of the same head, as many parameters, its constructors the container's, each argument corresponding.
    Record view_member (i' : nat) (Mi' : imember) (v : nat) (Mv : imember) : Prop := mk_view_member
      {
vm takes i' to v
        view_member_v : List.nth_error vm i' = Some v
        
v is a member of the group
      ; view_member_mem : ind_member D v = Some Mv
        
Both are record views, or neither
      ; view_member_is_rview : im_is_rview Mv = im_is_rview Mi'
        
Not a record view, it has the same head: both are the same declared member
      ; view_member_ref : im_is_rview Mi' = false -> im_ref p v Mv = im_ref qc i' Mi'
        
Not a record view, as many own uniform parameters
      ; view_member_unu : im_is_rview Mi' = false -> im_unu D Mv = im_unu Dq Mi'
        
A record view, a record view of the same record and fields
      ; view_member_rview : forall qr Is fl, im_head Mi' = mh_record qr Is fl -> exists Is', im_head Mv = mh_record qr Is' fl
        
The same constructor names, in order
      ; view_member_names : List.map fst (im_cons Mv) = List.map fst (im_cons Mi')
        
As many non-uniform parameters
      ; view_member_nn : im_nn Mv = im_nn Mi'
        
Each container constructor is one of Mv's, its arguments corresponding
      ; view_member_cons : forall c gs', ind_con Mi' c = Some gs' -> exists gs, ind_con Mv c = Some gs /\ cargs_correspond vm gs' gs
        
And conversely
      ; view_member_cons' : forall c gs, ind_con Mv c = Some gs -> exists gs', ind_con Mi' c = Some gs' /\ cargs_correspond vm gs' gs }.

      rewrite tele_rel_iff, List.length_rev, length_im_nparams, (view_member_nn _ _ _ _ Hv).

    Lemma nonrec_to : forall i' Mi' v Mv c ms ms' gs' gs gs1' gs1 vp vp' vs vs', ind_member Dq i' = Some Mi' ->
        view_member i' Mi' v Mv -> ind_con Mi' c = Some (gs1' ++ gs') -> ind_con Mv c = Some (gs1 ++ gs) ->
        cargs_correspond vm gs1' gs1 -> cargs_correspond vm gs' gs ->
        List.length vp = List.length gs1 -> List.length vp' = List.length gs1 ->
        List.length vs = List.length gs -> List.length vs' = List.length gs ->
        tele_rel (nr' i') (List.rev (im_nparams Mi')) ms ms' ->
        tele_prefix_rel (ar' i' ms c) (List.length (cargs_nonrec gs1')) (nonrec_arg_vals gs1' vp) (nonrec_arg_vals gs1' vp') ->
        tele_rel_from (ar' i' ms c) (List.length (cargs_nonrec gs1')) (nonrec_arg_vals gs1' vp) (nonrec_arg_vals gs' vs) (nonrec_arg_vals gs' vs') ->
        tele_rel_from (ar v ms c) (List.length (cargs_nonrec gs1)) (nonrec_arg_vals gs1 vp) (nonrec_arg_vals gs vs) (nonrec_arg_vals gs vs').
    Proof.
      intros i' Mi' v Mv c ms ms' gs'; induction gs' as [| g' gs' IH];
        intros gs0 gs1' gs1 vp vp' vs vs' HM Hv Hc' Hc Hcorr1 Hcorr Hlp Hlp' Hl Hl' Hms Hpre Hfr.
      - destruct gs0; cbn in Hcorr; try contradiction.
        destruct vs, vs'; cbn in *; try discriminate; exact I.
      - destruct gs0 as [| g gs]; [ destruct g'; contradiction |].
        destruct vs as [| x vs], vs' as [| x' vs']; cbn in Hl, Hl'; try discriminate.
        assert (Hlp1 : List.length (vp ++ x :: nil) = List.length (gs1 ++ g :: nil)) by (rewrite !List.length_app; cbn; lia).
        assert (Hlp1' : List.length (vp' ++ x' :: nil) = List.length (gs1 ++ g :: nil)) by (rewrite !List.length_app; cbn; lia).
        assert (Hc1' : ind_con Mi' c = Some ((gs1' ++ g' :: nil) ++ gs')) by (rewrite <- List.app_assoc; exact Hc').
        assert (Hc1 : ind_con Mv c = Some ((gs1 ++ g :: nil) ++ gs)) by (rewrite <- List.app_assoc; exact Hc).
        assert (Hlc : List.length vp = List.length gs1') by (rewrite Hlp; symmetry; exact (cargs_correspond_len _ _ _ Hcorr1)).
        assert (Hlc' : List.length vp' = List.length gs1') by (rewrite Hlp'; symmetry; exact (cargs_correspond_len _ _ _ Hcorr1)).
        assert (Hk' : List.length (nonrec_arg_vals gs1' vp) = List.length (cargs_nonrec gs1'))
          by exact (length_nonrec_arg_vals _ _ Hlc).
        destruct g' as [rA' A' | t'], g as [rA A | t]; cbn [cargs_correspond] in Hcorr; try contradiction;
          try (match type of Hcorr with _ = _ /\ cargs_correspond _ _ _ => destruct Hcorr as [Hrr Hcorr] end);
          cbn [nonrec_arg_vals tele_rel_from] in Hfr |- *.
        + (* non-recursive in both *)
          destruct Hfr as [Hx Hfr].
          split.
          * exact (proj1 (HA _ _ _ _ _ _ _ _ _ _ _ _ _ _ ms' _ vp' _ _ HM Hv Hc' Hc Hcorr1 Hlp Hlp' Hms Hpre) Hx).
          * pose proof (IH gs (gs1' ++ ca_arg rA' A' :: nil) (gs1 ++ ca_arg rA A :: nil) (vp ++ x :: nil) (vp' ++ x' :: nil) vs vs'
                          HM Hv Hc1' Hc1 (cargs_correspond_app _ _ _ (ca_arg rA' A' :: nil) (ca_arg rA A :: nil) Hcorr1 ltac:(cbn; tauto)) Hcorr Hlp1 Hlp1' ltac:(lia) ltac:(lia) Hms) as IH'.
            rewrite !cargs_nonrec_app, !nonrec_arg_vals_app, !List.length_app in IH' by lia;
            cbn [cargs_nonrec nonrec_arg_vals List.length List.app] in IH'; rewrite ?List.app_nil_r, ?Nat.add_0_r, ?Nat.add_1_r in IH'.
            apply IH'; [| exact Hfr ].
            apply tele_prefix_rel_snoc; [ exact Hpre |].
            rewrite <- Hk' in Hx |- *; exact Hx.
        + (* the view nests through it *)
          destruct Hfr as [Hx Hfr].
          pose proof (IH gs (gs1' ++ ca_arg rA' A' :: nil) (gs1 ++ ca_rec t :: nil) (vp ++ x :: nil) (vp' ++ x' :: nil) vs vs'
                        HM Hv Hc1' Hc1 (cargs_correspond_app _ _ _ (ca_arg rA' A' :: nil) (ca_rec t :: nil) Hcorr1 ltac:(cbn; tauto)) Hcorr Hlp1 Hlp1' ltac:(lia) ltac:(lia) Hms) as IH'.
          rewrite !cargs_nonrec_app, !nonrec_arg_vals_app, !List.length_app in IH' by lia;
            cbn [cargs_nonrec nonrec_arg_vals List.length List.app] in IH'; rewrite ?List.app_nil_r, ?Nat.add_0_r, ?Nat.add_1_r in IH'.
          apply IH'; [| exact Hfr ].
          apply tele_prefix_rel_snoc; [ exact Hpre |].
          rewrite <- Hk' in Hx |- *; exact Hx.
