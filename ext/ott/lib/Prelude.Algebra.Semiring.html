Prelude›Algebra›Semiring

Semirings: IsSemiring A eq plus mul zero one holds when plus and zero form a commutative monoid up to eq, plusIsCommutativeMonoid, and mul and one a monoid, mulIsMonoid, both on the one carrier A; mul distributes over plus on both sides, and zero absorbs on both sides. Each operation's congruence is in its own predicate. The bundle's with-area forwards the laws, under the names plusAssoc … mulRightId, and gives the two monoids, S.additiveMonoid and S.multiplicativeMonoid.

module Prelude›Algebra›Semiring (l : Level) where

import Prelude›Algebra›Monoid l use (IsMonoid; Monoid)
import Prelude›Algebra›CommutativeMonoid l
  use (IsCommutativeMonoid; CommutativeMonoid)

record IsSemiring (A : Type@{l}) (eq : ∀ (x : A) (y : A) → Type@{l})
    (plus : ∀ (x : A) (y : A) → A) (mul : ∀ (x : A) (y : A) → A) (zero : A)
    (one : A) : Type@{l} where
  plusIsCommutativeMonoid : IsCommutativeMonoid A eq plus zero;
  mulIsMonoid : IsMonoid A eq mul one;
  distribLeft : ∀ (x : A) (y : A) (z : A) →
    eq (mul x (plus y z)) (plus (mul x y) (mul x z));
  distribRight : ∀ (x : A) (y : A) (z : A) →
    eq (mul (plus x y) z) (plus (mul x z) (mul y z));
  zeroLeft : ∀ (x : A) → eq (mul zero x) zero;
  zeroRight : ∀ (x : A) → eq (mul x zero) zero
end

record Semiring : Type@{succl l} where
  carrier : Type@{l};
  eq : ∀ (x : carrier) (y : carrier) → Type@{l};
  plus : ∀ (x : carrier) (y : carrier) → carrier;
  mul : ∀ (x : carrier) (y : carrier) → carrier;
  zero : carrier;
  one : carrier;
  isSemiring : IsSemiring carrier eq plus mul zero one
with
  open isSemiring export (distribLeft; distribRight; zeroLeft; zeroRight)
  open isSemiring.plusIsCommutativeMonoid
    export (refl; sym; trans; opCong as plusCong; assoc as plusAssoc;
            leftId as plusLeftId; rightId as plusRightId; comm as plusComm)
  open isSemiring.mulIsMonoid
    export (opCong as mulCong; assoc as mulAssoc; leftId as mulLeftId;
            rightId as mulRightId)
  def additiveMonoid : CommutativeMonoid ≔
    record CommutativeMonoid where
      carrier ≔ carrier; eq ≔ eq; op ≔ plus; e ≔ zero;
      isCommutativeMonoid ≔ isSemiring.plusIsCommutativeMonoid
    end
  end
  def multiplicativeMonoid : Monoid ≔
    record Monoid where
      carrier ≔ carrier; eq ≔ eq; op ≔ mul; e ≔ one;
      isMonoid ≔ isSemiring.mulIsMonoid
    end
  end
end

end
