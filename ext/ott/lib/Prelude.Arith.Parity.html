Parity of natural numbers: Even and Odd computed by recursion, with doubling and the basic laws, among them double n = n + n and the distribution of double over plus.

HasParity n b says that n is even when b is tt, and odd when it is ff. It recurses on n and negates b at each step, so by conversion Even (succ n) is Odd n and Odd (succ n) is Even n. Even and Odd below unfold to it; it stays private.

private def HasParity : ∀ (n : ℕ) (b : Bool) → Type@0 ≔
  λ (n : ℕ) →
    elim n at y -> (∀ (b : Bool) → Type@0)
    | zero ⇒ λ (b : Bool) → IsTrue b
    | succ p, r ⇒ λ (b : Bool) → r (not b)
    end
end

def Even (n : ℕ) : Type@0 ≔ HasParity n tt end

def Odd (n : ℕ) : Type@0 ≔ HasParity n ff end

def double (n : ℕ) : ℕ ≔
  elim n at y -> ℕ
  | zero ⇒ 0
  | succ p, r ⇒ succ (succ r)
  end
end

Parity of doubles, successors and sums.

module Laws where

theorem evenDouble : ∀ (n : ℕ) → Even (double n) ≔
  λ (n : ℕ) →
    elim n at x -> Even (double x)
    | zero ⇒ unit
    | succ p, ih ⇒ ih
    end
end

theorem evenSuccOdd : ∀ (n : ℕ) (h : Even n) → Odd (succ n) ≔
  λ (n : ℕ) (h : Even n) → h
end

theorem oddSuccEven : ∀ (n : ℕ) (h : Odd n) → Even (succ n) ≔
  λ (n : ℕ) (h : Odd n) → h
end

Every number is even or odd: the step swaps the two sides.

def evenOrOdd : ∀ (n : ℕ) → Or (Even n) (Odd n) ≔
  λ (n : ℕ) →
    elim n at x -> Or (Even x) (Odd x)
    | zero ⇒ orInl (Even 0) (Odd 0) unit
    | succ p, ih ⇒ orSwap (Even p) (Odd p) ih
    end
end

No number is both even and odd. And lives in Type@1, so the negation is the one at level 1l.

theorem notEvenAndOdd : ∀ (n : ℕ) → Neg.Not 1l (And (Even n) (Odd n)) ≔
  λ (n : ℕ) →
    elim n at x -> (∀ (h : And (Even x) (Odd x)) → Void)
    | zero ⇒ λ (h : And Unit Void) → h Void (λ (a : Unit) (b : Void) → b)
    | succ p, ih ⇒ λ (h : And (Odd p) (Even p)) →
        ih (andSwap (Odd p) (Even p) h)
    end
end

The sum of two even numbers is even. The induction on a carries both "even a gives even plus a b" and "odd a gives odd plus a b", which the step swaps.

theorem evenPlus : ∀ (a : ℕ) (b : ℕ) (ha : Even a) (hb : Even b) →
    Even (plus a b) ≔
  λ (a : ℕ) (b : ℕ) (ha : Even a) (hb : Even b) →
    let both : And (∀ (h : Even a) → Even (plus a b))
                   (∀ (h : Odd a) → Odd (plus a b)) ≔
          elim a at x -> And (∀ (h : Even x) → Even (plus x b))
                                (∀ (h : Odd x) → Odd (plus x b))
          | zero ⇒ andIntro (∀ (h : Unit) → Even b) (∀ (h : Void) → Odd b)
                      (λ (h : Unit) → hb)
                      (λ (h : Void) → absurd (Odd b) h)
          | succ p, ih ⇒ andSwap (∀ (h : Even p) → Even (plus p b))
                                 (∀ (h : Odd p) → Odd (plus p b)) ih
          end
    in both (Even (plus a b)) (λ (f : ∀ (h : Even a) → Even (plus a b))
                                 (g : ∀ (h : Odd a) → Odd (plus a b)) → f ha)
    end
end

double n is n + n.

theorem doublePlus : ∀ (n : ℕ) → Eq (double n) (plus n n) ≔
  λ (n : ℕ) →
    elim n at x -> Eq (double x) (plus x x)
    | zero ⇒ unit
    | succ p, ih ⇒
        trans (succ (double p)) (succ (plus p p)) (plus p (succ p)) ih
          (sym (plus p (succ p)) (succ (plus p p)) (plusSucc p p))
    end
end

Doubling distributes over addition.

theorem doubleDistrib : ∀ (a : ℕ) (b : ℕ) →
    Eq (double (plus a b)) (plus (double a) (double b)) ≔
  λ (a : ℕ) (b : ℕ) →
    elim a at x -> Eq (double (plus x b)) (plus (double x) (double b))
    | zero ⇒ refl (double b)
    | succ p, ih ⇒ ih
    end
end
