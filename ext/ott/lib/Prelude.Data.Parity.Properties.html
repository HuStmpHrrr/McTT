Prelude›Data›Parity›Properties

Laws of Prelude›Data›Parity, stated with the Leibniz equality of Prelude›Logic›Leibniz at Parity: inv is an injective involution without fixed point; add is a commutative group with unit even in which every parity is its own inverse; mul is commutative, associative and idempotent, with unit odd and zero even, and distributes over add; toSign is an isomorphism taking add to the product of signs; and parity takes succ, plus and mult to inv, add and mul.

The laws of the operations are proved by cases. Two equations on inv are the steps of the homomorphisms: addInvLeft for plus and mulInvLeft for mult.

The proposition that a parity is even.

private def IsEven (p : Parity) : Type@0 ≔
  elim p at z -> Type@0 | even ⇒ Unit | odd ⇒ Void end
end

inv.

module Inv where

theorem invInv : ∀ (p : Parity) → Id Parity (inv (inv p)) p ≔
  λ (p : Parity) →
    elim p at z -> Id Parity (inv (inv z)) z
    | even ⇒ idRefl Parity even
    | odd ⇒ idRefl Parity odd
    end
end

theorem invInjective : ∀ (p : Parity) (q : Parity)
    (h : Id Parity (inv p) (inv q)) → Id Parity p q ≔
  λ (p : Parity) (q : Parity) (h : Id Parity (inv p) (inv q)) →
    idTrans Parity p (inv (inv q)) q
      (idTrans Parity p (inv (inv p)) (inv (inv q))
         (idSym Parity (inv (inv p)) p (invInv p))
         (idCong Parity Parity inv (inv p) (inv q) h))
      (invInv q)
end

No parity is its own opposite.

theorem parityNeInv : ∀ (p : Parity) → Not (Id Parity p (inv p)) ≔
  λ (p : Parity) →
    elim p at z -> Not (Id Parity z (inv z))
    | even ⇒ λ (h : Id Parity even odd) → h IsEven unit
    | odd ⇒ λ (h : Id Parity odd even) →
        h (λ (z : Parity) → IsEven (inv z)) unit
    end
end

end

add.

module Add where

open Inv use (invInv)

theorem addIdentityLeft : ∀ (p : Parity) → Id Parity (add even p) p ≔
  λ (p : Parity) → idRefl Parity p
end

theorem addIdentityRight : ∀ (p : Parity) → Id Parity (add p even) p ≔
  λ (p : Parity) →
    elim p at z -> Id Parity (add z even) z
    | even ⇒ idRefl Parity even
    | odd ⇒ idRefl Parity odd
    end
end

Every parity is its own inverse: add p p is even.

theorem addSelf : ∀ (p : Parity) → Id Parity (add p p) even ≔
  λ (p : Parity) →
    elim p at z -> Id Parity (add z z) even
    | even ⇒ idRefl Parity even
    | odd ⇒ idRefl Parity even
    end
end

A parity plus its opposite is odd.

theorem addInv : ∀ (p : Parity) → Id Parity (add p (inv p)) odd ≔
  λ (p : Parity) →
    elim p at z -> Id Parity (add z (inv z)) odd
    | even ⇒ idRefl Parity odd
    | odd ⇒ idRefl Parity odd
    end
end

inv on the left of a sum moves out of it.

theorem addInvLeft : ∀ (p : Parity) (q : Parity) →
    Id Parity (add (inv p) q) (inv (add p q)) ≔
  λ (p : Parity) (q : Parity) →
    elim p at z -> Id Parity (add (inv z) q) (inv (add z q))
    | even ⇒ idRefl Parity (inv q)
    | odd ⇒ idSym Parity (inv (inv q)) q (invInv q)
    end
end

theorem addComm : ∀ (p : Parity) (q : Parity) → Id Parity (add p q) (add q p) ≔
  λ (p : Parity) (q : Parity) →
    elim p at x -> Id Parity (add x q) (add q x)
    | even ⇒ idSym Parity (add q even) q (addIdentityRight q)
    | odd ⇒ elim q at y -> Id Parity (inv y) (add y odd)
            | even ⇒ idRefl Parity odd
            | odd ⇒ idRefl Parity even
            end
    end
end

theorem addAssoc : ∀ (p : Parity) (q : Parity) (r : Parity) →
    Id Parity (add (add p q) r) (add p (add q r)) ≔
  λ (p : Parity) (q : Parity) (r : Parity) →
    elim p at x -> Id Parity (add (add x q) r) (add x (add q r))
    | even ⇒ idRefl Parity (add q r)
    | odd ⇒ addInvLeft q r
    end
end

Adding the same parity on the left is injective.

theorem addCancelLeft : ∀ (p : Parity) (q : Parity) (r : Parity)
    (h : Id Parity (add p q) (add p r)) → Id Parity q r ≔
  λ (p : Parity) →
    elim p at x -> (∀ (q : Parity) (r : Parity)
                         (h : Id Parity (add x q) (add x r)) → Id Parity q r)
    | even ⇒ λ (q : Parity) (r : Parity) (h : Id Parity q r) → h
    | odd ⇒ λ (q : Parity) (r : Parity) (h : Id Parity (inv q) (inv r)) →
        Inv.invInjective q r h
    end
end

end

mul.

module Mul where

open Add use (addIdentityRight; addSelf)

theorem mulZeroLeft : ∀ (p : Parity) → Id Parity (mul even p) even ≔
  λ (p : Parity) → idRefl Parity even
end

theorem mulZeroRight : ∀ (p : Parity) → Id Parity (mul p even) even ≔
  λ (p : Parity) →
    elim p at z -> Id Parity (mul z even) even
    | even ⇒ idRefl Parity even
    | odd ⇒ idRefl Parity even
    end
end

theorem mulIdentityLeft : ∀ (p : Parity) → Id Parity (mul odd p) p ≔
  λ (p : Parity) → idRefl Parity p
end

theorem mulIdentityRight : ∀ (p : Parity) → Id Parity (mul p odd) p ≔
  λ (p : Parity) →
    elim p at z -> Id Parity (mul z odd) z
    | even ⇒ idRefl Parity even
    | odd ⇒ idRefl Parity odd
    end
end

theorem mulIdem : ∀ (p : Parity) → Id Parity (mul p p) p ≔
  λ (p : Parity) →
    elim p at z -> Id Parity (mul z z) z
    | even ⇒ idRefl Parity even
    | odd ⇒ idRefl Parity odd
    end
end

theorem mulComm : ∀ (p : Parity) (q : Parity) → Id Parity (mul p q) (mul q p) ≔
  λ (p : Parity) (q : Parity) →
    elim p at x -> Id Parity (mul x q) (mul q x)
    | even ⇒ idSym Parity (mul q even) even (mulZeroRight q)
    | odd ⇒ idSym Parity (mul q odd) q (mulIdentityRight q)
    end
end

theorem mulAssoc : ∀ (p : Parity) (q : Parity) (r : Parity) →
    Id Parity (mul (mul p q) r) (mul p (mul q r)) ≔
  λ (p : Parity) (q : Parity) (r : Parity) →
    elim p at x -> Id Parity (mul (mul x q) r) (mul x (mul q r))
    | even ⇒ idRefl Parity even
    | odd ⇒ idRefl Parity (mul q r)
    end
end

Multiplication distributes over addition on the left.

theorem mulDistribLeft : ∀ (p : Parity) (q : Parity) (r : Parity) →
    Id Parity (mul p (add q r)) (add (mul p q) (mul p r)) ≔
  λ (p : Parity) (q : Parity) (r : Parity) →
    elim p at x -> Id Parity (mul x (add q r)) (add (mul x q) (mul x r))
    | even ⇒ idRefl Parity even
    | odd ⇒ idRefl Parity (add q r)
    end
end

inv on the left of a product: mul (inv p) q is add q (mul p q).

theorem mulInvLeft : ∀ (p : Parity) (q : Parity) →
    Id Parity (mul (inv p) q) (add q (mul p q)) ≔
  λ (p : Parity) (q : Parity) →
    elim p at x -> Id Parity (mul (inv x) q) (add q (mul x q))
    | even ⇒ idSym Parity (add q even) q (addIdentityRight q)
    | odd ⇒ idSym Parity (add q q) even (addSelf q)
    end
end

Multiplication distributes over addition on the right.

theorem mulDistribRight : ∀ (p : Parity) (q : Parity) (r : Parity) →
    Id Parity (mul (add p q) r) (add (mul p r) (mul q r)) ≔
  λ (p : Parity) (q : Parity) (r : Parity) →
    elim p at x -> Id Parity (mul (add x q) r) (add (mul x r) (mul q r))
    | even ⇒ idRefl Parity (mul q r)
    | odd ⇒ mulInvLeft q r
    end
end

end

toSign and fromSign.

module Signs where

theorem toSignFromSign : ∀ (s : Sign) → Id Sign (toSign (fromSign s)) s ≔
  λ (s : Sign) →
    elim s at z -> Id Sign (toSign (fromSign z)) z
    | negative ⇒ idRefl Sign negative
    | positive ⇒ idRefl Sign positive
    end
end

theorem fromSignToSign : ∀ (p : Parity) → Id Parity (fromSign (toSign p)) p ≔
  λ (p : Parity) →
    elim p at z -> Id Parity (fromSign (toSign z)) z
    | even ⇒ idRefl Parity even
    | odd ⇒ idRefl Parity odd
    end
end

toSign takes the opposite parity to the opposite sign.

theorem invHomoOpposite : ∀ (p : Parity) →
    Id Sign (toSign (inv p)) (opposite (toSign p)) ≔
  λ (p : Parity) →
    elim p at z -> Id Sign (toSign (inv z)) (opposite (toSign z))
    | even ⇒ idRefl Sign negative
    | odd ⇒ idRefl Sign positive
    end
end

toSign takes the sum of parities to the product of signs.

theorem addHomoMul : ∀ (p : Parity) (q : Parity) →
    Id Sign (toSign (add p q)) (S.mul (toSign p) (toSign q)) ≔
  λ (p : Parity) (q : Parity) →
    elim p at x -> Id Sign (toSign (add x q)) (S.mul (toSign x) (toSign q))
    | even ⇒ idRefl Sign (toSign q)
    | odd ⇒ invHomoOpposite q
    end
end

end

parity is a homomorphism from the numbers.

module Homo where

open Inv use (invInv)

The opposite of the parity of succ n is the parity of n.

theorem succHomoInv : ∀ (n : ℕ) → Id Parity (inv (parity (succ n))) (parity n) ≔
  λ (n : ℕ) → invInv (parity n)
end

parity takes plus to add.

theorem plusHomoAdd : ∀ (m : ℕ) (n : ℕ) →
    Id Parity (parity (plus m n)) (add (parity m) (parity n)) ≔
  λ (m : ℕ) (n : ℕ) →
    elim m at x -> Id Parity (parity (plus x n)) (add (parity x) (parity n))
    | zero ⇒ idRefl Parity (parity n)
    | succ p, ih ⇒
        idTrans Parity (inv (parity (plus p n)))
          (inv (add (parity p) (parity n)))
          (add (inv (parity p)) (parity n))
          (idCong Parity Parity inv (parity (plus p n))
             (add (parity p) (parity n)) ih)
          (idSym Parity (add (inv (parity p)) (parity n))
             (inv (add (parity p) (parity n)))
             (Add.addInvLeft (parity p) (parity n)))
    end
end

parity takes mult to mul. In the step, mult (succ p) n is plus n (mult p n).

theorem multHomoMul : ∀ (m : ℕ) (n : ℕ) →
    Id Parity (parity (mult m n)) (mul (parity m) (parity n)) ≔
  λ (m : ℕ) (n : ℕ) →
    elim m at x -> Id Parity (parity (mult x n)) (mul (parity x) (parity n))
    | zero ⇒ idRefl Parity even
    | succ p, ih ⇒
        idTrans Parity (parity (plus n (mult p n)))
          (add (parity n) (mul (parity p) (parity n)))
          (mul (inv (parity p)) (parity n))
          (idTrans Parity (parity (plus n (mult p n)))
             (add (parity n) (parity (mult p n)))
             (add (parity n) (mul (parity p) (parity n)))
             (plusHomoAdd n (mult p n))
             (idCong Parity Parity (λ (q : Parity) → add (parity n) q)
                (parity (mult p n)) (mul (parity p) (parity n)) ih))
          (idSym Parity (mul (inv (parity p)) (parity n))
             (add (parity n) (mul (parity p) (parity n)))
             (Mul.mulInvLeft (parity p) (parity n)))
    end
end

end
