Prelude›Arith›Lattice

The natural numbers as a distributive lattice under Prelude›Arith›MinMax.min and Prelude›Arith›MinMax.max: associativity, absorption and distributivity, and the order-theoretic characterization of min and max as greatest lower and least upper bounds.

module Prelude›Arith›Lattice where

import Prelude›Data›Unit use (Unit; unit)
import Prelude›Data›Void use (Void)
import Prelude›Builtin›Nat use (succ)
import Prelude›Logic use (absurd)
import Prelude›Arith›Equality use (Eq; refl; sym; trans)
import Prelude›Arith›MinMax use (min; max)
open Prelude›Arith›MinMax.Laws use (minComm; minIdem)
import Prelude›Arith›Order use (Le)
open Prelude›Arith›Order.LeLaws use (leRefl; leTrans)

Associativity, absorption and distributivity.

module Algebra where

theorem minAssoc : ∀ (a : ℕ) (b : ℕ) (c : ℕ) →
    Eq (min (min a b) c) (min a (min b c)) ≔
  λ (a : ℕ) →
    elim a at x -> (∀ (b : ℕ) (c : ℕ) →
                         Eq (min (min x b) c) (min x (min b c)))
    | zero ⇒ λ (b : ℕ) (c : ℕ) → unit
    | succ p, ih ⇒ λ (b : ℕ) →
        elim b at y -> (∀ (c : ℕ) →
                             Eq (min (min (succ p) y) c)
                                (min (succ p) (min y c)))
        | zero ⇒ λ (c : ℕ) → unit
        | succ q, s ⇒ λ (c : ℕ) →
            elim c at z -> Eq (min (min (succ p) (succ q)) z)
                                 (min (succ p) (min (succ q) z))
            | zero ⇒ unit
            | succ r, t ⇒ ih q r
            end
        end
    end
end

theorem maxAssoc : ∀ (a : ℕ) (b : ℕ) (c : ℕ) →
    Eq (max (max a b) c) (max a (max b c)) ≔
  λ (a : ℕ) →
    elim a at x -> (∀ (b : ℕ) (c : ℕ) →
                         Eq (max (max x b) c) (max x (max b c)))
    | zero ⇒ λ (b : ℕ) (c : ℕ) → refl (max b c)
    | succ p, ih ⇒ λ (b : ℕ) →
        elim b at y -> (∀ (c : ℕ) →
                             Eq (max (max (succ p) y) c)
                                (max (succ p) (max y c)))
        | zero ⇒ λ (c : ℕ) → refl (max (succ p) c)
        | succ q, s ⇒ λ (c : ℕ) →
            elim c at z -> Eq (max (max (succ p) (succ q)) z)
                                 (max (succ p) (max (succ q) z))
            | zero ⇒ refl (max p q)
            | succ r, t ⇒ ih q r
            end
        end
    end
end

Absorption: the minimum of a and something above a is a.

theorem minMaxAbsorb : ∀ (a : ℕ) (b : ℕ) → Eq (min a (max a b)) a ≔
  λ (a : ℕ) →
    elim a at x -> (∀ (b : ℕ) → Eq (min x (max x b)) x)
    | zero ⇒ λ (b : ℕ) → unit
    | succ p, ih ⇒ λ (b : ℕ) →
        elim b at y -> Eq (min (succ p) (max (succ p) y)) (succ p)
        | zero ⇒ minIdem p
        | succ q, s ⇒ ih q
        end
    end
end

Absorption, dually: the maximum of a and something below a is a.

theorem maxMinAbsorb : ∀ (a : ℕ) (b : ℕ) → Eq (max a (min a b)) a ≔
  λ (a : ℕ) →
    elim a at x -> (∀ (b : ℕ) → Eq (max x (min x b)) x)
    | zero ⇒ λ (b : ℕ) → unit
    | succ p, ih ⇒ λ (b : ℕ) →
        elim b at y -> Eq (max (succ p) (min (succ p) y)) (succ p)
        | zero ⇒ refl p
        | succ q, s ⇒ ih q
        end
    end
end

The minimum distributes over the maximum.

theorem minMaxDistrib : ∀ (a : ℕ) (b : ℕ) (c : ℕ) →
    Eq (min a (max b c)) (max (min a b) (min a c)) ≔
  λ (a : ℕ) →
    elim a at x -> (∀ (b : ℕ) (c : ℕ) →
                         Eq (min x (max b c)) (max (min x b) (min x c)))
    | zero ⇒ λ (b : ℕ) (c : ℕ) → unit
    | succ p, ih ⇒ λ (b : ℕ) →
        elim b at y -> (∀ (c : ℕ) →
                             Eq (min (succ p) (max y c))
                                (max (min (succ p) y) (min (succ p) c)))
        | zero ⇒ λ (c : ℕ) → refl (min (succ p) c)
        | succ q, s ⇒ λ (c : ℕ) →
            elim c at z -> Eq (min (succ p) (max (succ q) z))
                                 (max (min (succ p) (succ q)) (min (succ p) z))
            | zero ⇒ refl (min p q)
            | succ r, t ⇒ ih q r
            end
        end
    end
end

The maximum distributes over the minimum. Two of the cases are instances of absorption, the second one up to minComm.

theorem maxMinDistrib : ∀ (a : ℕ) (b : ℕ) (c : ℕ) →
    Eq (max a (min b c)) (min (max a b) (max a c)) ≔
  λ (a : ℕ) →
    elim a at x -> (∀ (b : ℕ) (c : ℕ) →
                         Eq (max x (min b c)) (min (max x b) (max x c)))
    | zero ⇒ λ (b : ℕ) (c : ℕ) → refl (min b c)
    | succ p, ih ⇒ λ (b : ℕ) →
        elim b at y -> (∀ (c : ℕ) →
                             Eq (max (succ p) (min y c))
                                (min (max (succ p) y) (max (succ p) c)))
        | zero ⇒ λ (c : ℕ) →
            sym (min (succ p) (max (succ p) c)) (succ p)
              (minMaxAbsorb (succ p) c)
        | succ q, s ⇒ λ (c : ℕ) →
            elim c at z -> Eq (max (succ p) (min (succ q) z))
                                 (min (max (succ p) (succ q)) (max (succ p) z))
            | zero ⇒
                (* The goal computes to [Eq p (min (max p q) p)]. *)
                sym (min (max p q) p) p
                  (trans (min (max p q) p) (min p (max p q)) p
                     (minComm (max p q) p) (minMaxAbsorb p q))
            | succ r, t ⇒ ih q r
            end
        end
    end
end

end

min is the greatest lower bound and max the least upper bound.

module Bounds where

The minimum is a lower bound of its first argument.

theorem minLe : ∀ (a : ℕ) (b : ℕ) → Le (min a b) a ≔
  λ (a : ℕ) →
    elim a at x -> (∀ (b : ℕ) → Le (min x b) x)
    | zero ⇒ λ (b : ℕ) → unit
    | succ p, ih ⇒ λ (b : ℕ) →
        elim b at y -> Le (min (succ p) y) (succ p)
        | zero ⇒ unit
        | succ q, s ⇒ ih q
        end
    end
end

The minimum is a lower bound of its second argument.

theorem minLeRight : ∀ (a : ℕ) (b : ℕ) → Le (min a b) b ≔
  λ (a : ℕ) →
    elim a at x -> (∀ (b : ℕ) → Le (min x b) b)
    | zero ⇒ λ (b : ℕ) → unit
    | succ p, ih ⇒ λ (b : ℕ) →
        elim b at y -> Le (min (succ p) y) y
        | zero ⇒ unit
        | succ q, s ⇒ ih q
        end
    end
end

The maximum is an upper bound of its first argument.

theorem leMax : ∀ (a : ℕ) (b : ℕ) → Le a (max a b) ≔
  λ (a : ℕ) →
    elim a at x -> (∀ (b : ℕ) → Le x (max x b))
    | zero ⇒ λ (b : ℕ) → unit
    | succ p, ih ⇒ λ (b : ℕ) →
        elim b at y -> Le (succ p) (max (succ p) y)
        | zero ⇒ leRefl p
        | succ q, s ⇒ ih q
        end
    end
end

The maximum is an upper bound of its second argument.

theorem leMaxRight : ∀ (a : ℕ) (b : ℕ) → Le b (max a b) ≔
  λ (a : ℕ) →
    elim a at x -> (∀ (b : ℕ) → Le b (max x b))
    | zero ⇒ λ (b : ℕ) → leRefl b
    | succ p, ih ⇒ λ (b : ℕ) →
        elim b at y -> Le y (max (succ p) y)
        | zero ⇒ unit
        | succ q, s ⇒ ih q
        end
    end
end

theorem minLeMax : ∀ (a : ℕ) (b : ℕ) → Le (min a b) (max a b) ≔
  λ (a : ℕ) (b : ℕ) → leTrans (min a b) a (max a b) (minLe a b) (leMax a b)
end

The minimum is the greatest lower bound: a common lower bound is below it.

theorem leMin : ∀ (c : ℕ) (a : ℕ) (b : ℕ) (hca : Le c a) (hcb : Le c b) →
    Le c (min a b) ≔
  λ (c : ℕ) →
    elim c at w -> (∀ (a : ℕ) (b : ℕ) (hca : Le w a) (hcb : Le w b) →
                         Le w (min a b))
    | zero ⇒ λ (a : ℕ) (b : ℕ) (hca : Unit) (hcb : Unit) → unit
    | succ p, ih ⇒ λ (a : ℕ) →
        elim a at x -> (∀ (b : ℕ) (hca : Le (succ p) x)
                             (hcb : Le (succ p) b) →
                             Le (succ p) (min x b))
        | zero ⇒ λ (b : ℕ) (hca : Void) (hcb : Le (succ p) b) →
            absurd (Le (succ p) 0) hca
        | succ q, s ⇒ λ (b : ℕ) →
            elim b at y -> (∀ (hca : Le p q) (hcb : Le (succ p) y) →
                                 Le (succ p) (min (succ q) y))
            | zero ⇒ λ (hca : Le p q) (hcb : Void) → absurd (Le (succ p) 0) hcb
            | succ r, t ⇒ λ (hca : Le p q) (hcb : Le p r) → ih q r hca hcb
            end
        end
    end
end

The converse of leMin, for the first argument.

theorem leMinLeft : ∀ (c : ℕ) (a : ℕ) (b : ℕ) (h : Le c (min a b)) → Le c a ≔
  λ (c : ℕ) (a : ℕ) (b : ℕ) (h : Le c (min a b)) →
    leTrans c (min a b) a h (minLe a b)
end

The converse of leMin, for the second argument.

theorem leMinRight : ∀ (c : ℕ) (a : ℕ) (b : ℕ) (h : Le c (min a b)) → Le c b ≔
  λ (c : ℕ) (a : ℕ) (b : ℕ) (h : Le c (min a b)) →
    leTrans c (min a b) b h (minLeRight a b)
end

The maximum is the least upper bound: a common upper bound is above it.

theorem maxLe : ∀ (a : ℕ) (b : ℕ) (c : ℕ) (hac : Le a c) (hbc : Le b c) →
    Le (max a b) c ≔
  λ (a : ℕ) →
    elim a at x -> (∀ (b : ℕ) (c : ℕ) (hac : Le x c) (hbc : Le b c) →
                         Le (max x b) c)
    | zero ⇒ λ (b : ℕ) (c : ℕ) (hac : Unit) (hbc : Le b c) → hbc
    | succ p, ih ⇒ λ (b : ℕ) →
        elim b at y -> (∀ (c : ℕ) (hac : Le (succ p) c) (hbc : Le y c) →
                             Le (max (succ p) y) c)
        | zero ⇒ λ (c : ℕ) (hac : Le (succ p) c) (hbc : Unit) → hac
        | succ q, s ⇒ λ (c : ℕ) →
            elim c at w -> (∀ (hac : Le (succ p) w) (hbc : Le (succ q) w) →
                                 Le (max (succ p) (succ q)) w)
            | zero ⇒ λ (hac : Void) (hbc : Void) → absurd (Le (succ (max p q)) 0) hac
            | succ r, t ⇒ λ (hac : Le p r) (hbc : Le q r) → ih q r hac hbc
            end
        end
    end
end

The converse of maxLe, for the first argument.

theorem maxLeLeft : ∀ (a : ℕ) (b : ℕ) (c : ℕ) (h : Le (max a b) c) → Le a c ≔
  λ (a : ℕ) (b : ℕ) (c : ℕ) (h : Le (max a b) c) →
    leTrans a (max a b) c (leMax a b) h
end

The converse of maxLe, for the second argument.

theorem maxLeRight : ∀ (a : ℕ) (b : ℕ) (c : ℕ) (h : Le (max a b) c) → Le b c ≔
  λ (a : ℕ) (b : ℕ) (c : ℕ) (h : Le (max a b) c) →
    leTrans b (max a b) c (leMaxRight a b) h
end

Below its second argument, the minimum is the first.

theorem minEqLeft : ∀ (a : ℕ) (b : ℕ) (h : Le a b) → Eq (min a b) a ≔
  λ (a : ℕ) →
    elim a at x -> (∀ (b : ℕ) (h : Le x b) → Eq (min x b) x)
    | zero ⇒ λ (b : ℕ) (h : Unit) → unit
    | succ p, ih ⇒ λ (b : ℕ) →
        elim b at y -> (∀ (h : Le (succ p) y) → Eq (min (succ p) y) (succ p))
        | zero ⇒ λ (h : Void) → absurd (Eq 0 (succ p)) h
        | succ q, s ⇒ λ (h : Le p q) → ih q h
        end
    end
end

Below its second argument, the maximum is the second.

theorem maxEqRight : ∀ (a : ℕ) (b : ℕ) (h : Le a b) → Eq (max a b) b ≔
  λ (a : ℕ) →
    elim a at x -> (∀ (b : ℕ) (h : Le x b) → Eq (max x b) b)
    | zero ⇒ λ (b : ℕ) (h : Unit) → refl b
    | succ p, ih ⇒ λ (b : ℕ) →
        elim b at y -> (∀ (h : Le (succ p) y) → Eq (max (succ p) y) y)
        | zero ⇒ λ (h : Void) → absurd (Eq (succ p) 0) h
        | succ q, s ⇒ λ (h : Le p q) → ih q h
        end
    end
end

end

open Bounds use (minLe; minLeRight; leMax; leMaxRight; leMin; maxLe)

min and max are monotone.

module Mono where

The minimum is monotone in both arguments.

theorem minMono : ∀ (a : ℕ) (b : ℕ) (c : ℕ) (d : ℕ)
    (hab : Le a b) (hcd : Le c d) → Le (min a c) (min b d) ≔
  λ (a : ℕ) (b : ℕ) (c : ℕ) (d : ℕ) (hab : Le a b) (hcd : Le c d) →
    leMin (min a c) b d
      (leTrans (min a c) a b (minLe a c) hab)
      (leTrans (min a c) c d (minLeRight a c) hcd)
end

The maximum is monotone in both arguments.

theorem maxMono : ∀ (a : ℕ) (b : ℕ) (c : ℕ) (d : ℕ)
    (hab : Le a b) (hcd : Le c d) → Le (max a c) (max b d) ≔
  λ (a : ℕ) (b : ℕ) (c : ℕ) (d : ℕ) (hab : Le a b) (hcd : Le c d) →
    maxLe a c (max b d)
      (leTrans a b (max b d) hab (leMax b d))
      (leTrans c d (max b d) hcd (leMaxRight b d))
end

end
