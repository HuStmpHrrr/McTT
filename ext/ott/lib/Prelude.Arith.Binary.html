Prelude›Arith›Binary

Binary representation of numbers. half n and bit n are the quotient and the remainder of n by 2, so n is double (half n) + bit n (Halving.halfSpec). bits n lists the binary digits of n, least significant first, fromBits c k reads k digits back, and bitCount n counts the ones. powFast x n computes pow x n by repeated squaring, in about log n steps.

The recursions on the length of a representation run on fuel. They and the facts about their fuel live in the private module Digits; the public laws read them at enough fuel (Laws).

Half of n, rounded down. The recursion takes a flag b, 0 or 1, and computes half of n + b, so half (succ (succ n)) is succ (half n) by conversion. The flag is tested with pick, whose branches are only evaluated once chosen.

def half (n : ℕ) : ℕ ≔
  (elim n at y -> (∀ (b : ℕ) → ℕ)
   | zero ⇒ λ (b : ℕ) → 0
   | succ p, r ⇒ λ (b : ℕ) →
       pick b (λ (u : ℕ) → r 1) (λ (u : ℕ) → succ (r 0))
   end) 0
end

The low bit of n, 0 or 1. The flag b is flipped at each step, so bit (succ (succ n)) is bit n by conversion.

def bit (n : ℕ) : ℕ ≔
  (elim n at y -> (∀ (b : ℕ) → ℕ)
   | zero ⇒ λ (b : ℕ) → b
   | succ p, r ⇒ λ (b : ℕ) → pick b (λ (u : ℕ) → r 1) (λ (u : ℕ) → r 0)
   end) 0
end

Halving and the low bit.

module Halving where

theorem bitLe : ∀ (n : ℕ) → Le (bit n) 1 ≔
  twoStepInd (λ (n : ℕ) → Le (bit n) 1) unit unit
    (λ (n : ℕ) (h : Le (bit n) 1) → h)
end

Division by 2: n is twice its half, plus its low bit. Both sides of the step are succ (succ ...) of the hypothesis.

theorem halfSpec : ∀ (n : ℕ) → Eq (plus (double (half n)) (bit n)) n ≔
  twoStepInd (λ (n : ℕ) → Eq (plus (double (half n)) (bit n)) n) unit unit
    (λ (n : ℕ) (h : Eq (plus (double (half n)) (bit n)) n) → h)
end

Halving a positive number makes it smaller.

theorem halfLe : ∀ (n : ℕ) → Le (half (succ n)) n ≔
  twoStepInd (λ (n : ℕ) → Le (half (succ n)) n) unit unit
    (λ (n : ℕ) (h : Le (half (succ n)) n) → leStep (half (succ n)) n h)
end

theorem halfDouble : ∀ (n : ℕ) → Eq (half (double n)) n ≔
  λ (n : ℕ) →
    elim n at x -> Eq (half (double x)) x
    | zero ⇒ unit
    | succ p, ih ⇒ ih
    end
end

theorem bitDouble : ∀ (n : ℕ) → Eq (bit (double n)) 0 ≔
  λ (n : ℕ) →
    elim n at x -> Eq (bit (double x)) 0
    | zero ⇒ unit
    | succ p, ih ⇒ ih
    end
end

theorem halfSuccDouble : ∀ (n : ℕ) → Eq (half (succ (double n))) n ≔
  λ (n : ℕ) →
    elim n at x -> Eq (half (succ (double x))) x
    | zero ⇒ unit
    | succ p, ih ⇒ ih
    end
end

theorem bitSuccDouble : ∀ (n : ℕ) → Eq (bit (succ (double n))) 1 ≔
  λ (n : ℕ) →
    elim n at x -> Eq (bit (succ (double x))) 1
    | zero ⇒ unit
    | succ p, ih ⇒ ih
    end
end

halfSpec with the low bit first and the double spelled as a sum.

theorem bitPlusHalves : ∀ (n : ℕ) →
    Eq (plus (bit n) (plus (half n) (half n))) n ≔
  λ (n : ℕ) →
    let b ≔ bit n;
        h ≔ half n
    in trans (plus b (plus h h)) (plus (plus h h) b) n (plusComm b (plus h h))
         (trans (plus (plus h h) b) (plus (double h) b) n
            (cong (λ (y : ℕ) → plus y b) (plus h h) (double h)
               (sym (double h) (plus h h) (doublePlus h)))
            (halfSpec n))
    end
end

end

The recursions behind bits, fromBits and powFast, and the facts about their fuel.

private module Digits where

Digit i of n: the low bit after i halvings.

def digitAt (i : ℕ) : ∀ (n : ℕ) → ℕ ≔
  elim i at y -> (∀ (n : ℕ) → ℕ)
  | zero ⇒ λ (n : ℕ) → bit n
  | succ j, r ⇒ λ (n : ℕ) → r (half n)
  end
end

The number written by the first k digits of c: twice the number written by the digits after the first, plus the first.

def readFrom (k : ℕ) : ∀ (c : Bits) → ℕ ≔
  elim k at y -> (∀ (c : Bits) → ℕ)
  | zero ⇒ λ (c : Bits) → 0
  | succ p, r ⇒ λ (c : Bits) →
      plus (double (r (λ (i : ℕ) → c (succ i)))) (c 0)
  end
end

y, times x when the flag b is set.

def times (b : ℕ) (x : ℕ) (y : ℕ) : ℕ ≔
  pick b (λ (u : ℕ) → y) (λ (u : ℕ) → mult x y)
end

Exponentiation by squaring with fuel f: x to the n is the square of x to the half n, times x when the low bit of n is set. It stops when n reaches 0.

def powFrom (f : ℕ) : ∀ (x : ℕ) (n : ℕ) → ℕ ≔
  elim f at y -> (∀ (x : ℕ) (n : ℕ) → ℕ)
  | zero ⇒ λ (x : ℕ) (n : ℕ) → 1
  | succ g, r ⇒ λ (x : ℕ) (n : ℕ) →
      pick n (λ (u : ℕ) → 1)
        (λ (u : ℕ) → let y ≔ r x (half n) in times (bit n) x (mult y y) end)
  end
end

Fuel succ p is enough for n, so fuel p is enough for its half.

lemma halfFuel : ∀ (n : ℕ) (p : ℕ) (h : Le n (succ p)) → Le (half n) p ≔
  λ (n : ℕ) (p : ℕ) →
    elim n at x -> (∀ (h : Le x (succ p)) → Le (half x) p)
    | zero ⇒ λ (h : Unit) → unit
    | succ m, s ⇒ λ (h : Le m p) → leTrans (half (succ m)) m p (halfLe m) h
    end
end

lemma digitLe : ∀ (i : ℕ) (n : ℕ) → Le (digitAt i n) 1 ≔
  λ (i : ℕ) →
    elim i at x -> (∀ (n : ℕ) → Le (digitAt x n) 1)
    | zero ⇒ λ (n : ℕ) → bitLe n
    | succ j, ih ⇒ λ (n : ℕ) → ih (half n)
    end
end

With k digits or more, the digits of n write n. The digits after the first are those of half n, by conversion.

lemma readDigits : ∀ (k : ℕ) (n : ℕ) (h : Le n k) →
    Eq (readFrom k (λ (i : ℕ) → digitAt i n)) n ≔
  λ (k : ℕ) →
    elim k at x -> (∀ (n : ℕ) (h : Le n x) →
                         Eq (readFrom x (λ (i : ℕ) → digitAt i n)) n)
    | zero ⇒ λ (n : ℕ) →
        elim n at m -> (∀ (h : Le m 0) → Eq 0 m)
        | zero ⇒ λ (h : Unit) → unit
        | succ m, s ⇒ λ (h : Void) → h
        end
    | succ p, ih ⇒ λ (n : ℕ) (h : Le n (succ p)) →
        let rest ≔ readFrom p (λ (i : ℕ) → digitAt i (half n));
            hr : Eq rest (half n) ≔ ih (half n) (halfFuel n p h)
        in trans (plus (double rest) (bit n)) (plus (double (half n)) (bit n)) n
             (cong (λ (y : ℕ) → plus (double y) (bit n)) rest (half n) hr)
             (halfSpec n)
        end
    end
end

At most k of the first k digits are ones.

lemma countLe : ∀ (n : ℕ) (k : ℕ) →
    Le (sumTo (λ (i : ℕ) → digitAt i n) k) k ≔
  λ (n : ℕ) (k : ℕ) →
    elim k at x -> Le (sumTo (λ (i : ℕ) → digitAt i n) x) x
    | zero ⇒ unit
    | succ p, ih ⇒
        lePlusMono (digitAt p n) 1 (sumTo (λ (i : ℕ) → digitAt i n) p) p
          (digitLe p n) ih
    end
end

times b x y is pow x b times y, for a flag b.

lemma timesPow : ∀ (b : ℕ) (h : Le b 1) (x : ℕ) (y : ℕ) →
    Eq (times b x y) (mult (pow x b) y) ≔
  λ (b : ℕ) →
    elim b at k -> (∀ (h : Le k 1) (x : ℕ) (y : ℕ) →
                         Eq (times k x y) (mult (pow x k) y))
    | zero ⇒ λ (h : Unit) (x : ℕ) (y : ℕ) → sym (mult 1 y) y (multIdentityLeft y)
    | succ c, ih ⇒
        elim c at k -> (∀ (h : Le (succ k) 1) (x : ℕ) (y : ℕ) →
                             Eq (times (succ k) x y) (mult (pow x (succ k)) y))
        | zero ⇒ λ (h : Unit) (x : ℕ) (y : ℕ) →
            cong (λ (z : ℕ) → mult z y) x (mult x 1)
              (sym (mult x 1) x (multIdentityRight x))
        | succ e, s ⇒ λ (h : Void) (x : ℕ) (y : ℕ) →
            absurd (Eq (times (succ (succ e)) x y)
                       (mult (pow x (succ (succ e))) y)) h
        end
    end
end

With fuel f at least n, squaring computes the power. In the step, times contributes pow x (bit n), the square pow x (half n + half n), and the exponents add up to n by bitPlusHalves.

lemma powFromPow : ∀ (x : ℕ) (f : ℕ) (n : ℕ) (h : Le n f) →
    Eq (powFrom f x n) (pow x n) ≔
  λ (x : ℕ) (f : ℕ) →
    elim f at g -> (∀ (n : ℕ) (h : Le n g) → Eq (powFrom g x n) (pow x n))
    | zero ⇒ λ (n : ℕ) →
        elim n at m -> (∀ (h : Le m 0) → Eq 1 (pow x m))
        | zero ⇒ λ (h : Unit) → unit
        | succ m, s ⇒ λ (h : Void) → absurd (Eq 1 (pow x (succ m))) h
        end
    | succ g, ih ⇒ λ (n : ℕ) →
        elim n at m -> (∀ (h : Le m (succ g)) →
                             Eq (powFrom (succ g) x m) (pow x m))
        | zero ⇒ λ (h : Unit) → unit
        | succ m, s ⇒ λ (h : Le m g) →
            let k ≔ succ m;
                b ≔ bit k;
                hk ≔ half k;
                y ≔ powFrom g x hk;
                p ≔ pow x hk;
                hy : Eq y p ≔ ih hk (halfFuel k g h);
                flag : Eq (times b x (mult y y)) (mult (pow x b) (mult y y)) ≔
                  timesPow b (bitLe k) x (mult y y);
                square : Eq (mult y y) (pow x (plus hk hk)) ≔
                  trans (mult y y) (mult p p) (pow x (plus hk hk))
                    (trans (mult y y) (mult p y) (mult p p)
                       (cong (λ (z : ℕ) → mult z y) y p hy)
                       (cong (λ (z : ℕ) → mult p z) y p hy))
                    (sym (pow x (plus hk hk)) (mult p p) (powPlus x hk hk));
                whole : Eq (mult (pow x b) (pow x (plus hk hk)))
                           (pow x (plus b (plus hk hk))) ≔
                  sym (pow x (plus b (plus hk hk)))
                    (mult (pow x b) (pow x (plus hk hk)))
                    (powPlus x b (plus hk hk));
                exponent : Eq (pow x (plus b (plus hk hk))) (pow x k) ≔
                  cong (λ (z : ℕ) → pow x z) (plus b (plus hk hk)) k
                    (bitPlusHalves k)
            in trans (times b x (mult y y)) (mult (pow x b) (mult y y))
                 (pow x k) flag
                 (trans (mult (pow x b) (mult y y))
                    (mult (pow x b) (pow x (plus hk hk))) (pow x k)
                    (cong (λ (z : ℕ) → mult (pow x b) z)
                       (mult y y) (pow x (plus hk hk)) square)
                    (trans (mult (pow x b) (pow x (plus hk hk)))
                       (pow x (plus b (plus hk hk))) (pow x k)
                       whole exponent))
            end
        end
    end
end

The binary digits of n, least significant first; from index n on they are all 0.

def bits (n : ℕ) : Bits ≔ λ (i : ℕ) → Digits.digitAt i n end

def bitCount (n : ℕ) : ℕ ≔ sumTo (bits n) n end

Reading the digits back, shifting them, counting them, and the correctness of powFast.

module Laws where

The first k digits of n write n, when k is at least n.

theorem fromBitsBits : ∀ (n : ℕ) (k : ℕ) (h : Le n k) →
    Eq (fromBits (bits n) k) n ≔
  λ (n : ℕ) (k : ℕ) (h : Le n k) → Digits.readDigits k n h
end

n digits are enough for n.

theorem fromBitsSelf : ∀ (n : ℕ) → Eq (fromBits (bits n) n) n ≔
  λ (n : ℕ) → fromBitsBits n n (leRefl n)
end

Doubling writes a 0 as the lowest digit and shifts the others up.

theorem bitsDoubleZero : ∀ (n : ℕ) → Eq (bits (double n) 0) 0 ≔
  λ (n : ℕ) → Halving.bitDouble n
end

theorem bitsDoubleSucc : ∀ (n : ℕ) (i : ℕ) →
    Eq (bits (double n) (succ i)) (bits n i) ≔
  λ (n : ℕ) (i : ℕ) →
    cong (λ (m : ℕ) → bits m i) (half (double n)) n (halfDouble n)
end

Every digit is a 0 or a 1, so there are at most n ones.

theorem bitCountLe : ∀ (n : ℕ) → Le (bitCount n) n ≔
  λ (n : ℕ) → Digits.countLe n n
end

Repeated squaring computes the power.

theorem powFastPow : ∀ (x : ℕ) (n : ℕ) → Eq (powFast x n) (pow x n) ≔
  λ (x : ℕ) (n : ℕ) → Digits.powFromPow x n n (leRefl n)
end
