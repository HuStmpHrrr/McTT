Prelude›Arith›Monotone

Monotonicity of addition, multiplication and subtraction with respect to Prelude›Arith›Order.Le, transitivity laws for the strict order, and the interaction of Prelude›Arith›Sub.sub with the order.

module Prelude›Arith›Monotone where

import Prelude›Data›Unit use (Unit; unit)
import Prelude›Data›Void use (Void)
import Prelude›Builtin›Nat use (succ)
import Prelude›Logic use (absurd)
import Prelude›Arith›Equality use (Eq; refl; trans)
import Prelude›Arith›Plus use (plus)
open Prelude›Arith›Plus.Basic use (plusIdentityRight; plusSucc)

open Prelude›Arith›Order.LeLaws

Addition and multiplication are monotone, and addition cancels.

module Mono where

Addition is monotone in both arguments. The induction is on a with b generalized; at a = 0 the bound on c is weakened by lePlusLeft.

theorem lePlusMono : ∀ (a : ℕ) (b : ℕ) (c : ℕ) (d : ℕ)
    (hab : Le a b) (hcd : Le c d) → Le (plus a c) (plus b d) ≔
  λ (a : ℕ) →
    elim a at x -> (∀ (b : ℕ) (c : ℕ) (d : ℕ)
                         (hab : Le x b) (hcd : Le c d) →
                         Le (plus x c) (plus b d))
    | zero ⇒ λ (b : ℕ) (c : ℕ) (d : ℕ) (hab : Unit) (hcd : Le c d) →
        leTrans c d (plus b d) hcd (lePlusLeft b d)
    | succ p, ih ⇒ λ (b : ℕ) →
        elim b at z -> (∀ (c : ℕ) (d : ℕ)
                             (hab : Le (succ p) z) (hcd : Le c d) →
                             Le (plus (succ p) c) (plus z d))
        | zero ⇒ λ (c : ℕ) (d : ℕ) (hab : Void) (hcd : Le c d) →
            absurd (Le (succ (plus p c)) d) hab
        | succ q, s ⇒ λ (c : ℕ) (d : ℕ) (hab : Le p q) (hcd : Le c d) →
            ih q c d hab hcd
        end
    end
end

Addition is monotone in its first argument.

theorem lePlusMonoLeft : ∀ (a : ℕ) (b : ℕ) (c : ℕ) (h : Le a b) →
    Le (plus a c) (plus b c) ≔
  λ (a : ℕ) (b : ℕ) (c : ℕ) (h : Le a b) → lePlusMono a b c c h (leRefl c)
end

Addition is monotone in its second argument.

theorem lePlusMonoRight : ∀ (a : ℕ) (b : ℕ) (c : ℕ) (h : Le b c) →
    Le (plus a b) (plus a c) ≔
  λ (a : ℕ) (b : ℕ) (c : ℕ) (h : Le b c) → lePlusMono a a b c (leRefl a) h
end

A common first summand cancels: Le (plus (succ p) b) (plus (succ p) c) computes to Le (plus p b) (plus p c).

theorem lePlusCancel : ∀ (a : ℕ) (b : ℕ) (c : ℕ)
    (h : Le (plus a b) (plus a c)) → Le b c ≔
  λ (a : ℕ) (b : ℕ) (c : ℕ) →
    elim a at x -> (∀ (h : Le (plus x b) (plus x c)) → Le b c)
    | zero ⇒ λ (h : Le b c) → h
    | succ p, ih ⇒ λ (h : Le (plus p b) (plus p c)) → ih h
    end
end

Multiplication is monotone in both arguments: the step is lePlusMono applied to the bound on the factors and the induction hypothesis.

theorem leMultMono : ∀ (a : ℕ) (b : ℕ) (c : ℕ) (d : ℕ)
    (hab : Le a b) (hcd : Le c d) → Le (mult a c) (mult b d) ≔
  λ (a : ℕ) →
    elim a at x -> (∀ (b : ℕ) (c : ℕ) (d : ℕ)
                         (hab : Le x b) (hcd : Le c d) →
                         Le (mult x c) (mult b d))
    | zero ⇒ λ (b : ℕ) (c : ℕ) (d : ℕ) (hab : Unit) (hcd : Le c d) → unit
    | succ p, ih ⇒ λ (b : ℕ) →
        elim b at z -> (∀ (c : ℕ) (d : ℕ)
                             (hab : Le (succ p) z) (hcd : Le c d) →
                             Le (mult (succ p) c) (mult z d))
        | zero ⇒ λ (c : ℕ) (d : ℕ) (hab : Void) (hcd : Le c d) →
            absurd (Le (plus c (mult p c)) 0) hab
        | succ q, s ⇒ λ (c : ℕ) (d : ℕ) (hab : Le p q) (hcd : Le c d) →
            lePlusMono c d (mult p c) (mult q d) hcd (ih q c d hab hcd)
        end
    end
end

end

The strict order: transitivity, mixed with Le, and sums.

module Strict where

A strict bound gives a non-strict one.

theorem ltLe : ∀ (a : ℕ) (b : ℕ) (h : Lt a b) → Le a b ≔
  λ (a : ℕ) (b : ℕ) (h : Lt a b) → leTrans a (succ a) b (leSucc a) h
end

def ltLeTrans : ∀ (a : ℕ) (b : ℕ) (c : ℕ) (hab : Lt a b) (hbc : Le b c) →
    Lt a c ≔
  λ (a : ℕ) (b : ℕ) (c : ℕ) (hab : Lt a b) (hbc : Le b c) →
    leTrans (succ a) b c hab hbc
end

Le (succ a) (succ b) computes to Le a b, so hab bounds succ a.

theorem leLtTrans : ∀ (a : ℕ) (b : ℕ) (c : ℕ) (hab : Le a b) (hbc : Lt b c) →
    Lt a c ≔
  λ (a : ℕ) (b : ℕ) (c : ℕ) (hab : Le a b) (hbc : Lt b c) →
    leTrans (succ a) (succ b) c hab hbc
end

theorem ltTrans : ∀ (a : ℕ) (b : ℕ) (c : ℕ) (hab : Lt a b) (hbc : Lt b c) →
    Lt a c ≔
  λ (a : ℕ) (b : ℕ) (c : ℕ) (hab : Lt a b) (hbc : Lt b c) →
    ltLeTrans a b c hab (ltLe b c hbc)
end

theorem ltSucc : ∀ (n : ℕ) → Lt n (succ n) ≔
  λ (n : ℕ) → leRefl n
end

theorem ltPlus : ∀ (a : ℕ) (b : ℕ) → Lt a (plus a (succ b)) ≔
  λ (a : ℕ) (b : ℕ) →
    elim a at x -> Lt x (plus x (succ b))
    | zero ⇒ unit
    | succ p, ih ⇒ ih
    end
end

Addition preserves a strict bound in its second argument.

def ltPlusMonoRight : ∀ (m : ℕ) (j : ℕ) (n : ℕ) (h : Lt j n) →
    Lt (plus m j) (plus m n) ≔
  λ (m : ℕ) (j : ℕ) (n : ℕ) (h : Lt j n) →
    elim m at x -> Lt (plus x j) (plus x n)
    | zero ⇒ h
    | succ p, ih ⇒ ih
    end
end

end

Truncated subtraction: bounds, monotonicity and cancellation.

module SubLaws where

Truncated subtraction never exceeds its first argument.

def subLe : ∀ (a : ℕ) (b : ℕ) → Le (sub a b) a ≔
  λ (a : ℕ) (b : ℕ) →
    (elim b at y -> (∀ (x : ℕ) → Le (sub x y) x)
     | zero ⇒ λ (x : ℕ) → leRefl x
     | succ q, ih ⇒ λ (x : ℕ) →
         elim x at z -> Le (sub z (succ q)) z
         | zero ⇒ unit
         | succ p, s ⇒ leStep (sub p q) p (ih p)
         end
     end) a
end

Subtraction is monotone in its first argument.

theorem subMonoLeft : ∀ (a : ℕ) (b : ℕ) (c : ℕ) (h : Le a b) →
    Le (sub a c) (sub b c) ≔
  λ (a : ℕ) (b : ℕ) (c : ℕ) →
    (elim c at w -> (∀ (x : ℕ) (y : ℕ) (h : Le x y) →
                          Le (sub x w) (sub y w))
     | zero ⇒ λ (x : ℕ) (y : ℕ) (h : Le x y) → h
     | succ r, ih ⇒ λ (x : ℕ) →
         elim x at u -> (∀ (y : ℕ) (h : Le u y) →
                              Le (sub u (succ r)) (sub y (succ r)))
         | zero ⇒ λ (y : ℕ) (h : Unit) → unit
         | succ p, s ⇒ λ (y : ℕ) →
             elim y at v -> (∀ (h : Le (succ p) v) →
                                  Le (sub (succ p) (succ r)) (sub v (succ r)))
             | zero ⇒ λ (h : Void) → absurd (Le (sub p r) 0) h
             | succ q, t ⇒ λ (h : Le p q) → ih p q h
             end
         end
     end) a b
end

Subtraction is antitone in its second argument.

theorem subMonoRight : ∀ (a : ℕ) (b : ℕ) (c : ℕ) (h : Le b c) →
    Le (sub a c) (sub a b) ≔
  λ (a : ℕ) (b : ℕ) →
    (elim b at y -> (∀ (x : ℕ) (c : ℕ) (h : Le y c) →
                          Le (sub x c) (sub x y))
     | zero ⇒ λ (x : ℕ) (c : ℕ) (h : Unit) → subLe x c
     | succ q, ih ⇒ λ (x : ℕ) (c : ℕ) →
         elim c at w -> (∀ (h : Le (succ q) w) →
                              Le (sub x w) (sub x (succ q)))
         | zero ⇒ λ (h : Void) → absurd (Le x (sub x (succ q))) h
         | succ r, t ⇒ λ (h : Le q r) →
             elim x at u -> Le (sub u (succ r)) (sub u (succ q))
             | zero ⇒ unit
             | succ p, s ⇒ ih p r h
             end
         end
     end) a
end

Below a, subtracting b and adding it back on the right gives a.

theorem plusSubCancel : ∀ (a : ℕ) (b : ℕ) (h : Le b a) →
    Eq (plus (sub a b) b) a ≔
  λ (a : ℕ) (b : ℕ) →
    (elim b at y -> (∀ (x : ℕ) (h : Le y x) → Eq (plus (sub x y) y) x)
     | zero ⇒ λ (x : ℕ) (h : Unit) → plusIdentityRight x
     | succ q, ih ⇒ λ (x : ℕ) →
         elim x at z -> (∀ (h : Le (succ q) z) →
                              Eq (plus (sub z (succ q)) (succ q)) z)
         | zero ⇒ λ (h : Void) → absurd (Eq (plus 0 (succ q)) 0) h
         | succ p, s ⇒ λ (h : Le q p) →
             trans (plus (sub p q) (succ q)) (succ (plus (sub p q) q)) (succ p)
               (plusSucc (sub p q) q) (ih p h)
         end
     end) a
end

Below a, adding b on the left of sub a b gives a.

theorem subPlusCancel : ∀ (a : ℕ) (b : ℕ) (h : Le b a) →
    Eq (plus b (sub a b)) a ≔
  λ (a : ℕ) (b : ℕ) →
    (elim b at y -> (∀ (x : ℕ) (h : Le y x) → Eq (plus y (sub x y)) x)
     | zero ⇒ λ (x : ℕ) (h : Unit) → refl x
     | succ q, ih ⇒ λ (x : ℕ) →
         elim x at z -> (∀ (h : Le (succ q) z) →
                              Eq (plus (succ q) (sub z (succ q))) z)
         | zero ⇒ λ (h : Void) → absurd (Eq (succ (plus q 0)) 0) h
         | succ p, s ⇒ λ (h : Le q p) → ih p h
         end
     end) a
end

A strict bound leaves a positive difference.

theorem subPos : ∀ (a : ℕ) (b : ℕ) (h : Lt b a) → Lt 0 (sub a b) ≔
  λ (a : ℕ) (b : ℕ) →
    (elim b at y -> (∀ (x : ℕ) (h : Lt y x) → Lt 0 (sub x y))
     | zero ⇒ λ (x : ℕ) (h : Lt 0 x) → h
     | succ q, ih ⇒ λ (x : ℕ) →
         elim x at z -> (∀ (h : Lt (succ q) z) → Lt 0 (sub z (succ q)))
         | zero ⇒ λ (h : Void) → absurd (Lt 0 0) h
         | succ p, s ⇒ λ (h : Lt q p) → ih p h
         end
     end) a
end
