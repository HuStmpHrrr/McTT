Prelude›Algebra›AbelianGroup

Abelian groups: IsAbelianGroup A eq op e inv holds when the operations form a group up to eq, its predicate isGroup, and op is commutative. The predicate's with-area gives the commutative monoid predicate, isCommutativeMonoid; the bundle's forwards the laws and gives the group, A.group, and the commutative monoid, A.commutativeMonoid.

module Prelude›Algebra›AbelianGroup (l : Level) where

import Prelude›Algebra›Group l use (IsGroup; Group)
import Prelude›Algebra›CommutativeMonoid l
  use (IsCommutativeMonoid; CommutativeMonoid)

record IsAbelianGroup (A : Type@{l}) (eq : ∀ (x : A) (y : A) → Type@{l})
    (op : ∀ (x : A) (y : A) → A) (e : A) (inv : ∀ (x : A) → A) :
    Type@{l} where
  isGroup : IsGroup A eq op e inv;
  comm : ∀ (x : A) (y : A) → eq (op x y) (op y x)
with
  open isGroup
    export (refl; sym; trans; opCong; assoc; leftId; rightId; invCong; leftInv;
            rightInv)
  def isCommutativeMonoid : IsCommutativeMonoid A eq op e ≔
    record IsCommutativeMonoid A eq op e where
      isMonoid ≔ isGroup.isMonoid; comm ≔ comm
    end
  end
end

record AbelianGroup : Type@{succl l} where
  carrier : Type@{l};
  eq : ∀ (x : carrier) (y : carrier) → Type@{l};
  op : ∀ (x : carrier) (y : carrier) → carrier;
  e : carrier;
  inv : ∀ (x : carrier) → carrier;
  isAbelianGroup : IsAbelianGroup carrier eq op e inv
with
  open isAbelianGroup
    export (refl; sym; trans; opCong; assoc; leftId; rightId; invCong; leftInv;
            rightInv; comm)
  def group : Group ≔
    record Group where
      carrier ≔ carrier; eq ≔ eq; op ≔ op; e ≔ e; inv ≔ inv;
      isGroup ≔ isAbelianGroup.isGroup
    end
  end
  def commutativeMonoid : CommutativeMonoid ≔
    record CommutativeMonoid where
      carrier ≔ carrier; eq ≔ eq; op ≔ op; e ≔ e;
      isCommutativeMonoid ≔ isAbelianGroup.isCommutativeMonoid
    end
  end
end
