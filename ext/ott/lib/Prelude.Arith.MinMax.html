Prelude›Arith›MinMax

Minimum and maximum, by recursion on both arguments, with their laws stated with Prelude›Arith›Equality.

module Prelude›Arith›MinMax where

import Prelude›Data›Unit use (unit)
import Prelude›Builtin›Nat use (succ)
import Prelude›Arith›Equality use (Eq; refl; sym; trans)
import Prelude›Arith›Plus use (plus)
open Prelude›Arith›Plus.Basic use (plusIdentityRight; plusSucc)

def min (a : ℕ) (b : ℕ) : ℕ ≔
  (elim a at y -> (∀ (x : ℕ) → ℕ)
   | zero ⇒ λ (x : ℕ) → 0
   | succ p, r ⇒ λ (x : ℕ) →
       elim x at z -> ℕ
       | zero ⇒ 0
       | succ q, s ⇒ succ (r q)
       end
   end) b
end

def max (a : ℕ) (b : ℕ) : ℕ ≔
  (elim a at y -> (∀ (x : ℕ) → ℕ)
   | zero ⇒ λ (x : ℕ) → x
   | succ p, r ⇒ λ (x : ℕ) →
       elim x at z -> ℕ
       | zero ⇒ succ p
       | succ q, s ⇒ succ (r q)
       end
   end) b
end

Commutativity, idempotence, zero, and the sum of minimum and maximum.

module Laws where

theorem minComm : ∀ (a : ℕ) (b : ℕ) → Eq (min a b) (min b a) ≔
  λ (a : ℕ) →
    elim a at x -> (∀ (b : ℕ) → Eq (min x b) (min b x))
    | zero ⇒ λ (b : ℕ) →
        elim b at y -> Eq (min 0 y) (min y 0)
        | zero ⇒ unit
        | succ q, s ⇒ unit
        end
    | succ p, ih ⇒ λ (b : ℕ) →
        elim b at y -> Eq (min (succ p) y) (min y (succ p))
        | zero ⇒ unit
        | succ q, s ⇒ ih q
        end
    end
end

theorem maxComm : ∀ (a : ℕ) (b : ℕ) → Eq (max a b) (max b a) ≔
  λ (a : ℕ) →
    elim a at x -> (∀ (b : ℕ) → Eq (max x b) (max b x))
    | zero ⇒ λ (b : ℕ) →
        elim b at y -> Eq (max 0 y) (max y 0)
        | zero ⇒ unit
        | succ q, s ⇒ refl q
        end
    | succ p, ih ⇒ λ (b : ℕ) →
        elim b at y -> Eq (max (succ p) y) (max y (succ p))
        | zero ⇒ refl p
        | succ q, s ⇒ ih q
        end
    end
end

theorem minIdem : ∀ (n : ℕ) → Eq (min n n) n ≔
  λ (n : ℕ) →
    elim n at x -> Eq (min x x) x
    | zero ⇒ unit
    | succ p, ih ⇒ ih
    end
end

theorem minZeroRight : ∀ (n : ℕ) → Eq (min n 0) 0 ≔
  λ (n : ℕ) →
    elim n at x -> Eq (min x 0) 0
    | zero ⇒ unit
    | succ p, ih ⇒ unit
    end
end

theorem maxIdem : ∀ (n : ℕ) → Eq (max n n) n ≔
  λ (n : ℕ) →
    elim n at x -> Eq (max x x) x
    | zero ⇒ unit
    | succ p, ih ⇒ ih
    end
end

theorem maxIdentityRight : ∀ (n : ℕ) → Eq (max n 0) n ≔
  λ (n : ℕ) →
    elim n at x -> Eq (max x 0) x
    | zero ⇒ unit
    | succ p, ih ⇒ refl p
    end
end

The minimum and the maximum add up to the sum. In the successor case both sides are moved past plusSucc and meet at the induction hypothesis.

theorem plusMinMax : ∀ (a : ℕ) (b : ℕ) →
    Eq (plus (min a b) (max a b)) (plus a b) ≔
  λ (a : ℕ) →
    elim a at x -> (∀ (b : ℕ) → Eq (plus (min x b) (max x b)) (plus x b))
    | zero ⇒ λ (b : ℕ) → refl b
    | succ p, ih ⇒ λ (b : ℕ) →
        elim b at y -> Eq (plus (min (succ p) y) (max (succ p) y))
                             (plus (succ p) y)
        | zero ⇒ sym (plus p 0) p (plusIdentityRight p)
        | succ q, s ⇒
            let m ≔ min p q;
                M ≔ max p q;
                leftStep : Eq (plus m (succ M)) (succ (plus m M)) ≔ plusSucc m M;
                rightStep : Eq (succ (plus p q)) (plus p (succ q)) ≔
                  sym (plus p (succ q)) (succ (plus p q)) (plusSucc p q)
            in trans (plus m (succ M)) (succ (plus p q)) (plus p (succ q))
                 (trans (plus m (succ M)) (succ (plus m M)) (succ (plus p q))
                    leftStep (ih q))
                 rightStep
            end
        end
    end
end
