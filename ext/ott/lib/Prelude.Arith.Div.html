Prelude›Arith›Div

Euclidean division. div n k and mod n k are the quotient and remainder of n by k; they are total, with div n 0 = 0 and mod n 0 = n. They are implemented by divSucc n d and modSucc n d, which divide by succ d, so div n (succ d) computes to divSucc n d. Both are primitive recursive in the dividend: the remainder counts up and wraps to 0 on reaching the divisor, and the quotient grows by one at each wrap. The theorems are stated for a positive divisor succ d.

Written with the long forms (abstract def, private def, ...) rather than theorem, fact and let, for comparison.

One step of the remainder modulo succ d: succ r, or 0 when r has reached d.

def wrap (r : ℕ) (d : ℕ) : ℕ ≔ if ℕ (eqb r d) 0 (succ r) end

One step of the quotient: it grows by one exactly when the remainder wraps.

private def carry (r : ℕ) (d : ℕ) (q : ℕ) : ℕ ≔
  if ℕ (eqb r d) (succ q) q
end

The remainder of n by succ d: n steps of wrap from 0.

def modSucc (n : ℕ) (d : ℕ) : ℕ ≔
  Generic.iterate 0l ℕ (λ (r : ℕ) → wrap r d) n 0
end

The quotient of n by succ d.

def divSucc (n : ℕ) (d : ℕ) : ℕ ≔
  elim n at y -> ℕ
  | zero ⇒ 0
  | succ p, q ⇒ carry (modSucc p d) d q
  end
end

The quotient of n by k, which is 0 when k is 0.

def div (n : ℕ) (k : ℕ) : ℕ ≔
  elim k at y -> ℕ
  | zero ⇒ 0
  | succ d, r ⇒ divSucc n d
  end
end

The remainder of n by k, which is n when k is 0.

def mod (n : ℕ) (k : ℕ) : ℕ ≔
  elim k at y -> ℕ
  | zero ⇒ n
  | succ d, r ⇒ modSucc n d
  end
end

Division by zero, and division of zero.

module Zero where

abstract def divZeroDivisor : ∀ (n : ℕ) → Eq (div n 0) 0 ≔
  λ (n : ℕ) → unit
end

abstract def modZeroDivisor : ∀ (n : ℕ) → Eq (mod n 0) n ≔
  λ (n : ℕ) → refl n
end

Zero divided by any divisor, zero included, has quotient 0.

abstract def divZero : ∀ (k : ℕ) → Eq (div 0 k) 0 ≔
  λ (k : ℕ) →
    elim k at x -> Eq (div 0 x) 0
    | zero ⇒ unit
    | succ d, ih ⇒ unit
    end
end

Zero divided by any divisor, zero included, has remainder 0.

abstract def modZero : ∀ (k : ℕ) → Eq (mod 0 k) 0 ≔
  λ (k : ℕ) →
    elim k at x -> Eq (mod 0 x) 0
    | zero ⇒ unit
    | succ d, ih ⇒ unit
    end
end

The division equation and the bounds of the remainder.

module Bounds where

The division equation n = div n (succ d) * succ d + mod n (succ d). In the step, a remainder that does not wrap adds one to the remainder; one that wraps is equal to d, and the added succ d moves into the quotient.

abstract def divModSpec : ∀ (n : ℕ) (d : ℕ) →
    Eq (plus (mult (div n (succ d)) (succ d)) (mod n (succ d))) n ≔
  λ (n : ℕ) (d : ℕ) →
    elim n at x ->
      Eq (plus (mult (div x (succ d)) (succ d)) (mod x (succ d))) x
    | zero ⇒ unit
    | succ p, ih ⇒
        let r ≔ mod p (succ d);
            q ≔ div p (succ d);
            m ≔ mult q (succ d)
        in eqbCases r d
             (λ (b : Bool) →
                Eq (plus (mult (if ℕ b (succ q) q) (succ d))
                         (if ℕ b 0 (succ r)))
                   (succ p))
             (λ (h : IsFalse (eqb r d)) →
                trans (plus m (succ r)) (succ (plus m r)) (succ p)
                  (plusSucc m r) ih)
             (λ (h : Eq r d) →
                let toSum : Eq (plus (plus d m) 0) (plus d m) ≔
                      plusIdentityRight (plus d m);
                    swap : Eq (plus d m) (plus m d) ≔ plusComm d m;
                    toRem : Eq (plus m d) (plus m r) ≔
                      cong (λ (y : ℕ) → plus m y) d r (sym r d h)
                in trans (plus (plus d m) 0) (plus m d) p
                     (trans (plus (plus d m) 0) (plus d m) (plus m d)
                        toSum swap)
                     (trans (plus m d) (plus m r) p toRem ih)
                end)
        end
    end
end

The remainder is below the divisor. Lt (mod n (succ d)) (succ d) computes to Le (mod n (succ d)) d.

abstract def modLt : ∀ (n : ℕ) (d : ℕ) → Lt (mod n (succ d)) (succ d) ≔
  λ (n : ℕ) (d : ℕ) →
    elim n at x -> Lt (mod x (succ d)) (succ d)
    | zero ⇒ unit
    | succ p, ih ⇒
        let r ≔ mod p (succ d)
        in eqbCases r d
             (λ (b : Bool) → Le (if ℕ b 0 (succ r)) d)
             (λ (h : IsFalse (eqb r d)) → leEqbFalse r d ih h)
             (λ (h : Eq r d) → unit)
        end
    end
end

A dividend below the divisor is its own remainder.

abstract def modSmall : ∀ (n : ℕ) (d : ℕ) (h : Lt n (succ d)) →
    Eq (mod n (succ d)) n ≔
  λ (n : ℕ) (d : ℕ) →
    elim n at x -> (∀ (h : Lt x (succ d)) → Eq (mod x (succ d)) x)
    | zero ⇒ λ (h : Unit) → unit
    | succ p, ih ⇒ λ (h : Le (succ p) d) →
        let hp : Eq (mod p (succ d)) p ≔
              ih (leTrans p (succ p) d (leSucc p) h);
            atP : Eq (wrap p d) (succ p) ≔
              eqbCases p d
                (λ (b : Bool) → Eq (if ℕ b 0 (succ p)) (succ p))
                (λ (e : IsFalse (eqb p d)) → refl p)
                (λ (e : Eq p d) → ltNotEq p d h e)
        in transport (λ (x : ℕ) → Eq (wrap x d) (succ p)) p (mod p (succ d))
             (sym (mod p (succ d)) p hp) atP
        end
    end
end

A dividend below the divisor has quotient 0.

abstract def divSmall : ∀ (n : ℕ) (d : ℕ) (h : Lt n (succ d)) →
    Eq (div n (succ d)) 0 ≔
  λ (n : ℕ) (d : ℕ) →
    elim n at x -> (∀ (h : Lt x (succ d)) → Eq (div x (succ d)) 0)
    | zero ⇒ λ (h : Unit) → unit
    | succ p, ih ⇒ λ (h : Le (succ p) d) →
        let hp : Le p d ≔ leTrans p (succ p) d (leSucc p) h;
            hq : Eq (div p (succ d)) 0 ≔ ih hp;
            hr : Eq (mod p (succ d)) p ≔ modSmall p d hp;
            atZero : Eq (carry p d 0) 0 ≔
              eqbCases p d
                (λ (b : Bool) → Eq (if ℕ b 1 0) 0)
                (λ (e : IsFalse (eqb p d)) → unit)
                (λ (e : Eq p d) → ltNotEq p d h e);
            atP : Eq (carry p d (div p (succ d))) 0 ≔
              transport (λ (y : ℕ) → Eq (carry p d y) 0) 0 (div p (succ d))
                (sym (div p (succ d)) 0 hq) atZero
        in transport (λ (x : ℕ) → Eq (carry x d (div p (succ d))) 0)
             p (mod p (succ d)) (sym (mod p (succ d)) p hr) atP
        end
    end
end

end

The remainder is periodic in the divisor.

module Periodic where

The divisor itself has remainder 0.

abstract def modSelf : ∀ (d : ℕ) → Eq (mod (succ d) (succ d)) 0 ≔
  λ (d : ℕ) →
    let hd : Eq (mod d (succ d)) d ≔ modSmall d d (leRefl d);
        atD : Eq (wrap d d) 0 ≔
          eqbCases d d
            (λ (b : Bool) → Eq (if ℕ b 0 (succ d)) 0)
            (λ (e : IsFalse (eqb d d)) → eqbFalse d d e (refl d))
            (λ (e : Eq d d) → unit)
    in transport (λ (x : ℕ) → Eq (wrap x d) 0) d (mod d (succ d))
         (sym (mod d (succ d)) d hd) atD
    end
end

The remainder of a sum is determined by the remainder of the second summand: mod (a + b) (succ d) runs a more steps of wrap from mod b (succ d).

abstract def modPlus : ∀ (a : ℕ) (b : ℕ) (d : ℕ) →
    Eq (mod (plus a b) (succ d))
       (Generic.iterate 0l ℕ (λ (r : ℕ) → wrap r d) a (mod b (succ d))) ≔
  λ (a : ℕ) (b : ℕ) (d : ℕ) →
    let module Wrap ≔ IterateLaws (λ (r : ℕ) → wrap r d)
    in Wrap.iteratePlus a b 0
    end
end

The remainder is periodic: adding the divisor does not change it. The proof commutes the succ d extra steps of wrap to the start, where they return to 0 by modSelf.

abstract def modPlusDivisor : ∀ (n : ℕ) (d : ℕ) →
    Eq (mod (plus (succ d) n) (succ d)) (mod n (succ d)) ≔
  λ (n : ℕ) (d : ℕ) →
    let module Wrap ≔ IterateLaws (λ (r : ℕ) → wrap r d);
        module Steps ≔ Generic 0l ℕ;
        w ≔ λ (r : ℕ) → wrap r d;
        k ≔ succ d;
        split : Eq (mod (plus k n) k) (Steps.iterate w k (mod n k)) ≔
          modPlus k n d;
        comm : Eq (Steps.iterate w k (mod n k))
                  (Steps.iterate w n (mod k k)) ≔
          Wrap.iterateComm k n 0;
        back : Eq (Steps.iterate w n (mod k k)) (mod n k) ≔
          cong (λ (y : ℕ) → Steps.iterate w n y) (mod k k) 0 (modSelf d)
    in trans (mod (plus k n) k) (Steps.iterate w n (mod k k)) (mod n k)
         (trans (mod (plus k n) k) (Steps.iterate w k (mod n k))
            (Steps.iterate w n (mod k k)) split comm)
         back
    end
end

Adding a multiple of succ d does not change the remainder by succ d.

abstract def modPlusMult : ∀ (d : ℕ) (t : ℕ) (y : ℕ) →
    Eq (mod (plus (mult t (succ d)) y) (succ d)) (mod y (succ d)) ≔
  λ (d : ℕ) (t : ℕ) (y : ℕ) →
    elim t at x ->
      Eq (mod (plus (mult x (succ d)) y) (succ d)) (mod y (succ d))
    | zero ⇒ refl (mod y (succ d))
    | succ p, ih ⇒
        let k ≔ succ d;
            m ≔ mult p k;
            assoc : Eq (mod (plus (plus k m) y) k)
                       (mod (plus k (plus m y)) k) ≔
              cong (λ (z : ℕ) → mod z k) (plus (plus k m) y) (plus k (plus m y))
                (plusAssoc k m y)
        in trans (mod (plus (plus k m) y) k) (mod (plus k (plus m y)) k)
             (mod y k) assoc
             (trans (mod (plus k (plus m y)) k) (mod (plus m y) k) (mod y k)
                (modPlusDivisor (plus m y) d) ih)
        end
    end
end

end

Division by one.

module One where

Division by one: the remainder is 0.

abstract def modOne : ∀ (n : ℕ) → Eq (mod n 1) 0 ≔
  λ (n : ℕ) →
    elim n at x -> Eq (mod x 1) 0
    | zero ⇒ unit
    | succ p, ih ⇒
        transport (λ (x : ℕ) → Eq (wrap x 0) 0) 0 (mod p 1)
          (sym (mod p 1) 0 ih) unit
    end
end

Division by one: the quotient is the dividend, by divModSpec and modOne.

abstract def divOne : ∀ (n : ℕ) → Eq (div n 1) n ≔
  λ (n : ℕ) →
    let q ≔ div n 1;
        spec : Eq (plus (mult q 1) (mod n 1)) n ≔ divModSpec n 0;
        dropMod : Eq (plus (mult q 1) 0) (plus (mult q 1) (mod n 1)) ≔
          cong (λ (y : ℕ) → plus (mult q 1) y) 0 (mod n 1)
            (sym (mod n 1) 0 (modOne n));
        dropPlus : Eq (mult q 1) (plus (mult q 1) 0) ≔
          sym (plus (mult q 1) 0) (mult q 1) (plusIdentityRight (mult q 1));
        dropMult : Eq q (mult q 1) ≔ sym (mult q 1) q (multIdentityRight q)
    in trans q (mult q 1) n dropMult
         (trans (mult q 1) (plus (mult q 1) 0) n dropPlus
            (trans (plus (mult q 1) 0) (plus (mult q 1) (mod n 1)) n
               dropMod spec))
    end
end

end
