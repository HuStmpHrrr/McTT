Exponentiation by recursion on the exponent, with its laws stated with Prelude›Arith›Equality and proved from those of Prelude›Arith›Mult.

module Prelude›Arith›Pow where

import Prelude›Data›Unit use (unit)
import Prelude›Builtin›Nat use (succ)
import Prelude›Arith›Equality use (Eq; refl; sym; trans; cong)
import Prelude›Arith›Plus use (plus)
open Prelude›Arith›Plus.Basic use (plusIdentityRight)
import Prelude›Arith›Mult use (mult)
open Prelude›Arith›Mult.Basic use (multIdentityRight; multIdentityLeft)
open Prelude›Arith›Mult.Algebra use (multAssoc; multSwapMiddle)

pow a n is a to the power n: pow a 0 is 1 and pow a (succ p) is mult a (pow a p).

def pow (a : ℕ) (n : ℕ) : ℕ ≔
  elim n at y -> ℕ
  | zero ⇒ 1
  | succ p, r ⇒ mult a r
  end
end

Exponents zero, one and successor, and powers of one.

module Basic where

theorem powZero : ∀ (a : ℕ) → Eq (pow a 0) 1 ≔
  λ (a : ℕ) → unit
end

theorem powSucc : ∀ (a : ℕ) (n : ℕ) → Eq (pow a (succ n)) (mult a (pow a n)) ≔
  λ (a : ℕ) (n : ℕ) → refl (mult a (pow a n))
end

theorem powOne : ∀ (a : ℕ) → Eq (pow a 1) a ≔
  λ (a : ℕ) → multIdentityRight a
end

pow 1 (succ p) computes to plus (pow 1 p) 0.

theorem onePow : ∀ (n : ℕ) → Eq (pow 1 n) 1 ≔
  λ (n : ℕ) →
    elim n at x -> Eq (pow 1 x) 1
    | zero ⇒ unit
    | succ p, ih ⇒ trans (plus (pow 1 p) 0) (pow 1 p) 1 (plusIdentityRight (pow 1 p)) ih
    end
end

end

open Basic use (onePow)

The laws of exponents.

module Exp where

Powers turn sums of exponents into products. The step multiplies the induction hypothesis by a and reassociates.

theorem powPlus : ∀ (a : ℕ) (m : ℕ) (n : ℕ) →
    Eq (pow a (plus m n)) (mult (pow a m) (pow a n)) ≔
  λ (a : ℕ) (m : ℕ) (n : ℕ) →
    elim m at x -> Eq (pow a (plus x n)) (mult (pow a x) (pow a n))
    | zero ⇒ sym (mult 1 (pow a n)) (pow a n) (multIdentityLeft (pow a n))
    | succ p, ih ⇒
        let lhs ≔ mult a (pow a (plus p n));
            mid ≔ mult a (mult (pow a p) (pow a n));
            rhs ≔ mult (mult a (pow a p)) (pow a n);
            step : Eq lhs mid ≔
              cong (λ (y : ℕ) → mult a y) (pow a (plus p n))
                (mult (pow a p) (pow a n)) ih
        in trans lhs mid rhs step
             (sym rhs mid (multAssoc a (pow a p) (pow a n)))
        end
    end
end

A power of a product is the product of the powers.

theorem powMultBase : ∀ (a : ℕ) (b : ℕ) (n : ℕ) →
    Eq (pow (mult a b) n) (mult (pow a n) (pow b n)) ≔
  λ (a : ℕ) (b : ℕ) (n : ℕ) →
    elim n at x -> Eq (pow (mult a b) x) (mult (pow a x) (pow b x))
    | zero ⇒ unit
    | succ p, ih ⇒
        let lhs ≔ mult (mult a b) (pow (mult a b) p);
            mid ≔ mult (mult a b) (mult (pow a p) (pow b p));
            rhs ≔ mult (mult a (pow a p)) (mult b (pow b p));
            step : Eq lhs mid ≔
              cong (λ (y : ℕ) → mult (mult a b) y) (pow (mult a b) p)
                (mult (pow a p) (pow b p)) ih
        in trans lhs mid rhs step (multSwapMiddle a b (pow a p) (pow b p))
        end
    end
end

A power of a power multiplies the exponents. The step splits pow a (plus n (mult p n)) by powPlus and recombines the bases by powMultBase.

theorem powMult : ∀ (a : ℕ) (m : ℕ) (n : ℕ) →
    Eq (pow a (mult m n)) (pow (pow a m) n) ≔
  λ (a : ℕ) (m : ℕ) (n : ℕ) →
    elim m at x -> Eq (pow a (mult x n)) (pow (pow a x) n)
    | zero ⇒ sym (pow 1 n) 1 (onePow n)
    | succ p, ih ⇒
        let lhs ≔ pow a (plus n (mult p n));
            split ≔ mult (pow a n) (pow a (mult p n));
            mid ≔ mult (pow a n) (pow (pow a p) n);
            rhs ≔ pow (mult a (pow a p)) n;
            step : Eq split mid ≔
              cong (λ (y : ℕ) → mult (pow a n) y) (pow a (mult p n))
                (pow (pow a p) n) ih;
            back : Eq mid rhs ≔ sym rhs mid (powMultBase a (pow a p) n)
        in trans lhs split rhs (powPlus a n (mult p n))
             (trans split mid rhs step back)
        end
    end
end
