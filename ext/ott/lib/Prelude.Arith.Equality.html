Prelude›Arith›Equality

Equality of natural numbers, computed by recursion on both sides, with its basic laws.

module Prelude›Arith›Equality where

import Prelude›Data›Unit use (Unit; unit)
import Prelude›Data›Void use (Void)
import Prelude›Builtin›Nat use (succ)
import Prelude›Logic use (Not; absurd)

def Eq : ∀ (n : ℕ) (m : ℕ) → Type@0 ≔
  λ (n : ℕ) →
    elim n at y -> (∀ (m : ℕ) → Type@0)
    | zero ⇒ λ (m : ℕ) →
        elim m at z -> Type@0 | zero ⇒ Unit | succ q, s ⇒ Void end
    | succ p, r ⇒ λ (m : ℕ) →
        elim m at z -> Type@0 | zero ⇒ Void | succ q, s ⇒ r q end
    end
end

theorem refl : ∀ (n : ℕ) → Eq n n ≔
  λ (n : ℕ) →
    elim n at x -> Eq x x
    | zero ⇒ unit
    | succ p, ih ⇒ ih
    end
end

theorem zeroNeSucc : ∀ (n : ℕ) → Not (Eq 0 (succ n)) ≔
  λ (n : ℕ) (h : Eq 0 (succ n)) → h
end

theorem sym : ∀ (a : ℕ) (b : ℕ) (h : Eq a b) → Eq b a ≔
  λ (a : ℕ) →
    elim a at x -> (∀ (b : ℕ) (h : Eq x b) → Eq b x)
    | zero ⇒ λ (b : ℕ) →
        elim b at z -> (∀ (h : Eq 0 z) → Eq z 0)
        | zero ⇒ λ (h : Unit) → h
        | succ q, s ⇒ λ (h : Eq 0 (succ q)) → absurd (Eq (succ q) 0) h
        end
    | succ p, ih ⇒ λ (b : ℕ) →
        elim b at z -> (∀ (h : Eq (succ p) z) → Eq z (succ p))
        | zero ⇒ λ (h : Eq (succ p) 0) → h
        | succ q, s ⇒ λ (h : Eq (succ p) (succ q)) → ih q h
        end
    end
end

theorem trans : ∀ (a : ℕ) (b : ℕ) (c : ℕ) (hab : Eq a b) (hbc : Eq b c) →
    Eq a c ≔
  λ (a : ℕ) →
    elim a at x -> (∀ (b : ℕ) (c : ℕ) (hab : Eq x b) (hbc : Eq b c) → Eq x c)
    | zero ⇒ λ (b : ℕ) →
        elim b at z -> (∀ (c : ℕ) (hab : Eq 0 z) (hbc : Eq z c) → Eq 0 c)
        | zero ⇒ λ (c : ℕ) (hab : Unit) (hbc : Eq 0 c) → hbc
        | succ q, s ⇒ λ (c : ℕ) (hab : Eq 0 (succ q)) (hbc : Eq (succ q) c) →
            absurd (Eq 0 c) hab
        end
    | succ p, ih ⇒ λ (b : ℕ) →
        elim b at z -> (∀ (c : ℕ) (hab : Eq (succ p) z) (hbc : Eq z c) →
                             Eq (succ p) c)
        | zero ⇒ λ (c : ℕ) (hab : Eq (succ p) 0) (hbc : Eq 0 c) →
            absurd (Eq (succ p) c) hab
        | succ q, s ⇒ λ (c : ℕ) →
            elim c at w -> (∀ (hab : Eq (succ p) (succ q))
                                 (hbc : Eq (succ q) w) →
                                 Eq (succ p) w)
            | zero ⇒ λ (hab : Eq p q) (hbc : Eq (succ q) 0) →
                absurd (Eq (succ p) 0) hbc
            | succ r, t ⇒ λ (hab : Eq p q) (hbc : Eq q r) → ih q r hab hbc
            end
        end
    end
end

Every function on numbers respects equality. The proof is by induction on both numbers, with the function generalized so that the successor case can use the hypothesis at λ y → f (succ y).

theorem cong : ∀ (f : ∀ (x : ℕ) → ℕ) (a : ℕ) (b : ℕ) (h : Eq a b) →
    Eq (f a) (f b) ≔
  λ (f : ∀ (x : ℕ) → ℕ) (a : ℕ) →
    (elim a at x -> (∀ (g : ∀ (y : ℕ) → ℕ) (b : ℕ) (h : Eq x b) →
                          Eq (g x) (g b))
     | zero ⇒ λ (g : ∀ (y : ℕ) → ℕ) (b : ℕ) →
         elim b at z -> (∀ (h : Eq 0 z) → Eq (g 0) (g z))
         | zero ⇒ λ (h : Unit) → refl (g 0)
         | succ q, s ⇒ λ (h : Eq 0 (succ q)) → absurd (Eq (g 0) (g (succ q))) h
         end
     | succ p, ih ⇒ λ (g : ∀ (y : ℕ) → ℕ) (b : ℕ) →
         elim b at z -> (∀ (h : Eq (succ p) z) → Eq (g (succ p)) (g z))
         | zero ⇒ λ (h : Eq (succ p) 0) → absurd (Eq (g (succ p)) (g 0)) h
         | succ q, s ⇒ λ (h : Eq p q) → ih (λ (y : ℕ) → g (succ y)) q h
         end
     end) f
end

Equal numbers satisfy the same predicates, of any level u.

module Transport (u : Level) where

open Prelude›Logic.Negation u use (absurd)

def transport : ∀ (P : ∀ (n : ℕ) → Type@{u}) (x : ℕ) (y : ℕ)
    (h : Eq x y) (px : P x) → P y ≔
  λ (P : ∀ (n : ℕ) → Type@{u}) (x : ℕ) →
    (elim x at v -> (∀ (Q : ∀ (n : ℕ) → Type@{u}) (y : ℕ)
                          (h : Eq v y) (pv : Q v) → Q y)
     | zero ⇒ λ (Q : ∀ (n : ℕ) → Type@{u}) (y : ℕ) →
         elim y at w -> (∀ (h : Eq 0 w) (pv : Q 0) → Q w)
         | zero ⇒ λ (h : Unit) (pv : Q 0) → pv
         | succ q, s ⇒ λ (h : Eq 0 (succ q)) (pv : Q 0) → absurd (Q (succ q)) h
         end
     | succ p, ih ⇒ λ (Q : ∀ (n : ℕ) → Type@{u}) (y : ℕ) →
         elim y at w -> (∀ (h : Eq (succ p) w) (pv : Q (succ p)) → Q w)
         | zero ⇒ λ (h : Eq (succ p) 0) (pv : Q (succ p)) → absurd (Q 0) h
         | succ q, s ⇒ λ (h : Eq p q) (pv : Q (succ p)) →
             ih (λ (n : ℕ) → Q (succ n)) q h pv
         end
     end) P
end

end
