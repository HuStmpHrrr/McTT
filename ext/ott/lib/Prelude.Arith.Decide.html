Prelude›Arith›Decide

Decidability of equality and order on natural numbers, the boolean equality test eqb with its specification and case analysis, and the conditional pick on whether a number is zero.

Decision procedures for equality and order.

module Deciders where

Equality of numbers is decidable. In the step, Eq (succ p) (succ q) computes to Eq p q, so the hypothesis already has the goal's type.

def decEq : ∀ (n : ℕ) (m : ℕ) → Dec (Eq n m) ≔
  λ (n : ℕ) →
    elim n at x -> (∀ (m : ℕ) → Dec (Eq x m))
    | zero ⇒ λ (m : ℕ) →
        elim m at z -> Dec (Eq 0 z)
        | zero ⇒ orInl Unit (Not Unit) unit
        | succ q, s ⇒ orInr Void (Not Void) (λ (h : Void) → h)
        end
    | succ p, ih ⇒ λ (m : ℕ) →
        elim m at z -> Dec (Eq (succ p) z)
        | zero ⇒ orInr Void (Not Void) (λ (h : Void) → h)
        | succ q, s ⇒ ih q
        end
    end
end

The order is decidable, by the same recursion as decEq.

def decLe : ∀ (n : ℕ) (m : ℕ) → Dec (Le n m) ≔
  λ (n : ℕ) →
    elim n at x -> (∀ (m : ℕ) → Dec (Le x m))
    | zero ⇒ λ (m : ℕ) → orInl Unit (Not Unit) unit
    | succ p, ih ⇒ λ (m : ℕ) →
        elim m at z -> Dec (Le (succ p) z)
        | zero ⇒ orInr Void (Not Void) (λ (h : Void) → h)
        | succ q, s ⇒ ih q
        end
    end
end

The strict order is decidable, as an instance of decLe.

def decLt (n : ℕ) (m : ℕ) : Dec (Lt n m) ≔ decLe (succ n) m end

end

The conditional on whether a number is zero: pick b z s is z 0 when b is 0, and s 0 otherwise. The branches are functions of a dummy argument, so only the chosen one is computed. The conditional on a boolean is Prelude›Data›Bool.cond.

def pick (b : ℕ) (z : ∀ (u : ℕ) → ℕ) (s : ∀ (u : ℕ) → ℕ) : ℕ ≔
  (elim b at y -> (∀ (u : ℕ) → ℕ)
   | zero ⇒ z
   | succ k, t ⇒ s
   end) 0
end

The equality test: eqb n m is tt when n and m are equal, and ff otherwise.

def eqb : ∀ (n : ℕ) (m : ℕ) → Bool ≔
  λ (n : ℕ) →
    elim n at y -> (∀ (m : ℕ) → Bool)
    | zero ⇒ λ (m : ℕ) →
        elim m at z -> Bool | zero ⇒ tt | succ q, s ⇒ ff end
    | succ p, r ⇒ λ (m : ℕ) →
        elim m at z -> Bool | zero ⇒ ff | succ q, s ⇒ r q end
    end
end

The boolean equality test eqb agrees with Eq.

module EqbLaws where

eqb answers tt only on equal numbers.

theorem eqbSound : ∀ (n : ℕ) (m : ℕ) (h : IsTrue (eqb n m)) → Eq n m ≔
  λ (n : ℕ) →
    elim n at x -> (∀ (m : ℕ) (h : IsTrue (eqb x m)) → Eq x m)
    | zero ⇒ λ (m : ℕ) →
        elim m at z -> (∀ (h : IsTrue (eqb 0 z)) → Eq 0 z)
        | zero ⇒ λ (h : Unit) → h
        | succ q, s ⇒ λ (h : Void) → h
        end
    | succ p, ih ⇒ λ (m : ℕ) →
        elim m at z -> (∀ (h : IsTrue (eqb (succ p) z)) → Eq (succ p) z)
        | zero ⇒ λ (h : Void) → h
        | succ q, s ⇒ λ (h : IsTrue (eqb p q)) → ih q h
        end
    end
end

eqb answers tt on all equal numbers.

theorem eqbComplete : ∀ (n : ℕ) (m : ℕ) (h : Eq n m) → IsTrue (eqb n m) ≔
  λ (n : ℕ) →
    elim n at x -> (∀ (m : ℕ) (h : Eq x m) → IsTrue (eqb x m))
    | zero ⇒ λ (m : ℕ) →
        elim m at z -> (∀ (h : Eq 0 z) → IsTrue (eqb 0 z))
        | zero ⇒ λ (h : Unit) → h
        | succ q, s ⇒ λ (h : Void) → h
        end
    | succ p, ih ⇒ λ (m : ℕ) →
        elim m at z -> (∀ (h : Eq (succ p) z) → IsTrue (eqb (succ p) z))
        | zero ⇒ λ (h : Void) → h
        | succ q, s ⇒ λ (h : Eq p q) → ih q h
        end
    end
end

The specification of eqb: it answers tt exactly on equal numbers.

def eqbSpec (n : ℕ) (m : ℕ) : Iff (IsTrue (eqb n m)) (Eq n m) ≔
  iffIntro (IsTrue (eqb n m)) (Eq n m) (eqbSound n m) (eqbComplete n m)
end

When eqb answers ff, the numbers differ.

theorem eqbFalse : ∀ (n : ℕ) (m : ℕ) (h : IsFalse (eqb n m)) → Not (Eq n m) ≔
  λ (n : ℕ) →
    elim n at x -> (∀ (m : ℕ) (h : IsFalse (eqb x m)) → Not (Eq x m))
    | zero ⇒ λ (m : ℕ) →
        elim m at z -> (∀ (h : IsFalse (eqb 0 z)) → Not (Eq 0 z))
        | zero ⇒ λ (h : Void) (e : Unit) → h
        | succ q, s ⇒ λ (h : Unit) (e : Void) → e
        end
    | succ p, ih ⇒ λ (m : ℕ) →
        elim m at z -> (∀ (h : IsFalse (eqb (succ p) z)) →
                             Not (Eq (succ p) z))
        | zero ⇒ λ (h : Unit) (e : Void) → e
        | succ q, s ⇒ λ (h : IsFalse (eqb p q)) → ih q h
        end
    end
end

theorem eqbRefl : ∀ (n : ℕ) → IsTrue (eqb n n) ≔
  λ (n : ℕ) →
    elim n at x -> IsTrue (eqb x x)
    | zero ⇒ unit
    | succ p, ih ⇒ ih
    end
end

Case analysis on eqb a b for a predicate P on booleans: it suffices to prove P ff when eqb a b is ff, and P tt when a and b are equal.

def eqbCases : ∀ (a : ℕ) (b : ℕ) (P : ∀ (c : Bool) → Type@0)
    (hf : ∀ (h : IsFalse (eqb a b)) → P ff) (ht : ∀ (h : Eq a b) → P tt) →
    P (eqb a b) ≔
  λ (a : ℕ) →
    elim a at x -> (∀ (b : ℕ) (P : ∀ (c : Bool) → Type@0)
        (hf : ∀ (h : IsFalse (eqb x b)) → P ff) (ht : ∀ (h : Eq x b) → P tt) →
        P (eqb x b))
    | zero ⇒ λ (b : ℕ) →
        elim b at z -> (∀ (P : ∀ (c : Bool) → Type@0)
            (hf : ∀ (h : IsFalse (eqb 0 z)) → P ff)
            (ht : ∀ (h : Eq 0 z) → P tt) → P (eqb 0 z))
        | zero ⇒ λ (P : ∀ (c : Bool) → Type@0)
            (hf : ∀ (h : Void) → P ff) (ht : ∀ (h : Unit) → P tt) → ht unit
        | succ q, s ⇒ λ (P : ∀ (c : Bool) → Type@0)
            (hf : ∀ (h : Unit) → P ff) (ht : ∀ (h : Void) → P tt) → hf unit
        end
    | succ p, ih ⇒ λ (b : ℕ) →
        elim b at z -> (∀ (P : ∀ (c : Bool) → Type@0)
            (hf : ∀ (h : IsFalse (eqb (succ p) z)) → P ff)
            (ht : ∀ (h : Eq (succ p) z) → P tt) →
            P (eqb (succ p) z))
        | zero ⇒ λ (P : ∀ (c : Bool) → Type@0)
            (hf : ∀ (h : Unit) → P ff) (ht : ∀ (h : Void) → P tt) → hf unit
        | succ q, s ⇒ λ (P : ∀ (c : Bool) → Type@0)
            (hf : ∀ (h : IsFalse (eqb p q)) → P ff)
            (ht : ∀ (h : Eq p q) → P tt) →
            ih q P hf ht
        end
    end
end

A number at most b for which eqb answers ff is strictly below b.

theorem leEqbFalse : ∀ (a : ℕ) (b : ℕ) (h : Le a b) (e : IsFalse (eqb a b)) →
    Lt a b ≔
  λ (a : ℕ) →
    elim a at x -> (∀ (b : ℕ) (h : Le x b) (e : IsFalse (eqb x b)) → Lt x b)
    | zero ⇒ λ (b : ℕ) →
        elim b at z -> (∀ (h : Le 0 z) (e : IsFalse (eqb 0 z)) → Lt 0 z)
        | zero ⇒ λ (h : Unit) (e : Void) → e
        | succ q, s ⇒ λ (h : Unit) (e : Unit) → unit
        end
    | succ p, ih ⇒ λ (b : ℕ) →
        elim b at z -> (∀ (h : Le (succ p) z)
                             (e : IsFalse (eqb (succ p) z)) → Lt (succ p) z)
        | zero ⇒ λ (h : Void) (e : Unit) → h
        | succ q, s ⇒ λ (h : Le p q) (e : IsFalse (eqb p q)) → ih q h e
        end
    end
end
