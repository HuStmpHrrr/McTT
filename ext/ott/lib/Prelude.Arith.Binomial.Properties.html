Values at the edges, and Pascal's rule.

module Basic where

theorem chooseZero : ∀ (n : ℕ) → Eq (choose n 0) 1 ≔
  λ (n : ℕ) →
    elim n at x -> Eq (choose x 0) 1
    | zero ⇒ unit
    | succ p, ih ⇒ unit
    end
end

Pascal's rule, which holds by computation.

theorem pascal : ∀ (n : ℕ) (k : ℕ) →
    Eq (choose (succ n) (succ k)) (plus (choose n k) (choose n (succ k))) ≔
  λ (n : ℕ) (k : ℕ) → refl (plus (choose n k) (choose n (succ k)))
end

Above the diagonal, the coefficients vanish. In the step, Lt (succ p) (succ q) computes to Lt p q, which bounds both summands of Pascal's rule.

theorem chooseOver : ∀ (n : ℕ) (k : ℕ) (h : Lt n k) → Eq (choose n k) 0 ≔
  λ (n : ℕ) →
    elim n at x -> (∀ (k : ℕ) (h : Lt x k) → Eq (choose x k) 0)
    | zero ⇒ λ (k : ℕ) →
        elim k at z -> (∀ (h : Lt 0 z) → Eq (choose 0 z) 0)
        | zero ⇒ λ (h : Void) → h
        | succ q, s ⇒ λ (h : Unit) → unit
        end
    | succ p, ih ⇒ λ (k : ℕ) →
        elim k at z -> (∀ (h : Lt (succ p) z) → Eq (choose (succ p) z) 0)
        | zero ⇒ λ (h : Void) → h
        | succ q, s ⇒ λ (h : Lt p q) →
            plusCongBoth (choose p q) (choose p (succ q)) 0 0
              (ih q h) (ih (succ q) (leStep (succ p) q h))
        end
    end
end

The diagonal: choose n n is 1, since choose p (succ p) vanishes.

theorem chooseSelf : ∀ (n : ℕ) → Eq (choose n n) 1 ≔
  λ (n : ℕ) →
    elim n at x -> Eq (choose x x) 1
    | zero ⇒ unit
    | succ p, ih ⇒
        plusCongBoth (choose p p) (choose p (succ p)) 1 0
          ih (chooseOver p (succ p) (leRefl p))
    end
end

theorem chooseOne : ∀ (n : ℕ) → Eq (choose n 1) n ≔
  λ (n : ℕ) →
    elim n at x -> Eq (choose x 1) x
    | zero ⇒ unit
    | succ p, ih ⇒
        plusCongBoth (choose p 0) (choose p 1) 1 p (chooseZero p) ih
    end
end

end

open Basic use (chooseZero; chooseOver; chooseSelf)

Choosing a out of a + b, as a function of the two sizes, and its symmetry. Only Symmetry below uses it.

private module Split where

The coefficient choose (plus a b) a, indexed by the sizes a of the chosen part and b of the rest.

def split (a : ℕ) (b : ℕ) : ℕ ≔ choose (plus a b) a end

lemma zeroLeft : ∀ (b : ℕ) → Eq (split 0 b) 1 ≔
  λ (b : ℕ) → chooseZero b
end

lemma zeroRight : ∀ (a : ℕ) → Eq (split a 0) 1 ≔
  λ (a : ℕ) →
    trans (choose (plus a 0) a) (choose a a) 1
      (cong (λ (y : ℕ) → choose y a) (plus a 0) a (plusIdentityRight a))
      (chooseSelf a)
end

Pascal's rule for split: split (succ a) (succ b) is plus (split a (succ b)) (split (succ a) b). The second summand needs plus a (succ b) = succ (plus a b).

lemma succSucc : ∀ (a : ℕ) (b : ℕ) →
    Eq (split (succ a) (succ b)) (plus (split a (succ b)) (split (succ a) b)) ≔
  λ (a : ℕ) (b : ℕ) →
    cong (λ (y : ℕ) → plus (split a (succ b)) y)
      (choose (plus a (succ b)) (succ a)) (choose (succ (plus a b)) (succ a))
      (cong (λ (y : ℕ) → choose y (succ a)) (plus a (succ b)) (succ (plus a b))
         (plusSucc a b))
end

split is symmetric. The induction on a and then on b follows the rule succSucc, which is symmetric up to the order of the summands.

lemma symm : ∀ (a : ℕ) (b : ℕ) → Eq (split a b) (split b a) ≔
  λ (a : ℕ) →
    elim a at x -> (∀ (b : ℕ) → Eq (split x b) (split b x))
    | zero ⇒ λ (b : ℕ) →
        trans (split 0 b) 1 (split b 0) (zeroLeft b)
          (sym (split b 0) 1 (zeroRight b))
    | succ p, ih ⇒ λ (b : ℕ) →
        elim b at y -> Eq (split (succ p) y) (split y (succ p))
        | zero ⇒
            trans (split (succ p) 0) 1 (split 0 (succ p)) (zeroRight (succ p))
              (sym (split 0 (succ p)) 1 (zeroLeft (succ p)))
        | succ q, s ⇒
            let lhs ≔ split (succ p) (succ q);
                la ≔ split p (succ q);
                lb ≔ split (succ p) q;
                ra ≔ split q (succ p);
                rb ≔ split (succ q) p;
                rhs ≔ split (succ q) (succ p);
                mid : Eq (plus la lb) (plus rb ra) ≔
                  plusCongBoth la lb rb ra (ih (succ q)) s;
                swap : Eq (plus rb ra) (plus ra rb) ≔ plusComm rb ra
            in trans lhs (plus la lb) rhs (succSucc p q)
                 (trans (plus la lb) (plus rb ra) rhs mid
                    (trans (plus rb ra) (plus ra rb) rhs swap
                       (sym rhs (plus ra rb) (succSucc q p))))
            end
        end
    end
end

end

Symmetry of the binomial coefficients.

module Symmetry where

Symmetry: choosing a of plus a b is choosing the b left out.

theorem chooseSymm : ∀ (a : ℕ) (b : ℕ) →
    Eq (choose (plus a b) a) (choose (plus a b) b) ≔
  λ (a : ℕ) (b : ℕ) →
    trans (choose (plus a b) a) (choose (plus b a) b) (choose (plus a b) b)
      (Split.symm a b)
      (cong (λ (y : ℕ) → choose y b) (plus b a) (plus a b) (plusComm b a))
end

Symmetry with subtraction: below n, choose n k is choose n (n - k).

theorem chooseSymmSub : ∀ (n : ℕ) (k : ℕ) (h : Le k n) →
    Eq (choose n k) (choose n (sub n k)) ≔
  λ (n : ℕ) (k : ℕ) (h : Le k n) →
    let m ≔ sub n k;
        back : Eq (plus m k) n ≔ plusSubCancel n k h;
        forth : Eq n (plus m k) ≔ sym (plus m k) n back
    in trans (choose n k) (choose (plus m k) k) (choose n m)
         (cong (λ (y : ℕ) → choose y k) n (plus m k) forth)
         (trans (choose (plus m k) k) (choose (plus m k) m) (choose n m)
            (sym (choose (plus m k) m) (choose (plus m k) k) (chooseSymm m k))
            (cong (λ (y : ℕ) → choose y m) (plus m k) n back))
    end
end

end

Sums of the rows of Pascal's triangle.

module Row where

A prefix of row succ p sums to the same prefix of row p plus the prefix of row p one shorter, by Pascal's rule on each entry.

fact rowPrefixSucc : ∀ (p : ℕ) (m : ℕ) →
    Eq (sumTo (choose (succ p)) (succ m))
       (plus (sumTo (choose p) (succ m)) (sumTo (choose p) m)) ≔
  λ (p : ℕ) (m : ℕ) →
    elim m at x -> Eq (sumTo (choose (succ p)) (succ x))
                         (plus (sumTo (choose p) (succ x)) (sumTo (choose p) x))
    | zero ⇒
        (* The goal computes to [Eq 1 (plus (plus (choose p 0) 0) 0)]. *)
        sym (plus (plus (choose p 0) 0) 0) 1
          (trans (plus (plus (choose p 0) 0) 0) (plus (choose p 0) 0) 1
             (plusIdentityRight (plus (choose p 0) 0))
             (trans (plus (choose p 0) 0) (choose p 0) 1 (plusIdentityRight (choose p 0))
                (chooseZero p)))
    | succ q, ih ⇒
        let a ≔ choose p q;
            b ≔ choose p (succ q);
            c ≔ sumTo (choose p) (succ q);
            d ≔ sumTo (choose p) q;
            lhs ≔ plus (plus a b) (sumTo (choose (succ p)) (succ q));
            step : Eq lhs (plus (plus a b) (plus c d)) ≔
              cong (λ (y : ℕ) → plus (plus a b) y)
                (sumTo (choose (succ p)) (succ q)) (plus c d) ih;
            swap : Eq (plus (plus a b) (plus c d))
                      (plus (plus b a) (plus c d)) ≔
              cong (λ (y : ℕ) → plus y (plus c d)) (plus a b) (plus b a)
                (plusComm a b)
        in trans lhs (plus (plus a b) (plus c d)) (plus (plus b c) (plus a d))
             step
             (trans (plus (plus a b) (plus c d)) (plus (plus b a) (plus c d))
                (plus (plus b c) (plus a d)) swap (plusSwapMiddle b a c d))
        end
    end
end

Row n of Pascal's triangle sums to pow 2 n. Row succ p sums to twice row p by rowPrefixSucc, since choose p (succ p) vanishes; and pow 2 (succ p) computes to plus P (plus P 0).

theorem rowSum : ∀ (n : ℕ) → Eq (sumTo (choose n) (succ n)) (pow 2 n) ≔
  λ (n : ℕ) →
    elim n at x -> Eq (sumTo (choose x) (succ x)) (pow 2 x)
    | zero ⇒ unit
    | succ p, ih ⇒
        let s ≔ sumTo (choose p) (succ p);
            P ≔ pow 2 p;
            longer ≔ sumTo (choose p) (succ (succ p));
            dropTop : Eq longer s ≔
              cong (λ (y : ℕ) → plus y s) (choose p (succ p)) 0
                (chooseOver p (succ p) (leRefl p));
            first : Eq longer P ≔ trans longer s P dropTop ih;
            second : Eq s (plus P 0) ≔
              trans s P (plus P 0) ih (sym (plus P 0) P (plusIdentityRight P))
        in trans (sumTo (choose (succ p)) (succ (succ p))) (plus longer s)
             (plus P (plus P 0))
             (rowPrefixSucc p (succ p))
             (plusCongBoth longer s P (plus P 0) first second)
        end
    end
end

end
