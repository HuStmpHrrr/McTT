A Kripke weakening is a substitution, in the form the gluing proofs use. It is not Γ ⊢ʷ φ : Δ; see Definitions.
Lemma kripke_escape : forall Γ Δ φ,
    Γ ⊢ᵏ φ : Δ ->
    Γ ⊢ˢ (ι φ) : Δ.
Proof.
  intros * H; induction H;
    saturate_ctx_sub;
    match goal with H : wk_eq _ _ |- _ => rewrite H end.
  - rewrite sb_of_wk_id; eassumption.
  - rewrite sb_of_wk_compose, sb_of_wk_shift.
    saturate_sub.
    rewrite <- (sb_compose_id_left (Wk ⨟ (ι ψ))).
    mauto 4.
Qed.

End Fixed_GCtx.

Ltac saturate_kripke_escape :=
  match_by_head wk_kripke ltac:(fun H => pose proof (kripke_escape _ _ _ H));
  clear_dups.

Lemma kripke_id : forall Γ, ⊢ Γ -> Γ ⊢ᵏ wk_id : Γ.
Proof. intros; eapply wk_kripke_id; [ mauto 2 | reflexivity ]. Qed.

Lemma kripke_shift : forall Γ κ A, ⊢ Γ ▹{κ} A -> Γ ▹{κ} A ⊢ᵏ ↑ : Γ.
Proof.
  intros * H; eapply wk_kripke_shift;
    [ apply kripke_id; eassumption
    | mauto 3
    | symmetry; apply wk_compose_id_right ].
Qed.

The codomain may always be coarsened: this is the closure property the subtyping cases need, and the reason for the refinement premise.
Lemma kripke_ctxsub : forall Γ Δ Δ' φ,
    Γ ⊢ᵏ φ : Δ ->
    Δ ⊆ Δ' ->
    Γ ⊢ᵏ φ : Δ'.
Proof.
  intros * H ?; destruct H.
  - eapply wk_kripke_id; [ etransitivity; eassumption | eassumption ].
  - eapply wk_kripke_shift; [ eassumption | etransitivity; eassumption | eassumption ].
Qed.

Lemma kripke_compose : forall Γ Δ Θ φ ψ,
    Γ ⊢ᵏ ψ : Δ ->
    Δ ⊢ᵏ φ : Θ ->
    Γ ⊢ᵏ φ ⊙ ψ : Θ.
Proof.
  intros * Hψ Hφ; revert Γ ψ Hψ.
  induction Hφ; intros ? ? Hψ;
    match goal with H : wk_eq _ _ |- _ => rewrite H end.
  - rewrite wk_compose_id_left.
    eapply kripke_ctxsub; eassumption.
  - rewrite wk_compose_assoc.
    eapply wk_kripke_shift; [ | eassumption | reflexivity ].
    eauto.
Qed.

Every Kripke weakening is ⇑^n, where n is the difference of the two lengths. The gluing model needs this because readback turns a de Bruijn level into an index by counting from the length of the target context.
Lemma kripke_shiftn : forall Γ Δ φ,
    Γ ⊢ᵏ φ : Δ ->
    length Δ <= length Γ /\ wk_eq φ (wk_shiftn (length Γ - length Δ)).
Proof.
  intros * H; induction H as [ | ? ? ? ? ? ? ? [Hle Hψ] ];
    match goal with H : ctx_sub _ _ _ _ |- _ => pose proof (ctx_sub_length _ _ _ _ H) end;
    match goal with H : wk_eq _ _ |- _ => rewrite H end;
    unfold_ops; simpl in *;
    (split; [ lia | intro x ]).
  - lia.
  - rewrite Hψ; lia.
Qed.

Corollary kripke_preserves_exp : forall Γ Δ A M φ,
    Δ ⊢ M : A ->
    Γ ⊢ᵏ φ : Δ ->
    Γ ⊢ M[φ]ʷ : A[φ]ʷ.
Proof.
  intros * ? ?%kripke_escape.
  rewrite <- (exp_sub_of_wk M φ), <- (exp_sub_of_wk A φ); mauto 2.

q φ is not a Kripke weakening, since it is not a shift. It is still a substitution, which is all the Π clauses of the gluing model need to type a codomain in the extended context.
The domain is a type of a sort Srt of the entry's kind κ (wf_ctx_extend).
Corollary kripke_q_escape : forall Γ Δ κ A φ Srt,
    Δ ⊢ A : Srt ->
    sort_kind Srt κ ->
    Γ ⊢ᵏ φ : Δ ->
    Γ ▹{κ} A[φ]ʷ ⊢ˢ (ι (wk_q φ)) : Δ ▹{κ} A.
Proof.
  intros * HA HS Hφ%kripke_escape.
  rewrite sb_q_of_wk, <- (exp_sub_of_wk A φ).
  eapply wf_sub_q; [ exact Hφ | exact HA | exact HS |].
  inversion HS; subst; cbn; [ change (Typeω@i) with ((Typeω@i : exp)[ι φ]) | change Propₘ with ((Propₘ : exp)[ι φ])
    | change LevelSort with ((LevelSort : exp)[ι φ]) ]; mauto 2.
Qed.

Corollary kripke_preserves_exp_q : forall Γ Δ κ A B M φ Srt,
    Δ ▹{κ} A ⊢ M : B ->
    Δ ⊢ A : Srt ->
    sort_kind Srt κ ->
    Γ ⊢ᵏ φ : Δ ->
    Γ ▹{κ} A[φ]ʷ ⊢ M[wk_q φ]ʷ : B[wk_q φ]ʷ.
Proof.
  intros.
  rewrite <- (exp_sub_of_wk M), <- (exp_sub_of_wk B).
  eauto using kripke_q_escape, sub_preserves_exp.
Qed.

Corollary kripke_preserves_typ_q : forall Γ Δ κ A B φ Srt j,
    Δ ▹{κ} A ⊢ B : Typeω@j ->
    Δ ⊢ A : Srt ->
    sort_kind Srt κ ->
    Γ ⊢ᵏ φ : Δ ->
    Γ ▹{κ} A[φ]ʷ ⊢ B[wk_q φ]ʷ : Typeω@j.
Proof.
  intros.
  assert (Γ ▹{κ} A[φ]ʷ ⊢ exp_wk B (wk_q φ) : exp_wk (a_typ j) (wk_q φ))
    by (eapply kripke_preserves_exp_q; eassumption).
  assumption.
Qed.

Extending by the new variable as a term, rather than by the variable entry of q φ: the two agree on every term, and a type of the extended context mentions no slot at the new position.
Corollary kripke_q_var_eq : forall Γ Δ κ A B M φ Srt,
    Δ ▹{κ} A ⊢ M : B ->
    Δ ⊢ A : Srt ->
    sort_kind Srt κ ->
    Γ ⊢ᵏ φ : Δ ->
    Γ ▹{κ} A[φ]ʷ ⊢ M[ι (φ ⊙ ↑),,#0] ≈ M[wk_q φ]ʷ : B[wk_q φ]ʷ.
Proof.
  intros * HM HA HS Hφ.
  rewrite <- (exp_sub_of_wk M), <- (exp_sub_of_wk B).
  apply wf_exp_eq_sym.
  pose proof (kripke_q_escape _ _ _ _ _ _ HA HS Hφ) as Hq.
  pose proof (wf_sub_dom _ _ _ _ _ Hq) as HC.
  assert (Hk : Γ ▹{κ} A[φ]ʷ ⊢ᵏ φ ⊙ ↑ : Δ) by mauto 3.
  pose proof (kripke_escape _ _ _ Hk) as Hs.
  assert (Hs' : Γ ▹{κ} A[φ]ʷ ⊢ˢ ι (φ ⊙ ↑),,#0 : Δ ▹{κ} A).
  { eapply wf_sub_extend; [ exact Hs | exact HA | exact HS |].
    rewrite exp_sub_of_wk, <- exp_wk_wk; mauto 3. }
  eapply sub_eq_preserves_exp; [ exact HM |].
  constructor; [ exact Hq | exact Hs' | |].
  - intros x C Hx.
    replace (#x[ι (φ ⊙ ↑),,#0]) with (#x[ι wk_q φ]) by (destruct x; reflexivity).
    apply wf_exp_eq_refl.
    exact (wf_sub_apply _ _ _ _ _ Hq _ _ Hx).
  - intros x U Hx; inversion Hx; subst.
    exact (wf_sub_mod_image _ _ _ _ _ _ _ Hq Hx).
Qed.

Corollary shift_var_eq : forall Δ κ A B M u,
    Δ ▹{κ} A ⊢ M : B ->
    Δ ⊢ A : abs_kind_sort κ (universe_index_term u) ->
    Δ ▹{κ} A ⊢ M[ι ↑,,#0] ≈ M : B.
Proof.
  intros * HM HA.
  assert (Hk : Δ ⊢ᵏ wk_id : Δ) by (apply kripke_id; mauto 2).
  pose proof (kripke_q_var_eq_univ _ _ _ _ _ _ _ _ HM HA Hk) as H.
  rewrite exp_wk_id, !(exp_wk_id_ext _ _ wk_q_id) in H.
  replace (M[ι ↑,,#0]) with (M[ι (wk_id ⊙ ↑),,#0]); [ exact H |].
  apply exp_sub_sb_eq; intros [| x]; reflexivity.
Qed.
