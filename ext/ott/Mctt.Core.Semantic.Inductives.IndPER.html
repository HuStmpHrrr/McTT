Mctt.Core.Semantic.Inductives.IndPER

Groups of Inductive Types in the PER Model, Canonically

The clause of per_univ_elem_core for a member of a group takes the relations of the members' telescopes as arguments. Here they are fixed: the relation at a position of a member's telescope is the head PER of the type there, read in the environment of the left values (per_head, on the diagonal), which the PER model determines. per_univ_elem_indty_canon builds the clause from the relatedness of the types alone, and the views' fresh-variable conditions at the two sides' uniform parameters, which the canonical construction of the group proves from the view equations.

The type at a position of a telescope, and of a constructor's non-recursive arguments, of the member m.
  Definition nth_ctyp (T : list centry) (k : nat) : typ :=
    match List.nth_error T k with Some (ce_ass _ A) => A | _ => a_prop end.

The non-uniform parameters of the member m, none if there is no such member.
  Definition member_nparams (D : ind_decl) (m : nat) : list centry :=
    match ind_member D m with Some Mm => im_nparams Mm | None => nil end.

The type of the k-th non-recursive argument of the constructor c of the member m.
  Definition nth_atyp (D : ind_decl) (m : nat) (c : String.string) (k : nat) : typ :=
    match ind_member D m with
    | Some Mm => match ind_con Mm c with Some gs => List.nth k (cargs_nonrec gs) a_prop | None => a_prop end
    | None => a_prop
    end.

The telescope of the non-recursive arguments of the constructor c of the member m, empty if there is no such constructor.
  Definition member_con_tele (D : ind_decl) (m : nat) (c : String.string) : list centry :=
    match ind_member D m with
    | Some Mm => match ind_con Mm c with Some gs => cargs_nonrec_tele gs | None => nil end
    | None => nil
    end.

The relation of a non-uniform parameter, after the values vs of those before it: the head PER of its type there, in the environment of the components (components_env).
  Definition canon_nparams_rel (D : ind_decl) (ups : list domain) : nat -> nat -> list domain -> relation domain :=
    fun m k vs => per_head (nth_ctyp (List.rev (member_nparams D m)) k) (nth_ctyp (List.rev (member_nparams D m)) k)
                    (components_env D ups (List.rev (member_nparams D m)) vs)
                    (components_env D ups (List.rev (member_nparams D m)) vs).

The relation of a non-recursive argument, after the non-uniform parameters ms and the arguments vs before it: the head PER of its type there.
  Definition canon_args_rel (D : ind_decl) (ups : list domain) : nat -> list domain -> String.string -> nat -> list domain -> relation domain :=
    fun m ms c k vs => per_head (nth_atyp D m c k) (nth_atyp D m c k)
                         (components_env D ups (List.rev (member_nparams D m) ++ member_con_tele D m c) (ms ++ vs))
                         (components_env D ups (List.rev (member_nparams D m) ++ member_con_tele D m c) (ms ++ vs)).

The relation of a domain of a shape, after the values pre of its components, of the telescope Ts: the head PER of the domain there.
  Definition canon_doms_rel (D : ind_decl) (ups : list domain) : nat -> list domain -> list centry -> exp -> relation domain :=
    fun _ pre Ts B => per_head B B (components_env D ups Ts pre) (components_env D ups Ts pre).

  Lemma nth_ctyp_eq : forall D m Mm k κ A, ind_member D m = Some Mm -> List.nth_error (List.rev (im_nparams Mm)) k = Some (ce_ass κ A) ->
      nth_ctyp (List.rev (member_nparams D m)) k = A.
  Proof. intros * HM H; unfold nth_ctyp, member_nparams; rewrite HM, H; reflexivity. Qed.

  Lemma nth_atyp_eq : forall D m Mm c gs k A, ind_member D m = Some Mm -> ind_con Mm c = Some gs ->
      List.nth_error (cargs_nonrec gs) k = Some A -> nth_atyp D m c k = A.
  Proof. intros * HM Hc H; unfold nth_atyp; rewrite HM, Hc; apply List.nth_error_nth; exact H. Qed.

  Lemma member_nparams_eq : forall D m Mm, ind_member D m = Some Mm -> member_nparams D m = im_nparams Mm.
  Proof. intros * HM; unfold member_nparams; rewrite HM; reflexivity. Qed.

  Lemma member_con_tele_eq : forall D m Mm c gs, ind_member D m = Some Mm -> ind_con Mm c = Some gs ->
      member_con_tele D m c = cargs_nonrec_tele gs.
  Proof. intros * HM Hc; unfold member_con_tele; rewrite HM, Hc; reflexivity. Qed.

The relation of a domain at the components before it is the head PER of the domain in the environment of all of a shape's components: the environment reads only as many entries of the telescope as there are values (components_env_shape_prefix).
  Lemma canon_doms_rel_full : forall D ups m (Mm : imember) gs1 t (ms ws ys : list domain) k B,
      List.length ms = List.length (im_nparams Mm) -> List.length ws = List.length (cargs_nonrec gs1) ->
      List.length ys = k -> k <= List.length (ptyp_doms t) ->
      canon_doms_rel D ups m (ms ++ ws ++ ys) (List.rev (im_nparams Mm) ++ cargs_nonrec_tele gs1 ++ List.firstn k (shape_doms_tele t)) B
      = per_head B B (components_env D ups (List.rev (im_nparams Mm) ++ cargs_nonrec_tele gs1 ++ shape_doms_tele t) (ms ++ ws ++ ys))
                     (components_env D ups (List.rev (im_nparams Mm) ++ cargs_nonrec_tele gs1 ++ shape_doms_tele t) (ms ++ ws ++ ys)).
  Proof. intros * Hm Hw Hy Hk; unfold canon_doms_rel; rewrite <- Hy, <- components_env_shape_prefix by lia; reflexivity. Qed.

The head PER on the diagonal is the relation of a type related to any other.
  Lemma per_head_diag : forall {S ρ i a a' R},
      ⟦ S ⟧ ρ ↘ a -> DF a ≈ a' ∈ per_univ_elem i ↘ R ->
      DF a ≈ a' ∈ per_univ_elem i ↘ (per_head S S ρ ρ).
  Proof.
    intros * Ha HR.
    assert (HRaa : DF a ≈ a ∈ per_univ_elem i ↘ R) by (eapply per_univ_trans; [ exact HR | apply per_univ_sym; exact HR ]).
    eapply per_univ_elem_resp_iff; [ exact HR |].
    eapply per_head_iff; eassumption.
  Qed.

The types of the members' telescopes at two related prefixes, at ups and at a second base u2: their values related.
    Definition nparams_related (u2 : list domain) : Prop :=
      forall m Mm k κ A vs vs', ind_member D m = Some Mm ->
        List.nth_error (List.rev (im_nparams Mm)) k = Some (ce_ass κ A) -> tele_prefix_rel (nr m) k vs vs' ->
        exists a a', ⟦ A ⟧ components_env D ups (List.rev (im_nparams Mm)) vs ↘ a /\
          ⟦ A ⟧ components_env D u2 (List.rev (im_nparams Mm)) vs' ↘ a' /\ Dom a ≈ a' ∈ per_univ u /\

          not_level_sort a.

The evidence at a position, at the head PER.
    Lemma canon_nparams_evals : forall m Mm k κ A vs vs', ind_member D m = Some Mm ->
        List.nth_error (List.rev (im_nparams Mm)) k = Some (ce_ass κ A) -> tele_prefix_rel (nr m) k vs vs' ->
        rel_mod_eval (per_univ_elem u) A (components_env D ups (List.rev (im_nparams Mm)) vs)
          A (components_env D ups' (List.rev (im_nparams Mm)) vs') (nr m k vs).
    Proof.
      intros * HM Hk Hp; destruct (Hn _ _ _ _ _ _ _ HM Hk Hp) as (a & a' & Ea & Ea' & [R HR] & Hna).
      econstructor; [ exact Ea | exact Ea' |].
      unfold nr, canon_nparams_rel; rewrite (nth_ctyp_eq _ _ _ _ _ _ HM Hk), (member_nparams_eq _ _ _ HM).
      exact (per_head_diag Ea HR).
    Qed.

    Lemma canon_nparams_per : nparams_per D nr.
    Proof.
      intros m Mm k κ A vs vs' HM Hk Hp; destruct (canon_nparams_evals _ _ _ _ _ _ _ HM Hk Hp) as [a a' _ _ HR].
      eapply per_elem_PER; exact HR.
    Qed.

Invariance, by induction on the position: the prefixes on the right are related to themselves, by the positions before.
    Lemma canon_nparams_invariant_all : forall m Mm, ind_member D m = Some Mm ->
        forall k, k <= List.length (List.rev (im_nparams Mm)) ->
        (forall j vs vs', j < k -> tele_prefix_rel (nr m) j vs vs' -> (nr m j vs <~> nr m j vs')).
    Proof.
      intros m Mm HM; induction k as [| k IH]; intros Hk j vs vs' Hj Hp; [ lia |].
      destruct (Nat.lt_ge_cases j k) as [Hjk | Hjk]; [ apply IH; [ lia | exact Hjk | exact Hp ] |].
      assert (j = k) as -> by lia.
      destruct (nparams_lt_ass D m Mm k Hass HM ltac:(lia)) as (κ & A & HA).
      assert (HP : forall j vs vs', j < k -> tele_prefix_rel (nr m) j vs vs' -> PER (nr m j vs)).
      { intros j0 ? ? Hj0 Hp0; destruct (nparams_lt_ass D m Mm j0 Hass HM ltac:(lia)) as (κ0 & A0 & HA0).
        eapply canon_nparams_per; eassumption. }
      pose proof (tele_prefix_rel_refl_r k (nr m) HP (IH ltac:(lia)) k vs vs' (le_n _) Hp) as Hp'.
      destruct (canon_nparams_evals _ _ _ _ _ _ _ HM HA Hp) as [a a' Ea Ea' HR1].
      destruct (canon_nparams_evals _ _ _ _ _ _ _ HM HA Hp') as [b b' Eb Eb' HR2].
      functional_eval_rewrite_clear.
      eapply per_univ_elem_left_irrel; [ exact HR1 | exact HR2 ].
    Qed.

    Lemma canon_nparams_invariant : nparams_inv D nr.
    Proof.
      intros m Mm k κ A vs vs' HM Hk Hp; assert (Hkl : k < List.length (List.rev (im_nparams Mm))) by (apply List.nth_error_Some; congruence).
      exact (canon_nparams_invariant_all m Mm HM _ (le_n _) k vs vs' Hkl Hp).
    Qed.

    Lemma canon_nparams_per' : forall m Mm k vs vs', ind_member D m = Some Mm -> k < List.length (List.rev (im_nparams Mm)) ->
        tele_prefix_rel (nr m) k vs vs' -> PER (nr m k vs).
    Proof. intros * HM Hk Hp; destruct (nparams_lt_ass D m Mm k Hass HM Hk) as (κ & A & HA); eapply canon_nparams_per; eassumption. Qed.

    Lemma canon_nparams_invariant' : forall m Mm k vs vs', ind_member D m = Some Mm -> k < List.length (List.rev (im_nparams Mm)) ->
        tele_prefix_rel (nr m) k vs vs' -> (nr m k vs <~> nr m k vs').
    Proof. intros * HM Hk Hp; destruct (nparams_lt_ass D m Mm k Hass HM Hk) as (κ & A & HA); eapply canon_nparams_invariant; eassumption. Qed.

    Lemma canon_nparams_rel_refl_r : forall m Mm a b, ind_member D m = Some Mm ->
        tele_rel (nr m) (List.rev (im_nparams Mm)) a b -> tele_rel (nr m) (List.rev (im_nparams Mm)) b b.
    Proof.
      intros * HM H.
      eapply (tele_rel_refl_r (List.length (List.rev (im_nparams Mm))) (nr m)
                (fun k vs vs' Hk => canon_nparams_per' m Mm k vs vs' HM Hk)
                (fun k vs vs' Hk => canon_nparams_invariant' m Mm k vs vs' HM Hk)); [ lia | exact H ].
    Qed.

    Lemma canon_args_evals : forall m Mm ms ms' c gs k A vs vs', ind_member D m = Some Mm ->
        tele_rel (nr m) (List.rev (im_nparams Mm)) ms ms' -> ind_con Mm c = Some gs ->
        List.nth_error (cargs_nonrec gs) k = Some A -> tele_prefix_rel (ar m ms c) k vs vs' ->
        rel_mod_eval (per_univ_elem u) A (components_env D ups (List.rev (im_nparams Mm) ++ cargs_nonrec_tele gs) (ms ++ vs))
          A (components_env D ups' (List.rev (im_nparams Mm) ++ cargs_nonrec_tele gs) (ms' ++ vs')) (ar m ms c k vs).
    Proof.
      intros * HM Hm Hc Hk Hp; destruct (Ha _ _ _ _ _ _ _ _ _ _ HM Hm Hc Hk Hp) as (a & a' & Ea & Ea' & [R HR] & Hna).
      econstructor; [ exact Ea | exact Ea' |].
      unfold ar, canon_args_rel; rewrite (nth_atyp_eq _ _ _ _ _ _ _ HM Hc Hk), (member_nparams_eq _ _ _ HM),
        (member_con_tele_eq _ _ _ _ _ HM Hc).
      exact (per_head_diag Ea HR).
    Qed.

      intros m Mm ms ms' c gs k A vs vs' HM Hm Hc Hk Hp; destruct (canon_args_evals _ _ _ _ _ _ _ _ _ _ HM Hm Hc Hk Hp) as [a a' _ _ HR].

    Lemma canon_args_invariant_all : forall m Mm ms ms' c gs, ind_member D m = Some Mm ->
        tele_rel (nr m) (List.rev (im_nparams Mm)) ms ms' -> ind_con Mm c = Some gs ->
        forall k, k <= List.length (cargs_nonrec gs) ->
        forall j vs vs', j < k -> tele_prefix_rel (ar m ms c) j vs vs' -> (ar m ms c j vs <~> ar m ms' c j vs').
    Proof.
      intros m Mm ms ms' c gs HM Hm Hc.
      pose proof (canon_nparams_rel_refl_r _ _ _ _ HM Hm) as Hm'.
      induction k as [| k IH]; intros Hk j vs vs' Hj Hp; [ lia |].
      destruct (Nat.lt_ge_cases j k) as [Hjk | Hjk]; [ apply IH; [ lia | exact Hjk | exact Hp ] |].
      assert (j = k) as -> by lia.
      destruct (nth_error_lt_some' (cargs_nonrec gs) k ltac:(lia)) as [A HA].
      assert (Hp' : tele_prefix_rel (ar m ms' c) k vs' vs').
      { eapply (tele_prefix_rel_cross_refl (ar m ms c) (ar m ms' c) k k); [ lia | | | exact Hp ].

      functional_eval_rewrite_clear.
      eapply per_univ_elem_left_irrel; [ exact HR1 | exact HR2 ].
    Qed.

    Lemma canon_args_rel_refl_r : forall m Mm ms ms' c gs k ws ws', ind_member D m = Some Mm ->
        tele_rel (nr m) (List.rev (im_nparams Mm)) ms ms' -> ind_con Mm c = Some gs ->

The relation of a domain at the components before it is the head PER of the domain in the environment of all of a shape's components: the environment reads only as many entries of the telescope as there are values (components_env_shape_prefix).
    Lemma canon_doms_rel_at : forall m (Mm : imember) (ns : list domain) c gs1 t (ms ws ws' ys ys' : list domain) k B,
        tele_rel (nr m) (List.rev (im_nparams Mm)) ms ns ->
        tele_prefix_rel (ar m ms c) (List.length (cargs_nonrec gs1)) ws ws' ->
        tele_prefix_rel (shape_doms_rel (dr m) (List.rev (im_nparams Mm)) (cargs_nonrec_tele gs1) ms ws t) k ys ys' ->
        k <= List.length (ptyp_doms t) ->
        dr m (ms ++ ws ++ ys) (List.rev (im_nparams Mm) ++ cargs_nonrec_tele gs1 ++ List.firstn k (shape_doms_tele t)) B
        = per_head B B (components_env D ups (List.rev (im_nparams Mm) ++ cargs_nonrec_tele gs1 ++ shape_doms_tele t) (ms ++ ws ++ ys))
                       (components_env D ups (List.rev (im_nparams Mm) ++ cargs_nonrec_tele gs1 ++ shape_doms_tele t) (ms ++ ws ++ ys)).
    Proof.
      intros * Hm Hw Hy Hk.
      pose proof (proj1 (tele_rel_iff _ _ _ _) Hm) as (Hlm & _ & _).
      pose proof (proj1 (tele_prefix_rel_iff _ _ _ _) Hw) as (Hlw & _ & _).
      pose proof (proj1 (tele_prefix_rel_iff _ _ _ _) Hy) as (Hly & _ & _).
      rewrite List.length_rev in Hlm.
      exact (canon_doms_rel_full D ups m Mm gs1 t ms ws ys k B Hlm Hlw Hly Hk).
    Qed.
