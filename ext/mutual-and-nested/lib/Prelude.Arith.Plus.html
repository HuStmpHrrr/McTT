Prelude›Arith›Plus

Addition and its laws, stated with Prelude›Arith›Equality.

module Prelude›Arith›Plus where

import Prelude›Builtin›Nat use (succ)
import Prelude›Logic use (absurd)
import Prelude›Arith›Equality use (Eq; refl; sym; trans; cong)

def plus (n : ℕ) (m : ℕ) : ℕ ≔
  elim n at y -> ℕ
  | zero ⇒ m
  | succ p, r ⇒ succ r
  end
end

Addition of zero and of a successor on the right.

module Basic where

theorem plusIdentityRight : ∀ (n : ℕ) → Eq (plus n 0) n ≔
  λ (n : ℕ) →
    elim n at x -> Eq (plus x 0) x
    | zero ⇒ ⋆
    | succ p, ih ⇒ ih
    end
end

theorem plusSucc : ∀ (n : ℕ) (m : ℕ) → Eq (plus n (succ m)) (succ (plus n m)) ≔
  λ (n : ℕ) (m : ℕ) →
    elim n at x -> Eq (plus x (succ m)) (succ (plus x m))
    | zero ⇒ refl m
    | succ p, ih ⇒ ih
    end
end

end

open Basic use (plusIdentityRight; plusSucc)

Commutativity and associativity, and their consequences.

module Algebra where

theorem plusComm : ∀ (a : ℕ) (b : ℕ) → Eq (plus a b) (plus b a) ≔
  λ (a : ℕ) (b : ℕ) →
    elim a at x -> Eq (plus x b) (plus b x)
    | zero ⇒ sym (plus b 0) b (plusIdentityRight b)
    | succ p, ih ⇒
        let back : Eq (succ (plus b p)) (plus b (succ p)) ≔
              sym (plus b (succ p)) (succ (plus b p)) (plusSucc b p)
        in trans (succ (plus p b)) (succ (plus b p)) (plus b (succ p)) ih back
        end
    end
end

theorem plusAssoc : ∀ (a : ℕ) (b : ℕ) (c : ℕ) →
    Eq (plus (plus a b) c) (plus a (plus b c)) ≔
  λ (a : ℕ) (b : ℕ) (c : ℕ) →
    elim a at x -> Eq (plus (plus x b) c) (plus x (plus b c))
    | zero ⇒ refl (plus b c)
    | succ p, ih ⇒ ih
    end
end

The first summand can be moved past the second.

theorem plusLeftComm : ∀ (a : ℕ) (b : ℕ) (c : ℕ) →
    Eq (plus a (plus b c)) (plus b (plus a c)) ≔
  λ (a : ℕ) (b : ℕ) (c : ℕ) →
    elim a at x -> Eq (plus x (plus b c)) (plus b (plus x c))
    | zero ⇒ refl (plus b c)
    | succ p, ih ⇒
        let back : Eq (succ (plus b (plus p c))) (plus b (succ (plus p c))) ≔
              sym (plus b (succ (plus p c))) (succ (plus b (plus p c)))
                (plusSucc b (plus p c))
        in trans (succ (plus p (plus b c))) (succ (plus b (plus p c)))
             (plus b (succ (plus p c))) ih back
        end
    end
end

The last two summands of a sum can be exchanged.

theorem plusRightComm : ∀ (a : ℕ) (b : ℕ) (c : ℕ) →
    Eq (plus (plus a b) c) (plus (plus a c) b) ≔
  λ (a : ℕ) (b : ℕ) (c : ℕ) →
    trans (plus (plus a b) c) (plus a (plus b c)) (plus (plus a c) b)
      (plusAssoc a b c)
      (trans (plus a (plus b c)) (plus a (plus c b)) (plus (plus a c) b)
         (cong (λ (y : ℕ) → plus a y) (plus b c) (plus c b) (plusComm b c))
         (sym (plus (plus a c) b) (plus a (plus c b)) (plusAssoc a c b)))
end

Equal summands have equal sums.

theorem plusCongBoth : ∀ (a : ℕ) (b : ℕ) (c : ℕ) (d : ℕ)
    (hac : Eq a c) (hbd : Eq b d) → Eq (plus a b) (plus c d) ≔
  λ (a : ℕ) (b : ℕ) (c : ℕ) (d : ℕ) (hac : Eq a c) (hbd : Eq b d) →
    trans (plus a b) (plus c b) (plus c d)
      (cong (λ (y : ℕ) → plus y b) a c hac)
      (cong (λ (y : ℕ) → plus c y) b d hbd)
end

The two middle summands of a sum of sums can be exchanged.

theorem plusSwapMiddle : ∀ (a : ℕ) (b : ℕ) (c : ℕ) (d : ℕ) →
    Eq (plus (plus a b) (plus c d)) (plus (plus a c) (plus b d)) ≔
  λ (a : ℕ) (b : ℕ) (c : ℕ) (d : ℕ) →
    let lhs ≔ plus (plus a b) (plus c d);
        mid ≔ plus a (plus b (plus c d));
        inner ≔ plus a (plus c (plus b d));
        rhs ≔ plus (plus a c) (plus b d);
        swap : Eq mid inner ≔
          cong (λ (y : ℕ) → plus a y) (plus b (plus c d)) (plus c (plus b d))
            (plusLeftComm b c d);
        back : Eq inner rhs ≔ sym rhs inner (plusAssoc a c (plus b d))
    in trans lhs mid rhs (plusAssoc a b (plus c d))
         (trans mid inner rhs swap back)
    end
end

end

Injectivity of the successor, cancellation, and zero sums.

module Cancel where

The successor is injective; since Eq (succ a) (succ b) computes to Eq a b, the proof is the hypothesis itself.

theorem succInj : ∀ (a : ℕ) (b : ℕ) (h : Eq (succ a) (succ b)) → Eq a b ≔
  λ (a : ℕ) (b : ℕ) (h : Eq (succ a) (succ b)) → h
end

theorem plusCancelLeft : ∀ (a : ℕ) (b : ℕ) (c : ℕ)
    (h : Eq (plus a b) (plus a c)) → Eq b c ≔
  λ (a : ℕ) (b : ℕ) (c : ℕ) →
    elim a at x -> (∀ (h : Eq (plus x b) (plus x c)) → Eq b c)
    | zero ⇒ λ (h : Eq b c) → h
    | succ p, ih ⇒ λ (h : Eq (plus p b) (plus p c)) → ih h
    end
end

Right cancellation, reduced to left cancellation by commutativity.

theorem plusCancelRight : ∀ (a : ℕ) (b : ℕ) (c : ℕ)
    (h : Eq (plus a c) (plus b c)) → Eq a b ≔
  λ (a : ℕ) (b : ℕ) (c : ℕ) (h : Eq (plus a c) (plus b c)) →
    let hb : Eq (plus a c) (plus c b) ≔
          trans (plus a c) (plus b c) (plus c b) h (plusComm b c);
        hc : Eq (plus c a) (plus c b) ≔
          trans (plus c a) (plus a c) (plus c b) (plusComm c a) hb
    in plusCancelLeft c a b hc
    end
end

If a sum is zero, so is its first summand.

theorem plusEqZero : ∀ (a : ℕ) (b : ℕ) (h : Eq (plus a b) 0) → Eq a 0 ≔
  λ (a : ℕ) (b : ℕ) →
    elim a at x -> (∀ (h : Eq (plus x b) 0) → Eq x 0)
    | zero ⇒ λ (h : Eq b 0) → ⋆
    | succ p, ih ⇒ λ (h : ⊥) → h
    end
end

If a sum is zero, so is its second summand.

theorem plusEqZeroRight : ∀ (a : ℕ) (b : ℕ) (h : Eq (plus a b) 0) → Eq b 0 ≔
  λ (a : ℕ) (b : ℕ) →
    elim a at x -> (∀ (h : Eq (plus x b) 0) → Eq b 0)
    | zero ⇒ λ (h : Eq b 0) → h
    | succ p, ih ⇒ λ (h : ⊥) → absurd (Eq b 0) h
    end
end

end
