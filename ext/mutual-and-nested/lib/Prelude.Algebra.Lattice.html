Prelude›Algebra›Lattice

Lattices: IsLattice A eq meet join holds when meet and join are semilattices up to eq on the one carrier A, meetIsSemilattice and joinIsSemilattice, tied by the two absorption laws. The bundle's with-area forwards the laws of both operations, under the names meetAssoc … joinIdem, and gives their semilattices, L.meetSemilattice and L.joinSemilattice.

module Prelude›Algebra›Lattice (c : Level) (r : Level) where

import Prelude›Algebra›Semilattice c r use (IsSemilattice; Semilattice)

record IsLattice (A : Type@{c}) (eq : ∀ (x : A) (y : A) → Type@{r})
    (meet : ∀ (x : A) (y : A) → A) (join : ∀ (x : A) (y : A) → A) :
    Type@{maxl c r} where
  meetIsSemilattice : IsSemilattice A eq meet;
  joinIsSemilattice : IsSemilattice A eq join;
  absorbMeet : ∀ (x : A) (y : A) → eq (meet x (join x y)) x;
  absorbJoin : ∀ (x : A) (y : A) → eq (join x (meet x y)) x
end

record Lattice : Type@{succl (maxl c r)} where
  carrier : Type@{c};
  eq : ∀ (x : carrier) (y : carrier) → Type@{r};
  meet : ∀ (x : carrier) (y : carrier) → carrier;
  join : ∀ (x : carrier) (y : carrier) → carrier;
  isLattice : IsLattice carrier eq meet join
with
  open isLattice export (absorbMeet; absorbJoin)
  open isLattice.meetIsSemilattice
    export (refl; sym; trans; opCong as meetCong; assoc as meetAssoc;
            comm as meetComm; idem as meetIdem)
  open isLattice.joinIsSemilattice
    export (opCong as joinCong; assoc as joinAssoc; comm as joinComm;
            idem as joinIdem)
  def meetSemilattice : Semilattice ≔
    record Semilattice where
      carrier ≔ carrier; eq ≔ eq; op ≔ meet;
      isSemilattice ≔ isLattice.meetIsSemilattice
    end
  end
  def joinSemilattice : Semilattice ≔
    record Semilattice where
      carrier ≔ carrier; eq ≔ eq; op ≔ join;
      isSemilattice ≔ isLattice.joinIsSemilattice
    end
  end
end

end
