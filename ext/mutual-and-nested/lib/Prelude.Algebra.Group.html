Prelude›Algebra›Group

Groups: a monoid with an inverse inv on both sides. The record's members carrier, op, e and the monoid laws are those of its monoid. The laws derived from these (cancellation, the uniqueness of inverses, the inverses of e, of an inverse and of a product) and the homomorphisms of groups are in Prelude›Algebra›Group›Properties; groups over a setoid are Prelude›Algebra›Setoid›Group.

module Prelude›Algebra›Group (u : Level) where

import Prelude›Logic›Leibniz u use (Id)
import Prelude›Algebra›Monoid u use (Monoid)

record Group : Type@{succl u} where
  monoid : Monoid;
  inv : ∀ (x : monoid.carrier) → monoid.carrier;
  leftInv : ∀ (x : monoid.carrier) →
    Id monoid.carrier (monoid.op (inv x) x) monoid.e;
  rightInv : ∀ (x : monoid.carrier) →
    Id monoid.carrier (monoid.op x (inv x)) monoid.e
with
  def carrier : Type@{u} ≔ monoid.carrier end
  def op (x : carrier) (y : carrier) : carrier ≔ monoid.op x y end
  def e : carrier ≔ monoid.e end
  def assoc : ∀ (x : carrier) (y : carrier) (z : carrier) →
      Id carrier (op (op x y) z) (op x (op y z)) ≔
    monoid.assoc
  end
  def leftId : ∀ (x : carrier) → Id carrier (op e x) x ≔ monoid.leftId end
  def rightId : ∀ (x : carrier) → Id carrier (op x e) x ≔ monoid.rightId end
end
end

end
