Prelude›Algebra›Group

Groups: IsGroup A eq op e inv holds when op and e form a monoid up to eq, its predicate isMonoid, and inv respects eq and is an inverse on both sides. The bundle's with-area forwards the laws and gives the monoid below, G.monoid. The derived laws (cancellation, the uniqueness of inverses, the inverses of e, of an inverse, of a product and of a power) are in Prelude›Algebra›Group›Properties, the homomorphisms in Prelude›Algebra›Group›Hom.

module Prelude›Algebra›Group (l : Level) where

import Prelude›Algebra›Monoid l use (IsMonoid; Monoid)

record IsGroup (A : Type@{l}) (eq : ∀ (x : A) (y : A) → Type@{l})
    (op : ∀ (x : A) (y : A) → A) (e : A) (inv : ∀ (x : A) → A) :
    Type@{l} where
  isMonoid : IsMonoid A eq op e;
  invCong : ∀ (a : A) (b : A) (h : eq a b) → eq (inv a) (inv b);
  leftInv : ∀ (x : A) → eq (op (inv x) x) e;
  rightInv : ∀ (x : A) → eq (op x (inv x)) e
with
  open isMonoid export (refl; sym; trans; opCong; assoc; leftId; rightId)
end

record Group : Type@{succl l} where
  carrier : Type@{l};
  eq : ∀ (x : carrier) (y : carrier) → Type@{l};
  op : ∀ (x : carrier) (y : carrier) → carrier;
  e : carrier;
  inv : ∀ (x : carrier) → carrier;
  isGroup : IsGroup carrier eq op e inv
with
  open isGroup
    export (refl; sym; trans; opCong; assoc; leftId; rightId; invCong; leftInv;
            rightInv)
  def monoid : Monoid ≔
    record Monoid where
      carrier ≔ carrier; eq ≔ eq; op ≔ op; e ≔ e; isMonoid ≔ isGroup.isMonoid
    end
  end
end

end
