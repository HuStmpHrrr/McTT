Prelude›Algebra›Semilattice

Semilattices: IsSemilattice A eq op holds when op is a semigroup up to eq, its predicate isSemigroup, commutative and idempotent, as the minimum of two numbers or the conjunction of two booleans. The bundle's with-area forwards the laws and gives the semigroup below. The order a semilattice defines is in Prelude›Algebra›Semilattice›Properties.

module Prelude›Algebra›Semilattice (c : Level) (r : Level) where

import Prelude›Algebra›Semigroup c r use (IsSemigroup; Semigroup)

record IsSemilattice (A : Type@{c}) (eq : ∀ (x : A) (y : A) → Type@{r})
    (op : ∀ (x : A) (y : A) → A) : Type@{maxl c r} where
  isSemigroup : IsSemigroup A eq op;
  comm : ∀ (x : A) (y : A) → eq (op x y) (op y x);
  idem : ∀ (x : A) → eq (op x x) x
with
  open isSemigroup export (refl; sym; trans; opCong; assoc)
end

record Semilattice : Type@{succl (maxl c r)} where
  carrier : Type@{c};
  eq : ∀ (x : carrier) (y : carrier) → Type@{r};
  op : ∀ (x : carrier) (y : carrier) → carrier;
  isSemilattice : IsSemilattice carrier eq op
with
  open isSemilattice export (refl; sym; trans; opCong; assoc; comm; idem)
  def semigroup : Semigroup ≔
    record Semigroup where
      carrier ≔ carrier; eq ≔ eq; op ≔ op;
      isSemigroup ≔ isSemilattice.isSemigroup
    end
  end
end

end
