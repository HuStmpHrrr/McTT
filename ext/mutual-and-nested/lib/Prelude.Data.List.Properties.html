Prelude›Data›List›Properties

Laws of Prelude›Data›List over a fixed element type A of Type@{u}. Equalities of lists, and of other values of Type@{u}, are the Leibniz equality of Prelude›Logic›Leibniz; equalities of lengths are Prelude›Arith›Equality.

append is a monoid with unit nil, and length takes it to plus; map is a functor, keeps the length and distributes over append; foldr over append folds the second list first; reverse turns append around, is an involution and keeps the length. The laws of sum are in Prelude›Data›List›Nat›Properties.

Each law is proved by induction on a list, so its step case is a congruence: through cons x for lists, through succ (by conversion) for lengths.

module Prelude›Data›List›Properties (u : Level) (A : Type@{u}) where

import Prelude›Builtin›Nat use (succ)
import Prelude›Arith›Equality use (Eq; refl; trans)
import Prelude›Arith›Plus use (plus)
open Prelude›Arith›Plus.Basic use (plusIdentityRight; plusSucc)
import Prelude›Data›List use (List)
open Prelude›Data›List.Ops as Ops
import Prelude›Logic›Leibniz u use (Id; idRefl; idSym; idTrans; idCong)

private inductive L ≔ List u A

private module Op ≔ Ops u A

Concatenation and length.

module Append where

nil is a right unit; it is a left unit by unfolding.

theorem appendNil : ∀ (l : L) → Id L (Op.append l L.nil) l ≔
  λ (l : L) →
    elim l at z -> Id L (Op.append z L.nil) z
    | nil ⇒ idRefl L L.nil
    | cons x xs, ih ⇒ idCong L L (L.cons x) (Op.append xs L.nil) xs ih
    end
end

theorem appendAssoc : ∀ (l : L) (m : L) (n : L) →
    Id L (Op.append (Op.append l m) n) (Op.append l (Op.append m n)) ≔
  λ (l : L) (m : L) (n : L) →
    elim l at z -> Id L (Op.append (Op.append z m) n)
                        (Op.append z (Op.append m n))
    | nil ⇒ idRefl L (Op.append m n)
    | cons x xs, ih ⇒
        idCong L L (L.cons x) (Op.append (Op.append xs m) n)
          (Op.append xs (Op.append m n)) ih
    end
end

In the step, both sides are successors, and Eq (succ a) (succ b) is Eq a b.

theorem lengthAppend : ∀ (l : L) (m : L) →
    Eq (Op.length (Op.append l m)) (plus (Op.length l) (Op.length m)) ≔
  λ (l : L) (m : L) →
    elim l at z -> Eq (Op.length (Op.append z m))
                        (plus (Op.length z) (Op.length m))
    | nil ⇒ refl (Op.length m)
    | cons x xs, ih ⇒ ih
    end
end

end

map: the functor laws, length and concatenation.

module Maps where

theorem mapId : ∀ (l : L) → Id L (Op.map A (λ (x : A) → x) l) l ≔
  λ (l : L) →
    elim l at z -> Id L (Op.map A (λ (x : A) → x) z) z
    | nil ⇒ idRefl L L.nil
    | cons x xs, ih ⇒
        idCong L L (L.cons x) (Op.map A (λ (y : A) → y) xs) xs ih
    end
end

Map fusion: mapping f then g is mapping their composite.

theorem mapComp : ∀ (B : Type@{u}) (C : Type@{u}) (f : ∀ (x : A) → B)
    (g : ∀ (y : B) → C) (l : L) →
    Id (List u C) (Ops.map u B C g (Op.map B f l))
      (Op.map C (λ (x : A) → g (f x)) l) ≔
  λ (B : Type@{u}) (C : Type@{u}) (f : ∀ (x : A) → B)
    (g : ∀ (y : B) → C) (l : L) →
    elim l at z -> Id (List u C) (Ops.map u B C g (Op.map B f z))
                        (Op.map C (λ (x : A) → g (f x)) z)
    | nil ⇒ idRefl (List u C) (List.nil u C)
    | cons x xs, ih ⇒
        idCong (List u C) (List u C) (List.cons u C (g (f x)))
          (Ops.map u B C g (Op.map B f xs))
          (Op.map C (λ (y : A) → g (f y)) xs) ih
    end
end

theorem lengthMap : ∀ (B : Type@{u}) (f : ∀ (x : A) → B) (l : L) →
    Eq (Ops.length u B (Op.map B f l)) (Op.length l) ≔
  λ (B : Type@{u}) (f : ∀ (x : A) → B) (l : L) →
    elim l at z -> Eq (Ops.length u B (Op.map B f z)) (Op.length z)
    | nil ⇒ ⋆
    | cons x xs, ih ⇒ ih
    end
end

theorem mapAppend : ∀ (B : Type@{u}) (f : ∀ (x : A) → B) (l : L) (m : L) →
    Id (List u B) (Op.map B f (Op.append l m))
      (Ops.append u B (Op.map B f l) (Op.map B f m)) ≔
  λ (B : Type@{u}) (f : ∀ (x : A) → B) (l : L) (m : L) →
    elim l at z -> Id (List u B) (Op.map B f (Op.append z m))
                        (Ops.append u B (Op.map B f z) (Op.map B f m))
    | nil ⇒ idRefl (List u B) (Op.map B f m)
    | cons x xs, ih ⇒
        idCong (List u B) (List u B) (List.cons u B (f x))
          (Op.map B f (Op.append xs m))
          (Ops.append u B (Op.map B f xs) (Op.map B f m)) ih
    end
end

end

foldr.

module Folds where

Folding a concatenation folds the second list, then the first from that result.

theorem foldrAppend : ∀ (B : Type@{u}) (f : ∀ (x : A) (b : B) → B) (z : B)
    (l : L) (m : L) →
    Id B (Op.foldr B f z (Op.append l m)) (Op.foldr B f (Op.foldr B f z m) l) ≔
  λ (B : Type@{u}) (f : ∀ (x : A) (b : B) → B) (z : B) (l : L) (m : L) →
    elim l at y -> Id B (Op.foldr B f z (Op.append y m))
                        (Op.foldr B f (Op.foldr B f z m) y)
    | nil ⇒ idRefl B (Op.foldr B f z m)
    | cons x xs, ih ⇒
        idCong B B (f x) (Op.foldr B f z (Op.append xs m))
          (Op.foldr B f (Op.foldr B f z m) xs) ih
    end
end

Folding with the constructors rebuilds the list.

theorem foldrCons : ∀ (l : L) →
    Id L (Op.foldr L (λ (x : A) (r : L) → L.cons x r) L.nil l) l ≔
  λ (l : L) →
    elim l at y -> Id L (Op.foldr L (λ (x : A) (r : L) → L.cons x r) L.nil y)
                        y
    | nil ⇒ idRefl L L.nil
    | cons x xs, ih ⇒
        idCong L L (L.cons x)
          (Op.foldr L (λ (w : A) (r : L) → L.cons w r) L.nil xs) xs ih
    end
end

map is a fold.

theorem mapFoldr : ∀ (B : Type@{u}) (f : ∀ (x : A) → B) (l : L) →
    Id (List u B) (Op.map B f l)
      (Op.foldr (List u B) (λ (x : A) (r : List u B) → List.cons u B (f x) r)
         (List.nil u B) l) ≔
  λ (B : Type@{u}) (f : ∀ (x : A) → B) (l : L) →
    elim l at y -> Id (List u B) (Op.map B f y)
                        (Op.foldr (List u B)
                           (λ (x : A) (r : List u B) → List.cons u B (f x) r)
                           (List.nil u B) y)
    | nil ⇒ idRefl (List u B) (List.nil u B)
    | cons x xs, ih ⇒
        idCong (List u B) (List u B) (List.cons u B (f x)) (Op.map B f xs)
          (Op.foldr (List u B)
             (λ (w : A) (r : List u B) → List.cons u B (f w) r)
             (List.nil u B) xs) ih
    end
end

end

reverse.

module Reverse where

open Append use (appendNil; appendAssoc; lengthAppend)
open Maps use (mapAppend)

The one-element list.

private def one (x : A) : L ≔ L.cons x L.nil end

Reversal turns a concatenation around. In the step, the hypothesis is appended one x, and appendAssoc moves the brackets.

theorem reverseAppend : ∀ (l : L) (m : L) →
    Id L (Op.reverse (Op.append l m))
      (Op.append (Op.reverse m) (Op.reverse l)) ≔
  λ (l : L) (m : L) →
    elim l at z -> Id L (Op.reverse (Op.append z m))
                        (Op.append (Op.reverse m) (Op.reverse z))
    | nil ⇒ idSym L (Op.append (Op.reverse m) L.nil) (Op.reverse m)
              (appendNil (Op.reverse m))
    | cons x xs, ih ⇒
        idTrans L
          (Op.append (Op.reverse (Op.append xs m)) (one x))
          (Op.append (Op.append (Op.reverse m) (Op.reverse xs)) (one x))
          (Op.append (Op.reverse m) (Op.append (Op.reverse xs) (one x)))
          (idCong L L (λ (w : L) → Op.append w (one x))
             (Op.reverse (Op.append xs m))
             (Op.append (Op.reverse m) (Op.reverse xs)) ih)
          (appendAssoc (Op.reverse m) (Op.reverse xs) (one x))
    end
end

Reversal is an involution: reverse (reverse (cons x xs)) is reverse (append (reverse xs) (one x)), which reverseAppend turns into cons x (reverse (reverse xs)).

theorem reverseReverse : ∀ (l : L) → Id L (Op.reverse (Op.reverse l)) l ≔
  λ (l : L) →
    elim l at z -> Id L (Op.reverse (Op.reverse z)) z
    | nil ⇒ idRefl L L.nil
    | cons x xs, ih ⇒
        idTrans L
          (Op.reverse (Op.append (Op.reverse xs) (one x)))
          (L.cons x (Op.reverse (Op.reverse xs)))
          (L.cons x xs)
          (reverseAppend (Op.reverse xs) (one x))
          (idCong L L (L.cons x) (Op.reverse (Op.reverse xs)) xs ih)
    end
end

In the step, length (append (reverse xs) (one x)) is plus (length (reverse xs)) 1, which is succ (length (reverse xs)).

theorem lengthReverse : ∀ (l : L) →
    Eq (Op.length (Op.reverse l)) (Op.length l) ≔
  λ (l : L) →
    elim l at z -> Eq (Op.length (Op.reverse z)) (Op.length z)
    | nil ⇒ ⋆
    | cons x xs, ih ⇒
        trans (Op.length (Op.append (Op.reverse xs) (one x)))
          (plus (Op.length (Op.reverse xs)) 1)
          (succ (Op.length xs))
          (lengthAppend (Op.reverse xs) (one x))
          (trans (plus (Op.length (Op.reverse xs)) 1)
             (succ (plus (Op.length (Op.reverse xs)) 0))
             (succ (Op.length xs))
             (plusSucc (Op.length (Op.reverse xs)) 0)
             (trans (plus (Op.length (Op.reverse xs)) 0)
                (Op.length (Op.reverse xs)) (Op.length xs)
                (plusIdentityRight (Op.length (Op.reverse xs))) ih))
    end
end

Reversal commutes with map.

theorem mapReverse : ∀ (B : Type@{u}) (f : ∀ (x : A) → B) (l : L) →
    Id (List u B) (Op.map B f (Op.reverse l)) (Ops.reverse u B (Op.map B f l)) ≔
  λ (B : Type@{u}) (f : ∀ (x : A) → B) (l : L) →
    elim l at z -> Id (List u B) (Op.map B f (Op.reverse z))
                        (Ops.reverse u B (Op.map B f z))
    | nil ⇒ idRefl (List u B) (List.nil u B)
    | cons x xs, ih ⇒
        idTrans (List u B)
          (Op.map B f (Op.append (Op.reverse xs) (one x)))
          (Ops.append u B (Op.map B f (Op.reverse xs))
             (List.cons u B (f x) (List.nil u B)))
          (Ops.append u B (Ops.reverse u B (Op.map B f xs))
             (List.cons u B (f x) (List.nil u B)))
          (mapAppend B f (Op.reverse xs) (one x))
          (idCong (List u B) (List u B)
             (λ (w : List u B) →
                Ops.append u B w (List.cons u B (f x) (List.nil u B)))
             (Op.map B f (Op.reverse xs)) (Ops.reverse u B (Op.map B f xs)) ih)
    end
end
