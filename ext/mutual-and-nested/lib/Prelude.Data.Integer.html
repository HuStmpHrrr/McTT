Prelude›Data›Integer

The integers as an inductive type: pos n is the integer n and negsuc n is -(n + 1), so every integer has exactly one form. Prelude›Algebra›Integers has another encoding, as pairs of numbers.

diff m n is the integer m - n of two numbers, by recursion on both: diff (succ m) (succ n) is diff m n. Addition add is by cases on the forms, through diff; negation neg, subtraction sub, successor succInt and predecessor predInt follow. Multiplication mul is by signs and absolute values: mul i j is signed (S.mul (sign i) (sign j)) (mult (abs i) (abs j)), where signed s n is the integer of sign s and absolute value n. The laws are in Prelude›Data›Integer›Properties.

module Prelude›Data›Integer where

import Prelude›Builtin›Nat use (succ)
import Prelude›Arith›Plus use (plus)
import Prelude›Arith›Mult use (mult)
import Prelude›Data›Sign as S use (Sign)
open Sign use (negative; positive)

inductive Int : Type@0 where
| pos (n : ℕ)
| negsuc (n : ℕ)
end

open Int use (pos; negsuc)

def neg (i : Int) : Int ≔
  elim i at z -> Int
  | pos n ⇒ elim n at y -> Int | zero ⇒ pos 0 | succ k, r ⇒ negsuc k end
  | negsuc n ⇒ pos (succ n)
  end
end

The integer m - n.

def diff (m : ℕ) : ∀ (n : ℕ) → Int ≔
  elim m at x -> (∀ (n : ℕ) → Int)
  | zero ⇒ λ (n : ℕ) →
      elim n at y -> Int | zero ⇒ pos 0 | succ q, s ⇒ negsuc q end
  | succ p, r ⇒ λ (n : ℕ) →
      elim n at y -> Int | zero ⇒ pos (succ p) | succ q, s ⇒ r q end
  end
end

def add (i : Int) (j : Int) : Int ≔
  elim i at z -> Int
  | pos m ⇒
      elim j at w -> Int
      | pos n ⇒ pos (plus m n)
      | negsuc n ⇒ diff m (succ n)
      end
  | negsuc m ⇒
      elim j at w -> Int
      | pos n ⇒ diff n (succ m)
      | negsuc n ⇒ negsuc (succ (plus m n))
      end
  end
end

def sub (i : Int) (j : Int) : Int ≔ add i (neg j) end

def succInt (i : Int) : Int ≔ add (pos 1) i end

def predInt (i : Int) : Int ≔ add (negsuc 0) i end

def abs (i : Int) : ℕ ≔
  elim i at z -> ℕ | pos n ⇒ n | negsuc n ⇒ succ n end
end

The sign; 0 counts as positive.

def sign (i : Int) : Sign ≔
  elim i at z -> Sign | pos n ⇒ positive | negsuc n ⇒ negative end
end

The integer of sign s and absolute value n; 0 has both signs.

def signed (s : Sign) (n : ℕ) : Int ≔
  elim n at y -> Int
  | zero ⇒ pos 0
  | succ k, r ⇒
      elim s at z -> Int | negative ⇒ negsuc k | positive ⇒ pos (succ k) end
  end
end

def mul (i : Int) (j : Int) : Int ≔
  signed (S.mul (sign i) (sign j)) (mult (abs i) (abs j))
end
