Prelude›Algebra›Integers

The integers, as differences of numbers. An integer is a record of two numbers, pos and neg, that stands for pos - neg; pair a b stands for a - b, and two pairs are the same integer, Same x y, when they stand for the same difference: a + d = c + b for (a, b) and (c, d). Addition is componentwise and negation swaps the pair. With ofNat 0 as zero they form an abelian group up to Same, additiveGroup, an instance of Prelude›Algebra›AbelianGroup; its laws are Additive, and its powers are the multiples of an integer.

Same computes, so a closed instance is ⊤ or ⊥. magnitude and negative read a pair back as a sign and a number.

A pair of numbers, standing for their difference pos - neg.

record Int : Type@0 where
  pos : ℕ;
  neg : ℕ
end

def pair (a : ℕ) (b : ℕ) : Int ≔ record Int where pos ≔ a; neg ≔ b end end

x and y stand for the same difference.

def Same (x : Int) (y : Int) : Type@0 ≔
  Eq (plus x.pos y.neg) (plus y.pos x.neg)
end

The number n as the integer n - 0.

def ofNat (n : ℕ) : Int ≔ pair n 0 end

def add (x : Int) (y : Int) : Int ≔
  pair (plus x.pos y.pos) (plus x.neg y.neg)
end

def negate (x : Int) : Int ≔ pair x.neg x.pos end

x - y.

def minus (x : Int) (y : Int) : Int ≔ add x (negate y) end

The absolute value of the difference, read back by truncated subtraction.

def magnitude (x : Int) : ℕ ≔
  plus (monus x.pos x.neg) (monus x.neg x.pos)
end

tt when the difference is negative, that is when x.neg is not at most x.pos, and ff otherwise.

def negative (x : Int) : Bool ≔ not (leb x.neg x.pos) end

Same is an equivalence.

module Setoid where

theorem sameRefl : ∀ (x : Int) → Same x x ≔
  λ (x : Int) → refl (plus x.pos x.neg)
end

theorem sameSym : ∀ (x : Int) (y : Int) (h : Same x y) → Same y x ≔
  λ (x : Int) (y : Int) (h : Same x y) →
    sym (plus x.pos y.neg) (plus y.pos x.neg) h
end

Adding y.neg to both sides of the goal, the two hypotheses rewrite it into an equation that holds by rearranging; y.neg then cancels.

theorem sameTrans : ∀ (x : Int) (y : Int) (z : Int)
    (hxy : Same x y) (hyz : Same y z) →
    Same x z ≔
  λ (x : Int) (y : Int) (z : Int) (hxy : Same x y) (hyz : Same y z) →
    let a ≔ x.pos;
        b ≔ x.neg;
        c ≔ y.pos;
        d ≔ y.neg;
        f ≔ z.pos;
        g ≔ z.neg;
        start : Eq (plus (plus a g) d) (plus (plus a d) g) ≔
          plusRightComm a g d;
        first : Eq (plus (plus a d) g) (plus (plus c b) g) ≔
          cong (λ (w : ℕ) → plus w g) (plus a d) (plus c b) hxy;
        middle : Eq (plus (plus c b) g) (plus (plus c g) b) ≔
          plusRightComm c b g;
        second : Eq (plus (plus c g) b) (plus (plus f d) b) ≔
          cong (λ (w : ℕ) → plus w b) (plus c g) (plus f d) hyz;
        finish : Eq (plus (plus f d) b) (plus (plus f b) d) ≔
          plusRightComm f d b
    in plusCancelRight (plus a g) (plus f b) d
         (trans (plus (plus a g) d) (plus (plus a d) g) (plus (plus f b) d)
            start
            (trans (plus (plus a d) g) (plus (plus c b) g) (plus (plus f b) d)
               first
               (trans (plus (plus c b) g) (plus (plus c g) b)
                  (plus (plus f b) d) middle
                  (trans (plus (plus c g) b) (plus (plus f d) b)
                     (plus (plus f b) d) second finish))))
    end
end

end

open Setoid use (sameRefl; sameSym; sameTrans)

Addition and negation respect Same, and satisfy the group laws.

module Addition where

theorem addCong : ∀ (a : Int) (b : Int) (c : Int) (d : Int)
    (hac : Same a c) (hbd : Same b d) →
    Same (add a b) (add c d) ≔
  λ (a : Int) (b : Int) (c : Int) (d : Int)
    (hac : Same a c) (hbd : Same b d) →
    let lhs ≔ plus (plus a.pos b.pos) (plus c.neg d.neg);
        rhs ≔ plus (plus c.pos d.pos) (plus a.neg b.neg)
    in trans lhs (plus (plus a.pos c.neg) (plus b.pos d.neg)) rhs
         (plusSwapMiddle a.pos b.pos c.neg d.neg)
         (trans (plus (plus a.pos c.neg) (plus b.pos d.neg))
            (plus (plus c.pos a.neg) (plus d.pos b.neg)) rhs
            (plusCongBoth (plus a.pos c.neg) (plus b.pos d.neg)
               (plus c.pos a.neg) (plus d.pos b.neg) hac hbd)
            (plusSwapMiddle c.pos a.neg d.pos b.neg))
    end
end

theorem addAssoc : ∀ (x : Int) (y : Int) (z : Int) →
    Same (add (add x y) z) (add x (add y z)) ≔
  λ (x : Int) (y : Int) (z : Int) →
    plusCongBoth (plus (plus x.pos y.pos) z.pos) (plus x.neg (plus y.neg z.neg))
      (plus x.pos (plus y.pos z.pos)) (plus (plus x.neg y.neg) z.neg)
      (plusAssoc x.pos y.pos z.pos)
      (sym (plus (plus x.neg y.neg) z.neg) (plus x.neg (plus y.neg z.neg))
         (plusAssoc x.neg y.neg z.neg))
end

Zero on the left holds by conversion: plus 0 n is n.

theorem addIdentityLeft : ∀ (x : Int) → Same (add (ofNat 0) x) x ≔
  λ (x : Int) → refl (plus x.pos x.neg)
end

theorem addIdentityRight : ∀ (x : Int) → Same (add x (ofNat 0)) x ≔
  λ (x : Int) →
    plusCongBoth (plus x.pos 0) x.neg x.pos (plus x.neg 0) (plusIdentityRight x.pos)
      (sym (plus x.neg 0) x.neg (plusIdentityRight x.neg))
end

theorem addComm : ∀ (x : Int) (y : Int) → Same (add x y) (add y x) ≔
  λ (x : Int) (y : Int) →
    plusCongBoth (plus x.pos y.pos) (plus y.neg x.neg)
      (plus y.pos x.pos) (plus x.neg y.neg)
      (plusComm x.pos y.pos) (plusComm y.neg x.neg)
end

theorem negateCong : ∀ (a : Int) (b : Int) (h : Same a b) →
    Same (negate a) (negate b) ≔
  λ (a : Int) (b : Int) (h : Same a b) →
    trans (plus a.neg b.pos) (plus b.pos a.neg) (plus b.neg a.pos)
      (plusComm a.neg b.pos)
      (trans (plus b.pos a.neg) (plus a.pos b.neg) (plus b.neg a.pos)
         (sym (plus a.pos b.neg) (plus b.pos a.neg) h) (plusComm a.pos b.neg))
end

(-x) + x stands for (x.neg + x.pos) - (x.pos + x.neg), which is zero.

theorem negateLeft : ∀ (x : Int) → Same (add (negate x) x) (ofNat 0) ≔
  λ (x : Int) →
    trans (plus (plus x.neg x.pos) 0) (plus x.neg x.pos) (plus x.pos x.neg)
      (plusIdentityRight (plus x.neg x.pos)) (plusComm x.neg x.pos)
end

theorem negateRight : ∀ (x : Int) → Same (add x (negate x)) (ofNat 0) ≔
  λ (x : Int) →
    trans (plus (plus x.pos x.neg) 0) (plus x.pos x.neg) (plus x.neg x.pos)
      (plusIdentityRight (plus x.pos x.neg)) (plusComm x.pos x.neg)
end

end

The integers under addition, up to Same: an abelian group.

def additiveGroup : AbelianGroup ≔
  record AbelianGroup where
    carrier ≔ Int; eq ≔ Same; op ≔ add; e ≔ ofNat 0; inv ≔ negate;
    isAbelianGroup ≔
      record IsAbelianGroup Int Same add (ofNat 0) negate where
        isGroup ≔
          record IsGroup Int Same add (ofNat 0) negate where
            isMonoid ≔
              record IsMonoid Int Same add (ofNat 0) where
                isSemigroup ≔
                  record IsSemigroup Int Same add where
                    isMagma ≔
                      record IsMagma Int Same add where
                        isEquivalence ≔
                          record IsEquivalence Int Same where
                            refl ≔ sameRefl; sym ≔ sameSym; trans ≔ sameTrans
                          end;
                        opCong ≔ addCong
                      end;
                    assoc ≔ addAssoc
                  end;
                leftId ≔ addIdentityLeft;
                rightId ≔ addIdentityRight
              end;
            invCong ≔ negateCong;
            leftInv ≔ negateLeft;
            rightInv ≔ negateRight
          end;
        comm ≔ addComm
      end
  end
end

The numbers embed into the integers.

module Embedding where

theorem ofNatAdd : ∀ (m : ℕ) (n : ℕ) →
    Same (ofNat (plus m n)) (add (ofNat m) (ofNat n)) ≔
  λ (m : ℕ) (n : ℕ) → refl (plus (plus m n) 0)
end

Different numbers are different integers.

theorem ofNatInj : ∀ (m : ℕ) (n : ℕ) (h : Same (ofNat m) (ofNat n)) →
    Eq m n ≔
  λ (m : ℕ) (n : ℕ) (h : Same (ofNat m) (ofNat n)) →
    trans m (plus m 0) n (sym (plus m 0) m (plusIdentityRight m))
      (trans (plus m 0) (plus n 0) n h (plusIdentityRight n))
end

end
