    ChurchPair

    Church-encoded pairs, passed as arguments to the program that uses them.

    module ChurchPair where

import Prelude›Builtin›Nat use (succ)

eval (
    (* Main example *)
    (
      λ (Pair : ∀ (A : Type@0) (B : Type@0) → Type@1)
        (pair : ∀ (A : Type@0) (B : Type@0) (a : A) (b : B) → Pair A B)
        (fst : ∀ (A : Type@0) (B : Type@0) (p : Pair A B) → A)
        (snd : ∀ (A : Type@0) (B : Type@0) (p : Pair A B) → B) →
        (
          λ (p : Pair ℕ (∀ (x : ℕ) → ℕ)) →
            snd ℕ (∀ (x : ℕ) → ℕ) p
              (fst ℕ (∀ (x : ℕ) → ℕ) p)
        )
        (pair ℕ (∀ (x : ℕ) → ℕ) 3 (λ (x : ℕ) → succ (succ x)))
    )
    (* Church pair *)
    (λ (A : Type@0) (B : Type@0) →
       ∀ (C : Type@0) (pair : ∀ (a : A) (b : B) → C) → C)
    (* Church pair constructor *)
    (λ (A : Type@0) (B : Type@0) (a : A) (b : B)
       (C : Type@0) (pair : ∀ (a : A) (b : B) → C) →
       pair a b)
    (* Church pair fst *)
    (λ (A : Type@0) (B : Type@0)
       (p : ∀ (C : Type@0) (pair : ∀ (a : A) (b : B) → C) → C) →
       p A (λ (a : A) (b : B) → a))
    (* Church pair snd *)
    (λ (A : Type@0) (B : Type@0)
       (p : ∀ (C : Type@0) (pair : ∀ (a : A) (b : B) → C) → C) →
       p B (λ (a : A) (b : B) → b))
  ) : ℕ
    output
    Evaluate (λ (x1 : ∀ (A1 : Type@0)
                    (A2 : Type@0)

            (x2 : ∀ (A3 : Type@0)
