Prelude›Algebra›Monoid›Properties

Laws of a monoid M, derived from the fields of Prelude›Algebra›Monoid and stated up to its relation M.eq: rewriting on each side of the operation (Cong), the uniqueness of the neutral element (Neutral), and the laws of the powers pow x n, x combined with itself n times (Power). The unit takes the monoid as a parameter: import Prelude›Algebra›Monoid›Properties 0l additive.monoid as P gives the laws of numbers under addition.

import Prelude›Algebra›Monoid
module Prelude›Algebra›Monoid›Properties (l : Level)
    (M : Prelude›Algebra›Monoid.Monoid l) where

import Prelude›Builtin›Nat use (succ)
import Prelude›Arith›Plus use (plus)
import Prelude›Arith›Mult use (mult)

open M
  use (carrier as C; eq as R; op; e; opCong; assoc; leftId; rightId;
       refl as rRefl; sym as rSym; trans as rTrans)

pow x n is x combined with itself n times: op x (op x ... e).

def pow (x : C) (n : ℕ) : C ≔
  elim n at y → C
  | zero ⇒ e
  | succ p, r ⇒ op x r
  end
end

Rewriting one side of the operation.

module Cong where

theorem congLeft : ∀ (a : C) (b : C) (c : C) (h : R a b) →
    R (op a c) (op b c) ≔
  λ (a : C) (b : C) (c : C) (h : R a b) → opCong a c b c h (rRefl c)
end

theorem congRight : ∀ (a : C) (b : C) (c : C) (h : R b c) →
    R (op a b) (op a c) ≔
  λ (a : C) (b : C) (c : C) (h : R b c) → opCong a b a c (rRefl a) h
end

Associativity read from right to left.

theorem assocBack : ∀ (x : C) (y : C) (z : C) →
    R (op x (op y z)) (op (op x y) z) ≔
  λ (x : C) (y : C) (z : C) →
    rSym (op (op x y) z) (op x (op y z)) (assoc x y z)
end

end

open Cong use (congLeft; congRight; assocBack)

The neutral element is unique.

module Neutral where

An element that is neutral on the left is e: both are op d e.

theorem leftIdUnique : ∀ (d : C) (h : ∀ (x : C) → R (op d x) x) → R d e ≔
  λ (d : C) (h : ∀ (x : C) → R (op d x) x) →
    rTrans d (op d e) e (rSym (op d e) d (rightId d)) (h e)
end

An element that is neutral on the right is e: both are op e d.

theorem rightIdUnique : ∀ (d : C) (h : ∀ (x : C) → R (op x d) x) → R d e ≔
  λ (d : C) (h : ∀ (x : C) → R (op x d) x) →
    rTrans d (op e d) e (rSym (op e d) d (leftId d)) (h e)
end

end

Laws of powers.

module Power where

theorem powOne : ∀ (x : C) → R (pow x 1) x ≔
  λ (x : C) → rightId x
end

Powers respect R.

theorem powCong : ∀ (x : C) (y : C) (h : R x y) (n : ℕ) →
    R (pow x n) (pow y n) ≔
  λ (x : C) (y : C) (h : R x y) (n : ℕ) →
    elim n at k → R (pow x k) (pow y k)
    | zero ⇒ rRefl e
    | succ p, ih ⇒ opCong x (pow x p) y (pow y p) h ih
    end
end

The last factor may be split off on the right as well as on the left.

theorem powSuccRight : ∀ (x : C) (n : ℕ) →
    R (pow x (succ n)) (op (pow x n) x) ≔
  λ (x : C) (n : ℕ) →
    elim n at k → R (pow x (succ k)) (op (pow x k) x)
    | zero ⇒ rTrans (op x e) x (op e x) (rightId x) (rSym (op e x) x (leftId x))
    | succ p, ih ⇒
        rTrans (op x (pow x (succ p))) (op x (op (pow x p) x))
          (op (op x (pow x p)) x)
          (congRight x (pow x (succ p)) (op (pow x p) x) ih)
          (assocBack x (pow x p) x)
    end
end

A power of a sum of exponents is the product of the powers.

theorem powPlus : ∀ (x : C) (m : ℕ) (n : ℕ) →
    R (pow x (plus m n)) (op (pow x m) (pow x n)) ≔
  λ (x : C) (m : ℕ) (n : ℕ) →
    elim m at k → R (pow x (plus k n)) (op (pow x k) (pow x n))
    | zero ⇒ rSym (op e (pow x n)) (pow x n) (leftId (pow x n))
    | succ p, ih ⇒
        rTrans (op x (pow x (plus p n))) (op x (op (pow x p) (pow x n)))
          (op (op x (pow x p)) (pow x n))
          (congRight x (pow x (plus p n)) (op (pow x p) (pow x n)) ih)
          (assocBack x (pow x p) (pow x n))
    end
end

A power of a product of exponents is a power of a power.

theorem powMult : ∀ (x : C) (m : ℕ) (n : ℕ) →
    R (pow x (mult m n)) (pow (pow x n) m) ≔
  λ (x : C) (m : ℕ) (n : ℕ) →
    elim m at k → R (pow x (mult k n)) (pow (pow x n) k)
    | zero ⇒ rRefl e
    | succ p, ih ⇒
        rTrans (pow x (plus n (mult p n))) (op (pow x n) (pow x (mult p n)))
          (op (pow x n) (pow (pow x n) p))
          (powPlus x n (mult p n))
          (congRight (pow x n) (pow x (mult p n)) (pow (pow x n) p) ih)
    end
end

Powers of the same element commute.

theorem powComm : ∀ (x : C) (m : ℕ) (n : ℕ) →
    R (op (pow x m) (pow x n)) (op (pow x n) (pow x m)) ≔
  λ (x : C) (m : ℕ) (n : ℕ) →
    elim m at k → R (op (pow x k) (pow x n)) (op (pow x n) (pow x k))
    | zero ⇒
        rTrans (op e (pow x n)) (pow x n) (op (pow x n) e)
          (leftId (pow x n))
          (rSym (op (pow x n) e) (pow x n) (rightId (pow x n)))
    | succ p, ih ⇒
        let a ≔ op (op x (pow x p)) (pow x n);
            b ≔ op x (op (pow x p) (pow x n));
            c ≔ op x (op (pow x n) (pow x p));
            d ≔ op (op x (pow x n)) (pow x p);
            f ≔ op (op (pow x n) x) (pow x p);
            g ≔ op (pow x n) (op x (pow x p));
            ab : R a b ≔ assoc x (pow x p) (pow x n);
            bc : R b c ≔
              congRight x (op (pow x p) (pow x n)) (op (pow x n) (pow x p)) ih;
            cd : R c d ≔ assocBack x (pow x n) (pow x p);
            df : R d f ≔
              congLeft (op x (pow x n)) (op (pow x n) x) (pow x p)
                (powSuccRight x n);
            fg : R f g ≔ assoc (pow x n) x (pow x p)
        in rTrans a b g ab
             (rTrans b c g bc (rTrans c d g cd (rTrans d f g df fg)))
        end
    end
end
