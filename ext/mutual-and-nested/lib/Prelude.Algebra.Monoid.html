Prelude›Algebra›Monoid

Monoids: IsMonoid A eq op e holds when op is a semigroup up to eq, its predicate isSemigroup, and e is neutral on both sides; the bundle Monoid is a carrier, a relation, op, e and that predicate. The bundle's with-area forwards the laws, M.refl … M.opCong, M.assoc, M.leftId and M.rightId, and gives the semigroup and the magma below. The derived laws are in Prelude›Algebra›Monoid›Properties, the homomorphisms in Prelude›Algebra›Monoid›Hom.

module Prelude›Algebra›Monoid (c : Level) (r : Level) where

import Prelude›Algebra›Semigroup c r use (IsSemigroup; Semigroup)

record IsMonoid (A : Type@{c}) (eq : ∀ (x : A) (y : A) → Type@{r})
    (op : ∀ (x : A) (y : A) → A) (e : A) : Type@{maxl c r} where
  isSemigroup : IsSemigroup A eq op;
  leftId : ∀ (x : A) → eq (op e x) x;
  rightId : ∀ (x : A) → eq (op x e) x
with
  open isSemigroup export (refl; sym; trans; opCong; assoc)
end

record Monoid : Type@{succl (maxl c r)} where
  carrier : Type@{c};
  eq : ∀ (x : carrier) (y : carrier) → Type@{r};
  op : ∀ (x : carrier) (y : carrier) → carrier;
  e : carrier;
  isMonoid : IsMonoid carrier eq op e
with
  open isMonoid export (refl; sym; trans; opCong; assoc; leftId; rightId)
  def semigroup : Semigroup ≔
    record Semigroup where
      carrier ≔ carrier; eq ≔ eq; op ≔ op; isSemigroup ≔ isMonoid.isSemigroup
    end
  end
  open semigroup export (magma)
end

end
