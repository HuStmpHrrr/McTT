Prelude›Function›Combinators

Generic combinators over a type parameter at any universe level, with their laws at ℕ.

module Prelude›Function›Combinators where

import Prelude›Builtin›Nat use (succ)
import Prelude›Arith›Equality use (Eq; refl; sym; trans; cong)
import Prelude›Arith›Plus use (plus)
open Prelude›Arith›Plus.Algebra use (plusComm)
import Prelude›Function›Iter as Iter

Combinators over an arbitrary type A of Type@{u}. The other types they mention are at the same level u; this loses nothing, since by cumulativity types of levels u and v are both in Type@{maxl u v}.

module Generic (u : Level) (A : Type@{u}) where

def id (a : A) : A ≔ a end

The function that ignores its argument and returns b.

def const (B : Type@{u}) (b : B) (a : A) : B ≔ b end

compose B C g f applies f, then g.

def compose (B : Type@{u}) (C : Type@{u}) (g : ∀ (b : B) → C)
    (f : ∀ (a : A) → B) (a : A) : C ≔
  g (f a)
end

iterate f n a applies f to a n times; it is Prelude›Function›Iter.

def iterate (f : ∀ (a : A) → A) (n : ℕ) (a : A) : A ≔
  Iter.iter u A f n a
end

Laws of const and compose at ℕ.

module GenericLaws where

theorem constApply : ∀ (c : ℕ) (a : ℕ) → Eq (Generic.const 0l ℕ ℕ c a) c ≔
  λ (c : ℕ) (a : ℕ) → refl c
end

theorem composeId : ∀ (f : ∀ (n : ℕ) → ℕ) (a : ℕ) →
    Eq (Generic.compose 0l ℕ ℕ ℕ f (Generic.id 0l ℕ) a) (f a) ≔
  λ (f : ∀ (n : ℕ) → ℕ) (a : ℕ) → refl (f a)
end

end

Laws of iterating a fixed function f on numbers. Each law takes f first, as a module argument: IterateLaws.iteratePlus f m n a.

module IterateLaws (f : ∀ (n : ℕ) → ℕ) where

Iterating m + n times is iterating n times, then m times.

theorem iteratePlus : ∀ (m : ℕ) (n : ℕ) (a : ℕ) →
    Eq (Generic.iterate 0l ℕ f (plus m n) a)
       (Generic.iterate 0l ℕ f m (Generic.iterate 0l ℕ f n a)) ≔
  λ (m : ℕ) (n : ℕ) (a : ℕ) →
    elim m at x -> Eq (Generic.iterate 0l ℕ f (plus x n) a)
                         (Generic.iterate 0l ℕ f x (Generic.iterate 0l ℕ f n a))
    | zero ⇒ refl (Generic.iterate 0l ℕ f n a)
    | succ p, ih ⇒
        cong f (Generic.iterate 0l ℕ f (plus p n) a)
          (Generic.iterate 0l ℕ f p (Generic.iterate 0l ℕ f n a)) ih
    end
end

One more step may be taken first instead of last.

theorem iterateSucc : ∀ (n : ℕ) (a : ℕ) →
    Eq (Generic.iterate 0l ℕ f (succ n) a) (Generic.iterate 0l ℕ f n (f a)) ≔
  λ (n : ℕ) (a : ℕ) →
    elim n at x -> Eq (Generic.iterate 0l ℕ f (succ x) a)
                         (Generic.iterate 0l ℕ f x (f a))
    | zero ⇒ refl (f a)
    | succ p, ih ⇒
        cong f (Generic.iterate 0l ℕ f (succ p) a)
          (Generic.iterate 0l ℕ f p (f a)) ih
    end
end

Iterations of the same function commute, by iteratePlus and plusComm.

theorem iterateComm : ∀ (m : ℕ) (n : ℕ) (a : ℕ) →
    Eq (Generic.iterate 0l ℕ f m (Generic.iterate 0l ℕ f n a))
       (Generic.iterate 0l ℕ f n (Generic.iterate 0l ℕ f m a)) ≔
  λ (m : ℕ) (n : ℕ) (a : ℕ) →
    let mn ≔ Generic.iterate 0l ℕ f m (Generic.iterate 0l ℕ f n a);
        nm ≔ Generic.iterate 0l ℕ f n (Generic.iterate 0l ℕ f m a);
        toPlus : Eq mn (Generic.iterate 0l ℕ f (plus m n) a) ≔
          sym (Generic.iterate 0l ℕ f (plus m n) a) mn (iteratePlus m n a);
        swap : Eq (Generic.iterate 0l ℕ f (plus m n) a)
                  (Generic.iterate 0l ℕ f (plus n m) a) ≔
          cong (λ (k : ℕ) → Generic.iterate 0l ℕ f k a) (plus m n) (plus n m)
            (plusComm m n)
    in trans mn (Generic.iterate 0l ℕ f (plus n m) a) nm
         (trans mn (Generic.iterate 0l ℕ f (plus m n) a)
            (Generic.iterate 0l ℕ f (plus n m) a) toPlus swap)
         (iteratePlus n m a)
    end
end
