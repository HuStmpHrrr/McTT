Prelude›Algebra›AbelianGroup

Abelian groups: a group whose operation is commutative. The record's members are those of its group, and commutativeMonoid is its underlying commutative monoid.

module Prelude›Algebra›AbelianGroup (u : Level) where

import Prelude›Logic›Leibniz u use (Id)
import Prelude›Algebra›Group u use (Group)
import Prelude›Algebra›CommutativeMonoid u use (CommutativeMonoid)

record AbelianGroup : Type@{succl u} where
  group : Group;
  comm : ∀ (x : group.carrier) (y : group.carrier) →
    Id group.carrier (group.op x y) (group.op y x)
with
  def carrier : Type@{u} ≔ group.carrier end
  def op (x : carrier) (y : carrier) : carrier ≔ group.op x y end
  def e : carrier ≔ group.e end
  def inv (x : carrier) : carrier ≔ group.inv x end
  def commutativeMonoid : CommutativeMonoid ≔
    record CommutativeMonoid where monoid ≔ group.monoid; comm ≔ comm end
  end
end
end

end
