Prelude›Algebra›Semilattice›Properties

The order of a semilattice S: Le x y when S.op x y is x up to S.eq, as min x y is x when x is at most y. It is reflexive by idempotence, antisymmetric by commutativity and transitive by associativity, and S.op x y is below both x and y.

import Prelude›Algebra›Semilattice
module Prelude›Algebra›Semilattice›Properties (l : Level)
    (S : Prelude›Algebra›Semilattice.Semilattice l) where

open S use (carrier as C; op; eq; refl; sym; trans; opCong; assoc; comm; idem)

def Le (x : C) (y : C) : Type@{l} ≔ eq (op x y) x end

module Order where

theorem leRefl : ∀ (x : C) → Le x x ≔ idem end

x = x y = y x = y.

theorem leAntisym : ∀ (x : C) (y : C) (hxy : Le x y) (hyx : Le y x) → eq x y ≔
  λ (x : C) (y : C) (hxy : Le x y) (hyx : Le y x) →
    trans x (op x y) y (sym (op x y) x hxy)
      (trans (op x y) (op y x) y (comm x y) hyx)
end

x z = (x y) z = x (y z) = x y = x.

theorem leTrans : ∀ (x : C) (y : C) (z : C) (hxy : Le x y) (hyz : Le y z) →
    Le x z ≔
  λ (x : C) (y : C) (z : C) (hxy : Le x y) (hyz : Le y z) →
    trans (op x z) (op (op x y) z) x
      (opCong x z (op x y) z (sym (op x y) x hxy) (refl z))
      (trans (op (op x y) z) (op x (op y z)) x (assoc x y z)
         (trans (op x (op y z)) (op x y) x
            (opCong x (op y z) x y (refl x) hyz) hxy))
end

(x y) x = x (y x) = x (x y) = (x x) y = x y.

theorem opLeLeft : ∀ (x : C) (y : C) → Le (op x y) x ≔
  λ (x : C) (y : C) →
    trans (op (op x y) x) (op x (op y x)) (op x y)
      (assoc x y x)
      (trans (op x (op y x)) (op x (op x y)) (op x y)
         (opCong x (op y x) x (op x y) (refl x) (comm y x))
         (trans (op x (op x y)) (op (op x x) y) (op x y)
            (sym (op (op x x) y) (op x (op x y)) (assoc x x y))
            (opCong (op x x) y x y (idem x) (refl y))))
end

(x y) y = x (y y) = x y.

theorem opLeRight : ∀ (x : C) (y : C) → Le (op x y) y ≔
  λ (x : C) (y : C) →
    trans (op (op x y) y) (op x (op y y)) (op x y)
      (assoc x y y) (opCong x (op y y) x y (refl x) (idem y))
end

end
