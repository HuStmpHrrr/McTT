Prelude›Data›Option›Properties

Laws of Prelude›Data›Option over a fixed element type A of Type@{u}, stated with the Leibniz equality of Prelude›Logic›Leibniz: map is a functor, bind with some a monad, map is a bind, and the laws of getOr, isSome and orElse.

A law that holds by unfolding on a constructor (bind on some) is proved by idRefl; the others by cases on the option.

module Prelude›Data›Option›Properties (u : Level) (A : Type@{u}) where

import Prelude›Data›Bool use (Bool)
import Prelude›Data›Option use (Option)
open Option use (none; some)
open Prelude›Data›Option.Ops as O
import Prelude›Logic›Leibniz u use (Id; idRefl)

private inductive OA ≔ Option u A

private module Op ≔ O u A

The functor laws.

module Functor where

theorem mapId : ∀ (o : OA) → Id OA (Op.map A (λ (x : A) → x) o) o ≔
  λ (o : OA) →
    elim o at (z : OA) -> Id OA (Op.map A (λ (x : A) → x) z) z
    | none ⇒ idRefl OA OA.none
    | some x ⇒ idRefl OA (OA.some x)
    end
end

Mapping f then g is mapping their composite.

theorem mapComp : ∀ (B : Type@{u}) (C : Type@{u}) (f : ∀ (x : A) → B)
    (g : ∀ (y : B) → C) (o : OA) →
    Id (Option u C) (O.map u B C g (Op.map B f o))
      (Op.map C (λ (x : A) → g (f x)) o) ≔
  λ (B : Type@{u}) (C : Type@{u}) (f : ∀ (x : A) → B) (g : ∀ (y : B) → C)
    (o : OA) →
    elim o at (z : OA) -> Id (Option u C) (O.map u B C g (Op.map B f z))
                        (Op.map C (λ (x : A) → g (f x)) z)
    | none ⇒ idRefl (Option u C) (none u C)
    | some x ⇒ idRefl (Option u C) (some u C (g (f x)))
    end
end

end

The monad laws, with some as the unit.

module Monad where

some is a left unit: bind on some a continues with a.

theorem bindSomeLeft : ∀ (B : Type@{u}) (f : ∀ (x : A) → Option u B)
    (a : A) → Id (Option u B) (Op.bind B f (OA.some a)) (f a) ≔
  λ (B : Type@{u}) (f : ∀ (x : A) → Option u B) (a : A) →
    idRefl (Option u B) (f a)
end

some is a right unit.

theorem bindSomeRight : ∀ (o : OA) →
    Id OA (Op.bind A (λ (x : A) → OA.some x) o) o ≔
  λ (o : OA) →
    elim o at (z : OA) -> Id OA (Op.bind A (λ (x : A) → OA.some x) z) z
    | none ⇒ idRefl OA OA.none
    | some x ⇒ idRefl OA (OA.some x)
    end
end

theorem bindAssoc : ∀ (B : Type@{u}) (C : Type@{u})
    (f : ∀ (x : A) → Option u B) (g : ∀ (y : B) → Option u C) (o : OA) →
    Id (Option u C) (O.bind u B C g (Op.bind B f o))
      (Op.bind C (λ (x : A) → O.bind u B C g (f x)) o) ≔
  λ (B : Type@{u}) (C : Type@{u}) (f : ∀ (x : A) → Option u B)
    (g : ∀ (y : B) → Option u C) (o : OA) →
    elim o at (z : OA) -> Id (Option u C) (O.bind u B C g (Op.bind B f z))
                        (Op.bind C (λ (x : A) → O.bind u B C g (f x)) z)
    | none ⇒ idRefl (Option u C) (none u C)
    | some x ⇒ idRefl (Option u C) (O.bind u B C g (f x))
    end
end

map f is bind with some after f.

theorem mapBind : ∀ (B : Type@{u}) (f : ∀ (x : A) → B) (o : OA) →
    Id (Option u B) (Op.map B f o)
      (Op.bind B (λ (x : A) → some u B (f x)) o) ≔
  λ (B : Type@{u}) (f : ∀ (x : A) → B) (o : OA) →
    elim o at (z : OA) -> Id (Option u B) (Op.map B f z)
                        (Op.bind B (λ (x : A) → some u B (f x)) z)
    | none ⇒ idRefl (Option u B) (none u B)
    | some x ⇒ idRefl (Option u B) (some u B (f x))
    end
end

end

The default, the test and the choice.

module Access where

import Prelude›Logic›Leibniz 0l as BoolId

getOr commutes with map: mapping the default too.

theorem getOrMap : ∀ (C : Type@{u}) (f : ∀ (x : A) → C) (d : A) (o : OA) →
    Id C (O.getOr u C (f d) (Op.map C f o)) (f (Op.getOr d o)) ≔
  λ (C : Type@{u}) (f : ∀ (x : A) → C) (d : A) (o : OA) →
    elim o at (z : OA) -> Id C (O.getOr u C (f d) (Op.map C f z))
                        (f (Op.getOr d z))
    | none ⇒ idRefl C (f d)
    | some x ⇒ idRefl C (f x)
    end
end

Mapping keeps definedness.

theorem isSomeMap : ∀ (C : Type@{u}) (f : ∀ (x : A) → C) (o : OA) →
    BoolId.Id Bool (O.isSome u C (Op.map C f o)) (Op.isSome o) ≔
  λ (C : Type@{u}) (f : ∀ (x : A) → C) (o : OA) →
    elim o at (z : OA) -> BoolId.Id Bool (O.isSome u C (Op.map C f z))
                        (Op.isSome z)
    | none ⇒ BoolId.idRefl Bool (Op.isSome OA.none)
    | some x ⇒ BoolId.idRefl Bool (Op.isSome (OA.some x))
    end
end

none is a right unit of orElse (none is a left unit by unfolding).

theorem orElseNone : ∀ (o : OA) → Id OA (Op.orElse o OA.none) o ≔
  λ (o : OA) →
    elim o at (z : OA) -> Id OA (Op.orElse z OA.none) z
    | none ⇒ idRefl OA OA.none
    | some x ⇒ idRefl OA (OA.some x)
    end
end

theorem orElseAssoc : ∀ (o : OA) (p : OA) (q : OA) →
    Id OA (Op.orElse (Op.orElse o p) q) (Op.orElse o (Op.orElse p q)) ≔
  λ (o : OA) (p : OA) (q : OA) →
    elim o at (z : OA) -> Id OA (Op.orElse (Op.orElse z p) q)
                        (Op.orElse z (Op.orElse p q))
    | none ⇒ idRefl OA (Op.orElse p q)
    | some x ⇒ idRefl OA (OA.some x)
    end
end

end
