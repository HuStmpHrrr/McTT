Prelude›Algebra›Ring›Properties

Laws of a ring R: the signs of products. Each is an instance of the uniqueness of inverses in the additive group of R: (-x) y is the inverse of x y since their sum is (x + (-x)) y, that is 0 y, which is 0.

import Prelude›Algebra›Ring
module Prelude›Algebra›Ring›Properties (u : Level)
    (R : Prelude›Algebra›Ring.Ring u) where

import Prelude›Logic›Leibniz u use (Id; idSym; idTrans; idCong)
import Prelude›Algebra›Group›Properties u R.additiveGroup.group as Additive

private def C : Type@{u} ≔ R.carrier end

theorem negMulLeft : ∀ (x : C) (y : C) →
    Id C (R.mul (R.neg x) y) (R.neg (R.mul x y)) ≔
  λ (x : C) (y : C) →
    Additive.Inverse.invUnique (R.mul x y) (R.mul (R.neg x) y)
      (idTrans C (R.add (R.mul x y) (R.mul (R.neg x) y))
         (R.mul (R.add x (R.neg x)) y) R.zero
         (idSym C (R.mul (R.add x (R.neg x)) y)
            (R.add (R.mul x y) (R.mul (R.neg x) y))
            (R.semiring.distribRight x (R.neg x) y))
         (idTrans C (R.mul (R.add x (R.neg x)) y) (R.mul R.zero y) R.zero
            (idCong C C (λ (w : C) → R.mul w y) (R.add x (R.neg x)) R.zero
               (R.negRight x))
            (R.semiring.zeroLeft y)))
end

theorem negMulRight : ∀ (x : C) (y : C) →
    Id C (R.mul x (R.neg y)) (R.neg (R.mul x y)) ≔
  λ (x : C) (y : C) →
    Additive.Inverse.invUnique (R.mul x y) (R.mul x (R.neg y))
      (idTrans C (R.add (R.mul x y) (R.mul x (R.neg y)))
         (R.mul x (R.add y (R.neg y))) R.zero
         (idSym C (R.mul x (R.add y (R.neg y)))
            (R.add (R.mul x y) (R.mul x (R.neg y)))
            (R.semiring.distribLeft x y (R.neg y)))
         (idTrans C (R.mul x (R.add y (R.neg y))) (R.mul x R.zero) R.zero
            (idCong C C (R.mul x) (R.add y (R.neg y)) R.zero (R.negRight y))
            (R.semiring.zeroRight x)))
end

(-x) (-y) = -(x (-y)) = -(-(x y)) = x y.

theorem negMulNeg : ∀ (x : C) (y : C) →
    Id C (R.mul (R.neg x) (R.neg y)) (R.mul x y) ≔
  λ (x : C) (y : C) →
    idTrans C (R.mul (R.neg x) (R.neg y)) (R.neg (R.mul x (R.neg y))) (R.mul x y)
      (negMulLeft x (R.neg y))
      (idTrans C (R.neg (R.mul x (R.neg y))) (R.neg (R.neg (R.mul x y)))
         (R.mul x y)
         (idCong C C R.neg (R.mul x (R.neg y)) (R.neg (R.mul x y))
            (negMulRight x y))
         (Additive.Inverse.invInv (R.mul x y)))
end

Multiplying by -1 negates.

theorem negOneMul : ∀ (x : C) → Id C (R.mul (R.neg R.one) x) (R.neg x) ≔
  λ (x : C) →
    idTrans C (R.mul (R.neg R.one) x) (R.neg (R.mul R.one x)) (R.neg x)
      (negMulLeft R.one x)
      (idCong C C R.neg (R.mul R.one x) x (R.semiring.mulLeftId x))
end
