Prelude›Algebra›Group›Hom

Homomorphisms of groups: Hom G H is the record of a map from the carrier of G to that of H that respects the relations and takes products to products, up to H.eq. Unlike a homomorphism of monoids, it needs no field for the neutral element: the local module Laws G H f derives that f takes G.e to H.e (mapE) and inverses to inverses (mapInv), and gives the homomorphism of the underlying monoids (toMonoid).

module Prelude›Algebra›Group›Hom (c : Level) (r : Level) where

import Prelude›Algebra›Group c r use (Group)
import Prelude›Algebra›Monoid›Hom c r as MonoidHom
import Prelude›Algebra›Group›Properties

record Hom (G : Group) (H : Group) : Type@{maxl c r} where
  map : ∀ (x : G.carrier) → H.carrier;
  mapCong : ∀ (x : G.carrier) (y : G.carrier) (h : G.eq x y) →
    H.eq (map x) (map y);
  mapOp : ∀ (x : G.carrier) (y : G.carrier) →
    H.eq (map (G.op x y)) (H.op (map x) (map y))
end

def identity (G : Group) : Hom G G ≔
  record Hom G G where
    map ≔ λ (x : G.carrier) → x;
    mapCong ≔ λ (x : G.carrier) (y : G.carrier) (h : G.eq x y) → h;
    mapOp ≔ λ (x : G.carrier) (y : G.carrier) → G.refl (G.op x y)
  end
end

Laws of a homomorphism f from G to H.

module Laws (G : Group) (H : Group) (f : Hom G H) where

open Prelude›Algebra›Group›Properties c r H as PH

f.map G.e times itself is f.map (G.op G.e G.e), that is f.map G.e, which is f.map G.e times H.e; cancel it on the left.

theorem mapE : H.eq (f.map G.e) H.e ≔
  let m : H.carrier ≔ f.map G.e;
      square : H.eq (H.op m m) (f.map (G.op G.e G.e)) ≔
        H.sym (f.map (G.op G.e G.e)) (H.op m m) (f.mapOp G.e G.e);
      idem : H.eq (f.map (G.op G.e G.e)) m ≔
        f.mapCong (G.op G.e G.e) G.e (G.leftId G.e);
      unit : H.eq m (H.op m H.e) ≔ H.sym (H.op m H.e) m (H.rightId m)
  in PH.Laws.leftCancel m m H.e
       (H.trans (H.op m m) (f.map (G.op G.e G.e)) (H.op m H.e) square
          (H.trans (f.map (G.op G.e G.e)) m (H.op m H.e) idem unit))
  end
end

f.map x times f.map (G.inv x) is f.map (G.op x (G.inv x)), that is f.map G.e, which is H.e: so f.map (G.inv x) is the inverse of f.map x.

theorem mapInv : ∀ (x : G.carrier) → H.eq (f.map (G.inv x)) (H.inv (f.map x)) ≔
  λ (x : G.carrier) →
    PH.Laws.invUnique (f.map x) (f.map (G.inv x))
      (H.trans (H.op (f.map x) (f.map (G.inv x))) (f.map (G.op x (G.inv x))) H.e
         (H.sym (f.map (G.op x (G.inv x))) (H.op (f.map x) (f.map (G.inv x)))
            (f.mapOp x (G.inv x)))
         (H.trans (f.map (G.op x (G.inv x))) (f.map G.e) H.e
            (f.mapCong (G.op x (G.inv x)) G.e (G.rightInv x))
            mapE))
end

The same map, as a homomorphism of the underlying monoids.

def toMonoid : MonoidHom.Hom G.monoid H.monoid ≔
  record MonoidHom.Hom G.monoid H.monoid where
    map ≔ f.map; mapCong ≔ f.mapCong; mapOp ≔ f.mapOp; mapE ≔ mapE
  end
end
