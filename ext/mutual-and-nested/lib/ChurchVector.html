    ChurchVector

    Church-encoded vectors, passed as arguments to the program that uses them, with a head function that is total on non-empty vectors.

    module ChurchVector where

import Prelude›Builtin›Nat use (zero; succ)

eval (
    (* Main example *)
    (
      λ (Vec : ∀ (A : Type@0) (n : ℕ) → Type@2)
        (nil : ∀ (A : Type@0) → Vec A 0)
        (cons : ∀ (A : Type@0) (n : ℕ) (head : A) (tail : Vec A n) →
                  Vec A (succ n))
        (vecRec : ∀ (A : Type@0) (n : ℕ) (vec : Vec A n)
                    (C : ∀ (l : ℕ) → Type@1)
                    (nil : C 0)
                    (cons : ∀ (l : ℕ) (a : A) (r : C l) → C (succ l)) →
                    C n) →
        (
          λ (totalHead : ∀ (A : Type@0) (n : ℕ) (vec : Vec A (succ n)) → A)
            (vec : Vec (∀ (n : ℕ) → ℕ) 3) →
            totalHead ((∀ (n : ℕ) → ℕ)) 2 vec 4
        )
        (* total head function *)
        (
          λ (A : Type@0) (n : ℕ) (vec : Vec A (succ n)) →
            vecRec A (succ n) vec
              (λ (l : ℕ) →
                 elim l at r -> Type@0 | zero ⇒ ℕ | succ l, r ⇒ A end)
              0
              (λ (l : ℕ) (a : A)
                 (r : elim l at r -> Type@0 | zero ⇒ ℕ | succ l, r ⇒ A end) →
                 a)
        )
        (* example vector *)
        (
          cons (∀ (n : ℕ) → ℕ) 2
            (λ (n : ℕ) → succ (succ (succ n)))
            (
              cons (∀ (n : ℕ) → ℕ) 1
                (λ (n : ℕ) → succ n)
                (
                  cons (∀ (n : ℕ) → ℕ) 0
                    (λ (n : ℕ) → succ (succ n))
                    (nil (∀ (n : ℕ) → ℕ))
                )
            )
        )
    )
    (* Church vector *)
    (λ (A : Type@0) (n : ℕ) →
       ∀ (C : ∀ (l : ℕ) → Type@1)
         (nil : C 0)
         (cons : ∀ (l : ℕ) (a : A) (r : C l) → C (succ l)) →
         C n)
    (* Church vector nil *)
    (λ (A : Type@0)
       (C : ∀ (l : ℕ) → Type@1)
       (nil : C 0)
       (cons : ∀ (l : ℕ) (a : A) (r : C l) → C (succ l)) →
       nil)
    (* Church vector cons *)
    (λ (A : Type@0)
       (n : ℕ)
       (head : A)
       (tail : ∀ (C : ∀ (l : ℕ) → Type@1)
                 (nil : C 0)
                 (cons : ∀ (l : ℕ) (a : A) (r : C l) → C (succ l)) →
                 C n)
       (C : ∀ (l : ℕ) → Type@1)
       (nil : C 0)
       (cons : ∀ (l : ℕ) (a : A) (r : C l) → C (succ l)) →
       cons n head (tail C nil cons))
    (* Church vector recursor *)
    (λ (A : Type@0)
       (n : ℕ)
       (vec : ∀ (C : ∀ (l : ℕ) → Type@1)
                (nil : C 0)
                (cons : ∀ (l : ℕ) (a : A) (r : C l) → C (succ l)) →
                C n)
       (C : ∀ (l : ℕ) → Type@1)
       (nil : C 0)
       (cons : ∀ (l : ℕ) (a : A) (r : C l) → C (succ l)) →
       vec C nil cons)
  ) : ℕ
    output
