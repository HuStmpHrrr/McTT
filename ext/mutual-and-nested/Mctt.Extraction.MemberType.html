Computing Descriptions

mdenote_impl decides mdenote, together with what units and entries denote (udenote, edenote), the expansion of bodies (body_xp) and what an open generates (items_gen), by recursion on a termination order (dn_order) that mirrors their recursion: a slot is read through the unit the context gives it, a selection through the entry the description of its module gives, and an open through its target in the context of the expanded body before it. Wherever a lookup succeeds, the order asks for an order of what it found, so every module expression that denotes has an order (dn_order_of_mdenote); so does a selection from a well-formed module expression (dn_order_mem_of_wf), since the entries of the description of a well-formed module expression denote.

The Termination Order


Inductive dn_order (Θ : gdeps) (Ξ : gstack) : ctx -> modexp -> Prop :=
| dno_unit : forall Γ H TΦ,
    gc_unit Θ Ξ H = Some TΦ ->
    dn_order Θ Ξ Γ H
The open frame of a member's module, named by its path
| dno_ind_frame : forall Γ H p r,
    mod_qname H = Some p -> gs_find_ind Ξ p = Some r ->
    dn_order Θ Ξ Γ H
The open frame of a record type's module, named by its path
| dno_record_frame : forall Γ H p R,
    mod_qname H = Some p -> gs_find_record Ξ p = Some R ->
    dn_order Θ Ξ Γ H
| dno_path : forall Γ fp,
    dn_order Θ Ξ Γ (me_unit fp)
| dno_var : forall Γ x,
    (forall U, ctx_find_mod Γ x = Some U -> un_order Θ Ξ Γ U) ->
    dn_order Θ Ξ Γ (me_var x)
| dno_lit : forall Γ U,
    un_order Θ Ξ Γ U ->
    dn_order Θ Ξ Γ (me_lit U)
| dno_app : forall Γ H N,
    dn_order Θ Ξ Γ H ->
    dn_order Θ Ξ Γ (me_app H N)
| dno_mem : forall Γ H y,
    dn_order Θ Ξ Γ H ->
    (forall R0 R0m Γy E rd, mdenote Θ Ξ Γ H R0 -> ds_mod Θ Ξ R0 = Some R0m ->
       ds_entry R0m y = Some (Γy, E, rd) -> en_order Θ Ξ (Γy ++ Γ) E) ->
    dn_order Θ Ξ Γ (me_mem H y)
with un_order (Θ : gdeps) (Ξ : gstack) : ctx -> gunit -> Prop :=
| uno_body : forall Γ Δ sm Φ,
    bx_order Θ Ξ (Δ ++ Γ) Φ ->
    un_order Θ Ξ Γ (gu_mk Δ (md_body sm Φ))
| uno_alias : forall Γ Δ E,
    dn_order Θ Ξ (Δ ++ Γ) E ->
    un_order Θ Ξ Γ (gu_mk Δ (md_alias E))
with en_order (Θ : gdeps) (Ξ : gstack) : ctx -> gentry -> Prop :=
| eno_def : forall Γ b pv A oM, en_order Θ Ξ Γ (ge_def b pv A oM)
| eno_mod : forall Γ pv U,
    un_order Θ Ξ Γ U ->
    en_order Θ Ξ Γ (ge_mod pv U)
| eno_ind : forall Γ pv p k D Φ, en_order Θ Ξ Γ (ge_ind pv p k D Φ)
| eno_ind_alias : forall Γ pv Δ pm pq k Ns, en_order Θ Ξ Γ (ge_ind_alias pv Δ pm pq k Ns)
A record entry: its type former, where it is stored
| eno_record : forall Γ pv pr R Φ, en_order Θ Ξ Γ (ge_record pv pr R Φ)
with bx_order (Θ : gdeps) (Ξ : gstack) : ctx -> gmod -> Prop :=
| bxo_nil : forall Γ, bx_order Θ Ξ Γ gm_nil
| bxo_ext : forall Γ Φ x E,
    bx_order Θ Ξ Γ Φ ->
    bx_order Θ Ξ Γ (gm_ext Φ x E)
| bxo_open : forall Γ Φ H its,
    bx_order Θ Ξ Γ Φ ->
    (forall Φx, body_xp Θ Ξ Γ Φ Φx -> dn_order Θ Ξ (body_ctx Φx ++ Γ) H) ->
    (forall Φx, body_xp Θ Ξ Γ Φ Φx -> ig_order Θ Ξ (body_ctx Φx ++ Γ) H its) ->
    bx_order Θ Ξ Γ (gm_open Φ H its)
with ig_order (Θ : gdeps) (Ξ : gstack) : ctx -> modexp -> list iitem -> Prop :=
| igo_nil : forall Γ H, ig_order Θ Ξ Γ H nil
| igo_self : forall Γ H d pv its,
    ig_order Θ Ξ Γ H its ->
    ig_order Θ Ξ Γ H ((None, d, pv) :: its)
| igo_name : forall Γ H n d pv its,
    dn_order Θ Ξ Γ (me_mem H n) ->
    ig_order Θ Ξ Γ H its ->
    ig_order Θ Ξ Γ H ((Some n, d, pv) :: its).

The Decision Procedure

The conditions of an open on its own items, decided.
Definition items_ok_dec (its : list iitem) : { items_ok its } + { ~ items_ok its }.
Proof.
  unfold items_ok.
  destruct (both_dup its) as [n |]; [ right; intros [H _]; discriminate |].
  destruct (name_dup (List.map iitem_name its)) as [n |]; [ right; intros [_ H]; discriminate | left; split; reflexivity ].
Defined.

Equality of references to members of groups, decided.
Definition indref_eq_dec (x y : (qname * nat * ind_decl)%type) : {x = y} + {x <> y}.
Proof.
  destruct x as [[p1 k1] D1], y as [[p2 k2] D2].
  destruct (ind_decl_eq_dec_g exp_eq_dec centry_eq_dec D1 D2) as [<- | HD]; [| right; congruence ].
  destruct (Nat.eq_dec k1 k2) as [<- | Hk]; [| right; congruence ].
  destruct (qname_eq_dec p1 p2) as [<- | Hp]; [ left; reflexivity | right; congruence ].
Defined.

Lemma gc_unit_var : forall Θ Ξ x, gc_unit Θ Ξ (me_var x) = None.
Proof. reflexivity. Qed.

Lemma gc_unit_lit : forall Θ Ξ U, gc_unit Θ Ξ (me_lit U) = None.
Proof. reflexivity. Qed.

Lemma gc_unit_app : forall Θ Ξ H N, gc_unit Θ Ξ (me_app H N) = None.
Proof. reflexivity. Qed.

Lemma mod_qname_mem_in : forall H y p, mod_qname H = Some p -> mod_qname (me_mem H y) = Some (qname_in p y).
Proof. intros * Hq; cbn; rewrite Hq; reflexivity. Qed.

Lemma gc_unit_mem_qname : forall Θ Ξ H y r, gc_unit Θ Ξ (me_mem H y) = Some r -> mod_qname H = None -> False.
Proof. intros * Hu Hq; unfold gc_unit in Hu; cbn in Hu; rewrite Hq in Hu; discriminate. Qed.

The member whose module's open frame a module expression names, if any.
Definition frame_ind_of (Ξ : gstack) (H : modexp) : option (qname * nat * ind_decl) :=
  match mod_qname H with
  | Some p => gs_find_ind Ξ p
  | None => None
  end.

Lemma frame_ind_of_some : forall Ξ H r, frame_ind_of Ξ H = Some r -> exists p, mod_qname H = Some p /\ gs_find_ind Ξ p = Some r.
Proof. intros * Hf; unfold frame_ind_of in Hf; destruct (mod_qname H) as [p |]; [ eauto | discriminate ]. Qed.

Lemma frame_ind_of_none : forall Ξ H p r, frame_ind_of Ξ H = None -> mod_qname H = Some p -> gs_find_ind Ξ p = Some r -> False.
Proof. intros * Hf Hq Hk; unfold frame_ind_of in Hf; rewrite Hq, Hk in Hf; discriminate. Qed.

Lemma dn_of_frame_ind : forall Θ Ξ Γ H pq k D, frame_ind_of Ξ H = Some (pq, k, D) -> mdenote Θ Ξ Γ H (ds_ind nil pq k nil).
Proof. intros * Hf; destruct (frame_ind_of_some _ _ _ Hf) as (p & Hq & Hk); eapply dn_ind_frame; eassumption. Qed.

The record type whose module's open frame a module expression names, if any.
Definition frame_record_of (Ξ : gstack) (H : modexp) : option (qname * record_decl) :=
  match mod_qname H with
  | Some p => option_map (fun R => (p, R)) (gs_find_record Ξ p)
  | None => None
  end.

Lemma frame_record_of_some : forall Ξ H p R, frame_record_of Ξ H = Some (p, R) -> mod_qname H = Some p /\ gs_find_record Ξ p = Some R.
Proof.
  intros * Hf; unfold frame_record_of in Hf; destruct (mod_qname H) as [p' |]; [| discriminate ].
  destruct (gs_find_record Ξ p') as [R' |] eqn:E; cbn in Hf; [ injection Hf as <- <-; auto | discriminate ].
Qed.

Lemma frame_record_of_none : forall Ξ H p R, frame_record_of Ξ H = None -> mod_qname H = Some p -> gs_find_record Ξ p = Some R -> False.
Proof. intros * Hf Hq Hk; unfold frame_record_of in Hf; rewrite Hq, Hk in Hf; discriminate. Qed.

Lemma dn_of_frame_record : forall Θ Ξ Γ H p R, frame_record_of Ξ H = Some (p, R) -> mdenote Θ Ξ Γ H (ds_record nil p nil).
Proof. intros * Hf; destruct (frame_record_of_some _ _ _ _ Hf) as (Hq & Hk); eapply dn_record_frame; eassumption. Qed.
